63 public function startStep(StartStepRequest $request, BatchStep $batchStep)
64 {
65 \Log::debug('startStep called', [
66 'step_id' => $batchStep->id,
67 'user_id' => $request->user()->id,
68 'selected_line_id' => $request->session()->get('selected_line_id'),
69 ]);
70
71 if (! $this->stepBelongsToSelectedLine($request, $batchStep)) {
72 \Log::warning('stepBelongsToSelectedLine failed', [

81 $picksByMaterial = $this->reshapePicks($request->validated()['picks'] ?? []);

83 \Log::debug('startStep succeeded', ['step_id' => $batchStep->id]);

90 } catch (\Exception $e) {

97 /**
98 * Reshape the validated picks payload into a material-keyed map for the
99 * allocation service: [material_id => [['material_lot_id'=>, 'picked_qty'=>], ...]].
100 *
101 * @param array<int, array{material_id: int, lots: array<int, array{material_lot_id: int, picked_qty: float}>}> $picks
102 * @return array<int, array<int, array{material_lot_id: int, picked_qty: float}>>
103 */
104 private function reshapePicks(array $picks): array
105 {
106 $out = [];
107 foreach ($picks as $row) {
108 $materialId = (int) ($row['material_id'] ?? 0);
109 if ($materialId <= 0) {
110 continue;
111 }
112 foreach ($row['lots'] ?? [] as $lot) {
113 $out[$materialId][] = [
114 'material_lot_id' => (int) $lot['material_lot_id'],
115 'picked_qty' => (float) $lot['picked_qty'],
116 ];
117 }
118 }
119
120 return $out;
121 }

226 /**
227 * Stream a step document's uploaded file to the operator so they can read it
228 * before validating (line-scoped). Range-enabled via response()->file().
229 */
230 public function showDocumentFile(Request $request, BatchStepDocument $batchStepDocument)
231 {
232 $batchStepDocument->loadMissing('batchStep');
233 $step = $batchStepDocument->batchStep;
234
235 if (! $step || ! $this->stepBelongsToSelectedLine($request, $step)) {
236 abort(403);
237 }
238 abort_unless($batchStepDocument->file_path && \Illuminate\Support\Facades\Storage::exists($batchStepDocument->file_path), 404);
239
240 // Only render a narrow safelist inline; anything else (HTML, SVG, ...) is
241 // forced to download, so an uploaded document can't run script in the
242 // operator's session. nosniff stops the browser second-guessing the type.
243 $inlineSafe = ['application/pdf', 'image/png', 'image/jpeg', 'image/webp', 'image/gif', 'video/mp4', 'video/webm'];
244 $mime = $batchStepDocument->mime_type ?? 'application/octet-stream';
245 $disposition = in_array($mime, $inlineSafe, true) ? 'inline' : 'attachment';
246
247 return response()->file(\Illuminate\Support\Facades\Storage::path($batchStepDocument->file_path), [
248 'Content-Type' => $mime,
249 'Content-Disposition' => $disposition.'; filename="'.addslashes($batchStepDocument->original_name ?? 'document').'"',
250 'X-Content-Type-Options' => 'nosniff',
251 'Cache-Control' => 'private, max-age=3600',
252 ]);

255 /**
256 * Toggle a work-instruction checklist item on a step: tick it (recording who
257 * and when) or un-tick it. The item is defined on the step's template; we
258 * verify it belongs to this step's template and step number before recording.
259 */
260 public function toggleChecklistItem(Request $request, BatchStep $batchStep, TemplateStepChecklistItem $checklistItem)
261 {
262 if (! $this->stepBelongsToSelectedLine($request, $batchStep)) {
263 return back()->with('error', 'This step does not belong to the selected line.');
264 }
265
266 // Anti-IDOR: the item must belong to this step's template and step number.
267 $templateId = $batchStep->batch?->workOrder?->process_snapshot['template_id'] ?? null;
268 $checklistItem->loadMissing('templateStep:id,step_number');
269 if ($checklistItem->process_template_id !== $templateId
270 || $checklistItem->templateStep?->step_number !== $batchStep->step_number) {
271 return back()->with('error', 'This checklist item does not belong to this step.');
272 }
273
274 $existing = BatchStepChecklistCompletion::where('batch_step_id', $batchStep->id)
275 ->where('checklist_item_id', $checklistItem->id)
276 ->first();
277
278 if ($existing) {
279 $existing->delete();
280
281 return back()->with('success', 'Checklist item unchecked.');
282 }
283
284 BatchStepChecklistCompletion::create([
285 'batch_step_id' => $batchStep->id,
286 'checklist_item_id' => $checklistItem->id,
287 'checked_by_id' => $request->user()->id,
288 'checked_at' => now(),
289 ]);
290
291 return back()->with('success', 'Checklist item checked.');
292 }

294 /**
295 * Record (or overwrite) the operator's value for a typed step output (#B).
296 * Scalars write the typed column; a `picture` output sanitises + stores the
297 * uploaded image on the private disk. Re-recording soft-deletes the prior
298 * value (audit preserved) and inserts a fresh one.
299 */
300 public function recordOutput(Request $request, BatchStep $batchStep, TemplateStepOutput $output, ImageSanitizer $sanitizer)
301 {
302 if (! $this->stepBelongsToSelectedLine($request, $batchStep)) {
303 return back()->with('error', 'This step does not belong to the selected line.');
304 }
305
306 // Anti-IDOR: the output must belong to this step's template + step number.
307 $templateId = $batchStep->batch?->workOrder?->process_snapshot['template_id'] ?? null;
308 $output->loadMissing('templateStep:id,step_number');
309 if ($output->process_template_id !== $templateId
310 || $output->templateStep?->step_number !== $batchStep->step_number) {
311 return back()->with('error', 'This output does not belong to this step.');
312 }
313
314 $attrs = [
315 'batch_step_id' => $batchStep->id,
316 'output_id' => $output->id,
317 'recorded_by_id' => $request->user()->id,
318 'recorded_at' => now(),
319 ];
320
321 try {
322 if ($output->value_type === TemplateStepOutput::TYPE_PICTURE) {
323 $validated = $request->validate(['value' => ['required', 'image', 'max:10240']]);
324 $clean = $sanitizer->sanitize($validated['value']->getRealPath());
325 $path = 'batch-step-outputs/'.\Illuminate\Support\Str::random(40).'.'.$clean['extension'];
326 // Pin to the private disk explicitly: the default disk follows
327 // FILESYSTEM_DISK, and a `public` default would web-expose the file.
328 \Illuminate\Support\Facades\Storage::disk('local')->put($path, $clean['bytes']);
329 $attrs += [
330 'file_path' => $path,
331 'original_name' => $validated['value']->getClientOriginalName(),
332 'mime_type' => $clean['mime'],
333 'file_size' => strlen($clean['bytes']),
334 ];
335 } else {
336 $attrs += $this->scalarOutputAttrs($request, $output);
337 }
338 } catch (\InvalidArgumentException $e) {
339 return back()->with('error', __('The uploaded file is not a valid image.'));
340 } catch (\Illuminate\Validation\ValidationException $e) {
341 throw $e;
342 }
343
344 // Overwrite: soft-delete any live value, then insert the new one (keeps the
345 // partial-unique index happy and preserves the prior value as audit). Both
346 // steps run in one transaction with the live rows locked, so a failed insert
347 // can't leave the output with no value and two concurrent posts can't both
348 // pass the delete and then collide on the partial-unique index.
349 \Illuminate\Support\Facades\DB::transaction(function () use ($batchStep, $output, $attrs) {
350 BatchStepOutputValue::where('batch_step_id', $batchStep->id)
351 ->where('output_id', $output->id)
352 ->lockForUpdate()
353 ->get()->each->delete();
354 BatchStepOutputValue::create($attrs);
355 });
356
357 return back()->with('success', __('Output recorded'));
358 }

360 /**
361 * Validate + map a scalar output value to its typed column.
362 *
363 * @return array<string, mixed>
364 */
365 private function scalarOutputAttrs(Request $request, TemplateStepOutput $output): array
366 {
367 return match ($output->value_type) {
368 TemplateStepOutput::TYPE_NUMBER => [
369 'value_number' => $request->validate(['value' => ['required', 'numeric']])['value'],
370 ],
371 TemplateStepOutput::TYPE_BOOLEAN => [
372 // Require the field: without it a missing value would silently record
373 // `false` and satisfy the required-output gate. The UI posts '1'/'0'.
374 'value_boolean' => (bool) $request->validate([
375 'value' => ['required', 'boolean'],
376 ])['value'],
377 ],
378 TemplateStepOutput::TYPE_DATE => [
379 'value_date' => $request->validate(['value' => ['required', 'date']])['value'],
380 ],
381 TemplateStepOutput::TYPE_SELECT => [
382 'value_text' => $request->validate([
383 'value' => ['required', \Illuminate\Validation\Rule::in($output->options ?? [])],
384 ])['value'],
385 ],
386 default => [
387 'value_text' => $request->validate(['value' => ['required', 'string', 'max:5000']])['value'],
388 ],
389 };
390 }

467 public function store(Request $request)
468 {
469 $request->validate([
470 'work_order_id' => 'required|exists:work_orders,id',
471 'target_qty' => 'required|numeric|min:0.01',
472 'workstation_id' => 'nullable|exists:workstations,id',
473 'lot_number' => 'nullable|string|max:50',
474 'auto_lot' => 'nullable|boolean',
475 ]);
476
477 $workOrder = WorkOrder::find($request->input('work_order_id'));
478
479 if ($workOrder->line_id != $request->session()->get('selected_line_id')) {
480 return back()->with('error', 'This work order does not belong to the selected line.');
481 }
482
483 try {
484 $lotNumber = $request->input('lot_number');
485
486 if ($request->boolean('auto_lot') && ! $lotNumber) {
487 $lotNumber = $this->lotService->generateLot($workOrder->productType);
488 }
489
490 $this->workOrderService->createBatch(
491 $workOrder,
492 $request->input('target_qty'),
493 $request->input('workstation_id'),
494 $lotNumber,
495 );
496
497 return redirect()->route('operator.work-order.detail', $workOrder)
498 ->with('success', $lotNumber ? __('Batch created (LOT: :lot)', ['lot' => $lotNumber]) : __('Batch created'));
499 } catch (\Exception $e) {
500 return back()->with('error', 'Failed to create batch: '.$e->getMessage())->withInput();
501 }

525 public function qualityCheck(Request $request, Batch $batch)
526 {
527 $request->validate([
528 'production_quantity' => 'nullable|numeric|min:0',
529 'pallet_id' => 'nullable|integer|exists:pallets,id',
530 'samples' => 'required|array|min:1',
531 'samples.*.sample_number' => 'required|integer',
532 'samples.*.parameter_name' => 'required|string',
533 'samples.*.parameter_type' => 'required|in:measurement,pass_fail',
534 'samples.*.value_numeric' => 'nullable|numeric',
535 'samples.*.value_boolean' => 'nullable',
536 'samples.*.is_passed' => 'nullable',
537 ]);
538
539 $samples = collect($request->input('samples'))->map(fn ($s) => [
540 'sample_number' => $s['sample_number'],
541 'parameter_name' => $s['parameter_name'],
542 'parameter_type' => $s['parameter_type'],
543 'value_numeric' => $s['value_numeric'] ?? null,
544 'value_boolean' => isset($s['value_boolean']) ? (bool) $s['value_boolean'] : null,
545 'is_passed' => isset($s['is_passed']) ? (bool) $s['is_passed'] : null,
546 ])->toArray();
547
548 // Optional pallet link (#106): the pallet must belong to the batch's work order.
549 $pallet = null;
550 if ($request->filled('pallet_id')) {
551 $pallet = \App\Models\Pallet::find($request->input('pallet_id'));
552 if ($pallet && $pallet->work_order_id !== $batch->work_order_id) {
553 return back()->with('error', __('That pallet belongs to a different work order.'));
554 }
555 }
556
557 $check = $this->qcService->performCheck($batch, $request->user(), $samples, $request->input('production_quantity'), null, null, $pallet);
558
559 return back()->with($check->all_passed ? 'success' : 'warning',
560 $check->all_passed ? 'Quality check passed.' : 'Quality check recorded — some samples failed.');
561 }
