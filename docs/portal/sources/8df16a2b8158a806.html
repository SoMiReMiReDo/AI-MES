129 private function updateLineStatusLocked(Request $request, WorkOrder $workOrder)
130 {
131 $workOrder->assertProductionAvailable();
132 $lineId = $request->session()->get('selected_line_id');
133
134 if ($workOrder->line_id != $lineId) {
135 return back()->with('error', 'This work order does not belong to the selected line.');
136 }
137
138 $validated = $request->validate([
139 'line_status_id' => 'nullable|exists:line_statuses,id',
140 'produced_qty' => 'nullable|numeric|min:0',
141 ]);
142
143 $updates = ['line_status_id' => $validated['line_status_id']];
144
145 // In board_status mode: if the selected status is a "done" status, complete the work order
146 if ($validated['line_status_id']) {
147 $settingRows = DB::table('system_settings')->get()->keyBy('key');
148 $workflowMode = json_decode($settingRows['workflow_mode']->value ?? '"status"', true) ?? 'status';
149
150 if ($workflowMode === 'board_status') {
151 $newStatus = LineStatus::find($validated['line_status_id']);
152 if ($newStatus && $newStatus->is_done_status) {
153 if ($workOrder->usesStepLedger()) {
154 return back()->with('error', __('Record production on the work order steps.'));
155 }
156 $updates['status'] = WorkOrder::STATUS_DONE;
157 $updates['completed_at'] = now();
158 if (isset($validated['produced_qty'])) {
159 $updates['produced_qty'] = $validated['produced_qty'];
160 }
161 }
162 }
163 }
164
165 $workOrder->update($updates);
166
167 return back()->with('success', 'Status updated.');
168 }

170 /**
171 * JSON endpoint for polling — returns current queue counts.
172 */
173 public function check(Request $request)
174 {
175 $lineId = $request->session()->get('selected_line_id');
176 if (! $lineId) {
177 return response()->json(['active' => 0, 'workstation' => 0]);
178 }
179
180 $activeCount = WorkOrder::availableForProduction()->where('line_id', $lineId)
181 ->whereIn('status', WorkOrder::ACTIVE_STATUSES)
182 ->count();
183
184 $workstationCount = 0;
185 $settingRows = DB::table('system_settings')->get()->keyBy('key');
186 $trackingMode = json_decode($settingRows['production_tracking_mode']->value ?? '"per_operation"', true) ?? 'per_operation';
187 $routingEnabled = json_decode($settingRows['workstation_routing_enabled']->value ?? 'false', true) ?? false;
188
189 // Same selection and order source as queue(): with routing enabled the
190 // workstation may sit on another line and steps route across lines.
191 $selection = app(OperatorWorkstationSelection::class);
192 $workstation = $selection->resolve($request, (int) $lineId, allowOtherLines: $routingEnabled);
193
194 if ($workstation && in_array($trackingMode, ['per_operation', 'hybrid'])) {
195 $source = WorkOrder::availableForProduction()->whereIn('status', WorkOrder::ACTIVE_STATUSES)
196 ->when(! $routingEnabled, fn ($query) => $query->where('line_id', $lineId))
197 ->get();
198
199 // Ready at the workstation or starting there — the queue's two lists.
200 $workstationCount = $selection->workOrdersAt($source, $workstation, includeNotStarted: true)->count();
201 }
202
203 return response()->json([
204 'active' => $activeCount,
205 'workstation' => $workstationCount,
206 'timestamp' => now()->timestamp,
207 ]);
208 }

423 /**
424 * The order's serial units, newest first (at most 500), each with its last
425 * history row: two queries whatever the count.
426 *
427 * @return array{total: int, units: array<int, array<string, mixed>>}
428 */
429 private function serialUnitsOf(WorkOrder $workOrder): array
430 {
431 $query = \App\Models\SerialUnit::where('work_order_id', $workOrder->id);
432 $total = (clone $query)->count();
433 if ($total === 0) {
434 return ['total' => 0, 'units' => []];
435 }
436 $units = $query->with(['carton:id,carton_no', 'pallet:id,pallet_no', 'material:id,code,name'])
437 ->orderByDesc('id')->limit(500)->get();
438 $lastIds = \App\Models\UnitStepHistory::whereIn('serial_unit_id', $units->pluck('id'))
439 ->selectRaw('MAX(id) as id')->groupBy('serial_unit_id')->pluck('id');
440 $last = \App\Models\UnitStepHistory::whereIn('id', $lastIds)->with('workstation:id,code,name')->get()->keyBy('serial_unit_id');
441
442 return ['total' => $total, 'units' => $units->map(function (\App\Models\SerialUnit $u) use ($last) {
443 $h = $last->get($u->id);
444
445 return [
446 'id' => $u->id,
447 'serial_no' => $u->serial_no,
448 'psn' => $u->psn,
449 'status' => $u->status,
450 'material' => $u->material?->name,
451 'carton' => $u->carton?->carton_no,
452 'pallet' => $u->pallet?->pallet_no,
453 'last' => $h ? [
454 'parameters' => $h->parameters,
455 'result' => $h->result,
456 'workstation' => $h->workstation?->name,
457 'at' => $h->processed_at?->toIso8601String(),
458 ] : null,
459 ];
460 })->values()->all()];
461 }
