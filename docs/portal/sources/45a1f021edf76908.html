27/**
28 * Per-step quantity ledger and the transfer production flow: pieces logged as
29 * good at one station open the next station while the first is still working;
30 * scrap logged at a step never reaches the next one; a step can only be
31 * finished once nothing more can arrive and nothing is left waiting.
32 */

50 /** One batch of 10 through three plain sequential steps. */
51 private function makeBatch(float $qty = 10): Batch
52 {
53 $pt = ProductType::factory()->create();
54 $template = ProcessTemplate::factory()->create(['product_type_id' => $pt->id, 'is_active' => true]);
55 foreach ([1 => 'Mix', 2 => 'Bake', 3 => 'Pack'] as $n => $name) {
56 TemplateStep::factory()->create(['process_template_id' => $template->id, 'step_number' => $n, 'name' => $name]);
57 }
58
59 $wo = WorkOrder::factory()->create([
60 'line_id' => $this->line->id,
61 'planned_qty' => $qty,
62 'process_snapshot' => $template->load('steps')->toSnapshot(),
63 ]);
64
65 return app(WorkOrderService::class)->createBatch($wo, $qty);
66 }
67
68 private function step(Batch $batch, int $number): BatchStep
69 {
70 return $batch->steps()->where('step_number', $number)->first();
71 }

421 public function test_break_beam_pulses_go_through_the_ledger_in_transfer_flow(): void
422 {
423 ProductionFlow::set(ProductionFlow::TRANSFER);
424 $batch = $this->makeBatch();
425 $wo = $batch->workOrder;
426 $wo->update(['status' => WorkOrder::STATUS_IN_PROGRESS, 'counting_source' => WorkOrder::COUNTING_MACHINE]);
427 $mappings = [];
428 $pulse = function (int $stepNumber) use ($batch, &$mappings): array {
429 if (! isset($mappings[$stepNumber])) {
430 $step = $this->step($batch, $stepNumber);
431 $ws = Workstation::factory()->create(['line_id' => $this->line->id]);
432 $step->update(['workstation_id' => $ws->id, 'status' => BatchStep::STATUS_IN_PROGRESS]);
433 $mapping = $this->countStepMapping(['step_number' => $stepNumber, 'also_count_work_order' => true]);
434 $counters = app(\App\Services\Machine\MachineCounterService::class);
435 $counters->configure($counters->forSource($mapping), ['workstation_id' => $ws->id, 'batch_step_id' => $step->id,
436 'mode' => 'pulse', 'kind' => 'good', 'note' => 'Test assignment'], $this->operator->id);
437 $mappings[$stepNumber] = $mapping;
438 }
439 $result = app(ActionExecutor::class)->executeSingle($mappings[$stepNumber], ['event_id' => (string) \Illuminate\Support\Str::uuid(), 'timestamp' => now()->toISOString()]);
440 $this->assertSame('ok', $result['status'], (string) $result['message']);
441
442 return json_decode($result['message'], true);
443 };
444
445 // A pulse at the first station passes a piece on; it isn't finished output.
446 $outcome = $pulse(1);
447 $this->assertEquals(1.0, $this->step($batch, 1)->passed_qty);
448 $this->assertSame(BatchStep::STATUS_READY, $this->step($batch, 2)->status);
449 $this->assertEquals(0.0, $wo->fresh()->produced_qty);
450 $this->assertEquals(1, $outcome['applied_qty']);
451
452 // Nothing is waiting at the last station yet: a pulse there counts nothing.
453 $outcome = $pulse(3);
454 $this->assertEquals(0, $outcome['applied_qty']);
455 $this->assertEquals(0.0, $this->step($batch, 3)->passed_qty);
456 $this->assertEquals(0.0, $wo->fresh()->produced_qty);
457
458 // Once a piece reaches it, the last station's pulse is the order's output — counted once.
459 app(BatchService::class)->recordMachinePass($this->step($batch, 2), 1);
460 $outcome = $pulse(3);
461 $this->assertEquals(1, $outcome['applied_qty']);
462 $this->assertEquals(1.0, $this->step($batch, 3)->passed_qty);
