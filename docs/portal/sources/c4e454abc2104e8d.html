13/**
14 * OpenMES → ERP: warehouse documents production generated (#212).
15 *
16 * This is the "create the release / receipt document in the ERP" half of the
17 * integration: the ERP polls for material releases and product receipts it has
18 * not booked yet (`unsynced_only=1`), books its own counterpart, then calls
19 * `acknowledge` with its document number so the entry leaves the backlog.
20 *
21 * Reading requires `erp:stock:read`, acknowledging `erp:stock:write`; both are
22 * rate limited via the `erp-read` / `erp-import` limiters.
23 */
24class StockDocumentExportController extends Controller
25{
26 use BuildsCursorMeta;
27
28 public function index(StockExportRequest $request): JsonResponse
29 {
30 $query = StockDocument::query()
31 ->with([
32 'warehouse:id,code,name,erp_code',
33 'workOrder:id,order_no',
34 'lines.material:id,code,name',
35 'lines.productType:id,code,name',
36 'lines.materialLot:id,lot_number',
37 ])
38 ->orderBy('id');
39
40 if ($since = $request->input('since')) {
41 $query->where('updated_at', '>=', $since);
42 }
43
44 if ($type = $request->input('type')) {
45 $query->where('type', $type);
46 }
47
48 // Default to posted documents: a draft is not yet a real stock movement,
49 // so booking it in the ERP would be premature.
50 $query->where('status', $request->input('status', StockDocument::STATUS_POSTED));
51
52 if ($warehouse = $request->input('warehouse')) {
53 $query->whereHas('warehouse', fn ($q) => $q->where('code', $warehouse)->orWhere('erp_code', $warehouse));
54 }
55
56 if ($request->boolean('unsynced_only')) {
57 $query->notSynced();
58 }
59
60 $page = $query->cursorPaginate($request->perPage())->withQueryString();
61
62 return response()->json([
63 'data' => collect($page->items())->map(fn (StockDocument $doc) => $this->present($doc)),
64 'meta' => $this->cursorMeta($page),
65 ]);
66 }
67
68 /**
69 * Record the ERP's own document number against ours.
70 *
71 * The id is resolved here instead of via route-model binding on purpose:
72 * SubstituteBindings runs before this route's API-key middleware, so a bound
73 * model would be looked up before TenantContext is set and could reach another
74 * tenant's document. See ProductionExportController::show().
75 */
76 public function acknowledge(
77 AcknowledgeStockDocumentRequest $request,
78 int $stockDocument,
79 StockDocumentService $service,
80 ): JsonResponse {
81 $document = StockDocument::findOrFail($stockDocument);
82
83 $service->acknowledge($document, $request->input('erp_reference'));
84
85 // present() reads each line's material / product / lot, so load them here
86 // rather than paying a query per line.
87 return response()->json(['data' => $this->present($document->fresh([
88 'warehouse', 'workOrder', 'lines.material', 'lines.productType', 'lines.materialLot',
89 ]))]);
90 }
91
92 private function present(StockDocument $doc): array
93 {
94 return [
95 'id' => $doc->id,
96 'document_no' => $doc->document_no,
97 'type' => $doc->type,
98 'status' => $doc->status,
99 'direction' => $doc->direction() > 0 ? 'in' : 'out',
100 'warehouse_code' => $doc->warehouse?->code,
101 'warehouse_erp_code' => $doc->warehouse?->erp_code,
102 'work_order_no' => $doc->workOrder?->order_no,
103 'notes' => $doc->notes,
104 'erp_reference' => $doc->erp_reference,
105 'erp_synced_at' => $doc->erp_synced_at?->toIso8601String(),
106 'posted_at' => $doc->posted_at?->toIso8601String(),
107 'updated_at' => $doc->updated_at?->toIso8601String(),
108 'lines' => $doc->lines->map(fn ($line) => [
109 'material_code' => $line->material?->code,
110 'product_type_code' => $line->productType?->code,
111 'lot_number' => $line->effectiveLotNumber(),
112 'quantity' => (float) $line->quantity,
113 'unit_of_measure' => $line->unit_of_measure,
114 'notes' => $line->notes,
115 ])->values(),
116 ];
117 }
