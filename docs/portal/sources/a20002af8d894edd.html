9// Pages come from two places: this app, and any module installed under the
10// repository's modules/ directory. Both globs are resolved by Vite at build
11// time; a glob that matches nothing yields {}, so a build with no modules
12// installed simply has an empty second map. The lookup itself lives in
13// lib/pageResolver.js, where it can be tested.
14const corePages = import.meta.glob('./Pages/**/*.jsx', { eager: true });
15const modulePages = import.meta.glob('../../modules/*/resources/js/Pages/**/*.jsx', { eager: true });
16
17/**
18 * Start fetching the frontends of modules installed after this build, at import
19 * time.
20 *
21 * Not in setup(): Inertia resolves the *initial* page's component before it ever
22 * calls setup, so a module loaded there would always be one render too late —
23 * a hard refresh onto a module page would fail while client-side visits worked.
24 *
25 * The list is read straight off the Inertia payload, the same one Inertia boots
26 * from, so no extra request is needed to find out what to load.
27 */
28const modulesReady = loadModules(initialProps()?.moduleAssets);
29
30function initialProps() {
31 try {
32 // Inertia renders the initial payload either as a JSON <script> or in the
33 // root element's data-page attribute, depending on version. Read whichever
34 // is there: getting this wrong is silent — module loading simply never
35 // starts, and every module page falls through to the missing-page screen.
36 const json = document.querySelector('script[type="application/json"][data-page]');
37 const raw = json ? json.textContent : document.getElementById('app')?.dataset.page;
38
39 return JSON.parse(raw ?? '{}').props ?? null;
40 } catch {
41 // A malformed payload is Inertia's problem to report, not ours; skip
42 // module loading rather than breaking boot with a parse error.
43 return null;
44 }
45}
46
47// Screens that post with fetch() read the CSRF token from <meta name="csrf-token">.
48// That tag is rendered once, with the first page; an Inertia visit replaces the
49// page but not the head. Signing in regenerates the session token, so without
50// this the first scan after logging in (landing on a station by an Inertia
51// redirect) was refused with 419 until a reload. Every page carries the current
52// token as a prop: keep the tag in step with it.
53function syncCsrfToken(token) {
54 if (!token || typeof document === 'undefined') return;
55 let meta = document.querySelector('meta[name="csrf-token"]');
56 if (!meta) {
57 meta = document.createElement('meta');
58 meta.setAttribute('name', 'csrf-token');
59 document.head.appendChild(meta);
60 }
61 meta.setAttribute('content', token);
62}
63router.on('navigate', (event) => syncCsrfToken(event.detail.page.props?.csrf_token));

65createInertiaApp({
66 resolve: async (name) => {
67 // Awaiting here rather than before createInertiaApp keeps the resolver
68 // honest: whatever a module registered is visible by the time the first
69 // lookup happens, however Inertia orders its own startup.
70 await modulesReady;
71
72 const page = resolvePage(name, corePages, modulePages, window.__OPENMES_PAGES__ ?? {});
73 if (page) {
74 return page;
75 }
76
77 // A route outliving its page — a module that isn't installed, or a stale
78 // link. This used to throw, which meant a white screen and the reason
79 // only in the console. Render something the user can act on instead.
80 console.warn(`Inertia page not found: ${name}`);
81
82 return import('./Pages/_MissingPage.jsx').then((module) => ({
83 ...module,
84 default: (props) => module.default({ ...props, __pageName: name }),
85 }));
86 },
87 async setup({ el, App, props }) {
88 // Load the active locale's translation chunk before the first render so
89 // __() is ready and there's no flash of untranslated/wrong-language text.
90 await loadLocale(props.initialPage.props.locale ?? 'en');
91 // Set the active timezone from the Inertia prop.
92 setTimezone(props.initialPage.props.timezone);
93 // Tenant key for Reverb channel names (null-safe → 'g'), mirrors TenantScope.
94 window.__TENANT__ = props.initialPage.props.auth?.user?.tenant_id ?? 'g';
95 syncCsrfToken(props.initialPage.props.csrf_token);
96 // The design-system package ships no words of its own: the labels its
97 // controls need (a search box placeholder, an empty-result note) come
98 // from here, translated, once for the whole app.
99 const uiLabels = { searchPlaceholder: __('Search…'), noResultsLabel: __('No matches'), clearLabel: __('Clear') };
100 createRoot(el).render(<UILabelsProvider labels={uiLabels}><App {...props} /></UILabelsProvider>);
101 },
102 progress: { color: '#1e40af' },
103});
