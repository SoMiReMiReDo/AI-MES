102 /**
103 * Which of these work orders current stock cannot cover, and by what.
104 *
105 * Every material across every snapshot is loaded in one pass, so a planner
106 * board can be flagged without a query per tile. Each order is measured on
107 * its own against on-hand — orders are not netted against each other, so
108 * this answers "could this one run today", not "can the whole week".
109 *
110 * @param Collection<int, WorkOrder> $workOrders
111 * @return array<int, list<array<string, mixed>>> short lines keyed by work
112 * order id; covered orders are absent
113 */
114 public function shortagesForWorkOrders(Collection $workOrders): array
115 {
116 $workOrders = $workOrders->reject(fn ($order) => in_array($order->status, ['DONE', 'CANCELLED', 'REJECTED'], true));
117 $boms = [];
118 $ids = [];
119 $codes = [];
120
121 foreach ($workOrders as $workOrder) {
122 $bom = $workOrder->process_snapshot['bom'] ?? [];
123 if (empty($bom)) {
124 continue;
125 }
126
127 $boms[$workOrder->id] = $bom;
128
129 foreach ($bom as $bomItem) {
130 if (! empty($bomItem['material_id'])) {
131 $ids[] = $bomItem['material_id'];
132 } elseif (! empty($bomItem['material_code'])) {
133 $codes[] = $bomItem['material_code'];
134 }
135 }
136 }
137
138 if ($boms === []) {
139 return [];
140 }
141
142 $byId = $ids !== []
143 ? Material::whereIn('id', array_unique($ids))->get()->keyBy('id')
144 : collect();
145 $byCode = $codes !== []
146 ? Material::whereIn('code', array_unique($codes))->get()->keyBy('code')
147 : collect();
148
149 $covered = MaterialAllocation::whereIn('work_order_id', $workOrders->pluck('id'))
150 ->where('status', '!=', MaterialAllocation::STATUS_RETURNED)
151 ->whereHas('batch', fn ($query) => $query->where('status', '!=', 'CANCELLED'))
152 ->get()->groupBy('work_order_id')->map(fn ($rows) => $rows->groupBy('material_id')
153 ->map(fn ($items) => $items->sum(fn ($item) => max(0, (float) $item->allocated_qty - (float) $item->returned_qty))));
154 $out = [];
155
156 foreach ($workOrders as $workOrder) {
157 $bom = $boms[$workOrder->id] ?? null;
158 if (! $bom) {
159 continue;
160 }
161
162 $short = [];
163 $remainingCovered = ($covered->get($workOrder->id) ?? collect())->all();
164
165 $requirements = [];
166 foreach ($bom as $bomItem) {
167 $material = ! empty($bomItem['material_id'])
168 ? $byId->get($bomItem['material_id'])
169 : null;
170 if (! $material && ! empty($bomItem['material_code'])) {
171 $material = $byCode->get($bomItem['material_code']);
172 }
173
174 $key = $material ? 'id:'.$material->id : 'missing:'.($bomItem['material_id'] ?? $bomItem['material_code'] ?? json_encode($bomItem));
175 $requirements[$key] ??= ['material' => $material, 'item' => $bomItem, 'quantity' => 0.0];
176 $requirements[$key]['quantity'] += $this->calculateRequiredQty($bomItem, (float) $workOrder->planned_qty);
177 }
178
179 foreach ($requirements as $requirement) {
180 $material = $requirement['material'];
181 $bomItem = $requirement['item'];
182 $required = max(0, $requirement['quantity'] - ($remainingCovered[$material?->id] ?? 0));
183 if ($required <= 0) {
184 continue;
185 }
186 // Keep the signed accounting balance; operators need usable stock.
187 $available = max(0, (float) ($material?->available_quantity ?? 0));
188
189 if ($material && $available >= $required) {
190 continue;
191 }
192
193 $short[] = [
194 'material_code' => $bomItem['material_code'] ?? $material?->code,
195 'material_name' => $bomItem['material_name'] ?? $material?->name,
196 'unit_of_measure' => $bomItem['unit_of_measure'] ?? $material?->unit_of_measure,
197 'required_qty' => round($required, 4),
198 'available_qty' => round($available, 4),
199 'missing_qty' => round(max(0, $required - $available), 4),
200 'material_exists' => $material !== null,
201 // Available is on-hand minus what other batches have already
202 // reserved, so a full store can still read as zero available.
203 // Carry both halves: without them the shortage looks like the
204 // system losing stock rather than stock being spoken for.
205 'on_hand_qty' => round((float) ($material?->stock_quantity ?? 0), 4),
206 'reserved_qty' => round((float) ($material?->reserved_quantity ?? 0), 4),
207 ];
208 }
209
210 if ($short !== []) {
211 $out[$workOrder->id] = $short;
212 }
213 }
214
215 return $out;
216 }

218 /**
219 * @param array<int, array<string, mixed>> $bom
220 */
221 private function previewBom(array $bom, float $quantity): array
222 {
223 $preview = [];
224
225 // Bulk load materials referenced by the BOM to avoid N+1. The
226 // preview is read-only, so we deliberately skip lockForUpdate
227 // (also: locking outside a transaction is a no-op on Postgres
228 // and only adds noise/contention on SQLite).
229 $materialIds = [];
230 $materialCodes = [];
231 foreach ($bom as $bomItem) {
232 if (! empty($bomItem['material_id'])) {
233 $materialIds[] = $bomItem['material_id'];
234 } elseif (! empty($bomItem['material_code'])) {
235 $materialCodes[] = $bomItem['material_code'];
236 }
237 }
238
239 $materialsById = ! empty($materialIds)
240 ? Material::whereIn('id', array_unique($materialIds))->get()->keyBy('id')
241 : collect();
242 $materialsByCode = ! empty($materialCodes)
243 ? Material::whereIn('code', array_unique($materialCodes))->get()->keyBy('code')
244 : collect();
245
246 foreach ($bom as $bomItem) {
247 $material = null;
248 if (! empty($bomItem['material_id'])) {
249 $material = $materialsById->get($bomItem['material_id']);
250 }
251 if (! $material && ! empty($bomItem['material_code'])) {
252 $material = $materialsByCode->get($bomItem['material_code']);
253 }
254 $requiredQty = $this->calculateRequiredQty($bomItem, $quantity);
255 $available = $material?->available_quantity ?? 0;
256
257 $preview[] = [
258 'material_name' => $bomItem['material_name'] ?? $material?->name,
259 'material_code' => $bomItem['material_code'] ?? $material?->code,
260 'unit_of_measure' => $bomItem['unit_of_measure'] ?? $material?->unit_of_measure,
261 'required_qty' => $requiredQty,
262 // Available = on-hand minus what is already reserved by other batches.
263 'available_qty' => $available,
264 'on_hand_qty' => (float) ($material?->stock_quantity ?? 0),
265 'reserved_qty' => (float) ($material?->reserved_quantity ?? 0),
266 'sufficient' => $material ? $available >= $requiredQty : false,
267 'material_exists' => $material !== null,
268 'consumed_at' => $bomItem['consumed_at'] ?? 'start',
269 'step_number' => $bomItem['step_number'] ?? null,
270 'estimated_cost' => $material?->unit_price
271 ? round((float) $material->unit_price * $requiredQty, 2)
272 : null,
273 'currency' => $material?->price_currency,
274 ];
275 }
276
277 return $preview;
278 }

280 /**
281 * Build the WO-time lot-picking proposal for starting a given step: per
282 * lot-tracked material that this step start would allocate, the required
283 * quantity, the system's proposed lot split, and the candidate lots. Returns
284 * an empty array when lot tracking is off or nothing needs picking - the UI
285 * then skips the modal and starts the step directly. Read-only.
286 *
287 * @return array<int, array{material_id: int, material_name: ?string, material_code: ?string, unit_of_measure: ?string, required_qty: float, strategy: string, proposed: array, candidates: array}>
288 */
289 public function pickPreviewForStep(BatchStep $step): array
290 {
291 if (! $this->lotPicking->isLotTrackingEnabled()) {
292 return [];
293 }
294
295 $batch = $step->batch;
296 $bom = $batch->workOrder->process_snapshot['bom'] ?? [];
297 if (empty($bom)) {
298 return [];
299 }
300
301 // Start-items only allocate on the first start (batch still PENDING),
302 // mirroring BatchService::startStep's $wasPending gate.
303 $batchPending = $batch->status === Batch::STATUS_PENDING;
304 $stepNumber = $step->step_number;
305
306 $out = [];
307 foreach ($bom as $bomItem) {
308 $isStart = $this->isStartItem($bomItem) && $batchPending;
309 $isDuringThisStep = $this->isDuringItem($bomItem)
310 && (int) ($bomItem['step_number'] ?? 0) === $stepNumber;
311
312 if (! $isStart && ! $isDuringThisStep) {
313 continue;
314 }
315
316 $material = $this->resolveMaterialReadonly($bomItem);
317 if (! $material) {
318 continue;
319 }
320
321 // Phase 1: lots/batches only - serial-tracked materials are handled elsewhere.
322 if ($material->tracking_type === 'serial') {
323 continue;
324 }
325
326 // Skip materials already allocated for this batch (mirror allocateMatching's guard).
327 $alreadyAllocated = MaterialAllocation::where('batch_id', $batch->id)
328 ->where('material_id', $material->id)
329 ->exists();
330 if ($alreadyAllocated) {
331 continue;
332 }
333
334 $requiredQty = $this->calculateRequiredQty($bomItem, (float) $batch->target_qty);
335 $proposal = $this->lotPicking->proposePicks($material, $requiredQty);
336
337 $out[] = [
338 'material_id' => $material->id,
339 'material_name' => $bomItem['material_name'] ?? $material->name,
340 'material_code' => $bomItem['material_code'] ?? $material->code,
341 'unit_of_measure' => $bomItem['unit_of_measure'] ?? $material->unit_of_measure,
342 'required_qty' => $requiredQty,
343 'strategy' => $proposal['strategy'],
344 'proposed' => $proposal['proposed'],
345 'candidates' => $proposal['candidates'],
346 ];
347 }
348
349 return $out;
350 }

352 /**
353 * Mark allocations as consumed when batch is completed. Reads each
354 * allocation's consumed_qty (set by recordConsumption) and falls back
355 * to allocated_qty for any rows where the operator did not record an
356 * explicit number. Also releases the reservation and applies any
357 * leftover difference (returned + scrap) back to stock.
358 */
359 public function consumeForBatch(Batch $batch): void
360 {
361 DB::transaction(function () use ($batch) {
362 $allocations = MaterialAllocation::where('batch_id', $batch->id)
363 ->where('status', MaterialAllocation::STATUS_ALLOCATED)
364 ->lockForUpdate()
365 ->with(['material', 'lotPicks'])
366 ->get();
367
368 foreach ($allocations as $allocation) {
369 // Bridge the picked lots into the ISA-95 genealogy table so
370 // forward/backward traceability reflects what was actually
371 // consumed. consumeForBatch runs once per batch (allocations
372 // flip to CONSUMED), so this never double-writes.
373 $this->writeGenealogy($allocation);
374
375 // Use the operator-declared quantity when consumption was recorded —
376 // including an explicit zero (nothing used, return everything). Only
377 // fall back to the planned quantity when nothing was ever declared.
378 $actualConsumed = $allocation->consumption_recorded
379 ? (float) $allocation->consumed_qty
380 : (float) $allocation->allocated_qty;
381
382 $leftoverToReturn = max(0, (float) $allocation->allocated_qty - $actualConsumed - (float) $allocation->scrap_qty);
383
384 $this->releaseReservation($allocation->material, (float) $allocation->allocated_qty);
385
386 if ($leftoverToReturn > 0 && $allocation->material) {
387 $this->stockMovements->record(
388 $allocation->material,
389 StockMovement::TYPE_RETURN,
390 $leftoverToReturn,
391 sourceType: StockMovement::SOURCE_BATCH,
392 sourceId: $batch->id,
393 reason: StockMovementReason::make('Batch #:batch completed — leftover returned to stock', ['batch' => $batch->id]),
394 );
395 }
396
397 if ((float) $allocation->scrap_qty > 0 && $allocation->material) {
398 $this->stockMovements->record(
399 $allocation->material,
400 StockMovement::TYPE_SCRAP,
401 0, // scrap is a status change, not a stock delta — already left stock at allocation time
402 sourceType: StockMovement::SOURCE_BATCH,
403 sourceId: $batch->id,
404 reason: StockMovementReason::make('Batch #:batch scrap qty recorded', ['batch' => $batch->id]),
405 );
406 }
407
408 // Finalise the location balance against the quantity that actually
409 // stands. Deducts only the part not already booked by the operator's
410 // own entries, so a batch whose consumption was recorded step by step
411 // is not deducted twice. Scrap counts: it physically left the store
412 // too — only the leftover being returned above stayed behind.
413 $this->consumptionLocation->deduct(
414 $allocation,
415 $actualConsumed + (float) $allocation->scrap_qty,
416 );
417
418 $allocation->update([
419 'status' => MaterialAllocation::STATUS_CONSUMED,
420 'consumed_qty' => $actualConsumed,
421 'consumed_at' => now(),
422 // Snapshot the price so historical cost reports stay stable.
423 'unit_price_snapshot' => $actualConsumed > 0 ? $allocation->material?->unit_price : null,
424 'price_currency_snapshot' => $actualConsumed > 0 ? $allocation->material?->price_currency : null,
425 ]);
426 }
427 });
428 }

430 public function returnForBatch(Batch $batch): void
431 {
432 DB::transaction(function () use ($batch) {
433 $allocations = MaterialAllocation::where('batch_id', $batch->id)
434 ->where('status', MaterialAllocation::STATUS_ALLOCATED)
435 ->lockForUpdate()
436 ->with('material')
437 ->get();
438
439 foreach ($allocations as $allocation) {
440 if ($allocation->material) {
441 $this->stockMovements->record(
442 $allocation->material,
443 StockMovement::TYPE_RETURN,
444 (float) $allocation->allocated_qty,
445 sourceType: StockMovement::SOURCE_BATCH,
446 sourceId: $allocation->batch_id,
447 reason: StockMovementReason::make('Batch #:batch cancelled — return to stock', ['batch' => $allocation->batch_id]),
448 );
449 $this->releaseReservation($allocation->material, (float) $allocation->allocated_qty);
450 }
451
452 // Give back anything already taken off the location: a cancelled batch
453 // consumed nothing, so a location that was debited by an operator's
454 // entry must be made whole again.
455 $this->consumptionLocation->reverse($allocation);
456
457 // Lot tracking: return picked qty back to each lot.
458 $this->lotPicking->returnPicksForAllocation($allocation);
459
460 $allocation->update([
461 'status' => MaterialAllocation::STATUS_RETURNED,
462 'returned_qty' => $allocation->allocated_qty,
463 ]);
464 }
465 });
466 }

468 /**
469 * Record actual consumed quantity for a single allocation. Operator
470 * calls this from the post-step UI. Optionally records scrap.
471 */
472 public function recordConsumption(
473 MaterialAllocation $allocation,
474 float $actualConsumed,
475 float $scrap = 0,
476 ?string $notes = null,
477 ): MaterialAllocation {
478 if ($allocation->status !== MaterialAllocation::STATUS_ALLOCATED) {
479 throw new \DomainException('Allocation must be in `allocated` status to record consumption.');
480 }
481 if ($actualConsumed < 0 || $scrap < 0) {
482 throw new \InvalidArgumentException('Consumed and scrap quantities must be non-negative.');
483 }
484
485 return DB::transaction(function () use ($allocation, $actualConsumed, $scrap) {
486 $allocation->update([
487 'consumed_qty' => $actualConsumed,
488 'consumption_recorded' => true,
489 'scrap_qty' => $scrap,
490 // Snapshot the price so historical cost reports stay stable.
491 'unit_price_snapshot' => $actualConsumed > 0 ? $allocation->material?->unit_price : null,
492 'price_currency_snapshot' => $actualConsumed > 0 ? $allocation->material?->price_currency : null,
493 ]);
494
495 // Take it off the location it came from — consumed plus scrap, since both
496 // physically left the store. Same transaction as the quantity itself, so a
497 // refused deduction (location short, plant blocks negatives) leaves no
498 // consumed_qty claiming material that never moved.
499 $this->consumptionLocation->deduct($allocation->fresh(), $actualConsumed + $scrap);
500
501 return $allocation->fresh();
502 });
503 }

505 /**
506 * Mid-batch material adjustment (e.g. "operator added 5kg extra").
507 * Decrements stock + reserved by the delta and bumps allocated_qty.
508 */
509 public function adjustAllocation(
510 MaterialAllocation $allocation,
511 float $deltaQty,
512 User $user,
513 ?string $reason = null,
514 ): MaterialAllocation {
515 if ($allocation->status !== MaterialAllocation::STATUS_ALLOCATED) {
516 throw new \DomainException('Can only adjust allocations in `allocated` status.');
517 }
518 if ($deltaQty === 0.0) {
519 return $allocation;
520 }
521
522 $newAllocated = (float) $allocation->allocated_qty + $deltaQty;
523 if ($newAllocated < 0) {
524 throw new \InvalidArgumentException('Adjustment would make allocated_qty negative.');
525 }
526
527 return DB::transaction(function () use ($allocation, $deltaQty, $user, $reason) {
528 $material = $allocation->material;
529
530 if (! $material) {
531 throw new \DomainException('Allocation has no associated material.');
532 }
533
534 $this->stockMovements->record(
535 $material,
536 StockMovement::TYPE_ADJUSTMENT,
537 -$deltaQty,
538 user: $user,
539 sourceType: StockMovement::SOURCE_BATCH,
540 sourceId: $allocation->batch_id,
541 reason: $reason ?? StockMovementReason::make('Adjustment on batch #:batch', ['batch' => $allocation->batch_id]),
542 );
543
544 if ($deltaQty > 0) {
545 $material->increment('reserved_quantity', $deltaQty);
546 } else {
547 $material->decrement('reserved_quantity', abs($deltaQty));
548 }
549 \App\Sync\CollectionBroadcaster::flush($material); // increment/decrement bypass model events
550
551 $allocation->update([
552 'allocated_qty' => (float) $allocation->allocated_qty + $deltaQty,
553 'adjustment_qty' => (float) $allocation->adjustment_qty + $deltaQty,
554 ]);
555
556 return $allocation->fresh();
557 });
558 }

560 /**
561 * Return a leftover quantity from an in-flight allocation to stock (#99) —
562 * e.g. the operator over-issued and hands the surplus back before the batch
563 * completes. Books TYPE_RETURN, releases the reservation and restores the
564 * picked lots for the returned quantity.
565 *
566 * Crucially it DECREMENTS allocated_qty by the returned amount so the
567 * completion reconciler (consumeForBatch: leftover = allocated − consumed −
568 * scrap) never returns the same quantity a second time.
569 *
570 * @throws \DomainException|\InvalidArgumentException
571 */
572 public function returnQuantity(
573 MaterialAllocation $allocation,
574 float $qty,
575 User $user,
576 ?string $reason = null,
577 ): MaterialAllocation {
578 if ($qty <= 0) {
579 throw new \InvalidArgumentException('Return quantity must be positive.');
580 }
581
582 return DB::transaction(function () use ($allocation, $qty, $user, $reason) {
583 // Lock and re-read inside the transaction so two concurrent returns can't
584 // both validate the same returnable quantity and over-return.
585 $allocation = MaterialAllocation::query()->lockForUpdate()->findOrFail($allocation->getKey());
586
587 if ($allocation->status !== MaterialAllocation::STATUS_ALLOCATED) {
588 throw new \DomainException('Can only return material from an `allocated` allocation.');
589 }
590
591 $returnable = (float) $allocation->allocated_qty
592 - (float) $allocation->consumed_qty
593 - (float) $allocation->scrap_qty;
594 if ($qty > $returnable + 1e-9) {
595 throw new \InvalidArgumentException('Return quantity exceeds the unconsumed allocated quantity.');
596 }
597
598 $material = $allocation->material;
599
600 if (! $material) {
601 throw new \DomainException('Allocation has no associated material.');
602 }
603
604 $this->stockMovements->record(
605 $material,
606 StockMovement::TYPE_RETURN,
607 $qty,
608 user: $user,
609 sourceType: StockMovement::SOURCE_BATCH,
610 sourceId: $allocation->batch_id,
611 reason: $reason ?? StockMovementReason::make('Batch #:batch — unused material returned to stock', ['batch' => $allocation->batch_id]),
612 );
613
614 $this->releaseReservation($material, $qty);
615
616 // Lot tracking: hand the returned quantity back to the picked lots.
617 $this->lotPicking->returnPartialForAllocation($allocation, $qty);
618
619 $allocation->update([
620 // Shrink the allocation so consumeForBatch's leftover calc excludes what we just returned.
621 'allocated_qty' => (float) $allocation->allocated_qty - $qty,
622 'returned_qty' => (float) $allocation->returned_qty + $qty,
623 ]);
624
625 return $allocation->fresh();
626 });
627 }

631 /**
632 * @param array<int, array<int, array{material_lot_id: int|string, picked_qty: int|float|string}>> $picksByMaterial
633 * @param int|null $stepId Drives the stock-movement source (during-items) and batch_step_id.
634 * @param int|null $attributeStepId Sets batch_step_id only (start/end-items), leaving the
635 * stock-movement source as the batch - keeps genealogy attributable without changing accounting.
636 */
637 private function allocateMatching(
638 Batch $batch,
639 User $user,
640 \Closure $filter,
641 ?int $stepId = null,
642 array $picksByMaterial = [],
643 ?int $attributeStepId = null,
644 ): Collection {
645 $bom = $batch->workOrder->process_snapshot['bom'] ?? [];
646
647 if (empty($bom)) {
648 return collect();
649 }
650
651 $blockNegative = $this->blockNegativeStockEnabled();
652 $genealogyStepId = $stepId ?? $attributeStepId;
653
654 return DB::transaction(function () use ($batch, $user, $bom, $filter, $stepId, $genealogyStepId, $picksByMaterial, $blockNegative) {
655 $allocations = collect();
656
657 foreach ($bom as $bomItem) {
658 if (! $filter($bomItem)) {
659 continue;
660 }
661
662 $existing = MaterialAllocation::where('batch_id', $batch->id)
663 ->where('material_id', $bomItem['material_id'] ?? null)
664 ->first();
665 if ($existing) {
666 $allocations->push($existing);
667
668 continue;
669 }
670
671 $material = $this->resolveMaterial($bomItem);
672 if (! $material) {
673 continue;
674 }
675
676 $requiredQty = $this->calculateRequiredQty($bomItem, (float) $batch->target_qty);
677
678 if ($blockNegative && $material->available_quantity < $requiredQty) {
679 throw new InsufficientStockException(
680 $material,
681 $requiredQty,
682 $material->available_quantity,
683 );
684 }
685
686 // Stock leaves the warehouse + reservation increases.
687 $this->stockMovements->record(
688 $material,
689 StockMovement::TYPE_ALLOCATION,
690 -$requiredQty,
691 user: $user,
692 sourceType: $stepId ? StockMovement::SOURCE_BATCH_STEP : StockMovement::SOURCE_BATCH,
693 sourceId: $stepId ?: $batch->id,
694 reason: $stepId
695 ? StockMovementReason::make('Allocated to batch #:batch (step :step)', ['batch' => $batch->id, 'step' => $stepId])
696 : StockMovementReason::make('Allocated to batch #:batch', ['batch' => $batch->id]),
697 );
698 $material->increment('reserved_quantity', $requiredQty);
699 \App\Sync\CollectionBroadcaster::flush($material); // increment bypasses model events
700
701 $newAllocation = MaterialAllocation::create([
702 'batch_id' => $batch->id,
703 'batch_step_id' => $genealogyStepId,
704 'material_id' => $material->id,
705 'work_order_id' => $batch->work_order_id,
706 'allocated_qty' => $requiredQty,
707 'expected_qty' => $requiredQty,
708 'status' => MaterialAllocation::STATUS_ALLOCATED,
709 'allocated_by' => $user->id,
710 'allocated_at' => now(),
711 ]);
712
713 // Lot picking (opt-in via setting). Errors here roll back
714 // the surrounding transaction so stock/reserved stay consistent.
715 // When the operator supplied an explicit pick for this material
716 // (WO-time "suggest + override"), honour it; otherwise auto-pick.
717 if ($this->lotPicking->isLotTrackingEnabled()) {
718 $chosen = $picksByMaterial[$material->id] ?? null;
719 if (! empty($chosen)) {
720 $this->lotPicking->pickManualForAllocation($newAllocation, $material, $requiredQty, $chosen);
721 } else {
722 $this->lotPicking->pickForAllocation($newAllocation, $material, $requiredQty);
723 }
724 }
725
726 $allocations->push($newAllocation);
727 }
728
729 return $allocations;
730 });
731 }
732
733 /**
734 * Record one BatchStepLotConsumption row per picked lot for this allocation.
735 * Requires a step to attribute to (batch_step_id is NOT NULL); allocations
736 * without a step or without picks are skipped - genealogy stays optional.
737 */
738 private function writeGenealogy(MaterialAllocation $allocation): void
739 {
740 if (! $allocation->batch_step_id || $allocation->lotPicks->isEmpty()) {
741 return;
742 }
743
744 foreach ($allocation->lotPicks as $pick) {
745 BatchStepLotConsumption::create([
746 'batch_step_id' => $allocation->batch_step_id,
747 'material_lot_id' => $pick->material_lot_id,
748 'sublot_id' => null, // sublots are phase 2
749 'quantity_consumed' => $pick->picked_qty,
750 'consumed_at' => now(),
751 'recorded_by_id' => null, // system-recorded at batch completion
752 ]);
753 }
754 }

756 private function releaseReservation(?Material $material, float $qty): void
757 {
758 if (! $material || $qty <= 0) {
759 return;
760 }
761 $material->decrement('reserved_quantity', $qty);
762 \App\Sync\CollectionBroadcaster::flush($material); // decrement bypasses model events
763 }

782 private function resolveMaterial(array $bomItem): ?Material
783 {
784 $query = Material::query()->lockForUpdate();
785
786 if (! empty($bomItem['material_id'])) {
787 return $query->find($bomItem['material_id']);
788 }
789
790 if (! empty($bomItem['material_code'])) {
791 return $query->where('code', $bomItem['material_code'])->first();
792 }
793
794 return null;
795 }
796
797 /** Resolve a BOM item's material without locking (for read-only previews). */
798 private function resolveMaterialReadonly(array $bomItem): ?Material
799 {
800 if (! empty($bomItem['material_id'])) {
801 return Material::find($bomItem['material_id']);
802 }
803 if (! empty($bomItem['material_code'])) {
804 return Material::where('code', $bomItem['material_code'])->first();
805 }
806
807 return null;
808 }
809
810 private function calculateRequiredQty(array $bomItem, float $targetQty): float
811 {
812 $baseQty = ($bomItem['quantity_per_unit'] ?? 0) * $targetQty;
813 $scrapQty = $baseQty * (($bomItem['scrap_percentage'] ?? 0) / 100);
814
815 return round($baseQty + $scrapQty, 4);
816 }

824 } catch (\Throwable) {
