12/**
13 * Versioning of work-order configurations (#182).
14 *
15 * `work_orders.process_snapshot` stays the ACTIVE configuration — batches read it
16 * when their steps are generated. This service keeps the history around it: version 1
17 * is what the order was released with, and every applied change appends the next
18 * version together with the point from which it takes effect.
19 *
20 * Snapshots are append-only. Nothing here ever edits or deletes a row, because a
21 * snapshot is the record of what the shop floor was told to build.
22 */
23class WorkOrderSnapshotService
24{
25 /**
26 * Make sure the order has a version 1 recording the configuration it was
27 * released with. Idempotent, and safe to call on orders created before change
28 * control existed — that is exactly what it is for.
29 */
30 public function ensureBaseline(WorkOrder $workOrder, ?User $user = null): WorkOrderSnapshot
31 {
32 $existing = $workOrder->snapshots()->where('version', 1)->first();
33
34 if ($existing) {
35 return $existing;
36 }
37
38 return WorkOrderSnapshot::create([
39 'work_order_id' => $workOrder->id,
40 'version' => 1,
41 'snapshot' => $workOrder->process_snapshot ?? [],
42 // Version 1 has always applied: there is no earlier configuration for it
43 // to start after.
44 'effective_from' => ChangeEffectivePoint::Immediate,
45 'created_by_id' => $user?->id,
46 'tenant_id' => $workOrder->tenant_id,
47 ]);
48 }
49
50 /**
51 * Append the next configuration version and make it the order's active one.
52 *
53 * Must be called inside the caller's transaction: the version number is derived
54 * from the current maximum, so the work-order row lock taken here is what stops
55 * two concurrent applies from both writing version 3.
56 *
57 * @param array<string, mixed> $snapshot The new frozen configuration.
58 */
59 public function createVersion(
60 WorkOrder $workOrder,
61 array $snapshot,
62 ChangeEffectivePoint $effectiveFrom,
63 ?WorkOrderChangeRequest $changeRequest = null,
64 ?User $user = null,
65 ?int $effectiveFromBatchId = null,
66 ): WorkOrderSnapshot {
67 return DB::transaction(function () use (
68 $workOrder, $snapshot, $effectiveFrom, $changeRequest, $user, $effectiveFromBatchId
69 ) {
70 WorkOrder::whereKey($workOrder->getKey())->lockForUpdate()->first();
71 $workOrder->refresh();
72
73 $this->ensureBaseline($workOrder, $user);
74
75 $version = (int) $workOrder->snapshots()->max('version') + 1;
76
77 $created = WorkOrderSnapshot::create([
78 'work_order_id' => $workOrder->id,
79 'version' => $version,
80 'snapshot' => $snapshot,
81 'effective_from' => $effectiveFrom,
82 // Only meaningful for REMAINING_QUANTITY, and it is the whole point of
83 // it: "units 1–35 ran under v1, 36–100 under v2" stays answerable.
84 'effective_from_qty' => $effectiveFrom === ChangeEffectivePoint::RemainingQuantity
85 ? (float) $workOrder->produced_qty
86 : null,
87 'effective_from_batch_id' => $effectiveFromBatchId,
88 'change_request_id' => $changeRequest?->id,
89 'created_by_id' => $user?->id,
90 'tenant_id' => $workOrder->tenant_id,
91 ]);
92
93 // The active configuration moves with the new version; the previous one
94 // stays readable as its own snapshot row.
95 $workOrder->update([
96 'process_snapshot' => $snapshot,
97 'snapshot_version' => $version,
98 ]);
99
100 return $created;
101 });
102 }
103
104 /**
105 * The configuration a given produced unit was made under — the query behind
106 * "unit 12 was built to revision B, unit 80 to revision C".
107 *
108 * Only REMAINING_QUANTITY versions carry a unit boundary; NEXT_BATCH versions are
109 * attributed through `batches.snapshot_version` instead, so they are skipped here.
110 */
111 public function versionForUnit(WorkOrder $workOrder, float $unitNumber): ?WorkOrderSnapshot
112 {
113 return $workOrder->snapshots()
114 ->where('effective_from', ChangeEffectivePoint::RemainingQuantity->value)
115 ->where('effective_from_qty', '<', $unitNumber)
116 ->reorder('version', 'desc')
117 ->first()
118 ?? $workOrder->snapshots()->reorder('version', 'asc')->first();
119 }
120
121 /**
122 * Material requirements for the quantity still to be produced, from a given
123 * configuration. This is the "remaining material requirements recalculated" part
124 * of an applied change — reported, never written over existing allocations, so
125 * material already consumed keeps its own history.
126 *
127 * @param array<string, mixed> $snapshot
128 * @return array<int, array<string, mixed>>
129 */
130 public function remainingRequirements(WorkOrder $workOrder, array $snapshot): array
131 {
132 $remaining = max(0.0, (float) $workOrder->planned_qty - (float) $workOrder->produced_qty);
133
134 return collect($snapshot['bom'] ?? [])
135 ->filter(fn (array $row) => isset($row['material_id']))
136 ->map(function (array $row) use ($remaining) {
137 $perUnit = (float) ($row['quantity_per_unit'] ?? 0);
138 $scrap = (float) ($row['scrap_percentage'] ?? 0);
139
140 return [
141 'material_id' => (int) $row['material_id'],
142 'material_code' => $row['material_code'] ?? null,
143 'quantity_per_unit' => $perUnit,
144 'remaining_qty' => round($remaining * $perUnit * (1 + $scrap / 100), 4),
145 'consumed_at' => $row['consumed_at'] ?? 'start',
146 'step_number' => $row['step_number'] ?? null,
147 ];
148 })
149 ->values()
150 ->all();
151 }
