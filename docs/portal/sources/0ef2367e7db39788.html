12/**
13 * Picks lots to satisfy an allocation. The picking strategy decides the
14 * order in which available lots are considered:
15 *
16 * FEFO — first expiring first out (default; right answer for food/medical)
17 * FIFO — oldest received first (right answer for non-perishables)
18 * LIFO — newest received first (rare, for some accounting scenarios)
19 * MANUAL — caller supplies lot ids + quantities
20 */
21class LotPickingService
22{
23 /** Floating-point tolerance for quantity comparisons (matches test deltas). */
24 private const EPSILON = 0.0001;
25
26 /**
27 * Pick lots for the given allocation/material/quantity. Decrements
28 * each picked lot's available_qty, marks depleted lots, and writes
29 * allocation_lot_picks rows. Returns the picks collection.
30 *
31 * @throws InsufficientStockException when total available across lots
32 * is less than required.
33 */
34 public function pickForAllocation(
35 MaterialAllocation $allocation,
36 Material $material,
37 float $requiredQty,
38 ?string $strategy = null,
39 ): array {
40 $strategy = $strategy ?? $this->defaultStrategy();
41
42 return DB::transaction(function () use ($allocation, $material, $requiredQty, $strategy) {
43 $candidates = $this->orderedAvailableLots($material->id, $strategy);
44
45 $totalAvailable = (float) $candidates->sum(fn ($l) => $l->quantity_available);
46 if ($totalAvailable < $requiredQty) {
47 throw new InsufficientStockException($material, $requiredQty, $totalAvailable);
48 }
49
50 $remaining = $requiredQty;
51 $picks = [];
52
53 foreach ($candidates as $lot) {
54 if ($remaining <= 0) {
55 break;
56 }
57
58 $take = min($remaining, (float) $lot->quantity_available);
59
60 $picks[] = AllocationLotPick::create([
61 'tenant_id' => $allocation->tenant_id,
62 'material_allocation_id' => $allocation->id,
63 'material_lot_id' => $lot->id,
64 'picked_qty' => $take,
65 'picking_strategy' => $strategy,
66 ]);
67
68 $lot->decrement('quantity_available', $take);
69 $lot->refresh()->markConsumedIfEmpty();
70 \App\Sync\CollectionBroadcaster::flush($lot); // decrement bypasses model events
71
72 $remaining -= $take;
73 }
74
75 return $picks;
76 });
77 }

79 /**
80 * Pick the exact lots + quantities the operator chose at WO time
81 * (ERP-aligned "suggest + override"). Validates that each lot belongs to
82 * the material, is released/available, the per-lot quantity fits, and the
83 * chosen quantities sum to the required amount. Decrements each lot and
84 * writes allocation_lot_picks rows with the MANUAL strategy.
85 *
86 * @param array<int, array{material_lot_id: int|string, picked_qty: int|float|string}> $chosen
87 * @return array<int, AllocationLotPick>
88 *
89 * @throws \DomainException when the chosen lots/quantities are invalid
90 * @throws InsufficientStockException when a lot can't cover its line
91 */
92 public function pickManualForAllocation(
93 MaterialAllocation $allocation,
94 Material $material,
95 float $requiredQty,
96 array $chosen,
97 ): array {
98 return DB::transaction(function () use ($allocation, $material, $requiredQty, $chosen) {
99 // Normalise + collapse duplicate lot lines (the unique index on
100 // (allocation, lot) forbids two rows for the same lot anyway).
101 $lines = [];
102 foreach ($chosen as $row) {
103 $lotId = (int) ($row['material_lot_id'] ?? 0);
104 $qty = round((float) ($row['picked_qty'] ?? 0), 4);
105 if ($lotId <= 0 || $qty <= 0) {
106 throw new \DomainException(__('Each lot pick must reference a lot and a positive quantity.'));
107 }
108 $lines[$lotId] = ($lines[$lotId] ?? 0) + $qty;
109 }
110
111 if (empty($lines)) {
112 throw new \DomainException(__('At least one lot must be picked.'));
113 }
114
115 if (abs(array_sum($lines) - $requiredQty) > self::EPSILON) {
116 throw new \DomainException(__('Quantities must sum to the required amount'));
117 }
118
119 // Lock the chosen lots, scoped to this material + released status.
120 $lots = MaterialLot::whereIn('id', array_keys($lines))
121 ->where('material_id', $material->id)
122 ->where('status', MaterialLot::STATUS_RELEASED)
123 ->lockForUpdate()
124 ->get()
125 ->keyBy('id');
126
127 $picks = [];
128 foreach ($lines as $lotId => $qty) {
129 $lot = $lots->get($lotId);
130 if (! $lot) {
131 throw new \DomainException(__('No lots available for this material'));
132 }
133 if ($qty > (float) $lot->quantity_available + self::EPSILON) {
134 throw new InsufficientStockException($material, $qty, (float) $lot->quantity_available);
135 }
136
137 $picks[] = AllocationLotPick::create([
138 'tenant_id' => $allocation->tenant_id,
139 'material_allocation_id' => $allocation->id,
140 'material_lot_id' => $lot->id,
141 'picked_qty' => $qty,
142 'picking_strategy' => AllocationLotPick::STRATEGY_MANUAL,
143 ]);
144
145 $lot->decrement('quantity_available', $qty);
146 $lot->refresh()->markConsumedIfEmpty();
147 \App\Sync\CollectionBroadcaster::flush($lot); // decrement bypasses model events
148 }
149
150 return $picks;
151 });
152 }

154 /**
155 * Read-only proposal for the WO-time picking UI: the strategy that would
156 * run, the proposed (lot, qty) split for the required quantity, and the
157 * full candidate lot list the operator can pick from. Performs no locking
158 * and no mutations (locking outside a transaction is a no-op on Postgres).
159 *
160 * @return array{strategy: string, proposed: array<int, array{material_lot_id: int, picked_qty: float}>, candidates: array<int, array{id: int, lot_number: string, quantity_available: float, expiry_date: ?string, received_at: ?string, status: string}>}
161 */
162 public function proposePicks(Material $material, float $requiredQty, ?string $strategy = null): array
163 {
164 $strategy = $strategy ?? $this->defaultStrategy();
165
166 // Manual strategy proposes nothing; still order candidates by FEFO so
167 // the operator sees the most sensible lots first.
168 $orderStrategy = $strategy === 'manual' ? 'fefo' : $strategy;
169 $lots = $this->applyStrategyOrder($this->availableLotsQuery($material->id), $orderStrategy)->get();
170
171 $proposed = [];
172 if ($strategy !== 'manual') {
173 $remaining = $requiredQty;
174 foreach ($lots as $lot) {
175 if ($remaining <= 0) {
176 break;
177 }
178 $take = min($remaining, (float) $lot->quantity_available);
179 $proposed[] = ['material_lot_id' => $lot->id, 'picked_qty' => round($take, 4)];
180 $remaining -= $take;
181 }
182 }
183
184 return [
185 'strategy' => $strategy,
186 'proposed' => $proposed,
187 'candidates' => $lots->map(fn ($lot) => [
188 'id' => $lot->id,
189 'lot_number' => $lot->lot_number,
190 'quantity_available' => (float) $lot->quantity_available,
191 'expiry_date' => $lot->expiry_date?->toDateString(),
192 'received_at' => $lot->received_at?->toDateString(),
193 'status' => $lot->status,
194 ])->values()->all(),
195 ];
196 }

198 /**
199 * Return lots to stock when an allocation is cancelled. Re-opens
200 * depleted lots back to available status.
201 */
202 public function returnPicksForAllocation(MaterialAllocation $allocation): void
203 {
204 DB::transaction(function () use ($allocation) {
205 $picks = $allocation->lotPicks()->with('lot')->lockForUpdate()->get();
206
207 foreach ($picks as $pick) {
208 if (! $pick->lot) {
209 continue;
210 }
211 $pick->lot->increment('quantity_available', (float) $pick->picked_qty);
212 if ($pick->lot->status === MaterialLot::STATUS_CONSUMED && (float) $pick->lot->fresh()->quantity_available > 0) {
213 $pick->lot->update(['status' => MaterialLot::STATUS_RELEASED]);
214 }
215 \App\Sync\CollectionBroadcaster::flush($pick->lot); // increment bypasses model events
216 }
217
218 $allocation->lotPicks()->delete();
219 });
220 }

222 /**
223 * Return only part of an allocation's picked quantity to its lots (#99) —
224 * used when the operator hands back surplus mid-batch. Walks the picks
225 * newest-first (LIFO), restoring each lot's available quantity and reopening
226 * a depleted lot, then shrinks or removes the pick row so
227 * Σ picked_qty stays equal to the (now reduced) allocated_qty.
228 */
229 public function returnPartialForAllocation(MaterialAllocation $allocation, float $qty): void
230 {
231 if ($qty <= 0 || ! $this->isLotTrackingEnabled()) {
232 return;
233 }
234
235 DB::transaction(function () use ($allocation, $qty) {
236 $picks = $allocation->lotPicks()->with('lot')->lockForUpdate()->get()->reverse();
237 $remaining = $qty;
238
239 foreach ($picks as $pick) {
240 if ($remaining <= self::EPSILON) {
241 break;
242 }
243 $take = min($remaining, (float) $pick->picked_qty);
244 if ($take <= 0) {
245 continue;
246 }
247
248 if ($pick->lot) {
249 $pick->lot->increment('quantity_available', $take);
250 if ($pick->lot->status === MaterialLot::STATUS_CONSUMED && (float) $pick->lot->fresh()->quantity_available > 0) {
251 $pick->lot->update(['status' => MaterialLot::STATUS_RELEASED]);
252 }
253 \App\Sync\CollectionBroadcaster::flush($pick->lot); // increment bypasses model events
254 }
255
256 $newPicked = (float) $pick->picked_qty - $take;
257 if ($newPicked <= self::EPSILON) {
258 $pick->delete();
259 } else {
260 $pick->update(['picked_qty' => $newPicked]);
261 }
262
263 $remaining -= $take;
264 }
265 });
266 }

279 public function defaultStrategy(): string
280 {
281 try {
282 $row = DB::table('system_settings')->where('key', 'lot_picking_strategy')->value('value');
283 $val = json_decode($row ?? '"fefo"', true);
284
285 return in_array($val, ['fefo', 'fifo', 'lifo', 'manual'], true) ? $val : 'fefo';
286 } catch (\Throwable) {
287 return 'fefo';

291 /**
292 * Locked, strategy-ordered available lots for the auto-pick path.
293 * 'manual' returns empty - the caller supplies lots via pickManualForAllocation().
294 *
295 * @return \Illuminate\Support\Collection<int, MaterialLot>
296 */
297 private function orderedAvailableLots(int $materialId, string $strategy): \Illuminate\Support\Collection
298 {
299 if ($strategy === 'manual') {
300 return collect(); // caller chooses
301 }
302
303 return $this->applyStrategyOrder(
304 $this->availableLotsQuery($materialId)->lockForUpdate(),
305 $strategy,
306 )->get();
307 }
308
309 /** Base query for released lots with stock on hand (no lock, no ordering). */
310 private function availableLotsQuery(int $materialId): \Illuminate\Database\Eloquent\Builder
311 {
312 return MaterialLot::where('material_id', $materialId)
313 ->where('status', MaterialLot::STATUS_RELEASED)
314 ->where('quantity_available', '>', 0);
315 }

317 /**
318 * Apply the picking-strategy ordering to a lot query.
319 *
320 * @param \Illuminate\Database\Eloquent\Builder<MaterialLot> $q
321 * @return \Illuminate\Database\Eloquent\Builder<MaterialLot>
322 */
323 private function applyStrategyOrder(\Illuminate\Database\Eloquent\Builder $q, string $strategy): \Illuminate\Database\Eloquent\Builder
324 {
325 return match ($strategy) {
326 'fifo' => $q->orderBy('received_at')->orderBy('id'),
327 'lifo' => $q->orderByDesc('received_at')->orderByDesc('id'),
328 // default FEFO: nulls last (no expiry → use later)
329 default => $q->orderByRaw('expiry_date IS NULL, expiry_date ASC')->orderBy('received_at'),
330 };
331 }
