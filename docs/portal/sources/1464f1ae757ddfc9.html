189 /**
190 * Labels of the step's required checklist items (defined on the template
191 * step, resolved by template id + step number) that have not been ticked on
192 * this batch step yet - the items that block completion. Empty when none.
193 *
194 * @return \Illuminate\Support\Collection<int, string>
195 */
196 public function pendingRequiredChecklistLabels(): \Illuminate\Support\Collection
197 {
198 $templateId = $this->batch?->workOrder?->process_snapshot['template_id'] ?? null;
199 if (! $templateId) {
200 return collect();
201 }
202
203 $required = TemplateStepChecklistItem::where('process_template_id', $templateId)
204 ->where('is_required', true)
205 ->whereHas('templateStep', fn ($q) => $q->where('step_number', $this->step_number))
206 ->pluck('label', 'id');
207 if ($required->isEmpty()) {
208 return collect();
209 }
210
211 $done = $this->checklistCompletions()->pluck('checklist_item_id')->all();
212
213 return $required->reject(fn ($label, $id) => in_array($id, $done, true))->values();
214 }
215
216 /**
217 * Labels of required typed outputs on this step that the operator has not yet
218 * recorded. Resolved live from the work-order snapshot's template, mirroring
219 * pendingRequiredChecklistLabels(). Empty when nothing is pending.
220 */
221 public function pendingRequiredOutputs(): \Illuminate\Support\Collection
222 {
223 $templateId = $this->batch?->workOrder?->process_snapshot['template_id'] ?? null;
224 if (! $templateId) {
225 return collect();
226 }
227
228 $required = TemplateStepOutput::where('process_template_id', $templateId)
229 ->where('is_required', true)
230 ->whereHas('templateStep', fn ($q) => $q->where('step_number', $this->step_number))
231 ->pluck('label', 'id');
232 if ($required->isEmpty()) {
233 return collect();
234 }
235
236 $recorded = $this->outputValues()->pluck('output_id')->all();
237
238 return $required->reject(fn ($label, $id) => in_array($id, $recorded, true))->values();
239 }

288 /**
289 * Whether this step's sequence prerequisites are met (so it may move from
290 * PENDING to READY): the first step, any step when sequential enforcement is
291 * off, or a step whose immediate predecessor is DONE/SKIPPED. Does NOT factor
292 * in work-order blocking — that's re-checked at start time.
293 */
294 public function isPacking(): bool
295 {
296 return $this->kind === TemplateStep::KIND_PACKING;
297 }
298
299 public function prerequisitesMet(): bool
300 {
301 if (! config('openmmes.force_sequential_steps', true)) {
302 return true;
303 }
304
305 // Transfer flow: the step opens as soon as pieces are waiting at it,
306 // whether or not the previous station has finished the whole batch —
307 // and in any case once that station has closed, even if nothing reached
308 // this step (everything scrapped upstream), so the batch can still be
309 // closed instead of stalling on a step nobody can start.
310 if (ProductionFlow::isTransfer()) {
311 return $this->inflowClosed() || $this->availableQty() > 0;
312 }
313
314 if ($this->step_number === 1) {
315 return true;
316 }
317
318 $previousStep = $this->batch->steps()
319 ->where('step_number', $this->step_number - 1)
320 ->first();
321
322 return $previousStep && in_array($previousStep->status, [self::STATUS_DONE, self::STATUS_SKIPPED], true);
323 }
324
325 // ── Quantity ledger ───────────────────────────────────────────────────
326 //
327 // Each step counts what left it as good (`passed_qty`) or as scrap
328 // (`scrap_qty`). What arrives at a step is whatever the previous
329 // non-skipped step passed (the batch quantity for the first step), and the
330 // difference is what is still waiting at the station.
331
332 /**
333 * The nearest earlier step that is not SKIPPED — the one this step
334 * receives its pieces from. Null for the first effective step.
335 */
336 public function previousEffectiveStep(): ?self
337 {
338 $steps = $this->batch->relationLoaded('steps')
339 ? $this->batch->steps
340 : $this->batch->steps()->get();
341
342 return $steps
343 ->filter(fn (self $s) => $s->step_number < $this->step_number && $s->status !== self::STATUS_SKIPPED)
344 ->sortByDesc('step_number')
345 ->first();
346 }
347
348 /** Pieces that have reached this step so far. */
349 public function incomingQty(): float
350 {
351 $previous = $this->previousEffectiveStep();
352
353 return $previous ? (float) $previous->passed_qty : (float) $this->batch->target_qty;
354 }
355
356 /** Pieces that reached this step and have neither passed it nor been scrapped. */
357 public function availableQty(): float
358 {
359 if ($this->status === self::STATUS_SKIPPED) {
360 return 0.0;
361 }
362
363 return max(0.0, round($this->incomingQty() - (float) $this->passed_qty - (float) $this->scrap_qty, 2));
364 }
365
366 /** True once nothing more can arrive: the feeding step is closed (or there is none). */
367 public function inflowClosed(): bool
368 {
369 $previous = $this->previousEffectiveStep();
370
371 return $previous === null || $previous->status === self::STATUS_DONE;
372 }

374 /** A production stop gates repeatable output logs as well as starting work. */
375 public function productionBlocker(): ?string
376 {
377 $order = $this->batch->workOrder;
378 if ($message = $order->plannedStartBlocker()) {
379 return $message;
380 }
381 if (ProductionFlow::isTransfer() && $order->isMachineCounted()
382 && app(\App\Services\Machine\MachineCountingCompatibility::class)->legacySources($order->line_id)) {
383 return __('Migrate legacy machine channels before recording transfer production.');
384 }
385 if (in_array($this->batch->status, [Batch::STATUS_CANCELLED, Batch::STATUS_DONE], true)
386 || in_array($order->status, [...WorkOrder::TERMINAL_STATUSES, WorkOrder::STATUS_PAUSED, WorkOrder::STATUS_CHANGE_HOLD], true)) {
387 return __('Production is stopped for this work order.');
388 }
389 if ($order->isBlocked()) {
390 return __('Resolve the blocking issues before recording production.');
391 }
392 if (QualityControlTask::hasOpenBlockingForBatch($this->batch_id)) {
393 return __('A required quality control is outstanding for this batch and must be completed first.');
394 }
395
396 return null;
397 }
398
399 /**
400 * In transfer flow, why this step cannot be finished yet (null when it can).
401 * Whole-batch flow never blocks here: finishing passes the remainder along.
402 */
403 public function completionBlocker(): ?string
404 {
405 if (! ProductionFlow::isTransfer()) {
406 return null;
407 }
408
409 if (! $this->inflowClosed()) {
410 return __('Pieces can still arrive from :step — finish that step first.', [
411 'step' => $this->previousEffectiveStep()?->name,
412 ]);
413 }
414
415 $waiting = $this->availableQty();
416 if ($waiting > 0) {
417 return __(':qty pieces are still waiting at this step — log them as good or scrap before finishing.', [
418 'qty' => rtrim(rtrim(number_format($waiting, 2, '.', ''), '0'), '.'),
419 ]);
420 }
421
422 return null;
423 }

425 /**
426 * Check if this step can be started: it must be READY (the normal case,
427 * after promotion) or a still-PENDING step whose prerequisites are already
428 * met (safety net for steps created outside the promotion path), the work
429 * order must not be blocked. The operator UI only offers Start on READY.
430 */
431 public function canStart(): bool
432 {
433 if (! in_array($this->status, [self::STATUS_READY, self::STATUS_PENDING], true)) {
434 return false;
435 }
436
437 if ($this->batch->workOrder->isBlocked() || $this->batch->workOrder->plannedStartBlocker()) {
438 return false;
439 }
440
441 return $this->prerequisitesMet();
442 }
443
444 /**
445 * Check if this step can be completed.
446 */
447 public function canComplete(): bool
448 {
449 return $this->status === self::STATUS_IN_PROGRESS;
450 }

452 /**
453 * Can this step be skipped? Only optional steps or members of a variant
454 * group, and only while still pending/in progress.
455 */
456 public function canSkip(): bool
457 {
458 return in_array($this->status, [self::STATUS_PENDING, self::STATUS_READY, self::STATUS_IN_PROGRESS], true)
459 && ($this->is_optional || $this->variant_group !== null);
460 }
