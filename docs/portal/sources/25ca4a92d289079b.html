54 /**
55 * Swap the bearer token for the next request.
56 *
57 * forgetGuards() is what makes switching users mid-test actually switch them:
58 * the guard resolves the user once and keeps it for the rest of the test,
59 * so without this an authorization test would silently re-use the first caller.
60 */
61 private function asUser(string $token): self
62 {
63 $this->app['auth']->forgetGuards();
64
65 return $this->withHeader('Authorization', "Bearer {$token}");
66 }
67
68 private function asSupervisor(): self
69 {
70 return $this->asUser($this->supervisorToken);
71 }
72
73 private function asOperator(): self
74 {
75 return $this->asUser($this->operatorToken);
76 }

78 /** Stop the order for an engineering change — the state a change request starts from. */
79 private function stopForChange(): void

81 $this->asSupervisor()->postJson("/api/v1/work-orders/{$this->workOrder->id}/stop", [
82 'type' => 'ENGINEERING_CHANGE',

88 private function createRequest(array $overrides = []): WorkOrderChangeRequest
89 {
90 $response = $this->asSupervisor()->postJson(
91 "/api/v1/work-orders/{$this->workOrder->id}/change-requests",
92 array_merge([
93 'title' => 'Increase planned quantity',
94 'reason' => 'Customer extended the order.',
95 'proposed' => ['planned_qty' => 150],
96 ], $overrides),
97 );
98
99 $response->assertStatus(201);
100
101 return WorkOrderChangeRequest::findOrFail($response->json('data.id'));
102 }
