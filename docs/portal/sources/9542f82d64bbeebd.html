1# Live-edit overlay for contributors who want to edit source files on the
2# server (or locally) without installing Node/PHP/Composer.
3#
4# Usage:
5# docker compose -f docker-compose.yml -f docker-compose.dev.yml up -d
6#
7# What it adds:
8# - `frontend` service running `vite build --watch` so .jsx/.css/.blade.php
9# edits in ./backend/ are rebuilt to ./backend/public/build/ within ~100ms.
10# - Bind mounts on the backend container so PHP/Blade edits are live too
11# (no image rebuild needed for application code changes).
12#
13# Not for production: the image's baked-in `composer install` + `npm run build`
14# already produce a self-contained production container. Use this overlay only
15# in development / FTP-edit-on-staging-server workflows.
16
17services:
18 backend:
19 # Run the SAME production server as base — RoadRunner Octane (image CMD) —
20 # so the dev/live-edit stack is concurrent and robust (no single-threaded
21 # `artisan serve` serialization hang, no buggy `php -S` multi-worker 502s).
22 # Live .jsx editing still works: the `frontend` watcher rebuilds public/build
23 # (bind-mounted below) and the FlushViteManifest Octane listener (enabled by
24 # OCTANE_LIVE_RELOAD) re-reads the manifest each request so rebuilds show on
25 # refresh. PHP/Blade edits need `php artisan octane:reload` (Octane keeps the
26 # app booted in memory). See CLAUDE.md → "Dev / live editing".
27 environment:
28 OCTANE_LIVE_RELOAD: "true"
29 volumes:
30 # Composer manifest — bind-mounted (read-write) so `composer require/remove`
31 # run INSIDE the container write straight through to the host repo (and host
32 # edits are seen by the container), instead of drifting in the image layer.
33 # vendor/ is intentionally NOT mounted (stays image-layer for speed), so after
34 # changing these re-run `composer install` in the container — same pattern as
35 # `npm ci` for the frontend.
36 - ./backend/composer.json:/var/www/html/composer.json
37 - ./backend/composer.lock:/var/www/html/composer.lock
38 # Live application code (PHP, Blade, JSX source, routes, config).
39 # Keep storage/ and bootstrap/cache/ as named volumes (defined in the
40 # base compose file) so they survive across rebuilds.
41 - ./backend/app:/var/www/html/app
42 - ./backend/bootstrap/app.php:/var/www/html/bootstrap/app.php
43 - ./backend/config:/var/www/html/config
44 - ./backend/database:/var/www/html/database
45 - ./backend/lang:/var/www/html/lang
46 - ./backend/modules:/var/www/html/modules
47 - ./backend/public:/var/www/html/public
48 - ./backend/resources:/var/www/html/resources
49 - ./backend/routes:/var/www/html/routes
50 - ./backend/tests:/var/www/html/tests
51
52 # Reverb sees live PHP (broadcast auth + CollectionBroadcaster); restart it
53 # after editing those (it doesn't hot-reload like Octane).
54 reverb:
55 environment:
56 E2E_CONTROL_ENABLED: "true"
57 volumes:
58 - ./backend/app:/var/www/html/app
59 - ./backend/bootstrap/app.php:/var/www/html/bootstrap/app.php
60 - ./backend/config:/var/www/html/config
61 - ./backend/database:/var/www/html/database
62 - ./backend/routes:/var/www/html/routes

64 frontend:
65 image: ghcr.io/mes-open/openmes:${OPENMES_VERSION:-latest}
66 build:
67 context: .
68 dockerfile: backend/Dockerfile
69 container_name: ${OPENMES_NAME_PREFIX:-openmmes}-frontend
70 working_dir: /var/www/html
71 # Bypass the image ENTRYPOINT (docker-entrypoint.sh runs migrations/seeders
72 # and needs DB creds) — the watcher only builds assets, no DB, no app boot.
73 entrypoint: ["sh", "-c"]
74 # The production image strips node_modules; restore them, then watch.
75 # vite build --watch rebuilds public/build on every .jsx/.css edit (~100ms).
76 # Single-element list so the whole script reaches `sh -c` as one argument.
77 command: ["npm ci && npm run watch"]
78 volumes:
79 # Source the watcher needs to see. modules/ too: app.jsx globs
80 # modules/*/resources/js/Pages, so without this mount every module page
81 # is missing from the bundle and its routes render _MissingPage.
82 - ./backend/resources:/var/www/html/resources
83 - ./backend/modules:/var/www/html/modules:ro
84 - ./backend/lang:/var/www/html/lang:ro
85 - ./backend/vite.config.js:/var/www/html/vite.config.js:ro
86 - ./backend/package.json:/var/www/html/package.json:ro
87 - ./backend/package-lock.json:/var/www/html/package-lock.json:ro
88 # Workspace UI package — backend/package.json depends on it via
89 # `file:../packages/ui` (resolves to /var/www/packages/ui from the WORKDIR).
90 # Required for the runtime `npm ci` here, and lets edits to @openmes/ui
91 # rebuild live just like backend/resources.
92 - ./packages:/var/www/packages
93 # Output the watcher writes — bind-mounted into the backend container too
94 # so the live HTTP server picks up rebuilt bundles immediately.
95 - ./backend/public/build:/var/www/html/public/build
96 restart: unless-stopped
97 networks:
98 - openmmes-network
