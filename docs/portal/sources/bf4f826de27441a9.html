5use App\Events\CollectionChanged;

12/**
13 * Collection deltas go out only once the write is committed. Model events fire
14 * mid-transaction, so a create that later rolls back used to push a row that
15 * never existed into every open list — and no later delta ever removed it.
16 */
17class CollectionBroadcastAfterCommitTest extends TestCase
18{
19 use RefreshDatabase;
20
21 /** @var array<int, CollectionChanged> */
22 private array $captured = [];
23
24 protected function setUp(): void
25 {
26 parent::setUp();
27 Event::listen(CollectionChanged::class, function (CollectionChanged $e) {
28 $this->captured[] = $e;
29 });
30 }
31
32 /** @return array<int, string> order numbers broadcast to work_orders_all */
33 private function broadcastOrderNos(): array
34 {
35 return collect($this->captured)
36 ->where('collection', 'work_orders_all')
37 ->pluck('row.order_no')
38 ->all();
39 }
40
41 public function test_a_rolled_back_create_is_never_broadcast(): void
42 {
43 try {
44 DB::transaction(function () {
45 WorkOrder::factory()->create(['order_no' => 'PHANTOM-1']);
46 throw new \RuntimeException('component generation failed');
47 });
48 } catch (\RuntimeException) {
49 }
50
51 $this->assertDatabaseMissing('work_orders', ['order_no' => 'PHANTOM-1']);
52 $this->assertNotContains('PHANTOM-1', $this->broadcastOrderNos());
53 }
54
55 public function test_a_committed_create_is_broadcast_after_the_commit(): void
56 {
57 DB::transaction(function () {
58 WorkOrder::factory()->create(['order_no' => 'REAL-1']);
59
60 $this->assertNotContains('REAL-1', $this->broadcastOrderNos(), 'Broadcast before the commit.');
61 });
62
63 $this->assertContains('REAL-1', $this->broadcastOrderNos());
64 }
65
66 public function test_a_write_outside_a_transaction_is_broadcast_immediately(): void
67 {
68 WorkOrder::factory()->create(['order_no' => 'DIRECT-1']);
69
70 $this->assertContains('DIRECT-1', $this->broadcastOrderNos());
71 }
