26 /**
27 * Start a batch step.
28 *
29 * @param array<int, array<int, array{material_lot_id: int|string, picked_qty: int|float|string}>> $picksByMaterial
30 * Operator-chosen lot picks keyed by material id (WO-time "suggest +
31 * override"). Empty → automatic FEFO/FIFO/LIFO picking as before.
32 *
33 * @throws \Exception
34 */
35 public function startStep(BatchStep $step, User $user, array $picksByMaterial = []): BatchStep
36 {
37 return DB::transaction(function () use ($step, $user, $picksByMaterial) {
38 $step = $this->lockProductionStep($step);
39 $step->batch->workOrder->assertProductionAvailable();
40 if (ProductionFlow::isTransfer() && ($blocker = $step->productionBlocker())) {
41 throw new \DomainException($blocker);
42 }
43
44 // Enforce workstation routing (if enabled)
45 $this->guardWorkstationRouting($step, $user);
46
47 // Hard gate: an outstanding blocking quality control must be done
48 // before more work happens on this batch (#105).
49 if (QualityControlTask::hasOpenBlockingForBatch($step->batch_id)) {
50 throw new \Exception(__('A required quality control is outstanding for this batch and must be completed first.'));
51 }
52
53 // Validate step can be started
54 if (! $step->canStart()) {
55 $this->throwValidationError($step);
56 }
57
58 $batch = $step->batch;
59 $wasPending = $batch->status === Batch::STATUS_PENDING;
60
61 // Production hold (setting): a batch does not start production while
62 // stock cannot cover its order - the planner's shortage check, enforced.
63 if ($wasPending && self::holdsOnMaterialShortage()) {
64 $this->assertMaterialAvailable($batch->workOrder);
65 }
66
67 // Start the step
68 $step->update([
69 'status' => BatchStep::STATUS_IN_PROGRESS,
70 'started_at' => now(),
71 'started_by_id' => $user->id,
72 ]);
73
74 // Update batch status
75 $this->updateBatchStatus($batch);
76
77 // Allocate materials when batch first transitions to IN_PROGRESS
78 // (covers BOM rows with consumed_at='start' or unspecified). Attribute
79 // these allocations to this (first) step for genealogy.
80 if ($wasPending && $batch->fresh()->status === Batch::STATUS_IN_PROGRESS) {
81 $this->allocationService->allocateForBatch($batch, $user, $picksByMaterial, attributeStepId: $step->id);
82 }
83
84 // Always check for BOM rows targeted at *this* step (consumed_at='during').
85 $this->allocationService->allocateForStep($step, $user, $picksByMaterial);
86
87 // Update work order status
88 $this->workOrderService->updateWorkOrderStatus($batch->workOrder);
89
90 // Quality-control triggers: batch just entered production (#105).
91 if ($wasPending && $batch->fresh()->status === Batch::STATUS_IN_PROGRESS) {
92 $this->qualityTriggerService->fireInProduction($batch->fresh());
93 }
94
95 return $step->fresh();
96 });
97 }

109 /** @throws \DomainException naming what is missing and how much */
110 private function assertMaterialAvailable(WorkOrder $workOrder): void
111 {
112 $short = $this->allocationService->shortagesForWorkOrders(collect([$workOrder]))[$workOrder->id] ?? [];
113 if ($short === []) {
114 return;
115 }
116 $items = collect($short)->map(fn ($line) => trim(($line['material_code'] ?? $line['material_name'] ?? '?').' ('.__('missing :qty :unit', [
117 'qty' => rtrim(rtrim(number_format((float) $line['missing_qty'], 3, '.', ''), '0'), '.'),
118 'unit' => $line['unit_of_measure'] ?? '',
119 ]).')'))->implode(', ');
120
121 throw new \DomainException(__('Production hold: stock does not cover order :order. Missing: :items.', ['order' => $workOrder->order_no, 'items' => $items]));

124 /**
125 * Complete a batch step.
126 *
127 * @throws \Exception
128 */
129 public function completeStep(BatchStep $step, User $user, array $data = []): BatchStep
130 {
131 return DB::transaction(function () use ($step, $user, $data) {
132 $step = $this->lockProductionStep($step);
133 $step->batch->workOrder->assertProductionAvailable();
134 if (ProductionFlow::isTransfer() && ($blocker = $step->productionBlocker())) {
135 throw new \DomainException($blocker);
136 }
137
138 // Enforce workstation routing (if enabled)
139 $this->guardWorkstationRouting($step, $user);
140
141 // Validate step can be completed
142 if (! $step->canComplete()) {
143 throw new \Exception('Step cannot be completed. Current status: '.$step->status);
144 }
145
146 // Transfer flow: nothing may still be on its way here, and nothing
147 // may be left waiting, before the step is closed.
148 if ($blocker = $step->completionBlocker()) {
149 throw new \DomainException($blocker);
150 }
151
152 // Document control: a mandatory, validatable document attached to this
153 // step must be validated before the step can be completed.
154 $pendingDocs = $step->blockingDocuments()->pluck('name');
155 if ($pendingDocs->isNotEmpty()) {
156 throw new \Exception(__(
157 'This step is blocked: the mandatory document(s) ":docs" must be validated before it can be completed.',
158 ['docs' => $pendingDocs->implode(', ')],
159 ));
160 }
161
162 // Work-instruction control: required checklist items on this step must
163 // be ticked off before it can be completed.
164 $pendingChecklist = $step->pendingRequiredChecklistLabels();
165 if ($pendingChecklist->isNotEmpty()) {
166 throw new \Exception(__(
167 'This step is blocked: the required checklist item(s) ":items" must be completed before it can be completed.',
168 ['items' => $pendingChecklist->implode(', ')],
169 ));
170 }
171
172 // Output control: required typed outputs on this step must be recorded
173 // by the operator before it can be completed.
174 $pendingOutputs = $step->pendingRequiredOutputs();
175 if ($pendingOutputs->isNotEmpty()) {
176 throw new \Exception(__(
177 'This step is blocked: the required output(s) ":items" must be recorded before it can be completed.',
178 ['items' => $pendingOutputs->implode(', ')],
179 ));
180 }
181
182 // Read-confirmation control: a step flagged as carrying critical
183 // instructions must be acknowledged (read-confirmed) by the operator
184 // before it can be completed.
185 if ($step->needsReadConfirmation()) {
186 throw new \Exception(__(
187 'This step is blocked: you must confirm you have read the critical instructions before it can be completed.'
188 ));
189 }
190
191 // Recorded time (ISA-95 L3 system value) — the wall-clock diff, kept for
192 // audit. Retained regardless of any operator-confirmed actuals below.
193 $durationMinutes = null;
194 if ($step->started_at) {
195 $durationMinutes = (int) abs(now()->diffInMinutes($step->started_at));
196 }
197
198 // Operator-confirmed actual times (ISA-95 L3), stored separately from the
199 // recorded value and authoritative for performance reporting (#52). The
200 // optional setup/run split must fit within the confirmed elapsed total.
201 $actualElapsed = $data['actual_elapsed_minutes'] ?? null;
202 $actualSetup = $data['actual_setup_minutes'] ?? null;
203 $actualRun = $data['actual_run_minutes'] ?? null;
204 // A setup/run split is only meaningful against a total: reject a split
205 // supplied without an elapsed value so reporting can always verify it.
206 if ($actualElapsed === null && ($actualSetup !== null || $actualRun !== null)) {
207 throw new \Exception(__('Actual elapsed time is required when setup or run time is provided.'));
208 }
209 if ($actualElapsed !== null && ((int) $actualSetup + (int) $actualRun) > (int) $actualElapsed) {
210 throw new \Exception(__('Actual setup + run time cannot exceed the actual elapsed time.'));
211 }
212
213 // Whole-batch flow: finishing a step passes along whatever was not
214 // scrapped, so the ledger stays consistent without any quantity
215 // logging. (Transfer flow arrives here with nothing left waiting.)
216 $remainder = $step->availableQty();
217
218 // Complete the step
219 $step->update([
220 'status' => BatchStep::STATUS_DONE,
221 'passed_qty' => (float) $step->passed_qty + $remainder,
222 'completed_at' => now(),
223 'completed_by_id' => $user->id,
224 'duration_minutes' => $durationMinutes,
225 'actual_elapsed_minutes' => $actualElapsed,
226 'actual_setup_minutes' => $actualSetup,
227 'actual_run_minutes' => $actualRun,
228 ]);
229
230 // Update batch status
231 $batch = $step->batch;
232 $this->updateBatchStatus($batch);
233
234 // The next step (prerequisites now met) becomes READY.
235 $batch->promoteReadySteps();
236
237 $this->finishBatchIfComplete($batch, $step, $user, $data);
238
239 // Update work order status
240 $this->workOrderService->updateWorkOrderStatus($batch->workOrder);
241
242 return $step->fresh();
243 });
244 }

252 /** Correct a running manual step without erasing the audit trail or consumed output. */
253 public function correctGoodQuantity(BatchStep $step, User $user, float $good, float $expected, string $reason): BatchStep
254 {
255 return DB::transaction(function () use ($step, $user, $good, $expected, $reason) {
256 if (! ProductionFlow::isTransfer()) {
257 throw new \DomainException(__('Step corrections require transfer flow.'));
258 }
259 if (! $this->manualCorrectionsEnabled()) {
260 throw new \DomainException(__('Step total corrections require the Full edit policy.'));
261 }
262 $step = $this->lockProductionStep($step);
263 $step->batch->workOrder->assertProductionAvailable();
264 $this->loadLedgerContext($step);
265 $this->guardWorkstationRouting($step, $user);
266 if ((int) $step->started_by_id !== (int) $user->id && ! $user->hasAnyRole(['Admin', 'Supervisor'])) {
267 throw new \DomainException(__('Only the operator who started this step or a supervisor can correct its total.'));
268 }
269 $batch = $step->batch;
270 $order = $batch->workOrder;
271 if ($order->counting_source !== 'operator' || $step->status !== BatchStep::STATUS_IN_PROGRESS) {
272 throw new \DomainException(__('Only running, manually counted steps can be corrected.'));
273 }
274 if ($blocker = $step->productionBlocker()) {
275 throw new \DomainException($blocker);
276 }
277 if (! is_finite($good) || ! is_finite($expected) || $good < 0 || $good > 99999999 || abs($good - round($good, 2)) > 0.000001 || trim($reason) === '') {
278 throw new \DomainException(__('Enter a valid quantity and correction reason.'));
279 }
280 if (abs((float) $step->passed_qty - $expected) > 0.001) {
281 throw new \DomainException(__('The quantity changed. Refresh and review it before correcting.'));
282 }
283 if ($good + (float) $step->scrap_qty > $step->incomingQty() + 0.001) {
284 throw new \DomainException(__('The correction exceeds the incoming quantity.'));
285 }
286 $downstream = $batch->steps->filter(fn ($s) => $s->step_number > $step->step_number && $s->status !== BatchStep::STATUS_SKIPPED)->sortBy('step_number');
287 $next = $downstream->first();
288 if ($downstream->contains('status', BatchStep::STATUS_DONE)
289 || ($next && $good + 0.001 < (float) $next->passed_qty + (float) $next->scrap_qty)) {
290 throw new \DomainException(__('Correct downstream quantities first; these pieces have already been processed.'));
291 }
292 $before = (float) $step->passed_qty;
293 if (abs($before - $good) < 0.001) {
294 return $step;
295 }
296 $step->update(['passed_qty' => $good]);
297 \App\Models\AuditLog::create([
298 'user_id' => $user->id, 'entity_type' => BatchStep::class, 'entity_id' => $step->id,
299 'action' => 'quantity_corrected', 'before_state' => ['passed_qty' => $before],
300 'after_state' => ['passed_qty' => $good, 'reason' => trim($reason)],
301 ]);
302 $batch->promoteReadySteps();
303 $this->rollUpProducedQty($batch);
304 $this->workOrderService->updateWorkOrderStatus($order->fresh());
305
306 return $step->fresh();
307 });
308 }

310 /**
311 * Log pieces leaving a step: `$good` passed to the next station, `$scrap`
312 * lost here (recorded as a scrap entry without a reason — the reason can be
313 * added later from the scrap report). In transfer flow this is what opens
314 * the next step and drives the batch / work-order produced quantity live.
315 *
316 * @throws \DomainException when the step is not running, nothing is logged,
317 * or more is logged than is waiting at the step
318 */
319 public function recordQuantity(BatchStep $step, User $user, float $good, float $scrap = 0, ?string $notes = null): BatchStep
320 {
321 return DB::transaction(function () use ($step, $user, $good, $scrap, $notes) {
322 // Serialise concurrent logs (two operators, or an operator and a
323 // sensor) against the same step so the available count can't go negative.
324 $step = $this->lockProductionStep($step);
325 $step->batch->workOrder->assertProductionAvailable();
326 $this->loadLedgerContext($step);
327
328 $this->guardWorkstationRouting($step, $user);
329
330 if ($step->status !== BatchStep::STATUS_IN_PROGRESS) {
331 throw new \DomainException(__('Start the step before logging quantities.'));
332 }
333
334 if ($blocker = $step->productionBlocker()) {
335 throw new \DomainException($blocker);
336 }
337
338 $good = round(max(0, $good), 2);
339 $scrap = round(max(0, $scrap), 2);
340 if ($good + $scrap <= 0) {
341 throw new \DomainException(__('Log at least one good or scrapped piece.'));
342 }
343
344 $available = $step->availableQty();
345 if ($good + $scrap > $available + 0.001) {
346 throw new \DomainException(__('Only :qty pieces are waiting at this step.', ['qty' => self::formatQty($available)]));
347 }
348
349 $step->update([
350 'passed_qty' => (float) $step->passed_qty + $good,
351 'scrap_qty' => (float) $step->scrap_qty + $scrap,
352 ]);
353
354 $batch = $step->batch;
355 $workOrder = $batch->workOrder;
356
357 if ($scrap > 0) {
358 ScrapEntry::create([
359 'work_order_id' => $workOrder->id,
360 'batch_step_id' => $step->id,
361 'scrap_reason_id' => null,
362 'quantity' => $scrap,
363 'shift_id' => Shift::current($workOrder->line_id)?->id,
364 'notes' => $notes,
365 'reported_by' => $user->id,
366 'reported_at' => now(),
367 ]);
368 }
369
370 // Pieces are now waiting at the next station: open it (transfer flow).
371 $batch->promoteReadySteps();
372
373 if (ProductionFlow::isTransfer()) {
374 $this->rollUpProducedQty($batch);
375 $this->workOrderService->updateWorkOrderStatus($workOrder->fresh());
376 }
377
378 return $step->fresh();
379 });
380 }

382 /**
383 * Log pieces leaving a station that owns several consecutive steps: the
384 * scrap is lost at `$step`, the good pieces are passed through it and then
385 * through every following step bound to the same workstation (starting each
386 * one as needed), so the operator confirms the station's work once.
387 */
388 public function recordQuantityThroughStation(BatchStep $step, User $user, float $good, float $scrap = 0, ?string $notes = null): BatchStep
389 {
390 return DB::transaction(function () use ($step, $user, $good, $scrap, $notes) {
391 $first = $this->recordQuantity($step, $user, $good, $scrap, $notes);
392
393 // Only transfer flow can pass pieces into a step whose predecessor is
394 // still running; in whole-batch flow the station's next step opens
395 // when this one is finished, so this is an ordinary log.
396 if ($good <= 0 || ! $first->workstation_id || ! ProductionFlow::isTransfer()) {
397 return $first;
398 }
399
400 $following = $first->batch->steps()
401 ->where('step_number', '>', $first->step_number)
402 ->where('status', '!=', BatchStep::STATUS_SKIPPED)
403 ->orderBy('step_number')
404 ->get();
405
406 foreach ($following as $next) {
407 if ((int) $next->workstation_id !== (int) $first->workstation_id) {
408 break; // the station's run of steps ends here
409 }
410
411 if (in_array($next->status, [BatchStep::STATUS_PENDING, BatchStep::STATUS_READY], true)) {
412 $next = $this->startStep($next, $user);
413 }
414
415 $this->recordQuantity($next, $user, $good);
416 }
417
418 return $first->fresh();
419 });
420 }

422 /**
423 * Pieces counted leaving a step by a machine (break-beam pulse, machine
424 * good-count). Whole-batch flow keeps the historical bare counter. Transfer
425 * flow treats the count like an operator's good log without the operator
426 * gates: it is capped at what is actually waiting at the step (so an
427 * over-count can't create pieces downstream), it opens the next station, and
428 * it rolls the batch / work-order produced quantity up from the ledger.
429 *
430 * @return float the quantity actually counted (after the cap)
431 */
432 public function recordMachinePass(BatchStep $step, float $qty): float
433 {
434 if ($qty <= 0) {
435 return 0.0;
436 }
437
438 if (! ProductionFlow::isTransfer()) {
439 $step->increment('passed_qty', $qty);
440
441 return $qty;
442 }
443
444 return DB::transaction(function () use ($step, $qty) {
445 $step = $this->lockProductionStep($step);
446 $step->batch->workOrder->assertProductionAvailable();
447 $this->loadLedgerContext($step);
448
449 if ($step->productionBlocker() || in_array($step->status, [BatchStep::STATUS_DONE, BatchStep::STATUS_SKIPPED], true)) {
450 return 0.0;
451 }
452
453 $counted = round(min($qty, $step->availableQty()), 2);
454 if ($counted <= 0) {
455 return 0.0;
456 }
457
458 $step->update(['passed_qty' => (float) $step->passed_qty + $counted]);
459
460 $batch = $step->batch->fresh();
461 $batch->promoteReadySteps();
462 $this->rollUpProducedQty($batch);
463
464 return $counted;
465 });
466 }

473 * @throws \Exception

476 {
477 return DB::transaction(function () use ($step, $user, $reason) {
478 $step = $this->lockProductionStep($step);
479 $step->batch->workOrder->assertProductionAvailable();
480 if (ProductionFlow::isTransfer() && ($blocker = $step->productionBlocker())) {

485
486 if (! $step->canSkip()) {

490 $this->guardUnprocessedRouting($step);
491

512 /**
513 * Choose a variant within a group: activate this step and skip its siblings.
514 * Lets the operator override the template's default variant.
515 *
516 * @throws \Exception
517 */
518 public function chooseVariant(BatchStep $step, User $user): BatchStep
519 {
520 return DB::transaction(function () use ($step, $user) {
521 $step = $this->lockProductionStep($step);
522 $step->batch->workOrder->assertProductionAvailable();
523 if (ProductionFlow::isTransfer() && ($blocker = $step->productionBlocker())) {
524 throw new \DomainException($blocker);
525 }
526
527 if ($step->variant_group === null) {
528 throw new \Exception('This step is not part of a variant group.');
529 }
530
531 if ($step->status === BatchStep::STATUS_DONE) {
532 throw new \Exception('This variant is already completed.');
533 }
534
535 $this->guardUnprocessedRouting($step);
536 foreach ($step->variantSiblings()->get() as $sibling) {
537 $this->guardUnprocessedRouting($sibling);
538 }
539
540 // Activate the chosen variant, skip every sibling not already done.
541 $step->update(['status' => BatchStep::STATUS_PENDING, 'skip_reason' => null]);
542
543 $step->variantSiblings()
544 ->where('status', '!=', BatchStep::STATUS_DONE)
545 ->update([
546 'status' => BatchStep::STATUS_SKIPPED,
547 'completed_at' => now(),
548 'completed_by_id' => $user->id,
549 ]);
550
551 $this->updateBatchStatus($step->batch);
552 // Promote the chosen variant to READY if it's next in line.
553 $step->batch->promoteReadySteps();
554
555 return $step->fresh();
556 });
557 }

571 /**
572 * Pool dispatch (#52): a supervisor assigns a specific workstation to a step
573 * that carries only an Equipment Class (workstation_type_id). Valid only while
574 * the step is still assignable (PENDING/READY) and the chosen workstation is
575 * active and of the required type. Once assigned, guardWorkstationRouting
576 * enforces that only operators on that workstation may start the step.
577 */
578 public function assignWorkstation(BatchStep $step, int $workstationId, User $user): BatchStep
579 {
580 return DB::transaction(function () use ($step, $workstationId, $user) {
581 // Lock the row and re-read its status inside the transaction so a step
582 // an operator starts concurrently cannot still receive a workstation.
583 $locked = BatchStep::whereKey($step->getKey())->lockForUpdate()->firstOrFail();
584
585 if (! in_array($locked->status, [BatchStep::STATUS_PENDING, BatchStep::STATUS_READY], true)) {
586 throw new \Exception(__('Only a pending step can be assigned a workstation.'));
587 }
588
589 $workstation = Workstation::find($workstationId);
590 if (! $workstation || ! $workstation->is_active) {
591 throw new \Exception(__('The selected workstation is not available.'));
592 }
593
594 if ($locked->workstation_type_id && (int) $workstation->workstation_type_id !== (int) $locked->workstation_type_id) {
595 throw new \Exception(__('The selected workstation is not of the required type for this step.'));
596 }
597
598 $locked->update([
599 'workstation_id' => $workstation->id,
600 'assigned_by_id' => $user->id,
601 'assigned_at' => now(),
602 ]);
603
604 return $locked->fresh();
605 });

608 private function finishBatchIfComplete(Batch $batch, BatchStep $step, User $user, array $data = []): void
609 {
610 // If batch is complete, update produced quantity and consume materials
611 if ($batch->status === Batch::STATUS_DONE) {
612 // End-of-batch BOM rows (consumed_at='end') get allocated now,
613 // immediately before everything is marked consumed. Attribute to
614 // the completing step so the genealogy bridge has a step to record.
615 $this->allocationService->allocateForBatchEnd($batch, $user, attributeStepId: $step->id);
616 // Transfer flow: the batch produced exactly what left its last step.
617 $producedQty = ProductionFlow::isTransfer()
618 ? (float) ($batch->lastEffectiveStep()?->passed_qty ?? 0)
619 : ($data['produced_qty'] ?? $batch->target_qty);
620 $this->completeBatch($batch, $producedQty);
621 $this->allocationService->consumeForBatch($batch);
622
623 // Quality-control triggers: every-N-units checks (#105).
624 $this->qualityTriggerService->fireForUnits($batch->fresh());
625 }

629 /**
630 * Update batch status based on steps.
631 */
632 protected function updateBatchStatus(Batch $batch): void
633 {
634 // Check if all steps are complete
635 if ($batch->allStepsComplete()) {
636 $batch->update([
637 'status' => Batch::STATUS_DONE,
638 'completed_at' => now(),
639 ]);
640
641 return;
642 }
643
644 // Check if any step is in progress
645 $hasInProgressStep = $batch->steps()
646 ->where('status', BatchStep::STATUS_IN_PROGRESS)
647 ->exists();
648
649 if ($hasInProgressStep && $batch->status !== Batch::STATUS_IN_PROGRESS) {
650 $batch->update([
651 'status' => Batch::STATUS_IN_PROGRESS,
652 'started_at' => $batch->started_at ?? now(),
653 ]);
654 }
655 }

657 /**
658 * Complete a batch and update produced quantity.
659 */
660 protected function completeBatch(Batch $batch, float $producedQty): void
661 {
662 // Update batch produced qty
663 $batch->update([
664 'produced_qty' => $producedQty,
665 ]);
666
667 if (ProductionFlow::isTransfer()) {
668 $this->rollUpProducedQty($batch);
669
670 return;
671 }
672
673 // Update work order produced qty
674 $workOrder = $batch->workOrder;
675 $totalProduced = $workOrder->batches()
676 ->where('status', Batch::STATUS_DONE)
677 ->sum('produced_qty');
678
679 $workOrder->update([
680 'produced_qty' => $totalProduced,
681 ]);
682 }
683
684 /**
685 * Transfer flow: a batch has produced whatever left its last step so far,
686 * and the work order the sum over its live batches — updated as pieces
687 * flow, not only when a batch closes.
688 */
689 protected function rollUpProducedQty(Batch $batch): void
690 {
691 $batch->update([
692 'produced_qty' => (float) ($batch->lastEffectiveStep()?->passed_qty ?? 0),
693 ]);
694
695 $workOrder = $batch->workOrder;
696 $workOrder->update([
697 'produced_qty' => $workOrder->batches()
698 ->where('status', '!=', Batch::STATUS_CANCELLED)
699 ->sum('produced_qty'),
700 ]);
701 }

737 /**
738 * Enforce workstation routing: when enabled, a workstation-bound operator
739 * may only start/complete steps assigned to their own workstation.
740 *
741 * Bypassed for Admins/Supervisors and for line-level operators (users with
742 * no workstation assigned). Steps without an assigned workstation are open
743 * to anyone. This is the single server-side chokepoint covering both the
744 * Livewire UI and the REST API, since both route through BatchService.
745 *
746 * @throws \Exception
747 */
748 protected function guardWorkstationRouting(BatchStep $step, User $user): void
749 {
750 $enabled = json_decode(
751 DB::table('system_settings')->where('key', 'workstation_routing_enabled')->value('value') ?? 'false',
752 true
753 ) ?? false;
754
755 if (! $enabled || ! $step->workstation_id) {
756 return;
757 }
758
759 // Admins and Supervisors can operate any workstation.
760 if ($user->hasRole('Admin') || $user->hasRole('Supervisor')) {
761 return;
762 }
763
764 // Line-level operators (no workstation assigned) are not restricted.
765 if (! $user->workstation_id) {
766 return;
767 }
768
769 if ((int) $step->workstation_id !== (int) $user->workstation_id) {
770 $stationName = $step->workstation?->name ?? __('another workstation');
771 throw new \Exception(
772 __('This step is assigned to :station and will appear in that workstation\'s queue.', ['station' => $stationName])
773 );
774 }
775 }

777 /**
778 * Throw appropriate validation error based on step state.
779 *
780 * @throws \Exception
781 */
782 protected function throwValidationError(BatchStep $step): void
783 {
784 if (! in_array($step->status, [BatchStep::STATUS_PENDING, BatchStep::STATUS_READY], true)) {
785 throw new \Exception("Step is already {$step->status}");
786 }
787
788 $workOrder = $step->batch->workOrder;
789 if ($workOrder->isBlocked()) {
790 $issues = $workOrder->openBlockingIssues();
791 $issueList = $issues->pluck('title')->join(', ');
792 throw new \Exception("Work order is blocked by issues: {$issueList}");
793 }
794
795 // Check sequential enforcement
796 if (config('openmmes.force_sequential_steps', true) && $step->step_number > 1) {
797 $previousStep = $step->batch->steps()
798 ->where('step_number', $step->step_number - 1)
799 ->first();
800
801 if (! $previousStep || ! in_array($previousStep->status, [BatchStep::STATUS_DONE, BatchStep::STATUS_SKIPPED])) {
802 $prevNum = $step->step_number - 1;
803 throw new \Exception(__('Complete step :step before starting this step.', ['step' => $prevNum]));
804 }
805 }
806
807 throw new \Exception('Step cannot be started');
808 }
