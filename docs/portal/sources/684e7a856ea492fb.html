8/**
9 * A read-only TanStack DB collection synced over Reverb. Used directly by
10 * ResourceTable, the shared LiveShapesProvider, and the useSyncedShape hooks.
11 *
12 * 1. fetch the snapshot from /api/collections/{name}; the response also names
13 * the channel to subscribe to (the server decides tenant-vs-global scoping),
14 * 2. subscribe to that channel and apply live deltas — "upsert" / "delete",
15 * 3. on subscription-confirmed (and every reconnect/resubscribe) re-fetch the
16 * snapshot, so any write that landed before we were subscribed is caught.
17 *
18 * Tracks its own key set so the server can send a plain "upsert" without knowing
19 * whether the client already holds the row. Read-only: writes go through Laravel.
20 */
21export function realtimeCollection(name, getKey = (row) => row.id) {
22 return createCollection({
23 id: name,
24 getKey,
25 gcTime: 1000,
26 sync: {
27 sync: ({ begin, write, commit, markReady }) => {
28 let alive = true;
29 let release = null;
30 const keys = new Set();
31
32 const onChanged = (e) => {
33 if (!alive || !e?.row) return;
34 const key = getKey(e.row);
35 begin();
36 if (e.op === 'delete') {
37 if (keys.has(key)) {
38 write({ type: 'delete', value: e.row });
39 keys.delete(key);
40 }
41 } else {
42 write({ type: keys.has(key) ? 'update' : 'insert', value: e.row });
43 keys.add(key);
44 }
45 commit();
46 };
47
48 const subscribe = (ch) => {
49 // Re-fetch once subscribed (and on every reconnect) so writes
50 // that landed before subscription aren't lost. Joining a channel
51 // another collection already holds fires no new subscription —
52 // the initial load() below covers that case.
53 release = channels.join(ch, '.changed', onChanged, () => { if (alive) load(); });
54 };
55
56 const apply = (rows) => {
57 const fresh = new Set();
58 begin();
59 for (const row of rows) {
60 const key = getKey(row);
61 write({ type: keys.has(key) ? 'update' : 'insert', value: row });
62 fresh.add(key);
63 }
64 for (const key of keys) {
65 if (!fresh.has(key)) write({ type: 'delete', value: { id: key } });
66 }
67 commit();
68 keys.clear();
69 for (const k of fresh) keys.add(k);
70 };
71
72 const load = async () => {
73 try {
74 const res = await fetch(`/api/collections/${name}`, {
75 headers: { Accept: 'application/json' },
76 credentials: 'same-origin',
77 });
78 if (!res.ok) throw new Error(`snapshot ${name}: HTTP ${res.status}`);
79 const { rows, channel } = await res.json();
80 if (!alive) return;
81 if (!release && channel) subscribe(channel);
82 apply(rows);
83 } catch (e) {
84 // Leave whatever we have; don't hang the UI. Surface the
85 // reason so a failed snapshot (e.g. a 401 auth issue) is
86 // diagnosable instead of a silently empty list. #193
87 console.warn(`[live-sync] snapshot for "${name}" failed:`, e?.message ?? e);
88 } finally {
89 if (alive) markReady();
90 }
91 };
92
93 load();
94
95 return () => {
96 alive = false;
97 release?.();
98 };
99 },
100 },
101 });
102}

104/**
105 * A TanStack DB collection fed by rows the page already has — Inertia props
106 * rather than a Reverb shape.
107 *
108 * `ResourceTable` renders every admin list, but its rows came exclusively from
109 * a synced collection, so a list whose records aren't in `ShapeRegistry` (a BOM
110 * belongs to one process template; nothing broadcasts it) had to fall back to a
111 * bare `DataTable` and lost the list chrome with it — breadcrumbs, the toolbar's
112 * create button, the action rail. This collection closes that gap: same live
113 * query, same table, rows pushed in instead of pulled down.
114 *
115 * `applyRows` is the push side, called whenever the props change (an Inertia
116 * visit after a create/edit/delete re-renders with a fresh array). It diffs
117 * against the keys already held so an unchanged row isn't rewritten.
118 */
119export function staticCollection(id, getKey = (row) => row.id) {
120 // Set by `sync` once TanStack subscribes. Rows handed over before that
121 // (the first render happens before the effect) wait here.
122 let api = null;
123 let pending = null;
124 const keys = new Set();
125
126 const applyRows = (rows) => {
127 if (!api) {
128 pending = rows;
129 return;
130 }
131 const { begin, write, commit } = api;
132 const fresh = new Set();
133 begin();
134 for (const row of rows) {
135 const key = getKey(row);
136 write({ type: keys.has(key) ? 'update' : 'insert', value: row });
137 fresh.add(key);
138 }
139 for (const key of keys) {
140 if (!fresh.has(key)) write({ type: 'delete', value: { id: key } });
141 }
142 commit();
143 keys.clear();
144 for (const k of fresh) keys.add(k);
145 };
146
147 const collection = createCollection({
148 id,
149 getKey,
150 gcTime: 1000,
151 sync: {
152 sync: ({ begin, write, commit, markReady }) => {
153 api = { begin, write, commit };
154 if (pending) {
155 applyRows(pending);
156 pending = null;
157 }
158 markReady();
159 return () => {
160 api = null;
161 };
162 },
163 },
164 });
165
166 collection.applyRows = applyRows;
167 return collection;
168}
