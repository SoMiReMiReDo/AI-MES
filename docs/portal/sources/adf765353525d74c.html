19/**
20 * Synced-collection channels (Reverb). Private channel per collection, namespaced
21 * by tenant so a user only receives their own tenant's rows (tenantKey = the
22 * user's tenant_id, or "g" for null — mirrors the null-safe TenantScope).
23 *
24 * This is the authorization for the read path: without it any authenticated
25 * user would receive every tenant's row deltas.
26 */
27Broadcast::channel('col.{tenant}.{collection}', function ($user, string $tenant, string $collection) {
28 if (! $user) {
29 return false;
30 }
31
32 $userTenant = $user->tenant_id === null ? 'g' : (string) $user->tenant_id;
33 // A user may subscribe to their own tenant channel AND the global ("g")
34 // channel: global tables (no tenant_id column — e.g. skills, issues) and
35 // global lookup rows are visible to every tenant, mirroring the null-safe
36 // TenantScope and CollectionController's snapshot. Without the "g" allowance
37 // a tenant user is denied live updates for every global collection, so rows
38 // created/deleted on those tables never reflect without a full reload.
39 if ($tenant !== $userTenant && $tenant !== 'g') {
40 return false;
41 }
42
43 // Admin lists are Admin/Supervisor only. Refine per-collection as operator
44 // screens move onto Reverb.
45 return $user->hasAnyRole(['Admin', 'Supervisor']);
46});

48/**
49 * Live shift monitor — one private channel per workstation, carrying the nudge
50 * that its shift changed (ShiftMonitorChanged).
51 *
52 * Same audience as the screen itself: the monitor is reachable by Admin and
53 * Supervisor, so subscribing to a station's channel must be too. The nudge
54 * carries no production data, but the snapshot it prompts does, and that fetch
55 * is authorized separately by the route.
56 */
57Broadcast::channel('shift-monitor.{workstation}', function ($user, string $workstation) {
58 if (! $user || ! ctype_digit($workstation)) {
59 return false;
60 }
61
62 return $user->hasAnyRole(['Admin', 'Supervisor']);
