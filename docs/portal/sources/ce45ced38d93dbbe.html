39 /**
40 * Execute one mapping rule. Returns a result descriptor for logging.
41 */
42 public function executeSingle(TopicMapping $mapping, array $parsedData): array
43 {
44 $result = [
45 'mapping_id' => $mapping->id,
46 'action_type' => $mapping->action_type,
47 'status' => 'skipped',
48 'message' => null,
49 ];
50
51 try {
52 // Resolve the primary field value
53 $fieldValue = $this->parser->resolvePath($mapping->field_path, $parsedData);
54
55 // Evaluate condition
56 if (! $this->parser->evaluateCondition($mapping->condition_expr, $fieldValue)) {
57 $result['message'] = 'Condition not met';
58
59 return $result;
60 }
61
62 $params = $mapping->action_params ?? [];
63 $outcome = match ($mapping->action_type) {
64 TopicMapping::ACTION_UPDATE_BATCH_STEP => $this->updateBatchStep($params, $parsedData, $fieldValue),
65 TopicMapping::ACTION_UPDATE_WORK_ORDER_QTY => $this->updateWorkOrderQty($mapping, $params, $parsedData, $fieldValue),
66 TopicMapping::ACTION_COUNT_STEP => $this->countStep($mapping, $params, $parsedData, $fieldValue),
67 TopicMapping::ACTION_CREATE_ISSUE => $this->createIssue($params, $parsedData, $fieldValue),
68 TopicMapping::ACTION_UPDATE_LINE_STATUS => $this->updateLineStatus($params, $parsedData, $fieldValue),
69 TopicMapping::ACTION_SET_WORK_ORDER_STATUS => $this->setWorkOrderStatus($params, $parsedData, $fieldValue),
70 TopicMapping::ACTION_WEBHOOK_FORWARD => $this->webhookForward($params, $parsedData),
71 TopicMapping::ACTION_LOG_EVENT => ['logged' => true],
72 default => throw new \InvalidArgumentException("Unknown action: {$mapping->action_type}"),
73 };
74
75 $result['status'] = 'ok';
76 $result['message'] = json_encode($outcome);
77 } catch (\Throwable $e) {
78 $result['status'] = 'error';
79 $result['message'] = $e->getMessage();
80 Log::warning('ActionExecutor error', [
81 'mapping_id' => $mapping->id,
82 'error' => $e->getMessage(),
83 ]);
84 }
85
86 return $result;
87 }

91 private function updateBatchStep(array $params, array $data, mixed $fieldValue): array
92 {
93 // params: { step_id_path, result_path, result (static), batch_id_path, step_order }
94 $stepId = $this->resolveParam($params, 'step_id_path', $data);
95 $batchId = $this->resolveParam($params, 'batch_id_path', $data);
96 $stepOrder = $this->resolveParam($params, 'step_order_path', $data) ?? ($params['step_order'] ?? null);
97 $result = $this->resolveParam($params, 'result_path', $data) ?? ($params['result'] ?? 'done');
98
99 $step = null;
100 if ($stepId) {
101 $step = BatchStep::find($stepId);
102 } elseif ($batchId && $stepOrder !== null) {
103 $step = BatchStep::where('batch_id', $batchId)
104 ->where('step_order', $stepOrder)
105 ->first();
106 }
107
108 if (! $step) {
109 throw new \RuntimeException("BatchStep not found (step_id={$stepId}, batch_id={$batchId})");
110 }
111
112 if ($step->batch?->workOrder?->usesStepLedger()) {
113 throw new \DomainException('Transfer steps must be completed through the production workflow after quantities are recorded.');
114 }
115
116 $newStatus = match ($result) {
117 'done', 'completed', '1', 'true' => 'done',
118 'failed', 'error', '0', 'false' => 'failed',
119 default => 'done',
120 };
121
122 $step->update([
123 'status' => $newStatus,
124 'completed_at' => $newStatus === 'done' ? now() : null,
125 ]);
126
127 return ['step_id' => $step->id, 'new_status' => $newStatus];
128 }

144 private function countReading(\App\Models\MachineCounter $counter, TopicMapping $mapping, array $data, mixed $value): array
145 {
146 // Payload routing hints never override the supervisor's explicit assignment.
147 $params = $mapping->action_params ?? [];
148 $eventId = $this->parser->resolvePath($params['event_id_path'] ?? '$.event_id', $data);
149 $timestamp = $this->parser->resolvePath($params['timestamp_path'] ?? '$.timestamp', $data);
150 $at = $timestamp ? \Illuminate\Support\Carbon::parse($timestamp) : null;
151 $reading = $this->counters->ingest($counter, $value, $at, is_scalar($eventId) ? (string) $eventId : null, $this->counters->sourceSnapshot($mapping));
152
153 return ['reading_id' => $reading->id, 'counter_status' => $reading->status, 'applied_qty' => (float) $reading->applied_qty];
154 }

156 private function createIssue(array $params, array $data, mixed $fieldValue): array
157 {
158 // params: { issue_type_id, work_order_no_path, description_path, description (static) }

160 $orderNo = $this->resolveParam($params, 'work_order_no_path', $data) ?? ($params['work_order_no'] ?? null);

165 if ($orderNo) {
166 $workOrderId = WorkOrder::where('order_no', $orderNo)->value('id');
167 }
168
169 $issue = Issue::create([
170 'issue_type_id' => $issueTypeId,
171 'work_order_id' => $workOrderId,
172 'description' => (string) $description,
173 'status' => 'open',
174 'reported_by' => null, // machine-generated
175 ]);
176
177 return ['issue_id' => $issue->id];

180 private function updateLineStatus(array $params, array $data, mixed $fieldValue): array
181 {
182 // params: { line_id (static), line_code_path, status_id (static), status_code_path }
183 $lineId = $this->resolveParam($params, 'line_id_path', $data) ?? ($params['line_id'] ?? null);
184 $lineCode = $this->resolveParam($params, 'line_code_path', $data) ?? ($params['line_code'] ?? null);
185 $statusId = $this->resolveParam($params, 'status_id_path', $data) ?? ($params['status_id'] ?? null);
186 $statusCode = $this->resolveParam($params, 'status_code_path', $data) ?? ($params['status_code'] ?? null);
187
188 $line = $lineId
189 ? Line::find($lineId)
190 : Line::where('code', $lineCode)->first();
191
192 if (! $line) {
193 throw new \RuntimeException("Line not found (id={$lineId}, code={$lineCode})");
194 }
195
196 $lineStatus = $statusId
197 ? LineStatus::find($statusId)
198 : LineStatus::where('code', $statusCode)->first();
199
200 if (! $lineStatus) {
201 throw new \RuntimeException("LineStatus not found (id={$statusId}, code={$statusCode})");
202 }
203
204 // Update most recent in-progress work order on this line
205 $workOrder = WorkOrder::where('line_id', $line->id)
206 ->where('status', 'in_progress')
207 ->latest()
208 ->first();
209
210 if ($workOrder) {
211 $workOrder->update(['line_status_id' => $lineStatus->id]);
212 }
213
214 return ['line_id' => $line->id, 'line_status_id' => $lineStatus->id];
215 }

217 private function setWorkOrderStatus(array $params, array $data, mixed $fieldValue): array
218 {
219 // params: { order_no_path, order_id (static), status (static), status_path }
220 $orderNo = $this->resolveParam($params, 'order_no_path', $data) ?? ($params['order_no'] ?? null);
221 $orderId = $this->resolveParam($params, 'order_id_path', $data) ?? ($params['order_id'] ?? null);
222 $status = $this->resolveParam($params, 'status_path', $data) ?? ($params['status'] ?? null);
223
224 $allowed = ['pending', 'accepted', 'in_progress', 'completed', 'paused', 'rejected'];
225 if (! in_array($status, $allowed)) {
226 throw new \RuntimeException("Invalid work order status: {$status}");
227 }
228
229 $workOrder = $orderNo
230 ? WorkOrder::where('order_no', $orderNo)->first()
231 : WorkOrder::find($orderId);
232
233 if (! $workOrder) {
234 throw new \RuntimeException('WorkOrder not found');
235 }
236
237 if ($workOrder->usesStepLedger()) {
238 throw new \DomainException('Transfer order status must be changed through the production workflow.');
239 }
240
241 $workOrder->update(['status' => $status]);
242
243 return ['order_no' => $workOrder->order_no, 'new_status' => $status];
244 }

246 private function webhookForward(array $params, array $data): array
247 {
248 // params: { url, method (GET/POST), headers (object) }
249 $url = $params['url'] ?? null;
250 $method = strtolower($params['method'] ?? 'post');
251 $headers = $params['headers'] ?? [];
252
253 if (! $url || ! filter_var($url, FILTER_VALIDATE_URL)) {
254 throw new \RuntimeException('Invalid or missing webhook URL');
255 }
256
257 // Block SSRF — reject requests to private/loopback/metadata addresses
258 $host = parse_url($url, PHP_URL_HOST);
259 if ($host) {
260 $ip = gethostbyname($host);
261 if (filter_var($ip, FILTER_VALIDATE_IP)) {
262 foreach ([
263 FILTER_FLAG_NO_PRIV_RANGE,
264 FILTER_FLAG_NO_RES_RANGE,
265 ] as $flag) {
266 if (! filter_var($ip, FILTER_VALIDATE_IP, $flag)) {
267 throw new \RuntimeException('Webhook URL resolves to a private/reserved address');
268 }
269 }
270 // Block AWS/GCP/Azure metadata endpoints explicitly
271 if (in_array($ip, ['169.254.169.254', '169.254.170.2', '100.100.100.200'])) {
272 throw new \RuntimeException('Webhook URL resolves to a private/reserved address');
273 }
274 }
275 }
276
277 $response = Http::withHeaders($headers)
278 ->timeout(5)
279 ->{$method}($url, $data);
280
281 return ['status_code' => $response->status(), 'ok' => $response->successful()];
282 }

286 /**
287 * Resolve a param value from action_params.
288 * If the key ends with '_path', resolve that path against payload data.

290 private function resolveParam(array $params, string $pathKey, array $data): mixed
291 {
292 if (! isset($params[$pathKey])) {
