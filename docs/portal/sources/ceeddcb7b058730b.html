16/**
17 * Owns the lifecycle of a stock document (#212): create as draft, post, cancel.
18 *
19 * Posting is the only thing that moves stock. It is deliberately concentrated
20 * here so every path — manual UI entry, ERP import, automatic generation from a
21 * work order — produces the same three effects:
22 *
23 * 1. the per-warehouse balance in warehouse_stocks moves,
24 * 2. for material lines, materials.stock_quantity and the stock_movements
25 * ledger move through StockMovementService (the single entry point for
26 * material stock), and the material lot's available quantity follows,
27 * 3. the document becomes immutable (posted).
28 *
29 * Cancelling a posted document reverses all of it, so a mis-posted release can
30 * be undone without editing balances by hand.
31 */
32class StockDocumentService
33{
34 /** How many times a generated document number is retried on a collision. */
35 private const NUMBER_ATTEMPTS = 5;
36
37 public function __construct(
38 private StockMovementService $stockMovements,
39 private WarehouseStockService $warehouseStock,
40 ) {}

42 /**
43 * Create a draft document with its lines.
44 *
45 * @param array<string, mixed> $attributes document columns; `lines` holds the line rows
46 */
47 public function createDraft(array $attributes, ?User $user = null): StockDocument
48 {
49 $lines = $attributes['lines'] ?? [];
50 unset($attributes['lines']);
51
52 if ($lines === []) {
53 throw ValidationException::withMessages([
54 'lines' => __('A stock document needs at least one line.'),
55 ]);
56 }
57
58 $type = $attributes['type'];
59 $warehouse = $this->resolveWarehouse($attributes['warehouse_id'] ?? null, $type);
60
61 // A generated number can collide with a concurrent create; the partial
62 // unique index catches it and the next attempt reads the new high-water
63 // mark. An explicitly supplied number is the caller's to own, so a clash
64 // there is a real error and is not retried.
65 $attempts = isset($attributes['document_no']) ? 1 : self::NUMBER_ATTEMPTS;
66
67 for ($attempt = 1; ; $attempt++) {
68 try {
69 return DB::transaction(function () use ($attributes, $lines, $type, $warehouse, $user) {
70 $document = StockDocument::create([
71 ...$attributes,
72 'warehouse_id' => $warehouse->id,
73 'document_no' => $attributes['document_no'] ?? $this->nextDocumentNumber($type),
74 'status' => StockDocument::STATUS_DRAFT,
75 'created_by_id' => $attributes['created_by_id'] ?? $user?->id,
76 ]);
77
78 foreach (array_values($lines) as $index => $line) {
79 $document->lines()->create([
80 ...$this->normaliseLine($line, $document),
81 'sort_order' => $line['sort_order'] ?? $index,
82 ]);
83 }
84
85 return $document->load('lines');
86 });
87 } catch (UniqueConstraintViolationException $e) {
88 if ($attempt >= $attempts) {
89 throw $e;
90 }
91 }
92 }
93 }

95 /**
96 * Apply a draft document to stock. Idempotent by status: a document that is
97 * already posted is returned untouched rather than double-counted.
98 */
99 public function post(StockDocument $document, ?User $user = null): StockDocument
100 {
101 return DB::transaction(function () use ($document, $user) {
102 // Re-read under a row lock and only then judge the status: checking it
103 // outside the transaction lets two concurrent posts both pass the guard
104 // and apply the same lines twice.
105 $locked = StockDocument::where('id', $document->getKey())->lockForUpdate()->first();
106
107 if (! $locked) {
108 throw ValidationException::withMessages([
109 'status' => __('That item was already removed.'),
110 ]);
111 }
112
113 if ($locked->isPosted()) {
114 return $locked;
115 }
116
117 if (! $locked->isDraft()) {
118 throw ValidationException::withMessages([
119 'status' => __('Only a draft document can be posted.'),
120 ]);
121 }
122
123 $document = $locked;
124 $document->load('lines');
125
126 if ($document->lines->isEmpty()) {
127 throw ValidationException::withMessages([
128 'lines' => __('A stock document needs at least one line.'),
129 ]);
130 }
131
132 foreach ($document->lines as $line) {
133 $this->applyLine($document, $line, reverse: false, user: $user);
134 }
135
136 $document->update([
137 'status' => StockDocument::STATUS_POSTED,
138 'posted_at' => now(),
139 'posted_by_id' => $user?->id,
140 ]);
141
142 return $document->refresh();
143 });
144 }

146 /**
147 * Reverse a posted document. A draft is cancelled without touching stock
148 * (nothing was applied yet).
149 */
150 public function cancel(StockDocument $document, ?User $user = null, ?string $reason = null): StockDocument
151 {
152 return DB::transaction(function () use ($document, $user, $reason) {
153 // Same reason as post(): the status decides whether stock is reversed,
154 // so it must be read under the lock that guards the reversal.
155 $locked = StockDocument::where('id', $document->getKey())->lockForUpdate()->first();
156
157 if (! $locked) {
158 throw ValidationException::withMessages([
159 'status' => __('That item was already removed.'),
160 ]);
161 }
162
163 if ($locked->status === StockDocument::STATUS_CANCELLED) {
164 return $locked;
165 }
166
167 $document = $locked;
168
169 if ($document->isPosted()) {
170 $document->load('lines');
171
172 foreach ($document->lines as $line) {
173 $this->applyLine($document, $line, reverse: true, user: $user);
174 }
175 }
176
177 $document->update([
178 'status' => StockDocument::STATUS_CANCELLED,
179 'notes' => $reason
180 ? trim(($document->notes ? $document->notes."\n" : '').__('Cancelled: ').$reason)
181 : $document->notes,
182 ]);
183
184 return $document->refresh();
185 });
186 }

202 /**
203 * Next free document number for a type: TYPE-PREFIX/YYYY/NNNN, sequential
204 * per year. Two concurrent creates can compute the same number; the partial
205 * unique index on (document_no, tenant) rejects the loser and createDraft()
206 * retries it (see NUMBER_ATTEMPTS).
207 */
208 public function nextDocumentNumber(string $type): string
209 {
210 $prefix = match ($type) {
211 StockDocument::TYPE_MATERIAL_ISSUE => 'MI',
212 StockDocument::TYPE_MATERIAL_RECEIPT => 'MR',
213 StockDocument::TYPE_PRODUCT_RECEIPT => 'PR',
214 StockDocument::TYPE_PRODUCT_ISSUE => 'PI',
215 default => 'SD',
216 };
217
218 $year = now()->year;
219 $pattern = "{$prefix}/{$year}/%";
220
221 $last = StockDocument::withTrashed()
222 ->where('type', $type)
223 ->where('document_no', 'like', $pattern)
224 ->orderByDesc('id')
225 ->value('document_no');
226
227 $sequence = $last ? ((int) substr($last, strrpos($last, '/') + 1)) + 1 : 1;
228
229 return sprintf('%s/%d/%04d', $prefix, $year, $sequence);
230 }

232 /**
233 * Move one line's quantity in every place stock is tracked.
234 *
235 * `reverse` flips the sign, which is all a cancellation needs.
236 */
237 private function applyLine(StockDocument $document, $line, bool $reverse, ?User $user): void
238 {
239 $signed = $document->direction() * (float) $line->quantity * ($reverse ? -1 : 1);
240
241 $material = $document->isMaterialDocument() && $line->material_id !== null
242 ? Material::find($line->material_id)
243 : null;
244
245 // Before anything moves: an issue the stock cannot cover is refused here, not
246 // rolled back after the balances have already been written.
247 if ($material) {
248 $this->guardNegativeStock($document, $material, $signed);
249 }
250
251 $this->adjustWarehouseStock($document, $line, $signed);
252
253 if (! $material) {
254 return;
255 }
256
257 // materials.stock_quantity + the stock_movements ledger. The movement
258 // type mirrors the direction so the ledger reads the same as a shop-floor
259 // consumption or an inbound receipt.
260 $this->stockMovements->record(
261 material: $material,
262 movementType: $signed < 0 ? StockMovement::TYPE_CONSUME : StockMovement::TYPE_RECEIPT,
263 signedQuantity: $signed,
264 user: $user,
265 sourceType: StockMovement::SOURCE_STOCK_DOCUMENT,
266 sourceId: $document->id,
267 reason: $document->document_no,
268 warehouseId: $document->warehouse_id,
269 );
270
271 if ($line->material_lot_id !== null) {
272 $this->adjustLot($line->material_lot_id, $signed, (int) $line->material_id);
273 }
274 }

276 /**
277 * Move the (warehouse, item[, lot]) balance, creating the row on first use.
278 * Lot-tracked material lines keep both a lot row and the warehouse total for
279 * that material, so a per-material view does not have to sum lots.
280 */
281 private function adjustWarehouseStock(StockDocument $document, $line, float $signed): void
282 {
283 $isMaterial = $document->isMaterialDocument();
284
285 $keys = [
286 'warehouse_id' => $document->warehouse_id,
287 'material_id' => $isMaterial ? $line->material_id : null,
288 'product_type_id' => $isMaterial ? null : $line->product_type_id,
289 ];
290
291 try {
292 if ($isMaterial && $line->material_lot_id !== null) {
293 $this->warehouseStock->adjust(
294 [...$keys, 'material_lot_id' => $line->material_lot_id],
295 $signed,
296 $line->unit_of_measure,
297 );
298 }
299
300 $this->warehouseStock->adjust([...$keys, 'material_lot_id' => null], $signed, $line->unit_of_measure);
301 } catch (\RuntimeException) {
302 // The shared service is used by non-HTTP callers too, so it reports a
303 // plain runtime failure; posting turns it back into a form error.
304 throw ValidationException::withMessages([
305 'lines' => __('Could not read the stock balance to update. Try again.'),
306 ]);
307 }
308 }

310 /** Keep the lot's remaining quantity in step with what was issued/returned. */
311 private function adjustLot(int $lotId, float $signed, int $materialId): void
312 {
313 $lot = MaterialLot::where('id', $lotId)->lockForUpdate()->first();
314
315 // A lot belonging to another material is not this line's stock to move —
316 // the form request rejects it, and a payload assembled elsewhere must not
317 // slip past that.
318 if (! $lot || (int) $lot->material_id !== $materialId) {
319 return;
320 }
321
322 $lot->quantity_available = max(0, round((float) $lot->quantity_available + $signed, 4));
323
324 // A lot drained by an issue is consumed; returning stock to an emptied
325 // lot makes it usable again.
326 if ($lot->quantity_available <= 0 && $lot->status === MaterialLot::STATUS_RELEASED) {
327 $lot->status = MaterialLot::STATUS_CONSUMED;
328 } elseif ($lot->quantity_available > 0 && $lot->status === MaterialLot::STATUS_CONSUMED) {
329 $lot->status = MaterialLot::STATUS_RELEASED;
330 }
331
332 $lot->save();
333 }

335 /**
336 * Honour the system-wide "block negative stock" setting, the same switch the
337 * material allocation and shop-floor consumption paths respect.
338 *
339 * Both views of the stock are checked, because either can be the short one: the
340 * plant may hold plenty of a material while the warehouse this document issues
341 * from holds none of it.
342 */
343 private function guardNegativeStock(StockDocument $document, Material $material, float $signed): void
344 {
345 if ($signed >= 0 || ! $this->warehouseStock->blocksNegativeStock()) {
346 return;
347 }
348
349 // Locked, not just read: the lock is held for the rest of this transaction, so
350 // the balance cannot be spent by a concurrent posting between the check here
351 // and the move that follows it.
352 $balance = $this->warehouseStock->lockOrCreate([
353 'warehouse_id' => $document->warehouse_id,
354 'material_id' => $material->id,
355 ]);
356
357 if ((float) $balance->quantity + $signed < 0) {
358 throw ValidationException::withMessages([
359 'lines' => __('Posting would drive :material below zero at :warehouse (:available available).', [
360 'material' => $material->code,
361 'warehouse' => $document->warehouse?->code ?? $document->warehouse_id,
362 'available' => (float) $balance->quantity,
363 ]),
364 ]);
365 }
366
367 if ((float) $material->stock_quantity + $signed < 0) {
368 throw ValidationException::withMessages([
369 'lines' => __('Posting would drive :material below zero stock (:available available).', [
370 'material' => $material->code,
371 'available' => (float) $material->stock_quantity,
372 ]),
373 ]);
374 }
375 }

377 /**
378 * Coerce a submitted line into the columns its document type uses, so a
379 * product line can never smuggle in a material id (and vice versa).
380 *
381 * @param array<string, mixed> $line
382 * @return array<string, mixed>
383 */
384 private function normaliseLine(array $line, StockDocument $document): array
385 {
386 $isMaterial = $document->isMaterialDocument();
387
388 return [
389 'material_id' => $isMaterial ? ($line['material_id'] ?? null) : null,
390 'product_type_id' => $isMaterial ? null : ($line['product_type_id'] ?? null),
391 'material_lot_id' => $isMaterial ? ($line['material_lot_id'] ?? null) : null,
392 'lot_number' => $line['lot_number'] ?? null,
393 'quantity' => abs((float) ($line['quantity'] ?? 0)),
394 'unit_of_measure' => $line['unit_of_measure'] ?? null,
395 'notes' => $line['notes'] ?? null,
396 ];
397 }

399 /**
400 * Resolve the warehouse to post against, falling back to the default for the
401 * document's kind. Fails loudly rather than silently posting to the wrong
402 * kind of warehouse.
403 */
404 private function resolveWarehouse(?int $warehouseId, string $type): Warehouse
405 {
406 $kind = StockDocument::warehouseKindFor($type);
407
408 $warehouse = $warehouseId
409 ? Warehouse::find($warehouseId)
410 : Warehouse::resolveDefault($kind);
411
412 if (! $warehouse) {
413 throw ValidationException::withMessages([
414 'warehouse_id' => __('No warehouse is configured for this document type.'),
415 ]);
416 }
417
418 $accepts = $kind === Warehouse::KIND_FINISHED_GOODS
419 ? $warehouse->acceptsProducts()
420 : $warehouse->acceptsMaterials();
421
422 if (! $accepts) {
423 throw ValidationException::withMessages([
424 'warehouse_id' => __('Warehouse :code cannot hold this kind of item.', ['code' => $warehouse->code]),
425 ]);
426 }
427
428 return $warehouse;
429 }
