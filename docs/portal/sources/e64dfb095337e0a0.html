13/**
14 * Computes the schedule-capacity grid: how many hours each resource has
15 * available (from its shift calendar) versus how many hours of work are
16 * planned against it, per time bucket (day or week).
17 *
18 * The *line* axis ({@see lineCapacity()}) supplies machine capacity: available
19 * hours from active shifts (line-specific plus global) minus scheduled
20 * maintenance, planned hours from active work orders (minute-plans distributed
21 * across buckets, else a best-effort estimate lumped on the placement date;
22 * unestimated orders counted separately rather than treated as zero load).
23 *
24 * The *crew* axis ({@see crewCapacity()}) supplies labor capacity: available
25 * hours from each crew's workers (their line's shifts minus approved absences
26 * and crew break windows), planned hours from the forward labor demand of the
27 * lines the crew staffs (explicit crew↔line assignment, else derived from
28 * worker → workstation → line), weighted by each order's required operators.
29 * See {@see crewCapacity()} for the demand model.
30 */
31class CapacityService
32{
33 public function __construct(private WorkforceProvider $availability) {}

35 /**
36 * Build the line-axis capacity grid for the given range and granularity.
37 *
38 * @return array{granularity: string, buckets: array<int, array>, resources: array<int, array>}
39 */
40 public function lineCapacity(Carbon $rangeStart, Carbon $rangeEnd, string $granularity = 'week'): array
41 {
42 $granularity = in_array($granularity, ['day', 'week'], true) ? $granularity : 'week';
43
44 $rangeStart = $rangeStart->copy()->startOfDay();
45 $rangeEnd = $rangeEnd->copy()->endOfDay();
46
47 $buckets = $this->buildBuckets($rangeStart, $rangeEnd, $granularity);
48
49 $lines = Line::where('is_active', true)->orderBy('name')->get();
50 $shiftIndex = $this->buildShiftIndex(Shift::where('is_active', true)->get());
51
52 $workOrders = $this->activeWorkOrdersInRange($rangeStart, $rangeEnd);
53
54 $maintenance = MaintenanceEvent::whereIn('status', ['pending', 'in_progress'])
55 ->whereNotNull('scheduled_at')
56 ->whereNotNull('line_id')
57 ->where('scheduled_at', '>=', $rangeStart)
58 ->where('scheduled_at', '<=', $rangeEnd)
59 ->get();
60
61 $resources = [];
62 foreach ($lines as $line) {
63 // Every segment an order runs on this line consumes time here —
64 // the primary placement plus any extra segments, each on its dates.
65 $lineOrders = $workOrders->flatMap(fn ($wo) => $this->projectionsForLine($wo, $line->id));
66 $lineMaint = $maintenance->where('line_id', $line->id);
67
68 $cells = [];
69 foreach ($buckets as $bucket) {
70 $available = $this->lineAvailableMinutes($shiftIndex, $line->id, $bucket, $rangeStart, $rangeEnd)
71 - $this->maintenanceMinutes($lineMaint, $bucket);
72
73 $cells[$bucket['key']] = [
74 'available_minutes' => max(0, $available),
75 'planned_minutes' => 0,
76 'unestimated_count' => 0,
77 ];
78 }
79
80 foreach ($lineOrders as $wo) {
81 $this->placeOrder($wo, $cells, $buckets);
82 }
83
84 $resources[] = [
85 'id' => $line->id,
86 'name' => $line->name,
87 'code' => $line->code,
88 'cells' => $this->finalizeCells($cells),
89 ];
90 }
91
92 return $this->grid($granularity, $buckets, $resources);
93 }

95 /**
96 * Build the crew-axis (labor) capacity grid.
97 *
98 * Available = the labor-hours each crew supplies per bucket (its workers'
99 * line shifts minus approved absences and crew break windows).
100 *
101 * Planned = the forward labor demand of the lines the crew staffs. A crew
102 * "staffs" the lines its explicit assignment lists (or, failing that, the
103 * lines its active workers' workstations belong to). Each such line's
104 * planned machine-hours are weighted into person-hours by its orders'
105 * duration-weighted operator count ({@see WorkOrder::operatorFactor()}),
106 * then split equally among the crews that staff the line so shared lines
107 * aren't double-counted. Demand on lines no crew staffs is surfaced in an
108 * "Unassigned" row (zero available, so it reads as over-capacity).
109 *
110 * @return array{granularity: string, buckets: array<int, array>, resources: array<int, array>}
111 */
112 public function crewCapacity(Carbon $rangeStart, Carbon $rangeEnd, string $granularity = 'week'): array
113 {
114 $granularity = in_array($granularity, ['day', 'week'], true) ? $granularity : 'week';
115
116 $rangeStart = $rangeStart->copy()->startOfDay();
117 $rangeEnd = $rangeEnd->copy()->endOfDay();
118
119 $buckets = $this->buildBuckets($rangeStart, $rangeEnd, $granularity);
120
121 // Crews are workforce administration, and not every installation records
122 // them. Asking the contract first keeps the model out of the picture
123 // entirely when there are none — but the rest of this method still runs,
124 // because demand on a line nobody staffs is exactly what the Unassigned
125 // row exists to show.
126 $crews = $this->availability->crewsForCapacity();
127
128 $shiftIndex = $this->buildShiftIndex(Shift::where('is_active', true)->get());
129
130 // Operator-weighted labor demand per line per bucket. Every segment an
131 // order runs demands labor on its line, each on its own dates.
132 $ordersByLine = [];
133 foreach ($this->activeWorkOrdersInRange($rangeStart, $rangeEnd) as $wo) {
134 $lineIds = array_unique(array_filter([$wo->line_id, ...$wo->extraPlacements->pluck('line_id')->all()]));
135 foreach ($lineIds as $lineId) {
136 foreach ($this->projectionsForLine($wo, $lineId) as $projection) {
137 $ordersByLine[$lineId][] = $projection;
138 }
139 }
140 }
141 $lineDemand = [];
142 foreach ($ordersByLine as $lineId => $orders) {
143 $lineDemand[$lineId] = $this->lineLaborMinutes(collect($orders), $buckets);
144 }
145
146 // Lines each crew staffs (resolved once per crew) and how many crews
147 // staff each line, to split a line's demand fairly.
148 $crewLineIdsById = [];
149 $crewsPerLine = [];
150 foreach ($crews as $crew) {
151 $crewLineIdsById[$crew->id] = $this->crewLineIds($crew);
152 foreach ($crewLineIdsById[$crew->id] as $lineId) {
153 $crewsPerLine[$lineId] = ($crewsPerLine[$lineId] ?? 0) + 1;
154 }
155 }
156
157 // Absent-worker IDs are resolved once per distinct day and reused.
158 $absentByDay = [];
159
160 $resources = [];
161 foreach ($crews as $crew) {
162 $cells = [];
163 foreach ($buckets as $bucket) {
164 // Float through the split; round once when converting to hours.
165 $planned = 0.0;
166 foreach ($crewLineIdsById[$crew->id] as $lineId) {
167 $lineMinutes = $lineDemand[$lineId][$bucket['key']] ?? 0;
168 $planned += $lineMinutes / max(1, $crewsPerLine[$lineId] ?? 1);
169 }
170
171 $cells[$bucket['key']] = [
172 'available_minutes' => $this->crewAvailableMinutes($crew, $shiftIndex, $bucket, $rangeStart, $rangeEnd, $absentByDay),
173 'planned_minutes' => $planned,
174 'unestimated_count' => 0,
175 ];
176 }
177
178 $resources[] = [
179 'id' => $crew->id,
180 'name' => $crew->name,
181 'code' => $crew->code,
182 'cells' => $this->finalizeCells($cells),
183 ];
184 }
185
186 // Labor demand on lines no crew staffs would otherwise vanish; surface
187 // it in an "Unassigned" row with zero available labor (reads as over
188 // capacity) so the planner sees unmet demand instead of idle crews.
189 $unassigned = [];
190 $hasUnassigned = false;
191 foreach ($buckets as $bucket) {
192 $minutes = 0;
193 foreach ($lineDemand as $lineId => $perBucket) {
194 if (($crewsPerLine[$lineId] ?? 0) === 0) {
195 $minutes += $perBucket[$bucket['key']] ?? 0;
196 }
197 }
198 $unassigned[$bucket['key']] = ['available_minutes' => 0, 'planned_minutes' => $minutes, 'unestimated_count' => 0];
199 $hasUnassigned = $hasUnassigned || $minutes > 0;
200 }
201 if ($hasUnassigned) {
202 $resources[] = [
203 'id' => 0,
204 'name' => __('Unassigned'),
205 'code' => null,
206 'cells' => $this->finalizeCells($unassigned),
207 ];
208 }
209
210 return $this->grid($granularity, $buckets, $resources);
211 }

213 /**
214 * Work orders contributing to a single line-axis cell (for the drill-down),
215 * each with the hours it contributes to that bucket. Uses the same
216 * {@see orderContributions()} as the grid, so the listed orders always
217 * reconcile with the cell total.
218 *
219 * @return array<int, array>
220 */
221 public function cellOrders(int $lineId, Carbon $bucketStart, Carbon $bucketEnd): array
222 {
223 $bucket = [
224 'key' => 'cell',
225 'start' => $bucketStart->copy()->startOfDay(),
226 'end' => $bucketEnd->copy()->endOfDay(),
227 ];
228
229 $orders = WorkOrder::with(['batches.steps', 'productType', 'extraPlacements'])
230 ->whereIn('status', WorkOrder::ACTIVE_STATUSES)
231 ->where(function ($q) use ($lineId) {
232 $q->where('line_id', $lineId)
233 ->orWhereHas('extraPlacements', fn ($q2) => $q2->where('line_id', $lineId));
234 })
235 ->where(function ($q) use ($bucket) {
236 $q->whereBetween('due_date', [$bucket['start'], $bucket['end']])
237 ->orWhereBetween('planned_start_at', [$bucket['start'], $bucket['end']])
238 ->orWhereHas('extraPlacements', fn ($q2) => $q2->whereBetween('due_date', [$bucket['start'], $bucket['end']]))
239 ->orWhere(function ($q2) use ($bucket) {
240 $q2->whereNotNull('planned_start_at')
241 ->whereNotNull('planned_end_at')
242 ->where('planned_start_at', '<', $bucket['end'])
243 ->where('planned_end_at', '>', $bucket['start']);
244 })
245 ->orWhereNull('due_date');
246 })
247 ->orderBy('priority', 'desc')
248 ->orderBy('due_date')
249 ->get();
250
251 $rows = [];
252 foreach ($orders as $raw) {
253 // Sum every segment the order runs on this line within the bucket.
254 $minutes = 0;
255 $unestimated = false;
256 $wo = $raw;
257 foreach ($this->projectionsForLine($raw, $lineId) as $projection) {
258 $contribution = $this->orderContributions($projection, [$bucket]);
259 $minutes += $contribution['minutes'][$bucket['key']] ?? 0;
260 $unestimated = $unestimated || $contribution['unestimatedKey'] === $bucket['key'];
261 $wo = $projection;
262 }
263 if ($minutes <= 0 && ! $unestimated) {
264 continue;
265 }
266
267 $rows[] = [
268 'id' => $wo->id,
269 'order_no' => $wo->order_no,
270 'product_name' => $wo->productType?->name,
271 'status' => $wo->status,
272 'line_id' => $wo->line_id,
273 'due_date' => $wo->due_date?->toDateString(),
274 'week_number' => $wo->week_number,
275 'shift_number' => $wo->shift_number,
276 'planned_start_at' => $wo->planned_start_at?->toIso8601String(),
277 'planned_end_at' => $wo->planned_end_at?->toIso8601String(),
278 'hours' => round($minutes / 60, 1),
279 'unestimated' => $unestimated,
280 'minute_planned' => $wo->hasMinutePlanning(),
281 ];
282 }
283
284 return $rows;
285 }

306 /**
307 * How a work order's planned hours spread across the buckets — computed
308 * once per order (bucket placement is resolved a single time, not per
309 * bucket). The single source of truth for both axes and the drill-down.
310 *
311 * @return array{minutes: array<string, int>, unestimatedKey: ?string}
312 * minutes: bucket key => planned minutes;
313 * unestimatedKey: the bucket an unestimable order is flagged in
314 */
315 private function orderContributions(WorkOrder $wo, array $buckets): array
316 {
317 $minutes = [];
318
319 if ($wo->hasMinutePlanning()) {
320 // An inverted/zero span (planned_end <= planned_start) is invalid but the
321 // data model permits it (plannedDurationMinutes() returns it unclamped).
322 // Surface it as unestimated in the bucket holding planned_start_at rather
323 // than silently dropping it (a hidden double-book).
324 if ($wo->planned_end_at->lessThanOrEqualTo($wo->planned_start_at)) {
325 foreach ($buckets as $bucket) {
326 if ($wo->planned_start_at->between($bucket['start'], $bucket['end'])) {
327 return ['minutes' => $minutes, 'unestimatedKey' => $bucket['key']];
328 }
329 }
330
331 return ['minutes' => $minutes, 'unestimatedKey' => null];
332 }
333
334 foreach ($buckets as $bucket) {
335 $overlap = $this->overlapMinutes($wo->planned_start_at, $wo->planned_end_at, $bucket);
336 if ($overlap > 0) {
337 $minutes[$bucket['key']] = $overlap;
338 }
339 }
340
341 return ['minutes' => $minutes, 'unestimatedKey' => null];
342 }
343
344 // Non-minute-planned orders belong to exactly one bucket (their placement
345 // date / week), resolved once here rather than per bucket.
346 $key = $this->bucketKeyForOrder($wo, $buckets);
347 if ($key === null) {
348 return ['minutes' => $minutes, 'unestimatedKey' => null];
349 }
350
351 $estimate = $wo->estimatedDurationMinutes();
352 if ($estimate === null) {
353 return ['minutes' => $minutes, 'unestimatedKey' => $key];
354 }
355
356 $minutes[$key] = $estimate;
357
358 return ['minutes' => $minutes, 'unestimatedKey' => null];
359 }

361 /**
362 * The work order's segments on one line, as schedulable proxies: the
363 * order itself when its primary placement is on the line, plus a detached
364 * copy per extra segment there. Extra segments are coarse-only — the
365 * minute plan belongs to the primary placement and never follows a copy.
366 *
367 * @return array<int, WorkOrder>
368 */
369 private function projectionsForLine(WorkOrder $wo, int $lineId): array
370 {
371 $out = [];
372 if ($wo->line_id === $lineId) {
373 $out[] = $wo;
374 }
375 foreach ($wo->extraPlacements as $p) {
376 if ($p->line_id !== $lineId) {
377 continue;
378 }
379 $proxy = clone $wo;
380 $proxy->due_date = $p->due_date;
381 $proxy->shift_number = $p->shift_number ?? $wo->shift_number;
382 $proxy->end_date = $p->end_date;
383 $proxy->end_shift_number = $p->end_shift_number;
384 $proxy->planned_start_at = null;
385 $proxy->planned_end_at = null;
386 $out[] = $proxy;
387 }
388
389 return $out;
390 }

406 /**
407 * Active, line-assigned work orders that fall in the visible range — the
408 * shared source of planned hours for both the line and crew axes.
409 */
410 private function activeWorkOrdersInRange(Carbon $rangeStart, Carbon $rangeEnd): \Illuminate\Database\Eloquent\Collection
411 {
412 return WorkOrder::with(['batches.steps', 'extraPlacements'])
413 ->whereIn('status', WorkOrder::ACTIVE_STATUSES)
414 ->whereNotNull('line_id')
415 ->where(function ($q) use ($rangeStart, $rangeEnd) {
416 $q->whereBetween('due_date', [$rangeStart, $rangeEnd])
417 ->orWhereBetween('planned_start_at', [$rangeStart, $rangeEnd])
418 ->orWhere(fn ($span) => $span->where('planned_start_at', '<=', $rangeEnd)->where('end_date', '>=', $rangeStart))
419 // Independently-scheduled extra segments count too.
420 ->orWhereHas('extraPlacements', fn ($q2) => $q2->whereBetween('due_date', [$rangeStart, $rangeEnd]))
421 ->orWhere(function ($q2) use ($rangeStart, $rangeEnd) {
422 // Minute-planned orders overlapping the visible range.
423 $q2->whereNotNull('planned_start_at')
424 ->whereNotNull('planned_end_at')
425 ->where('planned_start_at', '<', $rangeEnd)
426 ->where('planned_end_at', '>', $rangeStart);
427 })
428 ->orWhereNull('due_date'); // week-only orders
429 })
430 ->get();

433 /**
434 * Operator-weighted labor minutes per bucket for a single line's orders:
435 * each order's machine-minutes in a bucket × its duration-weighted operator
436 * count ({@see WorkOrder::operatorFactor()}). This is the crew-axis demand —
437 * the person-minutes the line's planned work needs. With every step needing
438 * one operator it equals machine-minutes. Kept as floats and rounded once at
439 * the hour-display step, so per-order rounding can't drift the bucket total.
440 *
441 * @return array<string, float> bucket key => labor minutes
442 */
443 private function lineLaborMinutes(Collection $orders, array $buckets): array
444 {
445 $labor = [];
446 foreach ($buckets as $bucket) {
447 $labor[$bucket['key']] = 0.0;
448 }
449 foreach ($orders as $wo) {
450 $factor = $wo->operatorFactor();
451 foreach ($this->orderContributions($wo, $buckets)['minutes'] as $key => $minutes) {
452 $labor[$key] += $minutes * $factor;
453 }
454 }
455
456 return $labor;
457 }

501 /**
502 * @return array<int, array{key: string, label: string, start: Carbon, end: Carbon}>
503 */
504 private function buildBuckets(Carbon $start, Carbon $end, string $granularity): array
505 {
506 $buckets = [];
507
508 if ($granularity === 'day') {
509 $cursor = $start->copy()->startOfDay();
510 while ($cursor->lte($end)) {
511 $buckets[] = [
512 'key' => $cursor->format('Y-m-d'),
513 'label' => $cursor->translatedFormat('D d.m'),
514 'start' => $cursor->copy()->startOfDay(),
515 'end' => $cursor->copy()->endOfDay(),
516 ];
517 $cursor->addDay();
518 }
519
520 return $buckets;
521 }
522
523 $cursor = $start->copy()->startOfWeek();
524 while ($cursor->lte($end)) {
525 $weekStart = $cursor->copy()->startOfWeek();
526 $weekEnd = $cursor->copy()->endOfWeek();
527 $buckets[] = [
528 'key' => $weekStart->format('o-\WW'),
529 'label' => __('Week').' '.$cursor->isoWeek(),
530 'start' => $weekStart,
531 'end' => $weekEnd,
532 ];
533 $cursor->addWeek();
534 }
535
536 return $buckets;
537 }

539 /**
540 * Pre-index active shift minutes by line and ISO weekday, so per-day
541 * availability is an O(1) lookup instead of re-scanning every shift for
542 * every (resource, bucket, day). Global shifts (line_id null) are kept
543 * separately and added to every line.
544 *
545 * @return array{global: array<int, int>, line: array<int, array<int, int>>}
546 */
547 private function buildShiftIndex(Collection $shifts): array
548 {
549 $index = ['global' => [], 'line' => []];
550
551 foreach ($shifts as $shift) {
552 $minutes = $this->shiftMinutes($shift);
553 for ($dow = 1; $dow <= 7; $dow++) {
554 if (! $this->shiftRunsOn($shift, $dow)) {
555 continue;
556 }
557 if ($shift->line_id === null) {
558 $index['global'][$dow] = ($index['global'][$dow] ?? 0) + $minutes;
559 } else {
560 $index['line'][$shift->line_id][$dow] = ($index['line'][$shift->line_id][$dow] ?? 0) + $minutes;
561 }
562 }
563 }
564
565 return $index;
566 }

568 /**
569 * Shift minutes available to a given line on an ISO weekday: global shifts
570 * plus that line's own shifts. A null line resolves to global shifts only.
571 */
572 private function shiftMinutesForLine(array $index, ?int $lineId, int $isoDow): int
573 {
574 $global = $index['global'][$isoDow] ?? 0;
575 $line = $lineId !== null ? ($index['line'][$lineId][$isoDow] ?? 0) : 0;
576
577 return $global + $line;
578 }

580 /**
581 * Walk each day of a bucket that falls within the visible range, summing
582 * the integer the callback returns for that day. Shared scaffolding for
583 * the line and crew availability calculations.
584 */
585 private function eachDayInBucket(array $bucket, Carbon $rangeStart, Carbon $rangeEnd, callable $perDay): int
586 {
587 $from = $bucket['start']->greaterThan($rangeStart) ? $bucket['start'] : $rangeStart;
588 $to = $bucket['end']->lessThan($rangeEnd) ? $bucket['end'] : $rangeEnd;
589
590 $total = 0;
591 $cursor = $from->copy()->startOfDay();
592 $last = $to->copy()->startOfDay();
593 while ($cursor->lte($last)) {
594 $total += $perDay($cursor, (int) $cursor->format('N'));
595 $cursor->addDay();
596 }
597
598 return $total;
599 }

614 /**
615 * Labor-minutes a crew supplies within a bucket: for each active,
616 * non-absent worker, their line's shift minutes on each working day, less
617 * the crew's break windows for that day (floored at zero per worker-day).
618 *
619 * @param array<string, array<int>> $absentByDay per-day absent-worker cache (by reference)
620 */
621 private function crewAvailableMinutes(object $crew, array $shiftIndex, array $bucket, Carbon $rangeStart, Carbon $rangeEnd, array &$absentByDay): int
622 {
623 if ($crew->workers->isEmpty()) {
624 return 0;
625 }
626
627 return $this->eachDayInBucket($bucket, $rangeStart, $rangeEnd, function (Carbon $date, int $dow) use ($crew, $shiftIndex, &$absentByDay) {
628 $dateKey = $date->format('Y-m-d');
629 if (! isset($absentByDay[$dateKey])) {
630 $absentByDay[$dateKey] = $this->availability->absentWorkerIds($date);
631 }
632 $absent = $absentByDay[$dateKey];
633 $breakMinutes = $this->crewBreakMinutes($crew, $date);
634
635 $dayTotal = 0;
636 foreach ($crew->workers as $worker) {
637 if (in_array($worker->id, $absent, true)) {
638 continue;
639 }
640 $shiftMinutes = $this->shiftMinutesForLine($shiftIndex, $worker->workstation?->line_id, $dow);
641 if ($shiftMinutes > 0) {
642 $dayTotal += max(0, $shiftMinutes - $breakMinutes);
643 }
644 }
645
646 return $dayTotal;
647 });
648 }

650 /**
651 * Whether a shift is active on the given ISO weekday (1=Mon .. 7=Sun).
652 * `days_of_week` is normalized to ISO integers by {@see \App\Casts\DaysOfWeek}.
653 * Matches {@see Shift::current()}: an unset (null) list runs every day, an
654 * explicitly empty list runs on no day.
655 */
656 private function shiftRunsOn(Shift $shift, int $isoDow): bool
657 {
658 $days = $shift->days_of_week;
659 if (! is_array($days)) {
660 return true; // null / unset = every day
661 }
662
663 return in_array($isoDow, $days, true); // empty array = no day
664 }

666 /**
667 * Duration of a single shift occurrence in minutes. Computed from the
668 * nominal clock times (not wall-clock instants), so an "8h shift" always
669 * counts 480 minutes regardless of timezone or a DST transition that day.
670 * A shift whose end is at or before its start is treated as crossing
671 * midnight; a zero-length shift (start == end) counts as zero, not a day.
672 */
673 private function shiftMinutes(Shift $shift): int
674 {
675 return $this->clockSpanMinutes($shift->start_time, $shift->end_time);
676 }

678 /**
679 * Total crew break-window minutes that apply on the given date. Defers the
680 * weekday-membership check to {@see \App\Models\CrewBreakWindow::appliesOn()}
681 * so the empty-days semantics ("never applies") stay consistent with the
682 * rest of the app rather than being re-implemented here.
683 */
684 private function crewBreakMinutes(object $crew, Carbon $date): int
685 {
686 $total = 0;
687 foreach ($crew->breakWindows as $window) {
688 if (! $window->appliesOn($date)) {
689 continue;
690 }
691 $total += $this->clockSpanMinutes($window->start_time, $window->end_time);
692 }
693
694 return $total;
695 }

697 /**
698 * Minutes between two nominal clock times ("HH:MM" / "HH:MM:SS"), wrapping
699 * past midnight when the end is at or before the start. Pure clock math —
700 * independent of timezone and DST, which is what nominal shift/break spans
701 * should be.
702 */
703 private function clockSpanMinutes(string $start, string $end): int
704 {
705 $minutes = $this->clockMinutes($end) - $this->clockMinutes($start);
706 if ($minutes < 0) {
707 $minutes += 24 * 60;
708 }
709
710 return $minutes;
711 }
712
713 private function clockMinutes(string $time): int
714 {
715 [$h, $m] = array_pad(explode(':', $time), 2, '0');
716
717 return ((int) $h) * 60 + (int) $m;
718 }

720 /**
721 * Minutes consumed by scheduled maintenance within the bucket. The event's
722 * span is clamped to the bucket so an event crossing a bucket boundary only
723 * subtracts the minutes that actually fall inside it.
724 */
725 private function maintenanceMinutes(Collection $events, array $bucket): int
726 {
727 $total = 0;
728 foreach ($events as $event) {
729 if (! $event->scheduled_at) {
730 continue;
731 }
732 $end = $event->scheduled_end_at ?? $event->scheduled_at->copy()->addMinutes(60);
733 $total += $this->overlapMinutes($event->scheduled_at, $end, $bucket);
734 }
735
736 return $total;
737 }
738
739 /**
740 * Minutes of [$start, $end] that fall within a bucket.
741 */
742 private function overlapMinutes(Carbon $start, Carbon $end, array $bucket): int
743 {
744 $overlapStart = $start->greaterThan($bucket['start']) ? $start : $bucket['start'];
745 $overlapEnd = $end->lessThan($bucket['end']) ? $end : $bucket['end'];
746 $minutes = (int) $overlapStart->diffInMinutes($overlapEnd, false);
747
748 return max(0, $minutes);
749 }

751 /**
752 * Resolve the bucket a non-minute-planned order belongs to, by its due
753 * date or — failing that — its ISO week number. Returns null when the
754 * placement date is not within any visible bucket.
755 */
756 private function bucketKeyForOrder(WorkOrder $wo, array $buckets): ?string
757 {
758 $date = $wo->planned_start_at ?? $wo->due_date;
759
760 if (! $date && $wo->week_number && ! empty($buckets)) {
761 $year = $wo->production_year ?? (int) $buckets[0]['start']->isoWeekYear;
762 $date = $buckets[0]['start']->copy()->setISODate($year, (int) $wo->week_number)->startOfWeek();
763 }
764
765 if (! $date) {
766 return null;
767 }
768
769 foreach ($buckets as $bucket) {
770 if ($date->between($bucket['start'], $bucket['end'])) {
771 return $bucket['key'];
772 }
773 }
774
775 return null;
776 }
