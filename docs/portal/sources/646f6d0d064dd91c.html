46 /**
47 * Register a sub-assembly by its own serial where it is made (a board, a
48 * module), so the product it goes into later is linked to a unit with a
49 * history - not to a bare number. It has no process serial and does not
50 * follow the product's serial format; it is complete once registered.
51 *
52 * @throws BindingException when the serial is already taken
53 */
54 public function registerSubassembly(string $serialNo, Material $material, WorkOrder $workOrder, User $operator, ?int $workstationId = null): SerialUnit
55 {
56 $serialNo = app(UnitSerialisation::class)->normalize($serialNo) ?? '';
57
58 return DB::transaction(function () use ($serialNo, $material, $workOrder, $operator, $workstationId) {
59 if ($serialNo === '') {
60 throw new BindingException(__('Scan the serial number of the sub-assembly.'));
61 }
62 $owner = SerialUnit::where('serial_no', $serialNo)->lockForUpdate()->first()
63 ?? SerialUnit::where('psn', $serialNo)->lockForUpdate()->first();
64 if ($owner) {
65 throw new BindingException(__('Serial :sn is already registered (:material, order :order).', [
66 'sn' => $serialNo,
67 'material' => $owner->material?->name ?? '—',
68 'order' => $owner->workOrder?->order_no ?? '—',
69 ]));
70 }
71
72 try {
73 $unit = SerialUnit::create([
74 'serial_no' => $serialNo,
75 'material_id' => $material->id,
76 'work_order_id' => $workOrder->id,
77 'status' => SerialUnit::STATUS_COMPLETED,
78 'produced_at' => now(),
79 ]);
80 } catch (\Illuminate\Database\UniqueConstraintViolationException) {
81 // The same label scanned at two benches at once: one wins, the other is told.
82 throw new BindingException(__('Serial :sn is already registered (:material, order :order).', ['sn' => $serialNo, 'material' => '—', 'order' => '—']));
83 }
84 $this->recordStep($unit, $operator, null, [
85 'workstation_id' => $workstationId,
86 'parameters' => ['event' => 'subassembly_registered', 'material' => $material->code],
87 ]);
88
89 return $unit->fresh();
90 });
91 }

93 /**
94 * Record a processing event for a unit at a workstation. The high-precision
95 * timestamp guarantees ordering even for sub-second consecutive steps.
96 */
97 public function recordStep(SerialUnit $unit, User $operator, ?BatchStep $step, array $data = []): UnitStepHistory
98 {
99 return DB::transaction(function () use ($unit, $operator, $step, $data) {
100 $entry = $unit->history()->create([
101 'batch_step_id' => $step?->id,
102 // `exact_workstation`: the caller knows the bench, or knows it is unknown
103 // (a tester log from an unregistered station) - no fallback to the operator's.
104 'workstation_id' => ! empty($data['exact_workstation'])
105 ? ($data['workstation_id'] ?? null)
106 : ($data['workstation_id'] ?? $step?->workstation_id ?? $operator->workstation_id),
107 'operator_id' => $operator->id,
108 'parameters' => $data['parameters'] ?? null,
109 'result' => $data['result'] ?? null,
110 'notes' => $data['notes'] ?? null,
111 // An import passes the tester's own time so history reads in the
112 // order things happened, not the order files were loaded.
113 // Stored in the app's timezone whatever zone the caller's stamp
114 // carries (a tester's "…Z"), so history sorts with station rows.
115 'processed_at' => ($data['processed_at'] ?? now())->copy()->setTimezone(config('app.timezone'))->format('Y-m-d H:i:s.u'),
116 ]);
117
118 if (($data['result'] ?? null) === 'fail' && ($data['parameters']['event'] ?? null) !== 'scrapped') {
119 $this->applyFailPolicy($unit, $entry, $operator);
120 }
121 // A hold from failed tests waits for a retest: a pass lifts it (policy
122 // `block`) and the unit goes back to where it was - packed units stay
123 // packed. A hold someone put on by hand (a non-conformity) is not a
124 // test's to lift.
125 $unit->refresh();
126 // Only a test's pass - not an inspection or step marked "pass" elsewhere.
127 if (($data['result'] ?? null) === 'pass' && ($data['parameters']['event'] ?? null) === 'test'
128 && $unit->status === SerialUnit::STATUS_BLOCKED
129 && ($unit->extra_data['hold']['source'] ?? 'test') !== 'manual'
130 && $this->passLiftsHold($unit->extra_data['hold'] ?? [], $entry)) {
131 $unit->update([
132 'status' => $unit->packed_at ? SerialUnit::STATUS_COMPLETED : SerialUnit::STATUS_IN_PRODUCTION,
133 'extra_data' => array_diff_key($unit->extra_data ?? [], ['hold' => true]) ?: null,
134 ]);
135 // The release shows in the history next to the pass that made it.
136 $this->recordStep($unit, $operator, null, [
137 'workstation_id' => $entry->workstation_id,
138 'exact_workstation' => true,
139 'processed_at' => $entry->processed_at,
140 'parameters' => array_filter(['event' => 'unblocked', 'hold_source' => 'test', 'released_by' => 'test', 'station' => $entry->parameters['station'] ?? null]),
141 ]);
142 }
143 // A verdict on a unit already on a pallet changes what that pallet may
144 // do. Read the pallet from the database: the caller's instance may
145 // predate the packing.
146 if (($data['result'] ?? null) !== null) {
147 $palletId = SerialUnit::whereKey($unit->id)->value('pallet_id');
148 if ($palletId) {
149 Pallet::find($palletId)?->recomputeQualityStatus();
150 }
151 }
152
153 return $entry;
154 });
155 }

157 /**
158 * What a failed verdict does to the unit is the plant's call (settings):
159 * scrap at once, count attempts and block once the limit is reached so a
160 * retest can still clear it, or just keep the verdict. Scrapped is final.
161 */
162 private function applyFailPolicy(SerialUnit $unit, UnitStepHistory $entry, User $operator): void
163 {
164 $settings = app(UnitSerialisation::class);
165 $policy = $settings->get('unit_test_fail_policy');
166
167 if ($policy === UnitSerialisation::FAIL_SCRAP) {
168 // A late or re-imported log does not scrap a unit that already left (shipped) or is scrapped.
169 $unit->refresh();
170 if (! in_array($unit->status, SerialUnit::TERMINAL_STATUSES, true)) {
171 $unit->update(['status' => SerialUnit::STATUS_SCRAPPED]);
172 }
173
174 return;
175 }
176 if ($policy !== UnitSerialisation::FAIL_BLOCK) {
177 return; // record only
178 }
179
180 $limit = max(1, (int) $settings->get('unit_test_max_attempts'));
181
182 $fails = $settings->get('unit_test_attempts_scope') === UnitSerialisation::ATTEMPTS_TEST
183 ? $this->failsSinceLastPassAt($unit, $entry)
184 : $unit->history()->where('result', 'fail')->count();
185 if ($fails >= $limit && ! in_array($unit->status, SerialUnit::TERMINAL_STATUSES, true)) {
186 $unit->refresh();
187 $wasBlocked = $unit->status === SerialUnit::STATUS_BLOCKED;
188 $unit->update([
189 'status' => SerialUnit::STATUS_BLOCKED,
190 // A hand-made hold stays what it is; otherwise the tests put it on -
191 // remembering which test and when, so only that test's later pass lifts it.
192 'extra_data' => array_merge($unit->extra_data ?? [], ['hold' => $unit->extra_data['hold'] ?? array_filter([
193 'source' => 'test',
194 'at' => $entry->processed_at?->toIso8601String() ?? now()->toIso8601String(),
195 'workstation_id' => $entry->workstation_id,
196 'station' => $entry->parameters['station'] ?? null,
197 ], fn ($v) => $v !== null)]),
198 ]);
199 // The hold is a moment in the unit's history, not only its status:
200 // it reads right after the fail that put it on, with the count.
201 if (! $wasBlocked) {
202 $this->recordStep($unit, $operator, null, [
203 'workstation_id' => $entry->workstation_id,
204 'exact_workstation' => true,
205 'processed_at' => $entry->processed_at,
206 'parameters' => array_filter([
207 'event' => 'blocked', 'hold_source' => 'test',
208 'station' => $entry->parameters['station'] ?? null,
209 'failed_attempts' => $fails, 'max_attempts' => $limit,
210 ], fn ($v) => $v !== null),
211 ]);
212 }
213 }
214 }

216 /**
217 * Failed verdicts of this test: at the same station as the verdict just
218 * recorded, after that station's last pass (by the tester's own time - logs
219 * can arrive out of order). A scrap is not a test and does not count.
220 */
221 private function failsSinceLastPassAt(SerialUnit $unit, UnitStepHistory $entry): int
222 {
223 $atStation = fn () => $this->sameTest($unit->history()->reorder(), $entry->workstation_id, $entry->parameters['station'] ?? null);
224
225 $lastPass = $atStation()->where('result', 'pass')->max('processed_at');
226
227 return $atStation()->where('result', 'fail')
228 ->where(fn ($q) => $q->whereNull('parameters->event')->orWhere('parameters->event', '!=', 'scrapped'))
229 ->when($lastPass, fn ($q) => $q->where('processed_at', '>', $lastPass))
230 ->count();
231 }

233 /**
234 * Rows of the same test: the same workstation, or - for a tester the plant
235 * has no workstation for - the same station code in its log.
236 */
237 private function sameTest($query, ?int $workstationId, ?string $station)
238 {
239 if ($workstationId) {
240 return $query->where('workstation_id', $workstationId);
241 }
242
243 return $query->whereNull('workstation_id')
244 ->when($station !== null, fn ($q) => $q->where('parameters->station', $station), fn ($q) => $q->whereNull('parameters->station'));
245 }

247 /**
248 * Whether a pass lifts a test hold. Counting per test, only a pass of the
249 * test that put the hold on, and no older than the failure that did; every
250 * failure of the unit counting, any pass does (a hold from before the test
251 * was remembered also lifts on any pass).
252 */
253 private function passLiftsHold(array $hold, UnitStepHistory $pass): bool
254 {
255 // By the tester's own time in either scope: an older pass arriving late
256 // (logs come out of order) does not undo a newer hold.
257 $newer = ! isset($hold['at']) || ! $pass->processed_at || $pass->processed_at->greaterThanOrEqualTo(\Carbon\Carbon::parse($hold['at']));
258 if (app(UnitSerialisation::class)->get('unit_test_attempts_scope') !== UnitSerialisation::ATTEMPTS_TEST
259 || (! isset($hold['workstation_id']) && ! isset($hold['station']))) {
260 return $newer;
261 }
262 $sameTest = isset($hold['workstation_id'])
263 ? (int) $pass->workstation_id === (int) $hold['workstation_id']
264 : $pass->workstation_id === null && ($pass->parameters['station'] ?? null) === $hold['station'];
265
266 return $sameTest && $newer;
267 }

304 /**
305 * Bind a unit's serial number to its process serial number at a station:
306 * the moment the pre-printed label goes onto the product. Registers the
307 * unit on first sight, records the event in its history, and applies the
308 * plant's rules - a process serial owned by another unit is refused when
309 * process serials are unique, and a unit already carrying a different
310 * process serial is only re-bound with `force`, which callers reserve for
311 * supervisors and record with a reason.
312 *
313 * @return array{unit: SerialUnit, created: bool, rebound: bool, serial_assigned: bool}
314 *
315 * @throws BindingException
316 */
317 public function bindProcessSerial(string $serialNo, ?string $psn, User $operator, array $context = []): array
318 {
319 $settings = app(UnitSerialisation::class);
320 $serialNo = $settings->normalize($serialNo);
321 $psn = $settings->normalize($psn);
322
323 return DB::transaction(function () use ($serialNo, $psn, $operator, $context, $settings) {
324 $unit = SerialUnit::where('serial_no', $serialNo)->lockForUpdate()->first();
325 // A unit started on its PSN earlier gets its SN now: the product label.
326 $waiting = $psn !== null
327 ? SerialUnit::where('psn', $psn)->whereNull('serial_no')->when($unit, fn ($q) => $q->whereKeyNot($unit->id))->lockForUpdate()->first()
328 : null;
329 if ($unit && $waiting) {
330 // Two units would claim one PSN: the one waiting for its label and
331 // the one that already carries this SN. A person has to sort that out.
332 throw new BindingException(__('Process serial :psn belongs to a unit still waiting for its serial number, and :sn is already another unit.', ['psn' => $psn, 'sn' => $serialNo]));
333 }
334 $gotSerial = false;
335 if (! $unit && $waiting) {
336 $unit = $waiting;
337 $unit->serial_no = $serialNo;
338 $gotSerial = true;
339 }
340
341 if ($unit && in_array($unit->status, SerialUnit::TERMINAL_STATUSES, true)) {
342 throw new BindingException(__('Unit :sn is :status and can no longer be bound.', ['sn' => $unit->display_id, 'status' => $unit->status_label]));
343 }
344 // A held unit does not move on to its label: that would read as the next
345 // step done while it waits for a retest or a release. A supervisor's
346 // re-binding (a mixed-up PSN) is a correction, not a step, and still goes.
347 if ($unit && $unit->status === SerialUnit::STATUS_BLOCKED && empty($context['force'])) {
348 // With a different PSN on the scan the station still offers the
349 // supervisor's re-bind, the one way this refusal can be overridden.
350 throw new BindingException(__('Unit :sn is blocked - it cannot move on until it passes a retest or a supervisor releases it.', ['sn' => $unit->display_id]),
351 rebindable: $psn !== null && $unit->psn !== null && $unit->psn !== $psn);
352 }
353
354 if ($psn !== null && $settings->get('unit_psn_unique')) {
355 $owner = SerialUnit::where('psn', $psn)->when($unit, fn ($q) => $q->whereKeyNot($unit->id))->first();
356 if ($owner) {
357 throw new BindingException(__('Process serial :psn already belongs to unit :sn.', ['psn' => $psn, 'sn' => $owner->display_id]));
358 }
359 }
360
361 $rebound = false;
362 $previousPsn = $unit?->psn;
363 if ($unit) {
364 if ($psn !== null && $unit->psn !== null && $unit->psn !== $psn) {
365 if (empty($context['force'])) {
366 throw new BindingException(__('Unit :sn is already bound to process serial :psn.', ['sn' => $serialNo, 'psn' => $unit->psn]), rebindable: true);
367 }
368 $rebound = true;
369 }
370 $unit->fill(array_filter([
371 'psn' => $psn ?? $unit->psn,
372 'work_order_id' => $unit->work_order_id ?: ($context['work_order_id'] ?? null),
373 ], fn ($v) => $v !== null));
374 $created = false;
375 } else {
376 $unit = $this->registerUnit($serialNo, [
377 'psn' => $psn,
378 'work_order_id' => $context['work_order_id'] ?? null,
379 'status' => SerialUnit::STATUS_IN_PRODUCTION,
380 ]);
381 $created = true;
382 }
383
384 $unit->extra_data = array_merge($unit->extra_data ?? [], ['label_applied_at' => now()->toIso8601String()]);
385 $unit->save();
386
387 $this->recordStep($unit, $operator, null, [
388 'workstation_id' => $context['workstation_id'] ?? null,
389 'parameters' => array_filter([
390 'event' => $rebound ? 'process_serial_rebound' : 'label_applied',
391 'psn' => $psn,
392 'previous_psn' => $rebound ? $previousPsn : null,
393 ]),
394 'notes' => $rebound ? ($context['reason'] ?? null) : null,
395 ]);
396
397 return ['unit' => $unit->fresh(), 'created' => $created, 'rebound' => $rebound, 'serial_assigned' => $gotSerial];
398 });
399 }

401 /**
402 * Bind a component onto a unit by the identifier on the component's label.
403 * The identifier is resolved to what the system already knows - another
404 * serialised unit, or a material lot - and kept as scanned either way, so a
405 * bought-in part with only a vendor serial still traces. A component that is
406 * itself a serialised unit can only be inside one parent at a time.
407 *
408 * @throws BindingException
409 */
410 public function bindComponent(SerialUnit $unit, string $identifier, User $operator, array $context = []): SerialUnitComponent
411 {
412 $identifier = app(UnitSerialisation::class)->normalize($identifier) ?? '';
413
414 return DB::transaction(function () use ($unit, $identifier, $operator, $context) {
415 if (in_array($unit->status, SerialUnit::TERMINAL_STATUSES, true)) {
416 throw new BindingException(__('Unit :sn is :status and can no longer be bound.', ['sn' => $unit->display_id, 'status' => $unit->status_label]));
417 }
418 if ($identifier === '' || $identifier === $unit->serial_no || $identifier === $unit->psn) {
419 throw new BindingException(__('A unit cannot be a component of itself.'));
420 }
421
422 $existing = $unit->components()->installed()->where('identifier', $identifier)->first();
423 if ($existing) {
424 return $existing; // the same scan twice is a confirmation, not a second part
425 }
426
427 // A sub-assembly is known by its SN, or by its PSN while it has none.
428 $componentUnit = SerialUnit::findByIdentifier($identifier);
429 // Lot numbers are stored as typed; compare both sides folded (case,
430 // spacing) so the match does not depend on the normalise setting.
431 $lot = $componentUnit ? null : MaterialLot::whereRaw("UPPER(REPLACE(lot_number, ' ', '')) = ?", [strtoupper(str_replace(' ', '', $identifier))])->first();
432
433 if ($componentUnit) {
434 // Locked: two benches scanning the same board into two products at
435 // once must not both find it free.
436 $componentUnit = SerialUnit::whereKey($componentUnit->id)->lockForUpdate()->first();
437 // A sub-assembly that is held, scrapped or gone does not go into a product.
438 if (in_array($componentUnit->status, [...SerialUnit::TERMINAL_STATUSES, SerialUnit::STATUS_BLOCKED], true)) {
439 throw new BindingException(__('Component :id is :status and cannot be installed.', ['id' => $identifier, 'status' => $componentUnit->status_label]));
440 }
441 $elsewhere = SerialUnitComponent::installed()->where('component_serial_unit_id', $componentUnit->id)->with('unit:id,serial_no,psn')->first();
442 if ($elsewhere) {
443 throw new BindingException(__('Component :id is already installed in unit :sn.', ['id' => $identifier, 'sn' => $elsewhere->unit->display_id]));
444 }
445 // A unit cannot end up inside something it contains (A ⊃ B ⊃ A).
446 if ($this->contains($componentUnit, $unit)) {
447 throw new BindingException(__('Unit :sn is already inside :id - it cannot also contain it.', ['sn' => $unit->display_id, 'id' => $identifier]));
448 }
449 }
450
451 $component = $unit->components()->create([
452 'identifier' => $identifier,
453 'component_serial_unit_id' => $componentUnit?->id,
454 'material_lot_id' => $lot?->id,
455 'material_id' => $context['material_id'] ?? $lot?->material_id ?? $componentUnit?->material_id,
456 'quantity' => $context['quantity'] ?? 1,
457 'batch_step_id' => $context['batch_step_id'] ?? null,
458 'workstation_id' => $context['workstation_id'] ?? null,
459 'bound_by_id' => $operator->id,
460 'bound_at' => now(),
461 ]);
462
463 $this->recordStep($unit, $operator, null, [
464 'workstation_id' => $context['workstation_id'] ?? null,
465 'parameters' => array_filter([
466 'event' => 'component_bound',
467 'identifier' => $identifier,
468 'material' => $component->material?->code,
469 'kind' => $componentUnit ? 'serial_unit' : ($lot ? 'material_lot' : 'identifier'),
470 ]),
471 ]);
472
473 return $component;
474 });
475 }

477 /**
478 * Scrap a unit - final. The reason goes into its history next to who did
479 * it and where; the row itself is never deleted, the label exists.
480 *
481 * @throws BindingException when the unit already left the line
482 */
483 public function scrapUnit(SerialUnit $unit, User $operator, string $reason, ?int $workstationId = null): SerialUnit
484 {
485 return DB::transaction(function () use ($unit, $operator, $reason, $workstationId) {
486 if ($unit->status === SerialUnit::STATUS_SHIPPED) {
487 throw new BindingException(__('Unit :sn is :status and can no longer be bound.', ['sn' => $unit->display_id, 'status' => $unit->status_label]));
488 }
489 if ($unit->status !== SerialUnit::STATUS_SCRAPPED) {
490 $unit->update(['status' => SerialUnit::STATUS_SCRAPPED]);
491 $this->recordStep($unit, $operator, null, [
492 'workstation_id' => $workstationId ?? $operator->workstation_id,
493 'parameters' => ['event' => 'scrapped'],
494 'result' => 'fail',
495 'notes' => $reason,
496 ]);
497 }
498
499 return $unit->fresh();
500 });
501 }

503 /**
504 * Hold a non-conforming unit with an error code: nothing downstream takes
505 * it (packing, the pallet's quality gate) until a supervisor releases it.
506 * A passing retest does not lift this hold - it was not the test's.
507 *
508 * @throws BindingException
509 */
510 public function blockUnit(SerialUnit $unit, User $operator, ScrapReason $reason, ?string $note = null, ?int $workstationId = null): SerialUnit
511 {
512 return DB::transaction(function () use ($unit, $operator, $reason, $note, $workstationId) {
513 $unit = SerialUnit::whereKey($unit->id)->lockForUpdate()->firstOrFail();
514 if (in_array($unit->status, SerialUnit::TERMINAL_STATUSES, true)) {
515 throw new BindingException(__('Unit :sn is :status and can no longer be bound.', ['sn' => $unit->display_id, 'status' => $unit->status_label]));
516 }
517 if ($unit->status === SerialUnit::STATUS_BLOCKED) {
518 throw new BindingException(__('Unit :sn is already blocked.', ['sn' => $unit->display_id]));
519 }
520 $unit->update([
521 'status' => SerialUnit::STATUS_BLOCKED,
522 'extra_data' => array_merge($unit->extra_data ?? [], ['hold' => [
523 'source' => 'manual', 'reason_code' => $reason->code, 'reason' => $reason->name,
524 'by_id' => $operator->id, 'at' => now()->toIso8601String(),
525 ]]),
526 ]);
527 $this->recordStep($unit, $operator, null, [
528 'workstation_id' => $workstationId ?? $operator->workstation_id,
529 'parameters' => ['event' => 'blocked', 'reason_code' => $reason->code, 'reason' => $reason->name, 'category' => $reason->category],
530 'notes' => $note,
531 ]);
532 $this->recomputePallet($unit);
533
534 return $unit->fresh();
535 });
536 }

538 /**
539 * Release a held unit (by hand or after failed tests) - a supervisor's call,
540 * with the reason. It goes back to where it was: packed units stay packed.
541 *
542 * @throws BindingException
543 */
544 public function unblockUnit(SerialUnit $unit, User $operator, string $note, ?int $workstationId = null): SerialUnit
545 {
546 return DB::transaction(function () use ($unit, $operator, $note, $workstationId) {
547 $unit = SerialUnit::whereKey($unit->id)->lockForUpdate()->firstOrFail();
548 if ($unit->status !== SerialUnit::STATUS_BLOCKED) {
549 throw new BindingException(__('Unit :sn is not blocked.', ['sn' => $unit->display_id]));
550 }
551 $hold = $unit->extra_data['hold'] ?? [];
552 $unit->update([
553 'status' => $unit->packed_at ? SerialUnit::STATUS_COMPLETED : SerialUnit::STATUS_IN_PRODUCTION,
554 'extra_data' => array_diff_key($unit->extra_data ?? [], ['hold' => true]) ?: null,
555 ]);
556 $this->recordStep($unit, $operator, null, [
557 'workstation_id' => $workstationId ?? $operator->workstation_id,
558 'parameters' => array_filter(['event' => 'unblocked', 'reason_code' => $hold['reason_code'] ?? null, 'hold_source' => $hold['source'] ?? 'test']),
559 'notes' => $note,
560 ]);
561 $this->recomputePallet($unit);
562
563 return $unit->fresh();
564 });
565 }

567 /** A unit's hold changes what the pallet it sits on may do. */
568 private function recomputePallet(SerialUnit $unit): void
569 {
570 $palletId = SerialUnit::whereKey($unit->id)->value('pallet_id');
571 if ($palletId) {
572 Pallet::find($palletId)?->recomputeQualityStatus();
573 }

576 /** Take a component out of a unit; the row stays, closed, with the reason. */
577 /** Whether $needle sits somewhere inside $haystack's installed sub-assemblies (any depth). */
578 private function contains(SerialUnit $haystack, SerialUnit $needle, int $depth = 0): bool
579 {
580 if ($depth > 50) {
581 return false;
582 }
583 $childIds = SerialUnitComponent::installed()->where('serial_unit_id', $haystack->id)->whereNotNull('component_serial_unit_id')->pluck('component_serial_unit_id');
584 if ($childIds->contains($needle->id)) {
585 return true;
586 }
587 foreach (SerialUnit::whereIn('id', $childIds)->get() as $child) {
588 if ($this->contains($child, $needle, $depth + 1)) {
589 return true;
590 }
591 }
592
593 return false;
594 }

596 /** @throws BindingException when the product has shipped or been scrapped - its genealogy is final */
597 public function unbindComponent(SerialUnitComponent $component, User $operator, ?string $reason = null, ?int $workstationId = null): SerialUnitComponent
598 {
599 return DB::transaction(function () use ($component, $operator, $reason, $workstationId) {
600 $component = SerialUnitComponent::whereKey($component->id)->lockForUpdate()->firstOrFail();
601 if ($component->unbound_at !== null) {
602 return $component;
603 }
604 // What left the plant (or was scrapped) keeps the parts it had: removing
605 // one would rewrite the record and free the part for another product.
606 if (in_array($component->unit->status, SerialUnit::TERMINAL_STATUSES, true)) {
607 throw new BindingException(__('Unit :sn is :status and can no longer be bound.', ['sn' => $component->unit->display_id, 'status' => $component->unit->status_label]));
608 }
609 $component->update(['unbound_at' => now(), 'unbound_by_id' => $operator->id, 'unbind_reason' => $reason]);
610 $this->recordStep($component->unit, $operator, null, [
611 'workstation_id' => $workstationId ?? $operator->workstation_id,
612 'parameters' => ['event' => 'component_unbound', 'identifier' => $component->identifier],
613 'notes' => $reason,
614 ]);
615
616 return $component;
617 });
618 }
