1/**
2 * Which component renders a given Inertia page name.
3 *
4 * Pages reach the browser from three places, in order of precedence:
5 *
6 * 1. this app, globbed at build time;
7 * 2. a module that was present when the app was built, globbed the same way —
8 * `backend/modules/`, the directory PHP resolves as base_path('modules');
9 * 3. a module installed afterwards, which registered itself at runtime.
10 *
11 * The first two maps are produced by `import.meta.glob` in app.jsx, so Vite
12 * decides them while compiling; a glob that matches nothing yields {}. That is
13 * why the third exists: a module installed through the panel is invisible to
14 * both, and nothing rebuilds the bundle on a running system.
15 *
16 * Plain JS and dependency-free on purpose: the lookup is the part worth testing,
17 * and app.jsx cannot be imported in a test (it boots the whole app on import).
18 */
19
20/** Core pages are keyed by their glob path, e.g. './Pages/admin/lines/Index.jsx'. */
21export function coreKey(name) {
22 return `./Pages/${name}.jsx`;
23}
24
25/**
26 * Module pages are keyed by a path we only know the tail of, because the module
27 * directory name is whatever the installer called it:
28 * '../../modules/<Anything>/resources/js/Pages/<name>.jsx'
29 */
30export function moduleKey(modulePages, name) {
31 const suffix = `/resources/js/Pages/${name}.jsx`;
32
33 return Object.keys(modulePages).find((path) => path.endsWith(suffix)) ?? null;
34}
35
36/**
37 * The page component, or null when nothing provides it.
38 *
39 * Core wins over a module: a module must not be able to shadow a core screen by
40 * naming a file after it. A compiled-in module wins over a runtime-registered
41 * one for the same reason — if both are present, the one the operator built
42 * into their image is the one they meant.
43 */
44export function resolvePage(name, corePages, modulePages = {}, runtimePages = {}) {
45 const core = corePages[coreKey(name)];
46 if (core) {
47 return core;
48 }
49
50 const key = moduleKey(modulePages, name);
51 if (key) {
52 return modulePages[key];
53 }
54
55 // A module installed after this build: its pages are not in either glob, so
56 // it registers them itself when its script runs (see lib/moduleLoader.js).
57 return runtimePages[name] ?? null;
58}
