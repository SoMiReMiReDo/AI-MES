40 public function test_posting_a_material_issue_reduces_balances_and_writes_the_ledger(): void
41 {
42 $warehouse = $this->rawWarehouse();
43 $material = Material::factory()->create(['code' => 'FLOUR-01', 'stock_quantity' => 500]);
44
45 WarehouseStock::factory()->create([
46 'warehouse_id' => $warehouse->id,
47 'material_id' => $material->id,
48 'quantity' => 500,
49 ]);
50
51 $document = $this->service->createDraft([
52 'type' => StockDocument::TYPE_MATERIAL_ISSUE,
53 'warehouse_id' => $warehouse->id,
54 'lines' => [['material_id' => $material->id, 'quantity' => 120.5, 'unit_of_measure' => 'kg']],
55 ]);
56
57 $this->assertTrue($document->isDraft());
58 // A draft has not moved anything yet.
59 $this->assertEquals(500, (float) $material->fresh()->stock_quantity);
60
61 $this->service->post($document);
62
63 $this->assertTrue($document->fresh()->isPosted());
64 $this->assertEquals(379.5, (float) WarehouseStock::where('warehouse_id', $warehouse->id)
65 ->where('material_id', $material->id)
66 ->value('quantity'));
67 // The global material balance follows, with an audit trail.
68 $this->assertEquals(379.5, (float) $material->fresh()->stock_quantity);
69 $this->assertDatabaseHas('stock_movements', [
70 'material_id' => $material->id,
71 'warehouse_id' => $warehouse->id,
72 'movement_type' => StockMovement::TYPE_CONSUME,
73 'source_type' => StockMovement::SOURCE_STOCK_DOCUMENT,
74 'source_id' => $document->id,
75 ]);
76 }

106 WarehouseStock::factory()->create([

109 'material_lot_id' => $lot->id,

133 $this->assertSame(MaterialLot::STATUS_CONSUMED, $lot->fresh()->status);

136 public function test_cancelling_a_posted_document_reverses_every_effect(): void
137 {
138 $warehouse = $this->rawWarehouse();
139 $material = Material::factory()->create(['stock_quantity' => 500]);
140
141 WarehouseStock::factory()->create([
142 'warehouse_id' => $warehouse->id,
143 'material_id' => $material->id,
144 'quantity' => 500,
145 ]);
146
147 $document = $this->service->createDraft([
148 'type' => StockDocument::TYPE_MATERIAL_ISSUE,
149 'warehouse_id' => $warehouse->id,
150 'lines' => [['material_id' => $material->id, 'quantity' => 200]],
151 ]);
152
153 $this->service->post($document);
154 $this->service->cancel($document->fresh());
155
156 $this->assertSame(StockDocument::STATUS_CANCELLED, $document->fresh()->status);
157 $this->assertEquals(500, (float) WarehouseStock::where('warehouse_id', $warehouse->id)
158 ->where('material_id', $material->id)
159 ->value('quantity'));
160 $this->assertEquals(500, (float) $material->fresh()->stock_quantity);
161 // Reversal is booked, not erased: two movements, netting to zero.
162 $this->assertSame(2, StockMovement::where('source_id', $document->id)->count());
163 $this->assertEquals(0, (float) StockMovement::where('source_id', $document->id)->sum('quantity'));
164 }

296

300 // Plenty in the plant, 5 in this store.
