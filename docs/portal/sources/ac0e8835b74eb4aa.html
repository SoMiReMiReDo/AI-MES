35 private function workOrderWithRecipe(float $planned = 100, float $produced = 0): WorkOrder
36 {
37 $product = ProductType::factory()->create(['code' => 'BREAD-01', 'unit_of_measure' => 'pcs']);
38 $template = ProcessTemplate::factory()->create(['product_type_id' => $product->id, 'is_active' => true]);
39 $flour = Material::factory()->create(['code' => 'FLOUR-01', 'unit_of_measure' => 'kg']);
40
41 BomItem::create([
42 'process_template_id' => $template->id,
43 'material_id' => $flour->id,
44 'quantity_per_unit' => 0.5,
45 'scrap_percentage' => 10,
46 ]);
47
48 return WorkOrder::factory()->create([
49 'line_id' => Line::factory(),
50 'product_type_id' => $product->id,
51 'planned_qty' => $planned,
52 'produced_qty' => $produced,
53 'status' => WorkOrder::STATUS_IN_PROGRESS,
54 ]);
55 }

57 public function test_completion_generates_a_material_release_and_a_product_receipt(): void
58 {
59 Warehouse::factory()->rawMaterial()->isDefault()->create(['code' => 'RAW-1']);
60 Warehouse::factory()->finishedGoods()->isDefault()->create(['code' => 'FG-1']);
61
62 $workOrder = $this->workOrderWithRecipe(planned: 100, produced: 80);
63
64 $created = $this->service->generateForCompletion($workOrder);
65
66 $this->assertCount(2, $created);
67
68 $issue = StockDocument::where('work_order_id', $workOrder->id)
69 ->where('type', StockDocument::TYPE_MATERIAL_ISSUE)
70 ->with('lines')
71 ->first();
72
73 // 0.5 kg per unit + 10% scrap, times the 80 produced.
74 $this->assertEquals(44.0, (float) $issue->lines->first()->quantity);
75 $this->assertTrue($issue->isDraft());
76
77 $receipt = StockDocument::where('work_order_id', $workOrder->id)
78 ->where('type', StockDocument::TYPE_PRODUCT_RECEIPT)
79 ->with('lines')
80 ->first();
81
82 $this->assertEquals(80.0, (float) $receipt->lines->first()->quantity);
83 $this->assertSame($workOrder->product_type_id, $receipt->lines->first()->product_type_id);
84 // Drafts only: generation must never move stock on its own.
85 $this->assertTrue($receipt->isDraft());
86 }

88 public function test_generation_is_idempotent(): void
