38 private function keyWith(array $scopes): string

59 public function test_import_creates_work_orders_scoped_to_key_tenant(): void
60 {
61 $key = $this->keyWith([ApiScope::OrdersImport]);
62
63 $response = $this->withHeader('X-Api-Key', $key)->postJson('/api/v1/erp/work-orders/import', [
64 'orders' => [
65 $this->validOrder(['order_no' => 'ERP-1001', 'customer_order_no' => 'PO-77', 'unit_price' => 12.5]),
66 $this->validOrder(['order_no' => 'ERP-1002', 'planned_qty' => 40]),
67 ],
68 ]);
69
70 $response->assertOk();
71 $this->assertSame(2, $response->json('data.imported'));
72 $this->assertSame(0, $response->json('data.updated'));
73 $this->assertSame([], $response->json('data.errors'));
74
75 $this->assertDatabaseHas('work_orders', [
76 'order_no' => 'ERP-1001',
77 'customer_order_no' => 'PO-77',
78 'tenant_id' => $this->tenantId,
79 'status' => 'PENDING',
80 ]);
81 }

83 public function test_import_returns_207_with_structured_row_errors(): void
84 {
85 $key = $this->keyWith([ApiScope::OrdersImport]);
86
87 $response = $this->withHeader('X-Api-Key', $key)->postJson('/api/v1/erp/work-orders/import', [
88 'orders' => [
89 $this->validOrder(['order_no' => 'ERP-2001']),
90 $this->validOrder(['order_no' => 'ERP-2002', 'product_type_code' => 'NOPE']),
91 ],
92 ]);
93
94 $response->assertStatus(207);
95 $this->assertSame(1, $response->json('data.imported'));
96 $this->assertSame(2, $response->json('data.errors.0.row'));
97 $this->assertSame('product_type_code', $response->json('data.errors.0.field'));
98 $this->assertDatabaseMissing('work_orders', ['order_no' => 'ERP-2002']);
99 }
