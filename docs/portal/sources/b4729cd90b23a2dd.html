16/**
17 * The controlled change workflow (#182).
18 *
19 * A structural change to a running order — revision, quantity, line, BOM, schedule —
20 * goes through a request that is reviewed and approved before it touches anything.
21 * Applying it appends a new configuration version rather than editing the old one, so
22 * what the shop floor was building before the change stays exactly as recorded.
23 *
24 * Everything already executed is off limits by construction: this service writes to
25 * the work order and to not-yet-started steps, and never to completed batch steps,
26 * consumed material, quality results or produced quantities.
27 */
28class ChangeRequestService
29{
30 /**
31 * Statuses a change may be applied to.
32 *
33 * The stopped states are the point of the workflow. PENDING and ACCEPTED are here
34 * because a change can legitimately be raised on an order that has not started
35 * yet — there is no execution to protect, and without them such a request would
36 * be approved and then permanently un-appliable (nothing can move a PENDING order
37 * into a stopped state).
38 */
39 private const APPLICABLE_STATUSES = [
40 WorkOrder::STATUS_PAUSED,
41 WorkOrder::STATUS_BLOCKED,
42 WorkOrder::STATUS_CHANGE_HOLD,
43 WorkOrder::STATUS_PENDING,
44 WorkOrder::STATUS_ACCEPTED,
45 ];
46
47 public function __construct(
48 protected WorkOrderService $workOrders,
49 protected WorkOrderSnapshotService $snapshots,
50 ) {}

52 /**
53 * Raise a draft change request against an order.
54 *
55 * @param array<string, mixed> $data
56 *
57 * @throws \DomainException When the request proposes nothing, or the order is terminal.
58 */
59 public function create(WorkOrder $workOrder, array $data, User $user): WorkOrderChangeRequest
60 {
61 if (in_array($workOrder->status, WorkOrder::TERMINAL_STATUSES, true)) {
62 throw new \DomainException('Change requests cannot be raised against a finished work order.');
63 }
64
65 $proposed = $this->sanitizeProposed($data['proposed'] ?? []);
66
67 if ($proposed === []) {
68 throw new \DomainException('A change request must propose at least one change.');
69 }
70
71 // Two creates racing for the same sequence number: the unique index rejects
72 // the loser, and it simply takes the next one. Bounded, because a persistent
73 // failure is a real error and must not spin.
74 for ($attempt = 1; ; $attempt++) {
75 try {
76 return $this->insert($workOrder, $data, $user, $proposed);
77 } catch (UniqueConstraintViolationException $e) {
78 if ($attempt >= 3 || ! $this->isDuplicateCode($e)) {
79 throw $e;
80 }
81 }
82 }
83 }

115 /**
116 * Edit a draft. The impact is recomputed here because it is what the approver
117 * will read, and a stale one would describe a change nobody is proposing.
118 *
119 * @param array<string, mixed> $data
120 *
121 * @throws \DomainException When the request is no longer a draft.
122 */
123 public function update(WorkOrderChangeRequest $changeRequest, array $data, User $user): WorkOrderChangeRequest
124 {
125 $this->assertEditable($changeRequest);
126
127 $attributes = array_intersect_key($data, array_flip([
128 'title', 'reason', 'effective_from', 'effective_from_batch_id',
129 'produced_disposition', 'material_disposition',
130 ]));
131
132 if (array_key_exists('proposed', $data)) {
133 $proposed = $this->sanitizeProposed($data['proposed']);
134
135 if ($proposed === []) {
136 throw new \DomainException('A change request must propose at least one change.');
137 }
138
139 $attributes['proposed'] = $proposed;
140 $attributes['impact'] = $this->analyzeImpact($this->workOrderOf($changeRequest), $proposed);
141 }
142
143 $changeRequest->update($attributes);
144
145 return $changeRequest->fresh();
146 }

204 /**
205 * Apply an approved change: freeze the before-state, rebuild the configuration,
206 * append it as a new version and move the order onto it.
207 *
208 * The order must be stopped. Applying a structural change to an order that is
209 * actively running would hand the shop floor a new configuration mid-step, which
210 * is the failure mode this whole workflow exists to prevent.
211 *
212 * @param array<string, mixed> $data
213 *
214 * @throws \DomainException
215 */
216 public function apply(WorkOrderChangeRequest $changeRequest, User $user, array $data = []): WorkOrderChangeRequest
217 {
218 $this->assertTransition($changeRequest, ChangeRequestStatus::Applied);
219
220 return DB::transaction(function () use ($changeRequest, $user, $data) {
221 // Re-read the request under a row lock and check the transition again. The
222 // assertion above is outside the transaction, so two concurrent applies on
223 // the same APPROVED request would both pass it; the work-order lock below
224 // only serializes them, it does not stop the second one from appending a
225 // second snapshot version and regenerating the same steps again.
226 $locked = WorkOrderChangeRequest::whereKey($changeRequest->getKey())->lockForUpdate()->first();
227
228 if ($locked === null) {
229 throw new \DomainException('The change request no longer exists.');
230 }
231
232 $changeRequest->setRawAttributes($locked->getAttributes(), true);
233 $this->assertTransition($changeRequest, ChangeRequestStatus::Applied);
234
235 $workOrder = WorkOrder::whereKey($changeRequest->work_order_id)->lockForUpdate()->first();
236
237 // Null when the order was deleted while the request sat in review. Refuse
238 // as a domain rule, or the refresh() below would be a fatal error the
239 // controllers cannot turn into a 422.
240 if ($workOrder === null) {
241 throw new \DomainException('The work order this change belongs to no longer exists.');
242 }
243
244 $workOrder->refresh();
245
246 if (! in_array($workOrder->status, self::APPLICABLE_STATUSES, true)) {
247 throw new \DomainException(
248 'A change can only be applied to a stopped work order, or to one that has not started production.'
249 );
250 }
251
252 $changes = $changeRequest->proposedChanges();
253 // The effective point may be confirmed or overridden at apply time;
254 // without an override the request's own choice stands.
255 $chosen = $data['effective_from'] ?? $changeRequest->effective_from;
256 $effectiveFrom = $chosen instanceof ChangeEffectivePoint
257 ? $chosen
258 : ChangeEffectivePoint::from($chosen);
259
260 $this->assertEffectivePointAllowed($workOrder, $effectiveFrom);
261 $this->assertQuantityNotBelowProduced($workOrder, $changes);
262
263 // The before-state, captured now rather than reconstructed later: this is
264 // what makes the request readable as a diff for the rest of its life.
265 // Point the request at the row we hold the lock on, so the before-state is
266 // read from the same instance the change is about to be written to.
267 $changeRequest->setRelation('workOrder', $workOrder);
268 $previous = $this->capturePrevious($changeRequest, $changes);
269
270 $snapshot = $this->buildSnapshot($workOrder, $changes);
271
272 $this->applyToWorkOrder($workOrder, $changes);
273
274 $version = $this->snapshots->createVersion(
275 $workOrder,
276 $snapshot,
277 $effectiveFrom,
278 $changeRequest,
279 $user,
280 $data['effective_from_batch_id'] ?? $changeRequest->effective_from_batch_id,
281 );
282
283 // Not-yet-started work is the only execution data a change may touch.
284 $this->regenerateNotStartedSteps($workOrder, $effectiveFrom, $version->version);
285
286 $changeRequest->update([
287 'status' => ChangeRequestStatus::Applied,
288 'previous_values' => $previous,
289 'effective_from' => $effectiveFrom,
290 'implementation_notes' => $data['implementation_notes'] ?? $changeRequest->implementation_notes,
291 'applied_by_id' => $user->id,
292 'applied_at' => now(),
293 'resulting_snapshot_version' => $version->version,
294 // Freeze what the remaining production now needs, so "requirements
295 // recalculated" is an answer on the record rather than a recomputation
296 // that would give a different number tomorrow.
297 'impact' => array_merge($changeRequest->impact ?? [], [
298 'remaining_requirements' => $this->snapshots->remainingRequirements($workOrder, $snapshot),
299 ]),
300 ]);
301
302 return $changeRequest->fresh();
303 });
304 }

323 /**
324 * What applying this request would mean for work already done — the analysis the
325 * approver sees before deciding.
326 *
327 * @param array<string, mixed> $proposed
328 * @return array<string, mixed>
329 */
330 public function analyzeImpact(WorkOrder $workOrder, array $proposed): array
331 {
332 $batches = $workOrder->batches()->get();
333 $batchIds = $batches->pluck('id');
334
335 $stepCounts = BatchStep::whereIn('batch_id', $batchIds)
336 ->selectRaw('status, COUNT(*) as total')
337 ->groupBy('status')
338 ->pluck('total', 'status');
339
340 $consumed = (float) $workOrder->materialAllocations()->sum('consumed_qty');
341 $allocated = (float) $workOrder->materialAllocations()->sum('allocated_qty');
342
343 $produced = (float) $workOrder->produced_qty;
344 $planned = (float) $workOrder->planned_qty;
345
346 return [
347 'produced_qty' => $produced,
348 'planned_qty' => $planned,
349 'remaining_qty' => max(0.0, $planned - $produced),
350 'batches' => [
351 'completed' => $batches->where('status', Batch::STATUS_DONE)->count(),
352 'active' => $batches->where('status', Batch::STATUS_IN_PROGRESS)->count(),
353 'pending' => $batches->where('status', Batch::STATUS_PENDING)->count(),
354 ],
355 'steps' => [
356 'completed' => (int) ($stepCounts[BatchStep::STATUS_DONE] ?? 0),
357 'in_progress' => (int) ($stepCounts[BatchStep::STATUS_IN_PROGRESS] ?? 0),
358 // Everything still ahead — the work a change is actually allowed to alter.
359 'not_started' => (int) ($stepCounts[BatchStep::STATUS_PENDING] ?? 0)
360 + (int) ($stepCounts[BatchStep::STATUS_READY] ?? 0),
361 ],
362 'materials' => [
363 'allocated_qty' => $allocated,
364 'consumed_qty' => $consumed,
365 ],
366 'revision' => $this->revisionImpact($workOrder, $proposed),
367 'documents' => $this->documentImpact($workOrder, $proposed),
368 'warnings' => $this->warnings($workOrder, $proposed, $batches, $consumed),
369 ];
370 }

372 /**
373 * Conflicts between what is proposed and what has already happened. These do not
374 * block on their own — some are legitimate and the approver decides — except
375 * where apply() enforces them as hard rules.
376 *
377 * @param array<string, mixed> $proposed
378 * @param \Illuminate\Support\Collection<int, Batch> $batches
379 * @return array<int, string>
380 */
381 protected function warnings(WorkOrder $workOrder, array $proposed, $batches, float $consumed): array
382 {
383 $warnings = [];
384 $produced = (float) $workOrder->produced_qty;
385
386 if (array_key_exists('planned_qty', $proposed) && (float) $proposed['planned_qty'] < $produced) {
387 $warnings[] = __('The proposed quantity is below the quantity already produced (:produced).', [
388 'produced' => $produced,
389 ]);
390 }
391
392 if (array_key_exists('product_revision_id', $proposed) && $produced > 0) {
393 $warnings[] = __(':produced units were already produced under the current revision and keep it.', [
394 'produced' => $produced,
395 ]);
396 }
397
398 if ($batches->where('status', Batch::STATUS_IN_PROGRESS)->isNotEmpty()) {
399 $warnings[] = __('A batch is still in progress and will finish on its current configuration.');
400 }
401
402 if (array_key_exists('bom_template_ids', $proposed) && $consumed > 0) {
403 $warnings[] = __('Material has already been consumed; existing consumption is not recalculated.');
404 }
405
406 if (array_key_exists('line_id', $proposed) && $batches->where('status', Batch::STATUS_IN_PROGRESS)->isNotEmpty()) {
407 $warnings[] = __('Moving the line while a batch is running may strand its workstation assignments.');
408 }
409
410 return $warnings;
411 }

434 /**
435 * Which frozen engineering documents (#179) the new configuration would replace.
436 *
437 * @param array<string, mixed> $proposed
438 * @return array<int, array<string, mixed>>
439 */
440 protected function documentImpact(WorkOrder $workOrder, array $proposed): array
441 {
442 if (! array_key_exists('product_revision_id', $proposed) && ! array_key_exists('bom_template_ids', $proposed)) {
443 return [];
444 }
445
446 $current = collect($workOrder->process_snapshot['engineering_documents'] ?? [])->keyBy('document_id');
447 $next = collect($this->buildSnapshot($workOrder, $proposed)['engineering_documents'] ?? [])->keyBy('document_id');
448
449 return $current->keys()->merge($next->keys())->unique()
450 ->map(fn ($id) => [
451 'document_id' => $id,
452 'from_revision' => $current->get($id)['revision'] ?? null,
453 'to_revision' => $next->get($id)['revision'] ?? null,
454 ])
455 ->filter(fn (array $row) => $row['from_revision'] !== $row['to_revision'])
456 ->values()
457 ->all();
458 }

460 /**
461 * Rebuild the frozen configuration the way release does, so a changed order ends
462 * up with exactly the snapshot shape it would have had if it had been released
463 * this way in the first place.
464 *
465 * @param array<string, mixed> $changes
466 * @return array<string, mixed>
467 */
468 protected function buildSnapshot(WorkOrder $workOrder, array $changes): array
469 {
470 $revisionId = array_key_exists('product_revision_id', $changes)
471 ? $changes['product_revision_id']
472 : $workOrder->product_revision_id;
473
474 $templateIds = array_key_exists('bom_template_ids', $changes)
475 ? (array) $changes['bom_template_ids']
476 : $workOrder->bomTemplates()->pluck('process_templates.id')->all();
477
478 // Nothing structural changed — carry the current configuration forward so the
479 // new version still records the full picture (a snapshot is never a delta).
480 if (! array_key_exists('product_revision_id', $changes) && ! array_key_exists('bom_template_ids', $changes)) {
481 return $workOrder->process_snapshot ?? [];
482 }
483
484 $snapshot = $this->workOrders->rebuildProcessSnapshot(
485 $workOrder->product_type_id,
486 $templateIds,
487 $revisionId,
488 );
489
490 return $snapshot ?? ($workOrder->process_snapshot ?? []);
491 }

493 /**
494 * Write the approved fields onto the order. Only the fields a request may touch
495 * reach this point — sanitizeProposed() dropped the rest.
496 *
497 * @param array<string, mixed> $changes
498 */
499 protected function applyToWorkOrder(WorkOrder $workOrder, array $changes): void
500 {
501 $attributes = array_intersect_key($changes, array_flip([
502 'product_revision_id', 'planned_qty', 'line_id', 'due_date', 'description',
503 ]));
504
505 // The order has no production_notes column; notes live in extra_data so the
506 // field the issue asks for is still changeable under control.
507 if (array_key_exists('production_notes', $changes)) {
508 $attributes['extra_data'] = array_merge(
509 $workOrder->extra_data ?? [],
510 ['production_notes' => $changes['production_notes']],
511 );
512 }
513
514 if ($attributes !== []) {
515 $workOrder->update($attributes);
516 }
517
518 if (array_key_exists('bom_template_ids', $changes)) {
519 $this->workOrders->syncBomSelection($workOrder, (array) $changes['bom_template_ids']);
520 }
521
522 $workOrder->refresh();
523 }

525 /**
526 * Rebuild the steps of batches that have not started, from the new configuration.
527 *
528 * Batches with any executed step are left alone entirely: rewriting them is the
529 * one thing a change request may never do. IMMEDIATE is only reachable when no
530 * batch has started at all (see assertEffectivePointAllowed), so under every
531 * effective point this only ever touches untouched work.
532 */
533 protected function regenerateNotStartedSteps(WorkOrder $workOrder, ChangeEffectivePoint $effectiveFrom, int $version): void
534 {
535 if ($effectiveFrom === ChangeEffectivePoint::NextBatch) {
536 // Only batches created from now on carry the new configuration; the ones
537 // already queued keep the version they were generated from.
538 return;
539 }
540
541 $batches = $workOrder->batches()
542 ->where('status', Batch::STATUS_PENDING)
543 ->get()
544 ->filter(fn (Batch $batch) => ! $this->hasExecution($batch));
545
546 foreach ($batches as $batch) {
547 // Per-model delete, not a builder mass-delete: BatchStep soft-deletes with
548 // audit, and both the `deleted_by_id` stamp and the cascade to its
549 // documents run in the model's deleted event, which a bulk UPDATE skips.
550 foreach ($batch->steps as $step) {
551 $step->delete();
552 }
553
554 $this->workOrders->regenerateBatchSteps($batch, $workOrder->process_snapshot ?? []);
555 $batch->update(['snapshot_version' => $version]);
556 }
557 }

559 /**
560 * Whether anything on this batch has been executed, and is therefore untouchable.
561 *
562 * A step SKIPPED *by somebody* counts as executed: skipping records who, when and
563 * why, and choosing a variant skips its siblings the same way — operator decisions
564 * with their own audit trail. Such a batch can still sit at PENDING, so checking
565 * the batch status alone would let regeneration erase them.
566 *
567 * Variant siblings skipped at generation time are NOT execution: they have no
568 * `completed_by_id`, and treating them as executed would freeze every batch with a
569 * variant group against any change.
570 */
571 protected function hasExecution(Batch $batch): bool
572 {
573 return (float) $batch->produced_qty > 0
574 || $batch->steps()
575 ->where(fn ($q) => $q
576 ->whereIn('status', [BatchStep::STATUS_IN_PROGRESS, BatchStep::STATUS_DONE])
577 ->orWhere(fn ($qq) => $qq
578 ->where('status', BatchStep::STATUS_SKIPPED)
579 ->whereNotNull('completed_by_id')
580 )
581 )
582 ->exists();
583 }

585 /**
586 * IMMEDIATE rewrites the steps of open batches, so it is only allowed while
587 * nothing has been executed. Anything else would hand the shop floor a
588 * configuration that contradicts what it has already built.
589 *
590 * @throws \DomainException
591 */
592 protected function assertEffectivePointAllowed(WorkOrder $workOrder, ChangeEffectivePoint $effectiveFrom): void
593 {
594 if (! $effectiveFrom->touchesOpenBatches()) {
595 return;
596 }
597
598 $executed = $workOrder->batches()
599 ->whereIn('status', [Batch::STATUS_PENDING, Batch::STATUS_IN_PROGRESS])
600 ->get()
601 ->contains(fn (Batch $batch) => $this->hasExecution($batch));
602
603 if ($executed) {
604 throw new \DomainException(
605 'IMMEDIATE is not allowed: production has already been executed under the current configuration. '
606 .'Use NEXT_BATCH or REMAINING_QUANTITY.'
607 );
608 }
609 }

669 /**
670 * The order a request belongs to, refusing as a domain rule when it has been
671 * deleted — otherwise a null would reach analyzeImpact() as a TypeError, which the
672 * controllers cannot turn into a 422.
673 *
674 * @throws \DomainException
675 */

705 /**
706 * Next human-facing code for the order's tenant, as CR/{year}/{sequence}.
707 *
708 * The sequence is derived from the highest EXISTING NUMBER, not from the highest
709 * string: codes sort lexicographically, so `CR/2026/10000` sorts below
710 * `CR/2026/9999` and an ordered-by-code lookup would hand out 10000 forever once
711 * a plant passes four digits.
712 *
713 * A counter table is deliberately not used — it would serialise every change
714 * request in the plant behind one row. The unique index is the real guard, and a
715 * create that loses a concurrent race retries with the next number.
716 */
717 protected function generateCode(WorkOrder $workOrder): string
718 {
719 $year = now()->year;
720 $prefix = "CR/{$year}/";
721
722 $highest = WorkOrderChangeRequest::where('code', 'like', $prefix.'%')
723 ->pluck('code')
724 ->map(fn (string $code) => (int) substr($code, strlen($prefix)))
725 ->max() ?? 0;
726
727 // Four digits is the shape a plant expects; longer sequences simply grow, and
728 // stay correctly ordered because the number, not the string, is authoritative.
729 return $prefix.str_pad((string) ($highest + 1), 4, '0', STR_PAD_LEFT);
730 }
