7 /**
8 * Parse raw MQTT payload according to the declared format.
9 *
10 * Returns an associative array on success, or ['_raw' => $payload] as fallback.
11 */
12 public function parse(string $payload, string $format): array
13 {
14 return match ($format) {
15 'json' => $this->parseJson($payload),
16 'plain' => ['value' => $payload],
17 'csv' => $this->parseCsv($payload),
18 'hex' => ['hex' => $payload, 'bytes' => $this->hexToBytes($payload)],
19 default => ['_raw' => $payload],
20 };
21 }

23 /**
24 * Resolve a field path from parsed data.
25 *
26 * Supports:
27 * - JSONPath: $.field, $.nested.field, $.arr.0
28 * - Literal (no $): returned as-is
29 * - Empty / null: returns entire parsed data
30 */
31 public function resolvePath(?string $path, array $data): mixed
32 {
33 if (empty($path)) {
34 return $data;
35 }
36
37 if (!str_starts_with($path, '$')) {
38 return $path; // literal value
39 }
40
41 if ($path === '$') {
42 return $data;
43 }
44
45 $keys = explode('.', substr($path, 2)); // strip '$.'
46
47 $value = $data;
48 foreach ($keys as $key) {
49 if (is_array($value) && array_key_exists($key, $value)) {
50 $value = $value[$key];
51 } else {
52 return null;
53 }
54 }
55
56 return $value;
57 }
58
59 /**
60 * Evaluate a simple condition expression.
61 *
62 * Supported operators: ==, !=, >, >=, <, <=, contains
63 * Format: "<value> <operator> <literal>"
64 * Example: "5 > 0" or "active == true"
65 *
66 * The $resolvedValue is the already-resolved field value from payload.
67 */
68 public function evaluateCondition(?string $expression, mixed $resolvedValue): bool
69 {
70 if (empty($expression)) {
71 return true;
72 }
73
74 // Parse "value <op> literal"
75 $pattern = '/^value\s*(==|!=|>=|<=|>|<|contains)\s*(.+)$/';
76 if (!preg_match($pattern, trim($expression), $matches)) {
77 return true; // unparseable — allow
78 }
79
80 [, $operator, $literal] = $matches;
81 $literal = trim($literal);
82
83 // Coerce literal type
84 $literal = match (strtolower($literal)) {
85 'true' => true,
86 'false' => false,
87 'null' => null,
88 default => is_numeric($literal) ? ($literal + 0) : $literal,
89 };
90
91 return match ($operator) {
92 '==' => $resolvedValue == $literal,
93 '!=' => $resolvedValue != $literal,
94 '>' => is_numeric($resolvedValue) && $resolvedValue > $literal,
95 '>=' => is_numeric($resolvedValue) && $resolvedValue >= $literal,
96 '<' => is_numeric($resolvedValue) && $resolvedValue < $literal,
97 '<=' => is_numeric($resolvedValue) && $resolvedValue <= $literal,
98 'contains' => is_string($resolvedValue) && str_contains($resolvedValue, (string) $literal),
99 default => true,
100 };
101 }

105 private function parseJson(string $payload): array
106 {
107 $decoded = json_decode($payload, true);
108 if (json_last_error() !== JSON_ERROR_NONE || !is_array($decoded)) {
109 return ['_raw' => $payload, '_error' => 'Invalid JSON'];
110 }
111 return $decoded;
112 }

114 private function parseCsv(string $payload): array
115 {
116 $rows = [];
117 foreach (explode("\n", trim($payload)) as $line) {
118 if ($line !== '') {
119 $rows[] = str_getcsv($line);
120 }
121 }
122 return ['rows' => $rows];
123 }

125 private function hexToBytes(string $hex): array
126 {
127 $hex = preg_replace('/\s+/', '', $hex);
128 $bytes = [];
129 for ($i = 0; $i < strlen($hex); $i += 2) {
130 $bytes[] = hexdec(substr($hex, $i, 2));
131 }
132 return $bytes;
