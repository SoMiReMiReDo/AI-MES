5use App\Http\Controllers\Controller;

7use App\Services\CsvImport\WorkOrderImportService;
8use Illuminate\Http\JsonResponse;

10/**
11 * ERP → OpenMES: bulk import of work orders. Accepts a canonical JSON payload
12 * (see ImportWorkOrdersRequest); each order resolves its line and product type
13 * by code. Well-formed payloads always return a per-order report — a bad
14 * reference in one order does not fail the batch. Requires the
15 * `erp:orders:import` scope; rate limited via the `erp-import` limiter.
16 */
17class WorkOrderImportController extends Controller
18{
19 public function store(ImportWorkOrdersRequest $request, WorkOrderImportService $service): JsonResponse
20 {
21 $result = $service->importErp($request->input('orders'), $request->strategy());
22
23 // 207 Multi-Status when some rows failed but the batch was processed;
24 // 200 when every order was applied cleanly. Malformed payloads never
25 // reach here — the form request rejects them with 422.
26 $status = $result['errors'] === [] ? 200 : 207;
27
28 return response()->json(['data' => $result], $status);
29 }
30}
