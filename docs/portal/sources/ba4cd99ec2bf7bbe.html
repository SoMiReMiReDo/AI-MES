95 /**
96 * Check if all steps are complete. Every step must be DONE or SKIPPED, and
97 * each variant group must have one executed (DONE) step — a fully-skipped
98 * group means no variant was chosen, so the batch isn't finished.
99 */
100 public function allStepsComplete(): bool
101 {
102 $pending = $this->steps()
103 ->whereNotIn('status', [BatchStep::STATUS_DONE, BatchStep::STATUS_SKIPPED])
104 ->exists();
105
106 if ($pending) {
107 return false;
108 }
109
110 $variantSteps = $this->steps()->whereNotNull('variant_group')->get(['variant_group', 'status']);
111
112 foreach ($variantSteps->groupBy('variant_group') as $rows) {
113 if (! $rows->contains(fn ($s) => $s->status === BatchStep::STATUS_DONE)) {
114 return false; // a variant group with nothing executed
115 }
116 }
117
118 return true;
119 }

157 /**
158 * Promote every PENDING step whose sequence prerequisites are now met to
159 * READY ("next in line"). Idempotent — safe to call after any step
160 * transition or right after a batch's steps are created.
161 */
162 public function promoteReadySteps(): void
163 {
164 // Evaluate every step against one snapshot of the batch's steps instead
165 // of letting each prerequisite check lazy-load the batch and re-query its
166 // steps. Promoting to READY never changes another step's prerequisites
167 // (they depend on DONE/SKIPPED and passed quantities), so the snapshot
168 // stays valid for the whole loop. A relation-free copy of the batch keeps
169 // the snapshot off this instance, which callers keep using afterwards.
170 $steps = $this->steps()->get();
171 $batch = $this->withoutRelations()->setRelation('steps', $steps);
172
173 $steps->where('status', BatchStep::STATUS_PENDING)
174 ->each(function (BatchStep $step) use ($batch) {
175 $step->setRelation('batch', $batch);
176 if ($step->prerequisitesMet()) {
177 $step->update(['status' => BatchStep::STATUS_READY]);
178 }
179 });
180 }

191 /**
192 * Get the current (in progress or next ready/pending) step.
193 *
194 * Uses the loaded `steps` relation when present, so a list checking many
195 * batches (the operator queue / workstation filter) doesn't query per batch.
196 */
197 public function currentStep()
198 {
199 if ($this->relationLoaded('steps')) {
200 // Same answer as the queries below. The relation is ordered by
201 // step_number, and that ordering is applied before the READY-first
202 // CASE, so step_number decides between a READY and a PENDING step.
203 $steps = $this->steps->sortBy('step_number');
204
205 return $steps->firstWhere('status', BatchStep::STATUS_IN_PROGRESS)
206 ?? $steps->first(fn ($step) => in_array($step->status, [BatchStep::STATUS_READY, BatchStep::STATUS_PENDING], true));
207 }
208
209 // First check for in-progress step
210 $inProgress = $this->steps()
211 ->where('status', BatchStep::STATUS_IN_PROGRESS)
212 ->first();
213
214 if ($inProgress) {
215 return $inProgress;
216 }
217
218 // Otherwise the next actionable step — READY first, then PENDING.
219 return $this->steps()
220 ->whereIn('status', [BatchStep::STATUS_READY, BatchStep::STATUS_PENDING])
221 ->orderByRaw("CASE status WHEN '".BatchStep::STATUS_READY."' THEN 0 ELSE 1 END")
222 ->orderBy('step_number')
223 ->first();
224 }
