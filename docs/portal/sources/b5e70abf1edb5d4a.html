1import { test, expect, type Page } from '@playwright/test';
2import { execFileSync } from 'node:child_process';
3
4/**
5 * The admin work-order list is a synced collection: a create or delete must show
6 * up without a browser refresh — in the tab that made it (whose drawer submit
7 * redirects back to the same list, remounting it) and in any other open tab.
8 *
9 * Regression: the remount built a new collection while the old one was still
10 * shutting down, and the old one's echo.leave() unsubscribed the channel both
11 * shared, so the list went deaf until a manual refresh.
12 */
13
14const backend = `${process.env.OPENMES_NAME_PREFIX || 'openmmes'}-backend`;
15if (process.env.OPENMES_BROWSER_STORAGE_STATE) {
16 test.use({ storageState: process.env.OPENMES_BROWSER_STORAGE_STATE });
17}
18
19// Digits sort before letters, so the order lands at the top of the list (order_no asc).
20const orderNo = `000-E2E-LIVE-${Date.now()}`;
21
22async function login(page: Page) {
23 await page.goto('/admin/work-orders');
24 if (page.url().includes('/login')) {
25 await page.locator('input[name=username]').fill(process.env.ADMIN_USERNAME || 'admin');
26 await page.locator('input[name=password]').fill(process.env.ADMIN_PASSWORD || 'Admin1234!');
27 await page.locator('button[type=submit]').click();
28 await page.waitForURL((url) => !url.pathname.startsWith('/login'));
29 }
30 await page.goto('/locale/en');
31}
32
33/** Open the list and mark the document, so a full reload would be detectable. */
34async function openList(page: Page) {
35 await page.goto('/admin/work-orders');
36 await expect(page.getByRole('button', { name: 'New Work Order', exact: true })).toBeVisible();
37 await expect(page.getByRole('row').nth(2)).toBeVisible();
38 // Let the snapshot land and the channel subscription settle.
39 await page.waitForLoadState('networkidle');
40 await page.evaluate(() => { (window as any).__noReload = true; });
41}
42
43const rowFor = (page: Page) => page.getByRole('row').filter({ hasText: orderNo });
44
45const dbOrderId = () => execFileSync('docker', [
46 'exec', backend, 'php', 'artisan', 'tinker', '--execute',
47 `echo App\\Models\\WorkOrder::where('order_no', '${orderNo}')->value('id');`,
48], { encoding: 'utf8' }).trim();
49
50/** A server-side change the lists can only learn about through a live delta. */
51const setPlannedQty = (qty: number) => execFileSync('docker', [
52 'exec', backend, 'php', 'artisan', 'tinker', '--execute',
53 `App\\Models\\WorkOrder::where('order_no', '${orderNo}')->firstOrFail()->update(['planned_qty' => ${qty}]);`,
54]);
55
56test.afterAll(() => {
57 // The UI delete is a soft delete; remove the test row for good so repeated runs
58 // leave nothing in the dev database or the Trash page.
59 execFileSync('docker', [
60 'exec', backend, 'php', 'artisan', 'tinker', '--execute',
61 `App\\Models\\WorkOrder::withTrashed()->where('order_no', '${orderNo}')->get()->each->forceDelete();`,
62 ]);
63});
64
65test('a created and deleted work order appears and disappears live in every open list', async ({ page, context }) => {
66 await login(page);
67 const observer = await context.newPage();
68
69 await openList(page);
70 await openList(observer);
71 await expect(rowFor(page)).toHaveCount(0);
72
73 // Create through the list's drawer — its submit redirects back to this list.
74 await page.getByRole('button', { name: 'New Work Order', exact: true }).click();
75 await page.locator('input[name=order_no]').fill(orderNo);
76 await page.locator('input[name=planned_qty]').fill('3');
77 await page.getByRole('button', { name: 'Create', exact: true }).click();
78
79 await expect(rowFor(page)).toHaveCount(1, { timeout: 10_000 });
80 await expect(rowFor(observer)).toHaveCount(1, { timeout: 10_000 });
81 expect(dbOrderId(), 'the row is committed, not just broadcast').toMatch(/^\d+$/);
82
83 // Every step below remounts a list (a redirect back to it, or a nav click to the
84 // page you're on). The old collection shuts down ~1s later and used to
85 // unsubscribe the channel the new one shares, so wait past that and then change
86 // the order on the server: only a live delta can bring the new value in, since
87 // no list reloads its snapshot on its own.
88 const settle = () => page.waitForTimeout(2500);
89
90 // The creating tab remounted on the create's redirect.
91 await settle();
92 setPlannedQty(7);
93 await expect(rowFor(page)).toContainText('/ 7', { timeout: 10_000 });
94 await expect(rowFor(observer)).toContainText('/ 7', { timeout: 10_000 });
95
96 // The observer re-opens the page it is on from the sidebar.
97 await observer.locator('a[href="/admin/work-orders"]').first().click();
98 await expect(rowFor(observer)).toContainText('/ 7');
99 await settle();
100 setPlannedQty(9);
101 await expect(rowFor(observer)).toContainText('/ 9', { timeout: 10_000 });
102 await expect(rowFor(page)).toContainText('/ 9', { timeout: 10_000 });
103
104 // Delete from the creating tab (router.delete redirects back — another remount).
105 const row = rowFor(page);
106 await row.getByRole('button', { name: 'More actions' }).click();
107 await page.getByRole('menuitem', { name: /Delete/ }).click();
108 await page.getByRole('button', { name: 'Delete order', exact: true }).click();
109
110 await expect(rowFor(page)).toHaveCount(0, { timeout: 10_000 });
111 // The observer only hears about it live.
112 await expect(rowFor(observer)).toHaveCount(0, { timeout: 10_000 });
113 expect(dbOrderId(), 'soft-deleted').toBe('');
114
115 // Neither tab was reloaded to get there.
116 expect(await page.evaluate(() => (window as any).__noReload)).toBe(true);
117 expect(await observer.evaluate(() => (window as any).__noReload)).toBe(true);
118});
