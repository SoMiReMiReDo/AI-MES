7/**
8 * Optional feature "modules" (#144) — whole admin areas an installation can
9 * switch off to stay lean. Each module key maps 1:1 to a TabRegistry tab; a
10 * disabled module hides its tab from navigation AND 404s its routes
11 * (TabAccessMiddleware). Core areas (dashboard, orders, production, admin, …)
12 * are not listed here and are always on.
13 *
14 * The enabled set is persisted system-wide in `system_settings.enabled_modules`
15 * (a JSON array of keys). When the key is absent every module is treated as
16 * enabled, so existing installations and fresh databases are unaffected.
17 *
18 * This is distinct from the per-role tab-access matrix (who may see a tab) and
19 * from the deprecated plugin ModuleManager (`modules/` packages).
20 */
21class ModuleRegistry
22{
23 public const SETTING_KEY = 'enabled_modules';

30 * the cost/scrap analytics. Core essentials (Product Types, Lines, LOT
31 * Sequences, the Planner, Orders, Admin) are never listed here — always on.
32 */

92 /**
93 * Onboarding presets — one-click starting points for the module set the
94 * admin chooses at first login. `custom` is not listed: it means "use the
95 * admin's own checkbox selection". Keys must be a subset of OPTIONAL.
96 */
97 public const PRESETS = [
98 // Minimal: core production tracking + the work-order history report only.
99 'light' => ['reports'],
100 // Shop-floor operations: reporting + product/material data + quality +
101 // machine connectivity + packaging. Leaves multi-site structure, HR and
102 // webhooks off (an admin adds those via Custom or later in Settings).
103 'advanced' => [
104 'reports', 'advanced_reports', 'materials', 'product_engineering',
105 'companies', 'quality', 'maintenance', 'connectivity', 'packaging',
106 ],

109 /** @return array<int, string> */
110 /**
111 * Every optional module, including any an installed module registered.
112 *
113 * The constant stays the definition — migrations, seeders and the access
114 * matrix all read the same shape. This is the one seam through which an
115 * installed module can add to it; with no modules the array is returned
116 * untouched.
117 *
118 * @return array<string, mixed>
119 */
120 public static function all(): array
121 {
122 return app(\App\Extension\FilterRegistry::class)->filter('modules.optional', self::OPTIONAL);
123 }
124
125 public static function optionalKeys(): array
126 {
127 return array_keys(self::all());
128 }
129
130 /**
131 * Resolve a preset name to its enabled-module set. `custom` (or any unknown
132 * name) returns the explicit selection passed in, filtered to valid keys.
133 *
134 * @param array<int, string> $customSelection
135 * @return array<int, string>
136 */
137 public static function modulesForPreset(string $preset, array $customSelection = []): array
138 {
139 if ($preset === 'custom') {
140 return array_values(array_intersect($customSelection, self::optionalKeys()));
141 }
142
143 return self::PRESETS[$preset] ?? self::optionalKeys();
144 }
145
146 /**
147 * Enabled optional-module keys. A missing/invalid setting means "all
148 * enabled" (back-compat); during install the DB may not exist yet.
149 *
150 * @return array<int, string>
151 */
152 public static function enabled(): array
153 {
154 try {
155 $row = DB::table('system_settings')->where('key', self::SETTING_KEY)->first();
156 } catch (\Throwable) {
157 return self::optionalKeys();
158 }
159
160 if (! $row) {
161 return self::optionalKeys();
162 }
163
164 $vals = json_decode($row->value, true);
165 if (! is_array($vals)) {
166 return self::optionalKeys();
167 }
168
169 return array_values(array_intersect($vals, self::optionalKeys()));
170 }
171
172 /** Core (non-optional) areas are always enabled. */
173 public static function isModuleEnabled(string $key): bool
174 {
175 if (! array_key_exists($key, self::all())) {
176 return true;
177 }
178
179 return in_array($key, self::enabled(), true);
180 }
181
182 /** A TabRegistry tab is enabled unless its (optional) module is switched off. */
183 public static function isTabEnabled(string $tab): bool
184 {
185 return self::isModuleEnabled($tab);
186 }
187
188 /**
189 * Persist the enabled set (invalid/core keys are ignored).
190 *
191 * @param array<int, string> $enabledKeys
192 */
193 public static function save(array $enabledKeys): void
194 {
195 $clean = array_values(array_intersect($enabledKeys, self::optionalKeys()));
196
197 DB::table('system_settings')->updateOrInsert(
198 ['key' => self::SETTING_KEY],
199 ['value' => json_encode($clean), 'updated_at' => now()],
200 );
201 }
202
203 /**
204 * Shape for a selection form: [{key, label, description, enabled}].
205 *
206 * @return array<int, array{key: string, label: string, description: string, enabled: bool}>
207 */
208 public static function forForm(): array
209 {
210 $enabled = self::enabled();
211
212 return array_map(fn (string $k) => [
213 'key' => $k,
214 'label' => self::all()[$k]['label'],
215 'description' => self::all()[$k]['description'],
216 'enabled' => in_array($k, $enabled, true),
217 ], self::optionalKeys());
218 }
