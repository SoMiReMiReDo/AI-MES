9/**
10 * Central write-path for Reverb sync: maps each synced collection to its model
11 * and (for filtered collections) a membership test, then registers model
12 * create/update/delete listeners that broadcast a CollectionChanged delta.
13 *
14 * One place to see "model → collection(s)", instead of a trait on every model.
15 * Booted from AppServiceProvider::boot(). Mirrors the read-side definitions in
16 * ShapeRegistry (table/columns/where).
17 *
18 * Op semantics: "upsert" (client inserts or updates by key) or "delete". For a
19 * filtered collection, a row that no longer matches its filter is broadcast as a
20 * delete (it left the set) — e.g. a work order going terminal leaves
21 * work_orders_active.
22 */
23class CollectionBroadcaster
24{

112 /**
113 * Manually broadcast a model's collection delta. Use after writes that
114 * bypass Eloquent events — increment()/decrement() and query-builder mass
115 * updates — so the change still reaches clients.
116 */
117 public static function flush($model): void
118 {
119 $class = get_class($model);
120 $fresh = $model->fresh() ?? $model;
121 $row = $fresh->attributesToArray();
122 $tenant = $fresh->getAttribute('tenant_id');
123
124 foreach (self::map() as $name => [$mclass, $member]) {
125 if ($mclass !== $class) {
126 continue;
127 }
128 $op = ($member !== null && ! $member($fresh)) ? 'delete' : 'upsert';
129 self::safeBroadcast($name, $op, $row, $tenant);
130 }
131 }

133 /** Column allowlist per collection, resolved once per request. */
134 private static array $columnCache = [];
135
136 /**
137 * Narrow a row to the columns its collection actually exposes.
138 *
139 * The Shape's column list is an allowlist — "never include password hashes,
140 * tokens, PII" — and it withholds real secrets: a webhook's custom auth
141 * headers, a delivery's request/response bodies. The snapshot endpoint has
142 * always honoured it, but the delta broadcast sent whatever
143 * attributesToArray() returned, so a model attribute outside the list
144 * reached every subscriber anyway. Project here so both paths expose the
145 * same fields, which is what Shape's own docblock promises.
146 */
147 private static function project(string $name, array $row): array
148 {
149 if (! array_key_exists($name, self::$columnCache)) {
150 self::$columnCache[$name] = app(ShapeRegistry::class)->find($name)?->columns();
151 }
152
153 $columns = self::$columnCache[$name];
154
155 // No shape (a collection broadcast but not registered for reading) —
156 // send nothing rather than guess at what is safe to expose.
157 return $columns === null ? [] : array_intersect_key($row, array_flip($columns));
158 }

160 /** Depth of nested muted() calls; > 0 suppresses every delta. */
161 private static int $muted = 0;
162
163 /**
164 * Run $callback with delta broadcasting suppressed.
165 *
166 * For writes that are going to be rolled back — the importer's dry run.
167 * The rows briefly exist inside the transaction and fire the usual model
168 * events, but broadcasting them would push rows to every open browser that
169 * are about to stop existing, and no later delta would take them back
170 * (TanStack DB keeps what it was told until something contradicts it).
171 *
172 * Nests safely, and restores the counter even when $callback throws —
173 * which the dry run relies on, since a rollback is signalled by throwing.
174 */
175 public static function muted(callable $callback)
176 {
177 self::$muted++;
178
179 try {
180 return $callback();
181 } finally {
182 self::$muted--;
183 }
184 }
185
186 /**
187 * Dispatch a CollectionChanged delta without ever letting a broadcast
188 * failure break the originating write. Live sync is best-effort: if the
189 * broadcaster (e.g. Reverb) is unreachable, the write must still succeed
190 * and clients fall back to polling (useSyncedShape). Failures are logged.
191 *
192 * Sent only once the surrounding transaction commits (immediately when there
193 * is none). Model events fire mid-transaction, so a write that later rolls
194 * back — e.g. a work order whose component generation fails validation —
195 * would otherwise push a row that never existed into every open list, and
196 * nothing would ever take it back out.
197 */
198 private static function safeBroadcast(string $name, string $op, array $row, $tenant): void
199 {
200 if (self::$muted > 0) {
201 return;
202 }
203
204 $event = new CollectionChanged($name, $op, self::project($name, $row), $tenant);
205
206 DB::afterCommit(function () use ($event) {
207 try {
208 event($event);
209 } catch (\Throwable $e) {
210 report($e);
211 }
212 });
213 }
214
215 public static function boot(): void
216 {
217 // Group collections by model so each model registers its events once.
218 $byModel = [];
219 foreach (self::map() as $name => [$model, $member]) {
220 if (class_exists($model)) {
221 $byModel[$model][$name] = $member;
222 }
223 }
224
225 foreach ($byModel as $model => $collections) {
226 // `deleted` also fires on soft delete, broadcasting the row's removal.
227 // `restored` (SoftDeletes models only) re-broadcasts it as an upsert.
228 $events = ['created', 'updated', 'deleted'];
229 if (method_exists($model, 'restored')) {
230 $events[] = 'restored';
231 }
232
233 foreach ($events as $event) {
234 $model::{$event}(function ($m) use ($collections, $event) {
235 $row = $m->attributesToArray();
236 $tenant = $m->getAttribute('tenant_id');
237
238 foreach ($collections as $name => $member) {
239 $op = ($event === 'deleted' || ($member !== null && ! $member($m))) ? 'delete' : 'upsert';
240 self::safeBroadcast($name, $op, $row, $tenant);
241 }
242 });
243 }
244 }
