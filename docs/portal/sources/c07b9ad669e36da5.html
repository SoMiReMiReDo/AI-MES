22 /**
23 * Import work orders from parsed CSV data.
24 *
25 * @param array $mappedData Parsed and mapped CSV data
26 * @param string $strategy Import strategy (update_or_create, skip_existing, error_on_duplicate)
27 * @return array Import results
28 */
29 public function import(array $mappedData, string $strategy, ?int $targetLineId = null): array
30 {
31 $successful = 0;
32 $updated = 0;
33 $failed = 0;
34 $skipped = 0;
35 $errorLog = [];
36
37 foreach ($mappedData as $row) {
38 try {
39 $result = $this->importRow($row, $strategy, $targetLineId);
40
41 if ($result['status'] === 'success') {
42 $successful++;
43
44 // importRow() has always said which it was; only the caller
45 // never asked, so a re-import of existing orders reported
46 // itself as N creations.
47 if (($result['action'] ?? null) === 'updated') {
48 $updated++;
49 }
50 } elseif ($result['status'] === 'skipped') {
51 $skipped++;
52 } else {
53 $failed++;
54 $errorLog[] = [
55 'row' => $row['row_number'],
56 'error' => $result['error'],
57 ];
58 }
59 } catch (\Exception $e) {
60 $failed++;
61 $errorLog[] = [
62 'row' => $row['row_number'],
63 'error' => $e->getMessage(),
64 ];
65
66 Log::error('CSV import row failed', [
67 'row' => $row['row_number'],
68 'error' => $e->getMessage(),
69 ]);
70 }
71 }
72
73 return [
74 // `successful` stays created + updated for existing callers.
75 'successful' => $successful,
76 'updated' => $updated,
77 'failed' => $failed,
78 'skipped' => $skipped,
79 'error_log' => $errorLog,
80 ];
81 }

83 /**
84 * Bulk-import work orders from an ERP payload (already-validated array of
85 * canonical rows, not CSV). Reuses the same per-row logic as the CSV import
86 * but returns an ERP-shaped result that distinguishes created vs updated and
87 * carries structured per-row errors ({row, field, message}).
88 *
89 * @param array<int, array<string, mixed>> $rows
90 * @return array{imported: int, updated: int, skipped: int, errors: array<int, array{row: int, field: string|null, message: string}>}
91 */
92 public function importErp(array $rows, string $strategy): array
93 {
94 $imported = 0;
95 $updated = 0;
96 $skipped = 0;
97 $errors = [];
98
99 foreach ($rows as $index => $row) {
100 $rowNumber = $index + 1;
101 $row['row_number'] = $rowNumber;
102
103 try {
104 $result = $this->importRow($row, $strategy);
105
106 if ($result['status'] === 'success') {
107 ($result['action'] ?? null) === 'updated' ? $updated++ : $imported++;
108 } elseif ($result['status'] === 'skipped') {
109 $skipped++;
110 } else {
111 $errors[] = [
112 'row' => $rowNumber,
113 'field' => $result['field'] ?? null,
114 'message' => $result['error'],
115 ];
116 }
117 } catch (\Throwable $e) {
118 $errors[] = ['row' => $rowNumber, 'field' => null, 'message' => $e->getMessage()];
119
120 Log::error('ERP work order import row failed', [
121 'row' => $rowNumber,
122 'error' => $e->getMessage(),
123 ]);
124 }
125 }
126
127 return [
128 'imported' => $imported,
129 'updated' => $updated,
130 'skipped' => $skipped,
131 'errors' => $errors,
132 ];
133 }

135 /**
136 * Import work orders from the unified file importer (Admin → Import).
137 *
138 * Looser than the ERP contract on purpose — this is the path a planner uses
139 * with a spreadsheet, not an ERP: only the order number and quantity are
140 * required, the line and product type are optional lookups, unknown columns
141 * can be kept as `custom:<key>` in extra_data, and a whole file can be pinned
142 * to one line and one planning period from the form.
143 *
144 * @param list<array<string, mixed>> $rows canonical rows from RowMapper
145 * @param array{strategy?: string, target_line_id?: int|string|null, import_week?: int|null, import_month?: int|null, production_year?: int|null} $options
146 * @return array{imported: int, updated: int, skipped: int, errors: array<int, array{row: int, field: string|null, message: string}>}
147 */
148 public function importFromFile(array $rows, array $options = []): array
149 {
150 $strategy = $options['strategy'] ?? 'update_or_create';
151 $targetLineId = ! empty($options['target_line_id']) ? (int) $options['target_line_id'] : null;
152 $targetLine = $targetLineId ? Line::find($targetLineId) : null;
153
154 $period = array_filter([
155 'week_number' => ! empty($options['import_week']) ? (int) $options['import_week'] : null,
156 'month_number' => ! empty($options['import_month']) ? (int) $options['import_month'] : null,
157 'production_year' => ! empty($options['production_year']) ? (int) $options['production_year'] : null,
158 ]);
159
160 $lines = Line::pluck('id', 'code');
161 $productTypes = ProductType::get(['id', 'code'])->keyBy('code');
162 $templates = []; // product type id => active template|null, resolved once
163
164 return $this->processRows($rows, function (array $row) use ($strategy, $targetLineId, $targetLine, $period, $lines, $productTypes, &$templates) {
165 $orderNo = trim((string) ($row['order_no'] ?? ''));
166
167 if ($orderNo === '') {
168 return $this->error('order_no', __('Order number is required'));
169 }
170
171 $qty = (float) ($row['quantity'] ?? $row['planned_qty'] ?? 0);
172
173 if ($qty <= 0) {
174 return $this->error('quantity', __('Planned quantity must be greater than 0'));
175 }
176
177 $data = ['planned_qty' => $qty];
178
179 if ($targetLineId !== null) {
180 if (! $targetLine) {
181 return $this->error('line_code', __('Target line #:id not found', ['id' => $targetLineId]));
182 }
183 $data['line_id'] = $targetLine->id;
184 } elseif (! empty($row['line_code'])) {
185 $lineId = $lines[$row['line_code']] ?? null;
186
187 if (! $lineId) {
188 return $this->error('line_code', __("Line ':code' not found", ['code' => $row['line_code']]));
189 }
190 $data['line_id'] = $lineId;
191 }
192
193 $productType = null;
194
195 if (! empty($row['product_type_code'])) {
196 $productType = $productTypes[$row['product_type_code']] ?? null;
197
198 if (! $productType) {
199 return $this->error('product_type_code', __("Product type ':code' not found", ['code' => $row['product_type_code']]));
200 }
201 $data['product_type_id'] = $productType->id;
202 }
203
204 foreach (['priority' => 'int', 'due_date' => null, 'description' => null, 'customer_order_no' => null, 'unit_price' => 'float'] as $field => $cast) {
205 if (! array_key_exists($field, $row) || $row[$field] === null || $row[$field] === '') {
206 continue;
207 }
208 $data[$field] = match ($cast) {
209 'int' => (int) $row[$field],
210 'float' => (float) $row[$field],
211 default => $row[$field],
212 };
213 }
214
215 $extra = $row['custom'] ?? [];
216
217 if (! empty($row['product_name'])) {
218 $extra['product_name'] = $row['product_name'];
219 }
220
221 $data = array_merge($data, $period);
222
223 $existing = WorkOrder::where('order_no', $orderNo)->first();
224
225 if ($existing) {
226 if ($strategy === 'skip_existing') {
227 return $this->skipped();
228 }
229
230 if ($strategy === 'error_on_duplicate') {
231 return $this->error('order_no', __('Duplicate order number: :order', ['order' => $orderNo]));
232 }
233
234 if (in_array($existing->status, [WorkOrder::STATUS_DONE, WorkOrder::STATUS_CANCELLED], true)) {
235 return $this->skipped();
236 }
237
238 if ($extra !== []) {
239 $data['extra_data'] = array_merge($existing->extra_data ?? [], $extra);
240 }
241
242 $existing->update($data);
243
244 return $this->updated();
245 }
246
247 $data['order_no'] = $orderNo;
248 $data['status'] = WorkOrder::STATUS_PENDING;
249 $data['produced_qty'] = 0;
250
251 if ($extra !== []) {
252 $data['extra_data'] = $extra;
253 }
254
255 // A product type with an active template gets its process frozen onto
256 // the order, as the API path does; without one the order is still
257 // created — a supervisor can attach the process when accepting it.
258 $template = null;
259
260 if ($productType) {
261 $templates[$productType->id] ??= $productType->processTemplates()->where('is_active', true)->first();
262 $template = $templates[$productType->id];
263 }
264
265 if ($template) {
266 $data['process_snapshot'] = $this->snapshotService->createSnapshot($template);
267 }
268
269 WorkOrder::create($data);
270
271 return $this->created();
272 });
273 }

275 /**
276 * Import a single row.
277 */
278 protected function importRow(array $row, string $strategy, ?int $targetLineId = null): array
279 {
280 // Validate required fields
281 if (empty($row['order_no'])) {
282 return ['status' => 'error', 'field' => 'order_no', 'error' => 'Order number is required'];
283 }
284
285 // Find the line: an explicit "assign all rows to this line" override wins
286 // over the per-row line_code column (mirrors the web importer).
287 if ($targetLineId !== null) {
288 $line = Line::find($targetLineId);
289 if (! $line) {
290 return ['status' => 'error', 'field' => 'line_code', 'error' => "Target line #{$targetLineId} not found"];
291 }
292 } else {
293 $line = Line::where('code', $row['line_code'] ?? null)->first();
294 if (! $line) {
295 return ['status' => 'error', 'field' => 'line_code', 'error' => "Line '".($row['line_code'] ?? '')."' not found"];
296 }
297 }
298
299 // Find product type by code
300 $productType = ProductType::where('code', $row['product_type_code'] ?? null)->first();
301 if (! $productType) {
302 return ['status' => 'error', 'field' => 'product_type_code', 'error' => "Product type '".($row['product_type_code'] ?? '')."' not found"];
303 }
304
305 // Validate planned quantity
306 if (empty($row['planned_qty']) || $row['planned_qty'] <= 0) {
307 return ['status' => 'error', 'field' => 'planned_qty', 'error' => 'Planned quantity must be greater than 0'];
308 }
309
310 // Check if work order exists
311 $existing = WorkOrder::where('order_no', $row['order_no'])->first();
312
313 if ($existing) {
314 return $this->handleExisting($existing, $row, $strategy, $line, $productType);
315 } else {
316 return $this->createNew($row, $line, $productType);
317 }
318 }

320 /**
321 * Handle existing work order based on strategy.
322 */
323 protected function handleExisting(
324 WorkOrder $existing,
325 array $row,
326 string $strategy,
327 Line $line,
328 ProductType $productType
329 ): array {
330 switch ($strategy) {
331 case 'update_or_create':
332 return $this->updateExisting($existing, $row, $line, $productType);
333
334 case 'skip_existing':
335 return ['status' => 'skipped', 'message' => 'Work order already exists'];
336
337 case 'error_on_duplicate':
338 return ['status' => 'error', 'field' => 'order_no', 'error' => "Duplicate order number: {$row['order_no']}"];
339
340 default:
341 return ['status' => 'error', 'error' => 'Invalid import strategy'];
342 }
343 }

345 /**
346 * Update existing work order.
347 */
348 protected function updateExisting(
349 WorkOrder $existing,
350 array $row,
351 Line $line,
352 ProductType $productType
353 ): array {
354 // Don't update if work order is already done or cancelled
355 if (in_array($existing->status, ['DONE', 'CANCELLED'])) {
356 return ['status' => 'skipped', 'message' => 'Work order already completed/cancelled'];
357 }
358
359 DB::transaction(function () use ($existing, $row, $line, $productType) {
360 $existing->update(array_merge([
361 'line_id' => $line->id,
362 'product_type_id' => $productType->id,
363 'planned_qty' => $row['planned_qty'],
364 'priority' => $row['priority'] ?? 0,
365 'due_date' => $row['due_date'] ?? null,
366 'description' => $row['description'] ?? null,
367 ], $this->optionalErpFields($row)));
368
369 Log::info('Work order updated via CSV import', [
370 'order_no' => $existing->order_no,
371 ]);
372 });
373
374 return ['status' => 'success', 'action' => 'updated'];
375 }

377 /**
378 * Create new work order.
379 */
380 protected function createNew(array $row, Line $line, ProductType $productType): array
381 {
382 DB::transaction(function () use ($row, $line, $productType) {
383 // Get active process template for product type
384 $processTemplate = $productType->processTemplates()
385 ->where('is_active', true)
386 ->first();
387
388 if (! $processTemplate) {
389 throw new \Exception("No active process template found for product type '{$productType->code}'");
390 }
391
392 // Generate process snapshot
393 $snapshot = $this->snapshotService->createSnapshot($processTemplate);
394
395 WorkOrder::create(array_merge([
396 'order_no' => $row['order_no'],
397 'line_id' => $line->id,
398 'product_type_id' => $productType->id,
399 'process_snapshot' => $snapshot,
400 'planned_qty' => $row['planned_qty'],
401 'produced_qty' => 0,
402 'status' => 'PENDING',
403 'priority' => $row['priority'] ?? 0,
404 'due_date' => $row['due_date'] ?? null,
405 'description' => $row['description'] ?? null,
406 ], $this->optionalErpFields($row)));
407
408 Log::info('Work order created via CSV import', [
409 'order_no' => $row['order_no'],
410 ]);
411 });
412
413 return ['status' => 'success', 'action' => 'created'];
414 }

416 /**
417 * ERP-only columns applied on top of the shared create/update set, and only
418 * when present in the row — so the CSV path (which never supplies them) is
419 * unaffected.
420 *
421 * @param array<string, mixed> $row
422 * @return array<string, mixed>
423 */
424 protected function optionalErpFields(array $row): array
425 {
426 $fields = [];
427
428 if (array_key_exists('customer_order_no', $row) && $row['customer_order_no'] !== null) {
429 $fields['customer_order_no'] = $row['customer_order_no'];
430 }
431
432 if (array_key_exists('unit_price', $row) && $row['unit_price'] !== null) {
433 $fields['unit_price'] = $row['unit_price'];
434 }
435
436 return $fields;
437 }
