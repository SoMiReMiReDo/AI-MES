5use App\Http\Controllers\Controller;
6use App\Support\SoftDeleteRegistry;
7use App\Sync\ShapeRegistry;
8use Illuminate\Http\JsonResponse;
9use Illuminate\Http\Request;
10use Illuminate\Support\Facades\DB;
11use Illuminate\Support\Facades\Schema;

13/**
14 * Initial snapshot for a synced collection (the "load once" half of the Reverb
15 * sync; live deltas then arrive via CollectionChanged on the channel).
16 *
17 * Reuses the existing collection definitions (table / columns / where) from
18 * ShapeRegistry and applies the same tenant scope as the read path.
19 *
20 * GET /api/collections/{name} → { rows: [...], at: <unix ts> }
21 */
22class CollectionController extends Controller
23{
24 public function index(Request $request, ShapeRegistry $registry, string $name): JsonResponse
25 {
26 $shape = $registry->find($name);
27 abort_unless($shape, 404, "Unknown collection: {$name}");
28
29 $user = $request->user();
30
31 $query = DB::table($shape->table())->select($shape->columns());
32
33 $where = $shape->where($user);
34 if ($where) {
35 $query->whereRaw($where);
36 }
37
38 // Soft-deleted rows never reach clients. DB::table() bypasses the
39 // Eloquent SoftDeletes scope, so filter explicitly here; the live path
40 // is covered by CollectionBroadcaster (the `deleted` event broadcasts
41 // a delete op on soft delete too).
42 if (SoftDeleteRegistry::isSoftDeletable($shape->table())) {
43 $query->whereNull('deleted_at');
44 }
45
46 // Same tenant scope as App\Scopes\TenantScope: only when the user has a
47 // tenant and the table carries tenant_id.
48 $scoped = $user?->tenant_id && $this->tableHasTenant($shape->table());
49 if ($scoped) {
50 $query->where('tenant_id', $user->tenant_id);
51 }
52
53 // Tell the client which channel to subscribe to. Global (non-tenant)
54 // tables live on "g" for everyone; tenant tables on the user's tenant —
55 // matching where CollectionBroadcaster publishes each row.
56 $tenantKey = $scoped ? $user->tenant_id : 'g';
57
58 return response()->json([
59 'rows' => $query->get(),
60 'channel' => "col.{$tenantKey}.{$name}",
61 'at' => now()->timestamp,
62 ]);
63 }
64
65 private function tableHasTenant(string $table): bool
66 {
67 static $cache = [];
68
69 return $cache[$table] ??= Schema::hasColumn($table, 'tenant_id');
70 }
71}
