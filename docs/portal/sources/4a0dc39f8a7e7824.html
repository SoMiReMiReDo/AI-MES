5use App\Events\WorkOrder\WorkOrderCompleted;

10/**
11 * Creates the draft warehouse documents a completed work order implies (#212).
12 *
13 * Off by default for installs that do not run warehousing: it does nothing unless
14 * the `warehouse` module is enabled, the auto-generation config flag is on, and
15 * warehouses actually exist (WorkOrderStockDocumentService returns null with no
16 * warehouse to post against). Never breaks the completion it reacts to — a
17 * failure here must not roll back a finished work order.
18 */
19class GenerateWorkOrderStockDocuments
20{
21 public function __construct(private WorkOrderStockDocumentService $documents) {}
22
23 public function handle(WorkOrderCompleted $event): void
24 {
25 if (! config('openmmes.warehouse.auto_documents', true)) {
26 return;
27 }
28
29 try {
30 if (! ModuleRegistry::isModuleEnabled('warehouse')) {
31 return;
32 }
33
34 $this->documents->generateForCompletion($event->workOrder);
35 } catch (\Throwable $e) {
36 Log::warning('Could not generate stock documents for completed work order', [
37 'work_order_id' => $event->workOrder->id,
38 'error' => $e->getMessage(),
39 ]);
40 }
41 }
42}
