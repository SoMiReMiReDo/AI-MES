13/**
14 * The two ends of a module's life that were not guarded.
15 *
16 * `requires_core` has been in every manifest from the start and nothing read
17 * it, so a module built against extension points this core does not have
18 * installed cleanly and then quietly did nothing. And uninstall deleted the
19 * directory without ever calling the module's own uninstall hook — which is the
20 * module's only chance to undo what it did outside its own tables.
21 */
22class ModuleLifecycleGuardsTest extends TestCase
23{
24 use RefreshDatabase;

30 parent::setUp();

33 mkdir($this->modulesPath);
34 $this->app->instance(ModuleManager::class, new class($this->modulesPath) extends ModuleManager
35 {
36 public function __construct(string $path)

38 parent::__construct();
39 $this->modulesPath = $path;
40 }

43 Event::fake();
44 ProbeInstaller::$calls = [];
45 ProbeInstaller::$throwOnUninstall = false;

70 private function moduleZip(string $name, ?string $requiresCore): UploadedFile
71 {
72 $manifest = [
73 'name' => $name,
74 'display_name' => "{$name} module",
75 'version' => '1.0.0',
76 'provider' => "Modules\\{$name}\\Providers\\{$name}ServiceProvider",
77 ];
78
79 if ($requiresCore !== null) {
80 $manifest['requires_core'] = $requiresCore;
81 }
82
83 $path = tempnam(sys_get_temp_dir(), 'mod').'.zip';
84 $zip = new \ZipArchive;
85 $zip->open($path, \ZipArchive::CREATE | \ZipArchive::OVERWRITE);
86 $zip->addFromString("{$name}/module.json", json_encode($manifest));
87 $zip->addFromString(
88 "{$name}/Providers/{$name}ServiceProvider.php",
89 "<?php\nnamespace Modules\\{$name}\\Providers;\nclass {$name}ServiceProvider extends \\Illuminate\\Support\\ServiceProvider {}\n",
90 );
91 $zip->close();
92
93 return new UploadedFile($path, 'module.zip', 'application/zip', null, true);
94 }

96 /** Put a module on disk without going through the upload form. */
97 private function installOnDisk(string $name, ?string $requiresCore): void
98 {
99 $manifest = ['name' => $name, 'display_name' => "{$name} module", 'version' => '1.0.0'];
100
101 if ($requiresCore !== null) {
102 $manifest['requires_core'] = $requiresCore;
103 }
104
105 mkdir("{$this->modulesPath}/{$name}");
106 file_put_contents("{$this->modulesPath}/{$name}/module.json", json_encode($manifest));
107 }

175 public function test_uninstall_runs_the_modules_own_hook_before_deleting_it(): void
176 {
177 // After the directory is gone the Installer class cannot be loaded, so
178 // the hook would never run at all — which is what used to happen.
179 $this->installOnDisk('Probe', null);
180
181 $this->actingAs($this->admin())

216/** Stands in for a module shipping `Modules\<Name>\Installer`. */
217class ProbeInstaller
218{
219 /** @var list<string> */
220 public static array $calls = [];
221
222 public static bool $throwOnUninstall = false;
223
224 public function install(): void
225 {
226 self::$calls[] = 'install';
227 }
228
229 public function uninstall(): void
230 {
231 self::$calls[] = 'uninstall';
232
233 if (self::$throwOnUninstall) {
234 throw new \RuntimeException('Probe refuses to go quietly');
235 }
236 }
