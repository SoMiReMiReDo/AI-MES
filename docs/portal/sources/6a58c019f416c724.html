3namespace App\Http\Middleware;
4
5use App\Models\ApiKey;
6use App\Support\TenantContext;
7use Closure;
8use Illuminate\Http\JsonResponse;
9use Illuminate\Http\Request;
10use Symfony\Component\HttpFoundation\Response;
11
12/**
13 * Authenticates ERP integration requests by API key. The key travels in the
14 * `X-Api-Key` header, or as `Authorization: Bearer <key>`. On success the
15 * resolved ApiKey is attached to the request (`api_key` attribute) for the
16 * scope middleware and controllers, and the request-scoped TenantContext is
17 * set so tenant-scoped models read/write within the key's tenant even though
18 * there is no logged-in user.
19 */
20class AuthenticateApiKey
21{
22 public function __construct(private readonly TenantContext $tenant) {}
23
24 public function handle(Request $request, Closure $next): Response
25 {
26 $plaintext = $this->extractKey($request);
27
28 if ($plaintext === null) {
29 return $this->unauthorized(__('API key missing.'));
30 }
31
32 $key = ApiKey::findByPlaintext($plaintext);
33
34 if ($key === null || ! $key->isUsable()) {
35 return $this->unauthorized(__('Invalid or expired API key.'));
36 }
37
38 if (! $key->ipAllowed($request->ip())) {
39 return $this->unauthorized(__('Source IP not allowed for this API key.'));
40 }
41
42 // Establish tenant scope for this headless request before any query runs.
43 $this->tenant->set($key->tenant_id);
44
45 $request->attributes->set('api_key', $key);
46 $key->markUsed();
47
48 return $next($request);
49 }
50
51 /**
52 * Clear the tenant context after the response is sent so a reused Octane
53 * worker never carries one request's tenant into the next.
54 */
55 public function terminate(Request $request, Response $response): void
56 {
57 $this->tenant->clear();
58 }
59
60 private function extractKey(Request $request): ?string
61 {
62 $header = $request->header('X-Api-Key');
63
64 if (is_string($header) && $header !== '') {
65 return trim($header);
66 }
67
68 $bearer = $request->bearerToken();
69
70 return $bearer !== null && str_starts_with($bearer, ApiKey::PREFIX) ? $bearer : null;
71 }
72
73 private function unauthorized(string $message): JsonResponse
74 {
75 return response()->json(['message' => $message], 401);
76 }
77}
