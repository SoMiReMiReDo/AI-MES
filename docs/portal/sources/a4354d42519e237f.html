20 /**
21 * Recompute the automatic priority score from the configured rules whenever
22 * a work order is saved (create or update). No-op until at least one active
23 * priority rule exists, so manual priorities are preserved until scoring is
24 * configured. Runs with persist=false because a save is already in flight.
25 */
26 protected static function booted(): void
27 {
28 static::saving(function (self $workOrder): void {
29 if ($workOrder->exists && $workOrder->isDirty(['planned_start_at', 'planned_end_at'])
30 && $workOrder->planned_start_at && $workOrder->planned_end_at
31 && $workOrder->planned_end_at->lte($workOrder->planned_start_at)) {
32 throw \Illuminate\Validation\ValidationException::withMessages([
33 'planned_start_at' => __('Planned end must be after planned start.'),
34 ]);
35 }
36 if ($workOrder->exists && $workOrder->isDirty('planned_start_at') && $workOrder->planned_start_at?->isFuture()) {
37 $started = ! in_array($workOrder->getOriginal('status'), [self::STATUS_PENDING, self::STATUS_ACCEPTED], true)
38 || (float) $workOrder->getOriginal('produced_qty') > 0
39 || $workOrder->batches()->whereHas('steps', fn ($q) => $q->whereNotNull('started_at'))->exists();
40 if ($started) {
41 throw \Illuminate\Validation\ValidationException::withMessages([
42 'planned_start_at' => __('An order that has started cannot be postponed into the future.'),
43 ]);
44 }
45 }
46 if ($workOrder->exists && (($workOrder->isDirty('status') && in_array($workOrder->status, [self::STATUS_IN_PROGRESS, self::STATUS_DONE], true))
47 || $workOrder->isDirty('produced_qty'))) {
48 $workOrder->assertProductionAvailable();
49 }
50
51 // Recompute only when a scoring-relevant field changed (or the row is
52 // new); other saves — status transitions, produced_qty ticks — skip
53 // the rule query. Time-based rules are kept fresh by priority:recalculate.
54 if ($workOrder->exists
55 && ! $workOrder->isDirty(['customer_id', 'planned_qty', 'due_date'])) {
56 return;
57 }
58
59 app(\App\Services\WorkOrder\PriorityScoringService::class)->apply($workOrder, persist: false);
60 });
61
62 // Roll a completed order into its customer's aggregate metrics (order
63 // count, revenue, auto tier promotion), once per order. Covers both a
64 // transition into DONE (updated) and an order inserted already DONE
65 // (created — e.g. a historical import). recordCompletion is idempotent.
66 $accrue = function (self $workOrder): void {
67 if ($workOrder->status === self::STATUS_DONE && ! $workOrder->customer_totals_counted) {
68 app(\App\Services\Customer\CustomerMetricsService::class)->recordCompletion($workOrder);
69 }
70 };
71
72 static::created($accrue);
73 static::updated(function (self $workOrder) use ($accrue): void {
74 if ($workOrder->wasChanged('status')) {
75 $accrue($workOrder);
76 }
77 });
78 }

250 /**
251 * Best-effort estimate of total planned work in minutes for orders that do
252 * NOT have minute-level planning timestamps (those use
253 * {@see plannedDurationMinutes()} instead). Falls back through concrete
254 * batch-step durations, then the product's process-snapshot step estimates.
255 *
256 * Returns null when neither source yields a positive duration
257 * ("unestimated") so callers can surface the order rather than silently
258 * treating it as zero load.
259 */
260 public function estimatedDurationMinutes(): ?int
261 {
262 // 1. Concrete batch-step durations (actual planned runtimes per step).
263 if ($this->relationLoaded('batches')) {
264 $sum = 0;
265 foreach ($this->batches as $batch) {
266 if (! $batch->relationLoaded('steps')) {
267 continue;
268 }
269 foreach ($batch->steps as $step) {
270 $sum += (int) ($step->duration_minutes ?? 0);
271 }
272 }
273 if ($sum > 0) {
274 return $sum;
275 }
276 }
277
278 // 2. Process-snapshot step estimates captured from the product template.
279 $sum = 0;
280 foreach ($this->process_snapshot['steps'] ?? [] as $step) {
281 $sum += (int) ($step['estimated_duration_minutes'] ?? 0);
282 }
283
284 return $sum > 0 ? $sum : null;
285 }

287 /**
288 * ISA-95 L4 standard production time (#52): from the frozen snapshot's standard
289 * times, Σ(setup + run_per_unit × planned_qty) across steps. This is the
290 * planning/costing target that flows down from the ERP BOM, distinct from the
291 * total estimate above. Null when no step carries standard times.
292 */
293 public function estimatedStandardProductionMinutes(): ?int
294 {
295 $steps = $this->process_snapshot['steps'] ?? [];
296 $qty = (float) ($this->planned_qty ?? 0);
297 $total = 0.0;
298 $hasStandard = false;
299
300 // Mirror batch materialization: within a variant group only the selected
301 // path runs (default flag, else the lowest step_number), so the standard
302 // must not sum skipped siblings — that would inflate the costing target.
303 $chosen = $this->selectedVariantSteps($steps);
304
305 foreach ($steps as $step) {
306 $group = $step['variant_group'] ?? null;
307 if ($group !== null && isset($chosen[$group]) && ($step['step_number'] ?? null) !== $chosen[$group]) {
308 continue;
309 }
310
311 $setup = $step['setup_time_minutes'] ?? null;
312 $run = $step['run_time_per_unit_minutes'] ?? null;
313 if ($setup === null && $run === null) {
314 continue;
315 }
316 $hasStandard = true;
317 $total += (float) ($setup ?? 0) + (float) ($run ?? 0) * $qty;
318 }
319
320 return $hasStandard ? (int) round($total) : null;
321 }
322
323 /**
324 * The pre-selected step_number per variant group: the one flagged default,
325 * else the lowest step_number in the group. Mirrors
326 * WorkOrderService::createBatchStepsFromSnapshot so estimates match execution.
327 *
328 * @param array<int, array<string, mixed>> $steps
329 * @return array<string, int>
330 */
331 private function selectedVariantSteps(array $steps): array
332 {
333 $chosen = [];
334 $explicit = [];
335 foreach ($steps as $s) {
336 $group = $s['variant_group'] ?? null;
337 if ($group === null) {
338 continue;
339 }
340 $num = $s['step_number'];
341 if (! empty($s['is_default_variant'])) {
342 if (empty($explicit[$group])) {
343 $chosen[$group] = $num;
344 $explicit[$group] = true;
345 }
346 } elseif (empty($explicit[$group])) {
347 $chosen[$group] = isset($chosen[$group]) ? min($chosen[$group], $num) : $num;
348 }
349 }
350
351 return $chosen;
352 }

354 /**
355 * The engineering documents (#179) frozen onto this order at release, as
356 * immutable snapshot references. Backfills the display fields
357 * (original_filename/entry_point/file_size) for orders snapshotted before
358 * those were frozen, by joining the live document by id (withTrashed).
359 *
360 * @return array<int, array<string, mixed>>
361 */
362 public function frozenEngineeringDocuments(): array
363 {
364 $refs = $this->process_snapshot['engineering_documents'] ?? [];
365
366 $missing = collect($refs)
367 ->reject(fn ($r) => array_key_exists('original_filename', $r))
368 ->pluck('document_id')
369 ->filter()
370 ->all();
371
372 if ($missing) {
373 $live = EngineeringDocument::withTrashed()->findMany($missing)->keyBy('id');
374 $refs = array_map(function (array $r) use ($live) {
375 if (! array_key_exists('original_filename', $r) && ($d = $live->get($r['document_id'] ?? null))) {
376 $r['original_filename'] = $d->original_filename;
377 $r['entry_point'] = $d->entry_point;
378 $r['file_size'] = $d->file_size;
379 }
380
381 return $r;
382 }, $refs);
383 }
384
385 return array_values($refs);
386 }

388 /**
389 * Labor multiplier turning this order's machine-hours into person-hours for
390 * the schedule-capacity crew axis: the duration-weighted average operators
391 * across its process-snapshot steps — i.e. Σ(step minutes × operators) ÷
392 * Σ(step minutes), so true person-hours = machine-hours × this factor.
393 *
394 * Falls back to the heaviest step's operator count when no step has a
395 * duration, and to 1 when the snapshot has no steps/operators. A step with
396 * a missing or zero operator count is treated as needing one operator.
397 */
398 public function operatorFactor(): float
399 {
400 $totalMinutes = 0;
401 $weighted = 0;
402 $maxOperators = 0;
403
404 foreach ($this->process_snapshot['steps'] ?? [] as $step) {
405 $operators = max(1, (int) ($step['required_operators'] ?? 1));
406 $minutes = (int) ($step['estimated_duration_minutes'] ?? 0);
407 $maxOperators = max($maxOperators, $operators);
408 if ($minutes > 0) {
409 $totalMinutes += $minutes;
410 $weighted += $minutes * $operators;
411 }
412 }
413
414 if ($totalMinutes > 0) {
415 return $weighted / $totalMinutes;
416 }
417
418 return max(1, $maxOperators);
419 }

669 /**
670 * Quality metric for this work order: the share of produced units that
671 * were not scrapped, as a percentage. Scrap entries automatically pull
672 * this down. Returns null when nothing has been produced yet.
673 */
674 public function qualityPct(): ?float
675 {
676 $produced = (float) $this->produced_qty;
677
678 if ($produced <= 0) {
679 return null;
680 }
681
682 $good = max(0.0, $produced - $this->totalScrapQty());
683
684 return round(($good / $produced) * 100, 2);
685 }
