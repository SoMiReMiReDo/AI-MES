3namespace Tests\Feature\Sync;
4
5use Illuminate\Http\Request;
6use Laravel\Sanctum\Http\Middleware\EnsureFrontendRequestsAreStateful;
7use Laravel\Sanctum\Sanctum;
8use Tests\TestCase;
9
10/**
11 * Regression guard for "POST create returns 200 but GET /api/collections returns
12 * 401" — the live-sync list read needs a Sanctum stateful session. The fix marks
13 * the app's own host stateful via currentRequestHost(), so same-origin reads
14 * authenticate on any host/port/domain without per-deploy SANCTUM_STATEFUL_DOMAINS.
15 */
16class StatefulCollectionAccessTest extends TestCase
17{
18 public function test_current_request_host_placeholder_is_in_stateful_domains(): void
19 {
20 $this->assertContains(
21 Sanctum::$currentRequestHostPlaceholder,
22 config('sanctum.stateful'),
23 'config/sanctum.php must include Sanctum::currentRequestHost() so same-origin reads stay stateful.',
24 );
25 }
26
27 public function test_same_origin_request_is_treated_as_stateful_on_any_host(): void
28 {
29 // App served on an arbitrary host the app URL doesn't cover.
30 $request = Request::create('http://mes.factory.local:8095/api/collections/work_orders_all');
31 $request->headers->set('Origin', 'http://mes.factory.local:8095');
32 $request->headers->set('Host', 'mes.factory.local:8095');
33
34 $this->assertTrue(
35 EnsureFrontendRequestsAreStateful::fromFrontend($request),
36 'Same-origin SPA requests must be stateful regardless of the host.',
37 );
38 }
39
40 public function test_cross_origin_request_is_not_stateful(): void
41 {
42 // Different Origin (attacker site) than the app host → must stay rejected.
43 $request = Request::create('http://mes.factory.local:8095/api/collections/work_orders_all');
44 $request->headers->set('Origin', 'http://evil.example.com');
45 $request->headers->set('Host', 'mes.factory.local:8095');
46
47 $this->assertFalse(
48 EnsureFrontendRequestsAreStateful::fromFrontend($request),
49 'Cross-origin requests must not become stateful (CSRF protection preserved).',
50 );
51 }
52}
