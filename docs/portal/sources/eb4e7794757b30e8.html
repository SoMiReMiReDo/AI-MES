44 /** One batch of 10 through three plain sequential steps. */
45 private function makeBatch(float $qty = 10): Batch
46 {
47 $pt = ProductType::factory()->create();
48 $template = ProcessTemplate::factory()->create(['product_type_id' => $pt->id, 'is_active' => true]);
49 foreach ([1 => 'Mix', 2 => 'Bake', 3 => 'Pack'] as $n => $name) {
50 TemplateStep::factory()->create(['process_template_id' => $template->id, 'step_number' => $n, 'name' => $name]);
51 }
52
53 $wo = WorkOrder::factory()->create([
54 'line_id' => $this->line->id,
55 'planned_qty' => $qty,
56 'process_snapshot' => $template->load('steps')->toSnapshot(),
57 ]);
58
59 return app(WorkOrderService::class)->createBatch($wo, $qty);
60 }

72 public function test_skipping_partially_scrapped_step_preserves_material_balance(): void
73 {
74 ProductionFlow::set(ProductionFlow::TRANSFER);
75 $batch = $this->makeBatch();
76 $service = app(BatchService::class);
77 $step = $this->step($batch, 1);
78 $step->update(['is_optional' => true]);
79 $service->startStep($step->fresh(), $this->operator);
80 $service->recordQuantity($step->fresh(), $this->operator, 4, 2);
81 try {
82 $service->skipStep($step->fresh(), $this->operator);
83 $this->fail('A processed step was skipped.');
84 } catch (\DomainException $e) {
85 $this->assertStringContainsString('cannot be skipped', $e->getMessage());
86 }
87 $this->assertSame(BatchStep::STATUS_IN_PROGRESS, $step->fresh()->status);
88 $this->assertLessThanOrEqual(8, $this->step($batch, 2)->incomingQty(), 'Skipping must not resurrect the 2 scrapped pieces');
89 }
