48 public function test_in_production_trigger_fires_once_when_batch_enters_production(): void
49 {
50 $trigger = QualityControlTrigger::factory()->create(); // in_production, active
51 $batch = Batch::factory()->inProgress()->create();
52
53 $this->service()->fireInProduction($batch);
54 $this->service()->fireInProduction($batch); // idempotent — no duplicate
55
56 $tasks = QualityControlTask::where('quality_control_trigger_id', $trigger->id)->get();
57 $this->assertCount(1, $tasks);
58 $this->assertSame($batch->id, $tasks->first()->batch_id);
59 $this->assertSame($batch->work_order_id, $tasks->first()->work_order_id);
60 $this->assertSame(QualityControlTask::STATUS_DUE, $tasks->first()->status);
61 }

213 public function test_performing_a_task_records_a_quality_check_against_the_batch(): void
214 {
215 $batch = Batch::factory()->inProgress()->create();

218 'work_order_id' => $batch->work_order_id,
219 ]);
220
221 $samples = [
222 ['sample_number' => 1, 'parameter_name' => 'Length', 'parameter_type' => 'measurement', 'value_numeric' => 9.9, 'is_passed' => true],
223 ];
224
225 $task = $this->service()->performTask($task->fresh(), $this->admin, $samples);
226
227 $this->assertSame(QualityControlTask::STATUS_DONE, $task->status);
228 $this->assertNotNull($task->quality_check_id);
229 $this->assertNull($task->issue_id);
230 $this->assertDatabaseHas('quality_checks', ['id' => $task->quality_check_id, 'batch_id' => $batch->id, 'all_passed' => true]);
