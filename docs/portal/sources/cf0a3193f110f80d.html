20 /**
21 * Discover all modules by scanning the modules/ directory.
22 */
23 public function discover(): Collection
24 {
25 if (! is_dir($this->modulesPath)) {
26 return collect();
27 }
28
29 $enabled = $this->enabledNames();
30 $modules = collect();
31
32 foreach (scandir($this->modulesPath) as $entry) {
33 if ($entry === '.' || $entry === '..') {
34 continue;
35 }
36
37 $manifestPath = "{$this->modulesPath}/{$entry}/module.json";
38 if (! is_file($manifestPath)) {
39 continue;
40 }
41
42 $manifest = json_decode(file_get_contents($manifestPath), true);
43 if (! $manifest || empty($manifest['name'])) {
44 continue;
45 }
46
47 $manifest['enabled'] = in_array($manifest['name'], $enabled);
48 $manifest['directory'] = $entry;
49 $manifest['has_error'] = false;
50
51 // Check if provider class exists (only if enabled)
52 if ($manifest['enabled'] && ! empty($manifest['provider'])) {
53 $manifest['has_error'] = ! class_exists($manifest['provider']);
54 }
55
56 $modules->push($manifest);
57 }
58
59 return $modules->sortBy('display_name');
60 }

132 /**
133 * Copy a module's compiled frontend into public/, where the browser can
134 * fetch it.
135 *
136 * A module's pages are not in this app's bundle — that was compiled before
137 * the module existed, and nothing rebuilds it on a running system. The
138 * module ships its own built file instead, and it has to be reachable over
139 * HTTP for the browser to load it; modules/ is not web-accessible.
140 *
141 * Silently does nothing for a module without a built frontend: plenty of
142 * modules are backend-only, and those are not broken.
143 */
144 public function publishAssets(string $name): void
145 {
146 $source = "{$this->modulesPath}/{$name}/public/{$name}.js";
147
148 if (! is_file($source)) {
149 return;
150 }
151
152 $targetDir = public_path('modules');
153 if (! is_dir($targetDir)) {
154 mkdir($targetDir, 0755, true);
155 }
156
157 copy($source, "{$targetDir}/{$name}.js");
158
159 // The sidecar records which runtime contract the module was built
160 // against; HandleInertiaRequests passes it on so the browser can refuse
161 // a module built for a core it no longer matches.
162 $manifest = "{$this->modulesPath}/{$name}/public/{$name}.runtime.json";
163 if (is_file($manifest)) {
164 copy($manifest, "{$targetDir}/{$name}.runtime.json");
165 }
166 }

168 /** Remove what publishAssets() put in public/. */
169 public function unpublishAssets(string $name): void
170 {
171 foreach (["{$name}.js", "{$name}.runtime.json"] as $file) {
172 $path = public_path("modules/{$file}");
173 if (is_file($path)) {
174 unlink($path);
175 }
176 }

179 /**
180 * Enabled modules that ship a frontend, as the browser needs them: a URL to
181 * load and the contract version the file was built against.
182 *
183 * @return array<int, array{name: string, url: string, requires_runtime: int|null}>
184 */
185 public function frontendAssets(): array
186 {
187 $assets = [];
188
189 foreach ($this->enabledNames() as $name) {
190 $path = public_path("modules/{$name}.js");
191
192 if (! is_file($path)) {
193 continue;
194 }
195
196 $requires = null;
197 $sidecar = public_path("modules/{$name}.runtime.json");
198 if (is_file($sidecar)) {
199 $decoded = json_decode(file_get_contents($sidecar), true);
200 $requires = is_array($decoded) ? ($decoded['requires_runtime'] ?? null) : null;
201 }
202
203 $assets[] = [
204 'name' => $name,
205 // Cache-busted on the file's contents, not its mtime: publishing
206 // copies the file, so mtime is "when it was copied" — it changes
207 // on every enable/disable cycle even when nothing about the
208 // module did, throwing away the browser's cache for no reason,
209 // and two publishes inside the same second collide.
210 'url' => "/modules/{$name}.js?v=".substr(md5_file($path), 0, 12),
211 'requires_runtime' => $requires,
212 ];
213 }
214
215 return $assets;
216 }

218 /**
219 * Install a module from a ZIP file.
220 * Returns the module name on success, throws on failure.
221 */
222 public function installFromZip(string $zipPath): string
223 {
224 if (! class_exists(\ZipArchive::class)) {
225 throw new \RuntimeException('ZipArchive PHP extension is required.');
226 }
227
228 $zip = new \ZipArchive;
229 if ($zip->open($zipPath) !== true) {
230 throw new \RuntimeException('Could not open ZIP file.');
231 }
232
233 // Find module.json inside the ZIP
234 $manifestContent = null;
235 $moduleRoot = null;
236
237 for ($i = 0; $i < $zip->numFiles; $i++) {
238 $name = $zip->getNameIndex($i);
239 if (str_ends_with($name, 'module.json')) {
240 $manifestContent = $zip->getFromIndex($i);
241 // Determine the root directory inside the ZIP
242 $parts = explode('/', $name);
243 $moduleRoot = count($parts) > 1 ? $parts[0] : null;
244 break;
245 }
246 }
247
248 if (! $manifestContent) {
249 $zip->close();
250 throw new \RuntimeException('No module.json found inside the ZIP.');
251 }
252
253 $manifest = json_decode($manifestContent, true);
254 if (! $manifest || empty($manifest['name'])) {
255 $zip->close();
256 throw new \RuntimeException('Invalid module.json: missing "name" field.');
257 }
258
259 // Validate required manifest fields
260 if (empty($manifest['version'])) {
261 $zip->close();
262 throw new \RuntimeException('Invalid module.json: missing "version" field.');
263 }
264 if (empty($manifest['provider'])) {
265 $zip->close();
266 throw new \RuntimeException('Invalid module.json: missing "provider" field.');
267 }
268
269 // Validate provider namespace
270 if (! str_starts_with($manifest['provider'], 'Modules\\')) {
271 $zip->close();
272 throw new \RuntimeException('Invalid provider namespace: must start with "Modules\\".');
273 }
274
275 // The core version the module was built against. Nothing read this
276 // until now, so a module installed on an older core simply did not
277 // work: the seams it relies on were absent, and the admin was left
278 // guessing. Refusing the install says so once, at the only moment
279 // anybody can act on it.
280 if (! $this->coreSatisfies($manifest['requires_core'] ?? null)) {
281 $zip->close();
282 throw new \RuntimeException(sprintf(
283 'Module "%s" requires OpenMES %s; this installation is %s.',
284 $manifest['name'],
285 $manifest['requires_core'],
286 config('version.current'),
287 ));
288 }
289
290 // Validate module name (alphanumeric + hyphens only, no path traversal)
291 if (! preg_match('/^[A-Za-z0-9_-]+$/', $manifest['name'])) {
292 $zip->close();
293 throw new \RuntimeException('Invalid module name: only alphanumeric characters, hyphens, and underscores are allowed.');
294 }
295
296 // Check for path traversal in ZIP entries
297 $dangerousFunctions = ['exec', 'system', 'passthru', 'shell_exec', 'proc_open', 'popen', 'eval'];
298 $dangerousPattern = '/\b('.implode('|', $dangerousFunctions).')\s*\(/i';
299
300 for ($i = 0; $i < $zip->numFiles; $i++) {
301 $entryName = $zip->getNameIndex($i);
302
303 // Path traversal check
304 $normalized = str_replace('\\', '/', $entryName);
305 if (str_contains($normalized, '..') || str_starts_with($normalized, '/')) {
306 $zip->close();
307 throw new \RuntimeException("ZIP contains suspicious path: {$entryName}");
308 }
309
310 // Scan PHP files for dangerous functions
311 if (str_ends_with(strtolower($entryName), '.php')) {
312 $content = $zip->getFromIndex($i);
313 if ($content !== false && preg_match($dangerousPattern, $content)) {
314 $zip->close();
315 throw new \RuntimeException("PHP file contains prohibited function call: {$entryName}");
316 }
317 }
318 }
319
320 $moduleName = $manifest['name'];
321 $destDir = "{$this->modulesPath}/{$moduleName}";
322
323 // Extract
324 if ($moduleRoot) {
325 // ZIP has a top-level directory — extract and rename it
326 $tmpDir = "{$this->modulesPath}/_tmp_{$moduleName}";
327 $zip->extractTo($tmpDir);
328 $zip->close();
329
330 if (is_dir($destDir)) {
331 $this->deleteDirectory($destDir);
332 }
333 rename("{$tmpDir}/{$moduleRoot}", $destDir);
334 $this->deleteDirectory($tmpDir);
335 } else {
336 // ZIP contains files at the root
337 $zip->extractTo($destDir);
338 $zip->close();
339 }
340
341 return $moduleName;
342 }

344 /**
345 * Delete an installed module directory.
346 *
347 * The module's own uninstall hook runs FIRST, while its classes are still
348 * on disk and autoloadable — after the directory is gone there is nothing
349 * left to call. It is the module's only chance to undo what it did outside
350 * its own tables: permissions it registered, settings it wrote.
351 *
352 * A throwing hook aborts the uninstall and leaves the directory in place.
353 * Deleting anyway would run half an uninstall and lose the other half with
354 * no way to retry; this way the administrator sees the failure and the
355 * module is still there to try again. The module's own migrations are NOT
356 * rolled back — see the note the controller shows.
357 */
358 public function uninstall(string $name): void
359 {
360 $this->runInstaller($name, 'uninstall');
361
362 $this->disable($name);
363 $dir = "{$this->modulesPath}/{$name}";
364 if (is_dir($dir)) {
365 $this->deleteDirectory($dir);
366 }
367 }

369 /**
370 * Load all enabled module ServiceProviders into the application.
371 * Called from AppServiceProvider::boot().
372 */
373 public function loadEnabled(\Illuminate\Foundation\Application $app): void
374 {
375 foreach ($this->enabledNames() as $name) {
376 $manifestPath = "{$this->modulesPath}/{$name}/module.json";
377 if (! is_file($manifestPath)) {
378 continue;
379 }
380
381 $manifest = json_decode(file_get_contents($manifestPath), true);
382 $provider = $manifest['provider'] ?? null;
383
384 if ($provider && class_exists($provider)) {
385 // Wrapped rather than registered directly: an exception from the
386 // provider's boot() is thrown while Laravel boots every provider,
387 // long after this method returned, so the caller's try/catch
388 // cannot see it — and one bad module would 502 the whole app.
389 //
390 // Registered with force, because the container deduplicates by
391 // class name: without it the second module's guard would be
392 // mistaken for the first one's and silently skipped.
393 $app->register(
394 new \App\Support\ModuleProviderGuard($app, new $provider($app), $name),
395 force: true,
396 );
397 }
398 }
399 }

415 /**
416 * Run a module's optional installer hook.
417 *
418 * Migrations cover tables, but a module usually has setup that is not schema:
419 * the permissions behind the tabs it registers, default settings, seed rows.
420 * A module provides it by shipping `Modules\<Name>\Installer` with an
421 * `install()` and/or `uninstall()` method; modules without one are the normal
422 * case and this does nothing.
423 *
424 * Exceptions are deliberately NOT swallowed — the caller decides what a
425 * failed install means, and enabling a module whose setup failed is worse
426 * than not enabling it.
427 *
428 * @param string $method install | uninstall
429 */
430 public function runInstaller(string $name, string $method = 'install'): void
431 {
432 $installer = "Modules\\{$name}\\Installer";
433
434 if (! class_exists($installer)) {
435 return;
436 }
437
438 $instance = app($installer);
439
440 if (method_exists($instance, $method)) {
441 $instance->{$method}();
442 }
443 }

455 protected function deleteDirectory(string $dir): void
456 {
457 if (! is_dir($dir)) {
458 return;
459 }
460 foreach (scandir($dir) as $item) {
461 if ($item === '.' || $item === '..') {
462 continue;
463 }
464 $path = "{$dir}/{$item}";
465 is_dir($path) ? $this->deleteDirectory($path) : unlink($path);
466 }
467 rmdir($dir);
468 }
