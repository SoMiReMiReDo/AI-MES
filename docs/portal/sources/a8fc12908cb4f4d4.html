17/**
18 * Unified material traceability / genealogy.
19 *
20 * Builds on the existing ISA-95 lot infrastructure (material_lots,
21 * batch_step_lot_consumption) and the formal batch-output link
22 * (material_lots.source_batch_id) to answer two questions:
23 *
24 * forward — "where did this material lot go?" (impact analysis on a bad
25 * supplier delivery: which finished work orders are affected)
26 * backward — "what fed into this finished lot?" (recall: trace a finished
27 * batch back to its ingredient lots and their suppliers)
28 */
29class TraceabilityService
30{
31 /** Hard cap on backward recursion to guard against cyclic / deep chains. */
32 private const MAX_DEPTH = 10;
33
34 /**
35 * Forward trace: every batch step / batch / work order that consumed the lot,
36 * plus - for a finished-goods lot (source_batch_id set) - the output pallet(s)
37 * it was packed onto and the customer order(s) it fulfils. Genealogy no longer
38 * dead-ends at the finished lot: it continues to pallet and customer order.
39 *
40 * @return array{lot: array, consumptions: \Illuminate\Support\Collection, work_orders: \Illuminate\Support\Collection}
41 */
42 public function forwardTrace(MaterialLot $lot): array
43 {
44 // Only the work-order chain is needed downstream (work_orders + total).
45 // Heavier relations (workstation, batch numbers, recordedBy) were trimmed
46 // because the caller discards them.
47 $consumptions = $lot->consumptions()
48 ->with([
49 'batchStep:id,batch_id',
50 'batchStep.batch:id,work_order_id',
51 'batchStep.batch.workOrder:id,order_no,product_type_id,status',
52 'batchStep.batch.workOrder.productType:id,name,code',
53 ])
54 ->orderByDesc('consumed_at')
55 ->get();
56
57 $workOrders = $consumptions
58 ->map(fn ($c) => $c->batchStep?->batch?->workOrder)
59 ->filter()
60 ->unique('id')
61 ->values();
62
63 $packing = $this->finishedGoodsPacking($lot);
64
65 return [
66 'lot' => $lot->only(['id', 'lot_number', 'material_id', 'status', 'supplier_lot_no', 'source_container_no']),
67 'consumptions' => $consumptions,
68 'work_orders' => $workOrders,
69 'total_consumed' => (float) $consumptions->sum('quantity_consumed'),
70 'is_finished_good' => $lot->source_batch_id !== null,
71 'pallets' => $packing['pallets'],
72 'customer_orders' => $packing['customer_orders'],
73 ];
74 }

76 /**
77 * The output pallet(s) a finished-goods lot was packed onto and the customer
78 * order(s) it fulfils. Derived through the producing batch: a pallet pins to
79 * one batch (pallets.batch_id) and a finished lot pins to the same batch
80 * (material_lots.source_batch_id), so the link is the batch they share.
81 *
82 * A raw/inbound lot (no source_batch_id) returns empty collections.
83 *
84 * @return array{pallets: \Illuminate\Support\Collection, customer_orders: \Illuminate\Support\Collection}
85 */
86 private function finishedGoodsPacking(MaterialLot $lot): array
87 {
88 if ($lot->source_batch_id === null) {
89 return ['pallets' => collect(), 'customer_orders' => collect()];
90 }
91
92 // withTrashed: a finished lot must still resolve its pallets / customer
93 // order if the pallet or the producing batch was later soft-deleted
94 // (recall readiness - the packed/shipped hop must not vanish).
95 $pallets = Pallet::withTrashed()
96 ->where('batch_id', $lot->source_batch_id)
97 ->with(['workOrder:id,order_no,customer_order_no'])
98 ->orderBy('pallet_no')
99 ->get();
100
101 $batch = Batch::withTrashed()
102 ->with(['workOrder' => fn ($q) => $q->withTrashed()->select('id', 'order_no', 'customer_order_no')])
103 ->find($lot->source_batch_id);
104
105 // Customer order(s): the producing batch's work order plus any pallet's
106 // own work order (normally the same), de-duplicated and null-stripped.
107 $customerOrders = collect([$batch?->workOrder?->customer_order_no])
108 ->merge($pallets->map(fn (Pallet $p) => $p->workOrder?->customer_order_no))
109 ->filter()
110 ->unique()
111 ->values();
112
113 $pallets = $pallets->map(fn (Pallet $p) => [
114 'pallet_no' => $p->pallet_no,
115 'status' => $p->status instanceof PalletStatus ? $p->status->value : $p->status,
116 'qty' => (int) $p->qty,
117 'location' => $p->location,
118 'shipped_at' => $p->shipped_at?->format('Y-m-d H:i'),
119 ])->values();
120
121 return ['pallets' => $pallets, 'customer_orders' => $customerOrders];
122 }

124 /**
125 * Reverse traceability for recall: starting from one or more source lots,
126 * walk forward through every batch that consumed them and collect the
127 * affected work orders plus the finished serial units produced under them -
128 * the "what do I need to pull off the shelf?" answer.
129 *
130 * The walk is transitive: a consuming batch may itself produce output lots
131 * (material_lots.source_batch_id), which can be consumed further downstream,
132 * so multi-stage builds surface every affected level. Bounded by MAX_DEPTH.
133 *
134 * @param Collection<int, MaterialLot> $sourceLots
135 * @return array{source_lots: array, work_orders: array, totals: array, truncated: bool}
136 */
137 public function recallImpact(Collection $sourceLots): array
138 {
139 $visitedLotIds = [];
140 $affected = []; // work_order_id => aggregate row
141 $affectedBatchIds = []; // batch_id => true (batches that consumed the recalled material)
142 $frontier = $sourceLots->pluck('id')->filter()->unique()->values()->all();
143 $truncated = false;
144
145 for ($depth = 0; ! empty($frontier); $depth++) {
146 if ($depth >= self::MAX_DEPTH) {
147 $truncated = true;
148 break;
149 }
150
151 foreach ($frontier as $id) {
152 $visitedLotIds[$id] = true;
153 }
154
155 $consumptions = BatchStepLotConsumption::query()
156 ->whereIn('material_lot_id', $frontier)
157 ->with([
158 'batchStep:id,batch_id',
159 // Recall must see through soft-deleted intermediates, otherwise
160 // a deleted batch/WO would hide downstream affected lots.
161 'batchStep.batch' => fn ($q) => $q->withTrashed()->select('id', 'batch_number', 'work_order_id'),
162 'batchStep.batch.workOrder' => fn ($q) => $q->withTrashed()->select('id', 'order_no', 'product_type_id', 'status'),
163 'batchStep.batch.workOrder.productType:id,name,code',
164 ])
165 ->get();
166
167 $consumingBatchIds = [];
168
169 foreach ($consumptions as $consumption) {
170 $batch = $consumption->batchStep?->batch;
171 if (! $batch) {
172 continue; // orphaned consumption (no batch step / batch at all)
173 }
174
175 // Track the batch for the downstream walk even if its WO is gone.
176 $consumingBatchIds[$batch->id] = true;
177 $affectedBatchIds[$batch->id] = true;
178
179 $workOrder = $batch->workOrder;
180 if (! $workOrder) {
181 continue; // WO unrecoverable; batch still walked downstream
182 }
183
184 $row = $affected[$workOrder->id] ?? [
185 'id' => $workOrder->id,
186 'order_no' => $workOrder->order_no,
187 'product' => $workOrder->productType?->name,
188 'status' => $workOrder->status,
189 'quantity_consumed' => 0.0,
190 'batches' => [],
191 ];
192 $row['quantity_consumed'] += (float) $consumption->quantity_consumed;
193 if ($batch->batch_number !== null && ! in_array($batch->batch_number, $row['batches'], true)) {
194 $row['batches'][] = $batch->batch_number;
195 }
196 $affected[$workOrder->id] = $row;
197 }
198
199 // Next level: output lots of the consuming batches we haven't walked yet.
200 $frontier = MaterialLot::withTrashed()
201 ->whereIn('source_batch_id', array_keys($consumingBatchIds))
202 ->pluck('id')
203 ->reject(fn ($id) => isset($visitedLotIds[$id]))
204 ->unique()
205 ->values()
206 ->all();
207 }
208
209 $serialsByWorkOrder = empty($affected)
210 ? collect()
211 : SerialUnit::query()
212 ->whereIn('work_order_id', array_keys($affected))
213 // Only units from batches that consumed the recalled material;
214 // null batch_id is the legacy fallback (pre-batch-linked serials).
215 ->where(function ($query) use ($affectedBatchIds) {
216 $query->whereIn('batch_id', array_keys($affectedBatchIds))
217 ->orWhereNull('batch_id');
218 })
219 ->orderByDesc('produced_at')
220 ->orderByDesc('id')
221 ->get(['id', 'serial_no', 'status', 'work_order_id', 'produced_at'])
222 ->groupBy('work_order_id');
223
224 $workOrders = collect($affected)->values()->map(function (array $row) use ($serialsByWorkOrder) {
225 $serials = ($serialsByWorkOrder[$row['id']] ?? collect())
226 ->map(fn (SerialUnit $unit) => [
227 'serial_no' => $unit->serial_no,
228 'status' => $unit->status,
229 'produced_at' => $unit->produced_at?->format('Y-m-d H:i'),
230 ])
231 ->values()
232 ->all();
233 $row['quantity_consumed'] = round($row['quantity_consumed'], 4);
234 $row['finished_serials'] = $serials;
235
236 return $row;
237 })->all();
238
239 return [
240 'source_lots' => $sourceLots->map(fn (MaterialLot $lot) => [
241 'lot_number' => $lot->lot_number,
242 'material' => $lot->material?->name,
243 'supplier_lot_no' => $lot->supplier_lot_no,
244 ])->values()->all(),
245 'work_orders' => $workOrders,
246 'truncated' => $truncated,
247 'totals' => [
248 'work_orders' => count($workOrders),
249 'finished_serials' => array_sum(array_map(fn ($w) => count($w['finished_serials']), $workOrders)),
250 'quantity_consumed' => round(array_sum(array_column($workOrders, 'quantity_consumed')), 4),
251 ],
252 ];
253 }

255 /**
256 * Recall impact for a serialised unit, resolved through its batch: a
257 * component serial is traced via the output lots its batch produced, which
258 * is what flows downstream into finished goods. A unit with no batch (or a
259 * top-level finished unit with no downstream consumption) yields an empty
260 * impact.
261 */
262 public function recallImpactForSerial(SerialUnit $unit): array
263 {
264 $sourceLots = $unit->batch_id
265 ? MaterialLot::with('material:id,name,code')
266 ->where('source_batch_id', $unit->batch_id)
267 ->get()
268 : collect();
269
270 return $this->recallImpact($sourceLots);
271 }

273 /**
274 * Diagnostic drill-down for a finished serial unit: for every component lot
275 * consumed to build it, the production lines and workstations that component
276 * passed through during its OWN manufacture (the steps of the batch that
277 * produced it). Answers "this finished piece is defective - which component,
278 * and on which line, was at fault?".
279 *
280 * A component with no producing batch (a raw inbound lot) is returned with
281 * an empty journey: it came from a supplier, not an internal line.
282 *
283 * @return array{components: array}
284 */
285 public function componentLineJourneys(SerialUnit $unit): array
286 {
287 $batch = $unit->batch_id ? Batch::find($unit->batch_id) : null;
288 if (! $batch) {
289 return ['components' => []];
290 }
291
292 $components = $this->batchInputLots($batch)
293 ->map(fn (MaterialLot $lot) => $this->lotLineJourney($lot))
294 ->values()
295 ->all();
296
297 return ['components' => $components];
298 }

300 /**
301 * The line / workstation path a single material lot travelled through during
302 * its own production, resolved from the steps of the batch that produced it
303 * (source_batch_id). Empty journey for a raw inbound lot.
304 *
305 * @return array{lot_number: ?string, material: ?string, material_code: ?string, supplier_lot_no: ?string, status: ?string, is_raw: bool, lines: array, steps: array}
306 */
307 private function lotLineJourney(MaterialLot $lot): array
308 {
309 $node = [
310 'lot_number' => $lot->lot_number,
311 'material' => $lot->material?->name,
312 'material_code' => $lot->material?->code,
313 'supplier_lot_no' => $lot->supplier_lot_no,
314 'status' => $lot->status,
315 'is_raw' => $lot->source_batch_id === null,
316 'lines' => [],
317 'steps' => [],
318 ];
319
320 if (! $lot->source_batch_id) {
321 return $node;
322 }
323
324 $batch = Batch::with([
325 'steps:id,batch_id,step_number,name,status,workstation_id,completed_by_id,completed_at',
326 'steps.workstation:id,name,code,line_id',
327 'steps.workstation.line:id,name,code',
328 'steps.completedBy:id,name',
329 ])->find($lot->source_batch_id);
330
331 if (! $batch) {
332 return $node;
333 }
334
335 $lines = []; // line_id => row, first occurrence wins (steps are step-ordered)
336
337 foreach ($batch->steps as $step) {
338 $line = $step->workstation?->line;
339 $node['steps'][] = [
340 'step_number' => $step->step_number,
341 'name' => $step->name,
342 'status' => $step->status,
343 'line' => $line?->name,
344 'workstation' => $step->workstation?->name,
345 'completed_by' => $step->completedBy?->name,
346 'completed_at' => $step->completed_at?->format('Y-m-d H:i'),
347 ];
348 if ($line && ! array_key_exists($line->id, $lines)) {
349 $lines[$line->id] = ['name' => $line->name, 'code' => $line->code];
350 }
351 }
352
353 $node['lines'] = array_values($lines);
354
355 return $node;
356 }

358 /**
359 * Backward trace from a material lot: the ingredient lots that fed into it.
360 *
361 * For an inbound raw lot this is terminal (supplier reference). For a
362 * batch-produced lot (source_batch_id set), it returns the lots consumed by
363 * that batch, recursing into each so the full ingredient tree is built.
364 */
365 public function backwardTraceLot(MaterialLot $lot, int $depth = 0): array
366 {
367 $node = [
368 'lot' => $lot->only(['id', 'lot_number', 'material_id', 'status']),
369 'material' => $lot->material?->only(['id', 'name', 'code']),
370 'supplier_lot_no' => $lot->supplier_lot_no,
371 'supplier_reference' => $lot->supplier_reference,
372 'source_container_no' => $lot->source_container_no,
373 'inspection_id' => $lot->inspection_id,
374 'source_batch_id' => $lot->source_batch_id,
375 'ingredients' => [],
376 'truncated' => false,
377 ];
378
379 if ($depth >= self::MAX_DEPTH) {
380 $node['truncated'] = true;
381
382 return $node;
383 }
384
385 if ($lot->source_batch_id) {
386 $batch = Batch::find($lot->source_batch_id);
387 if ($batch) {
388 $node['source_batch'] = $batch->only(['id', 'batch_number', 'lot_number', 'work_order_id']);
389 $node['ingredients'] = $this->batchInputLots($batch)
390 ->map(fn (MaterialLot $ingredient) => $this->backwardTraceLot($ingredient, $depth + 1))
391 ->values()
392 ->all();
393 }
394 }
395
396 return $node;
397 }

399 /**
400 * Full genealogy for a finished batch: which lots were consumed at each
401 * step, by which operator, when — plus the batch's own output lots.
402 */
403 public function batchGenealogy(Batch $batch): array
404 {
405 $batch->loadMissing([
406 'workOrder:id,order_no,product_type_id,status',
407 'workOrder.productType:id,name,code',
408 'steps:id,batch_id,name,step_number,status,workstation_id,started_by_id,completed_by_id,started_at,completed_at',
409 'steps.workstation:id,name,code,line_id',
410 'steps.completedBy:id,name',
411 'outputLots:id,lot_number,material_id,source_batch_id,status',
412 'outputLots.material:id,name,code',
413 ]);
414
415 $consumptions = BatchStepLotConsumption::query()
416 ->whereHas('batchStep', fn ($q) => $q->where('batch_id', $batch->id))
417 ->with([
418 'materialLot:id,lot_number,material_id,supplier_lot_no,source_container_no,source_batch_id,status',
419 'materialLot.material:id,name,code',
420 'batchStep:id,batch_id,name,step_number',
421 'recordedBy:id,name',
422 ])
423 ->orderBy('consumed_at')
424 ->get()
425 ->groupBy('batch_step_id');
426
427 return [
428 'batch' => $batch,
429 'consumptions_by_step' => $consumptions,
430 'distinct_input_lots' => $this->batchInputLots($batch),
431 ];
432 }

434 /**
435 * Distinct material lots consumed by any step of the given batch.

456 /**
457 * Full chain for a pallet: pallet → work order (incl. customer order) →

465 'workOrder.productType:id,name',
466 // withTrashed: recall must still resolve a pallet whose batch was later
467 // soft-deleted, otherwise the genealogy link silently disappears.
468 'batch' => fn ($q) => $q->withTrashed(),
469 ]);
470
471 return [
472 'pallet' => [
473 'id' => $pallet->id,
474 'pallet_no' => $pallet->pallet_no,
475 'status' => $pallet->status instanceof PalletStatus ? $pallet->status->value : $pallet->status,
476 'qty' => (int) $pallet->qty,
477 'location' => $pallet->location,
478 'created_at' => $pallet->created_at?->format('Y-m-d H:i'),
479 'shipped_at' => $pallet->shipped_at?->format('Y-m-d H:i'),
480 ],
481 'work_order' => $pallet->workOrder ? [
482 'order_no' => $pallet->workOrder->order_no,
483 'customer_order_no' => $pallet->workOrder->customer_order_no,
484 'product' => $pallet->workOrder->productType?->name,
485 ] : null,
486 'batch' => $pallet->batch ? $this->batchChain($pallet->batch) : null,
487 // Serialised units on the pallet, by carton - the shipment's exact contents.
488 'units' => SerialUnit::where('pallet_id', $pallet->id)
489 ->with('carton:id,carton_no')
490 ->orderBy('carton_id')->orderBy('serial_no')
491 ->get()
492 ->map(fn (SerialUnit $u) => [
493 'serial_no' => $u->serial_no,
494 'psn' => $u->psn,
495 'status' => $u->status,
496 'carton_no' => $u->carton?->carton_no,
497 'packed_at' => $u->packed_at?->format('Y-m-d H:i'),
498 ])->values(),
499 ];

502 /**
503 * The batch leg of the chain: per-step machine/line/operator + consumed
504 * lots, the distinct input lots, and the quality controls for the batch.
505 */
506 private function batchChain(Batch $batch): array
507 {
508 $genealogy = $this->batchGenealogy($batch);
509 $b = $genealogy['batch'];
510 $byStep = $genealogy['consumptions_by_step'];
511
512 // batchGenealogy already loaded steps, steps.workstation (incl. line_id)
513 // and completedBy. Only pull what's still missing - the workstation line,
514 // the step starter, the batch-level machine and the quality controls.
515 $b->loadMissing([
516 'steps.workstation.line:id,name',
517 'steps.startedBy:id,name',
518 'workstation:id,name',
519 'qualityChecks.checkedBy:id,name',
520 'qualityChecks.samples',
521 ]);
522
523 return [
524 'batch_number' => $b->batch_number,
525 'lot_number' => $b->lot_number,
526 'status' => $b->status,
527 'machine' => $b->workstation?->name,
528 'steps' => $b->steps->map(fn ($s) => [
529 'step_number' => $s->step_number,
530 'name' => $s->name,
531 'status' => $s->status,
532 'machine' => $s->workstation?->name,
533 'line' => $s->workstation?->line?->name,
534 'operator' => $s->completedBy?->name ?? $s->startedBy?->name,
535 'completed_at' => $s->completed_at ? Carbon::parse($s->completed_at)->format('Y-m-d H:i') : null,
536 'consumptions' => ($byStep[$s->id] ?? collect())->map(fn ($c) => [
537 'lot_number' => $c->materialLot?->lot_number,
538 'material' => $c->materialLot?->material?->name,
539 'quantity' => (float) $c->quantity_consumed,
540 ])->values(),
541 ])->values(),
542 'input_lots' => $genealogy['distinct_input_lots']->map(fn ($lot) => [
543 'material' => $lot->material?->name,
544 'lot_number' => $lot->lot_number,
545 'supplier_lot_no' => $lot->supplier_lot_no,
546 'status' => $lot->status,
547 ])->values(),
548 'quality_checks' => $b->qualityChecks->map(fn ($qc) => [
549 'all_passed' => (bool) $qc->all_passed,
550 'checked_by' => $qc->checkedBy?->name,
551 'checked_at' => $qc->checked_at ? Carbon::parse($qc->checked_at)->format('Y-m-d H:i') : null,
552 'samples' => $qc->samples->map(fn ($s) => [
553 'parameter' => $s->parameter_name,
554 'value' => $s->value_numeric ?? $s->value_boolean,
555 'passed' => $s->is_passed,
556 ])->values(),
557 ])->values(),
558 ];
559 }

590 /**
591 * Work orders with their pallets and batches, each batch carrying the lots it
592 * produced and the component lots it consumed.
593 *
594 * @param Builder<WorkOrder> $query
595 */
596 private function traceWorkOrders(Builder $query): Collection
597 {
598 $workOrders = $query
599 ->with([
600 'productType:id,name',
601 'pallets:id,work_order_id,batch_id,pallet_no,status',
602 'pallets.batch:id,lot_number',
603 'batches:id,work_order_id,batch_number,lot_number,status',
604 'batches.outputLots:id,source_batch_id,lot_number,material_id,status',
605 'batches.outputLots.material:id,name',
606 ])
607 ->orderByDesc('id')
608 ->get();
609
610 // Precompute the consumed (component) lots for every batch in one pass,
611 // instead of two queries per batch inside the map below (N+1).
612 $batchIds = $workOrders->pluck('batches')->flatten(1)->pluck('id')->filter()->unique()->all();
613 $inputLotsByBatch = $this->inputLotsByBatch($batchIds);
614
615 return $workOrders->map(fn ($wo) => [
616 'order_no' => $wo->order_no,
617 'product' => $wo->productType?->name,
618 'status' => $wo->status,
619 'pallets' => $wo->pallets->map(fn ($p) => [
620 'pallet_no' => $p->pallet_no,
621 'status' => $p->status instanceof PalletStatus ? $p->status->value : $p->status,
622 'batch_lot' => $p->batch?->lot_number,
623 ])->values(),
624 'batches' => $wo->batches->map(fn ($b) => [
625 'batch_number' => $b->batch_number,
626 'lot_number' => $b->lot_number,
627 'status' => $b->status,
628 'output_lots' => $b->outputLots->map(fn ($o) => [
629 'lot_number' => $o->lot_number,
630 'material' => $o->material?->name,
631 'status' => $o->status,
632 ])->values(),
633 'components' => collect($inputLotsByBatch[$b->id] ?? [])->map(fn ($lot) => [
634 'lot_number' => $lot->lot_number,
635 'material' => $lot->material?->name,
636 'supplier_lot_no' => $lot->supplier_lot_no,
637 'status' => $lot->status,
638 ])->values(),
639 ])->values(),
640 ])->values();
641 }

643 /**
644 * Distinct consumed (component) material lots for many batches at once,
645 * keyed by batch id - two queries total regardless of batch count.
646 *
647 * @param array<int, int> $batchIds
648 * @return array<int, array<int, MaterialLot>>
649 */
650 private function inputLotsByBatch(array $batchIds): array
651 {
652 if (empty($batchIds)) {
653 return [];
654 }
655
656 $pairs = BatchStepLotConsumption::query()
657 ->join('batch_steps', 'batch_step_lot_consumption.batch_step_id', '=', 'batch_steps.id')
658 ->whereNull('batch_steps.deleted_at')
659 ->whereIn('batch_steps.batch_id', $batchIds)
660 ->select('batch_steps.batch_id', 'batch_step_lot_consumption.material_lot_id')
661 ->distinct()
662 ->get();
663
664 $lots = MaterialLot::with('material:id,name,code')
665 ->whereIn('id', $pairs->pluck('material_lot_id')->unique())
666 ->get()
667 ->keyBy('id');
668
669 $map = [];
670 foreach ($pairs as $pair) {
671 $lot = $lots->get($pair->material_lot_id);
672 if ($lot) {
673 $map[$pair->batch_id][] = $lot;
674 }
675 }
676
677 return $map;
678 }

680 /**
681 * Resolve a free-text search to a result: a finished batch (by lot_number)
682 * or a material lot (by lot_number / supplier_lot_no / source_container_no).
683 *
684 * @return array{type: string, model: mixed}|null
685 */
686 public function resolve(string $term): ?array
687 {
688 $term = trim($term);
689 if ($term === '') {
690 return null;
691 }
692
693 // Pallet number (e.g. PAL-000001) - most specific, won't collide with lots.
694 $pallet = Pallet::where('pallet_no', $term)->first();
695 if ($pallet) {
696 return ['type' => 'pallet', 'model' => $pallet];
697 }
698
699 $batch = Batch::where('lot_number', $term)->first();
700 if ($batch) {
701 return ['type' => 'batch', 'model' => $batch];
702 }
703
704 // Grouped so the orWhere chain can't escape the global scopes
705 // (tenant + soft-delete) via AND/OR precedence.
706 $lot = MaterialLot::where(function ($query) use ($term) {
707 $query->where('lot_number', $term)
708 ->orWhere('supplier_lot_no', $term)
709 ->orWhere('source_container_no', $term);
710 })->first();
711 if ($lot) {
712 return ['type' => 'material_lot', 'model' => $lot];
713 }
714
715 return null;
716 }

718 /**
719 * Where a component identifier ended up: every unit it was scanned into,
720 * installed or since removed. This is the recall question for a bought-in
721 * part the system knows only by the serial on its label.
722 *
723 * @return array{identifier: string, units: Collection<int, array<string, mixed>>}
724 */
725 public function componentTrace(string $identifier): array
726 {
727 $bindings = SerialUnitComponent::where('identifier', $identifier)
728 ->with(['unit:id,serial_no,psn,status,work_order_id', 'unit.workOrder:id,order_no,product_type_id', 'unit.workOrder.productType:id,name', 'material:id,code,name', 'workstation:id,name', 'boundBy:id,name'])
729 ->orderByDesc('bound_at')->orderByDesc('id')
730 ->get();
731
732 return [
733 'identifier' => $identifier,
734 'material' => $bindings->first()?->material?->only(['code', 'name']),
735 'units' => $bindings->map(fn (SerialUnitComponent $c) => [
736 'serial_no' => $c->unit?->serial_no,
737 'psn' => $c->unit?->psn,
738 'status' => $c->unit?->status,
739 'work_order' => $c->unit?->workOrder?->order_no,
740 'product' => $c->unit?->workOrder?->productType?->name,
741 'installed' => $c->unbound_at === null,
742 'bound_at' => $c->bound_at?->format('Y-m-d H:i'),
743 'unbound_at' => $c->unbound_at?->format('Y-m-d H:i'),
744 'workstation' => $c->workstation?->name,
745 'bound_by' => $c->boundBy?->name,
746 ])->values(),
747 ];
748 }

750 /**
751 * The components inside a unit, as scanned: what each one resolved to, and
752 * the ones taken out since, so the trace shows the unit as built and as it is.
753 *
754 * @return Collection<int, array<string, mixed>>
755 */
756 public function installedComponents(SerialUnit $unit): Collection
757 {
758 return $unit->components()
759 ->with(['material:id,code,name', 'componentUnit:id,serial_no,status', 'materialLot:id,lot_number,supplier_lot_no', 'workstation:id,name', 'boundBy:id,name'])
760 ->orderBy('bound_at')
761 ->get()
762 ->map(fn (SerialUnitComponent $c) => [
763 'identifier' => $c->identifier,
764 'kind' => $c->component_serial_unit_id ? 'serial_unit' : ($c->material_lot_id ? 'material_lot' : 'identifier'),
765 'material' => $c->material?->name,
766 'material_code' => $c->material?->code,
767 'quantity' => (float) $c->quantity,
768 'component_status' => $c->componentUnit?->status,
769 'supplier_lot_no' => $c->materialLot?->supplier_lot_no,
770 'installed' => $c->unbound_at === null,
771 'bound_at' => $c->bound_at?->format('Y-m-d H:i'),
772 'unbound_at' => $c->unbound_at?->format('Y-m-d H:i'),
773 'unbind_reason' => $c->unbind_reason,
774 'workstation' => $c->workstation?->name,
775 'bound_by' => $c->boundBy?->name,
776 ]);
777 }
