92 /** Whether `$action` is legal from this order's current status. */
93 public static function canTransition(WorkOrder $workOrder, string $action): bool
94 {
95 $rule = self::transition($action);
96
97 return $rule !== null && in_array($workOrder->status, $rule['from'], true)
98 && (! in_array($rule['to'], [WorkOrder::STATUS_IN_PROGRESS, WorkOrder::STATUS_DONE], true) || ! $workOrder->plannedStartBlocker());
99 }

103 *
104 * @return array{ok: bool, message: string} `message` is the success line or

114 private static function applyTransitionLocked(WorkOrder $workOrder, string $action): array
115 {
116 $rule = self::transition($action);
117 if ($rule === null) {
118 return ['ok' => false, 'message' => 'Unknown action.'];
119 }
120
121 if (in_array($rule['to'], [WorkOrder::STATUS_IN_PROGRESS, WorkOrder::STATUS_DONE], true) && ($message = $workOrder->plannedStartBlocker())) {
122 return ['ok' => false, 'message' => $message];
123 }
124
125 if (! self::canTransition($workOrder, $action)) {
126 return ['ok' => false, 'message' => $rule['error']];
127 }
128
129 $workOrder->update(['status' => $rule['to']]);
130
131 return ['ok' => true, 'message' => __($rule['success'], ['order' => $workOrder->order_no])];
132 }

134 /**
135 * Apply one transition to many selected orders (a list's bulk-action bar).
136 *
137 * Orders the transition is illegal for are skipped rather than failing the
138 * whole request — a selection spanning mixed statuses is the normal case, and
139 * the caller is told how many were skipped. The whole batch is one
140 * transaction, so a mid-way failure leaves nothing half-applied.
141 *
142 * @param list<int> $ids
143 * @return string the message to flash back
144 */
145 public static function applyBulkTransition(array $ids, string $action): string
146 {
147 $rule = self::transitions()[$action];
148
149 [$total, $applied] = DB::transaction(function () use ($ids, $rule, $action) {
150 $orders = WorkOrder::whereIn('id', $ids)->orderBy('id')->lockForUpdate()->get();
151 $eligible = $orders->filter(fn (WorkOrder $w) => self::canTransition($w, $action));
152 // Individual updates preserve model events and live notifications.
153 $eligible->each(fn (WorkOrder $w) => $w->update(['status' => $rule['to']]));
154
155 return [$orders->count(), $eligible->count()];
156 });
157
158 $skipped = $total - $applied;
159 $message = __($rule['bulk'], ['count' => $applied]);
160
161 if ($skipped > 0) {
162 $message .= ' '.__(':count skipped (not applicable in their current status).', ['count' => $skipped]);
163 }
164
165 return $message;
166 }

255 /**
256 * Freeze the RELEASED engineering documents (#179) active for the order's
257 * product revision and product type at release time. Stored as compact
258 * references (id + revision + checksum + lifecycle) in the immutable process
259 * snapshot, so uploading a newer document revision later never rewrites this
260 * or any historical work order. Returns the snapshot unchanged when nothing
261 * is attached.
262 *
263 * @param array<string, mixed> $data
264 */
265 private function attachEngineeringSnapshot(?array $snapshot, array $data): ?array
266 {
267 $owners = [];
268 if (! empty($data['product_revision_id'])) {
269 $owners[] = ['product_revision', (int) $data['product_revision_id']];
270 }
271 if (! empty($data['product_type_id'])) {
272 $owners[] = ['product_type', (int) $data['product_type_id']];
273 }
274 if (! $owners) {
275 return $snapshot;
276 }
277
278 // Only freeze documents that are RELEASED and in their effectivity window
279 // at snapshot time — a future-dated or expired release is not active yet /
280 // any more and must not be captured onto the order. effective_from/to are
281 // DATE columns, so compare against today's DATE (not the full datetime), or
282 // a document effective "through today" is dropped once the clock passes
283 // midnight.
284 $today = now()->toDateString();
285 $docs = \App\Models\EngineeringDocument::query()
286 ->where('lifecycle_status', \App\Enums\EngineeringDocumentLifecycle::Released)
287 ->where(fn ($q) => $q->whereNull('effective_from')->orWhere('effective_from', '<=', $today))
288 ->where(fn ($q) => $q->whereNull('effective_to')->orWhere('effective_to', '>=', $today))
289 ->where(function ($q) use ($owners) {
290 foreach ($owners as [$type, $id]) {
291 $q->orWhere(fn ($qq) => $qq->where('entity_type', $type)->where('entity_id', $id));
292 }
293 })
294 ->get();
295
296 if ($docs->isEmpty()) {
297 return $snapshot;
298 }
299
300 $snapshot ??= [];
301 $snapshot['engineering_documents'] = $docs->map(fn ($d) => [
302 'document_id' => $d->id,
303 'entity_type' => $d->entity_type,
304 'entity_id' => $d->entity_id,
305 // Frozen display fields so the operator view is self-sufficient (no
306 // live lookup needed) and shows exactly what was released.
307 'original_filename' => $d->original_filename,
308 'entry_point' => $d->entry_point,
309 'file_size' => $d->file_size,
310 'revision' => $d->revision,
311 'package_type' => $d->package_type->value,
312 'checksum' => $d->checksum,
313 'lifecycle_at_release' => $d->lifecycle_status->value,
314 'released_at' => $d->released_at?->toIso8601String(),
315 ])->values()->all();
316 $snapshot['engineering_snapshotted_at'] = now()->toIso8601String();
317
318 return $snapshot;
319 }

321 /**
322 * Re-select the BOM(s) backing an existing work order and rebuild its
323 * snapshot. An empty selection falls back to the single active template for
324 * the product type. Blocked once production has started - the snapshot is
325 * the frozen recipe batches were built against, so changing it mid-flight
326 * would desync requirements from what was already allocated/consumed.
327 *
328 * @param array<int, int|string> $templateIds
329 *
330 * @throws \Exception
331 */
332 public function updateBomSelection(WorkOrder $workOrder, array $templateIds): WorkOrder
333 {
334 return DB::transaction(function () use ($workOrder, $templateIds) {
335 // Serialize against createBatch() on this work order: hold the row lock
336 // while checking for batches so a concurrent batch can't slip in between
337 // the check and the snapshot rewrite (the snapshot is the frozen recipe
338 // batches are built from).
339 WorkOrder::whereKey($workOrder->getKey())->lockForUpdate()->first();
340
341 if ($workOrder->batches()->exists()) {
342 throw new \Exception('Cannot change BOMs after production has started.');
343 }
344
345 $snapshot = $this->buildProcessSnapshot($workOrder->product_type_id, $templateIds);
346
347 // Preserve the immutable frozen blocks verbatim. The revision (#180) and
348 // engineering documents (#179) were snapshotted at work-order creation;
349 // BOM reselection rebuilds only the BOM/structure and must NOT re-query
350 // those from (possibly newer) live records, or the "immutable snapshot"
351 // guarantee would break. Copy them across unchanged.
352 $existing = $workOrder->process_snapshot ?? [];
353 foreach (['revision', 'engineering_documents', 'engineering_snapshotted_at'] as $key) {
354 if (array_key_exists($key, $existing)) {
355 $snapshot ??= [];
356 $snapshot[$key] = $existing[$key];
357 }
358 }
359
360 $workOrder->update(['process_snapshot' => $snapshot]);
361 $this->syncBomSelection($workOrder, $snapshot['bom_template_ids'] ?? []);
362
363 return $workOrder->fresh();
364 });
365 }

367 /**
368 * Build the immutable process snapshot for a work order from its BOM
369 * selection. The snapshot keeps the flat `bom` list every downstream
370 * consumer (requirements, allocation, costing) already reads - but that list
371 * is now the material-deduplicated UNION of all selected BOMs, so a material
372 * appearing in several BOMs contributes its summed requirement exactly once.
373 *
374 * Process structure (steps, timing) comes from the FIRST selected BOM (the
375 * "primary" template); the others contribute their materials only.
376 *
377 * @param array<int, int|string> $templateIds Explicit BOM (process template) selection; empty = legacy auto-pick.
378 * @return array<string, mixed>|null Null when no template applies (order without a BOM).
379 */
380 public function buildProcessSnapshot(?int $productTypeId, array $templateIds = []): ?array
381 {
382 $templates = $this->resolveBomTemplates($productTypeId, $templateIds);
383
384 if ($templates->isEmpty()) {
385 return null;
386 }
387
388 // Snapshot each selected template once, then stitch: structure from the
389 // primary, merged materials from all of them.
390 $snapshots = $templates->map(fn (ProcessTemplate $t) => $t->toSnapshot());
391
392 $snapshot = $snapshots->first();
393 $merged = $this->mergeBoms($snapshots->pluck('bom')->all());
394 // Steps come from the primary BOM only, so a `during` item pulled in from a
395 // secondary BOM may reference a step that isn't in this order's flow - it
396 // would then never be allocated. Degrade such items to `start` so the
397 // material is still consumed (single-BOM orders are unaffected: their
398 // `during` items always reference an existing step).
399 $stepNumbers = array_column($snapshot['steps'] ?? [], 'step_number');
400 $snapshot['bom'] = $this->degradeOrphanDuringItems($merged, $stepNumbers);
401 $snapshot['bom_template_ids'] = $templates->pluck('id')->values()->all();
402 $snapshot['bom_templates'] = $templates->map(fn (ProcessTemplate $t) => [
403 'id' => $t->id,
404 'name' => $t->name,
405 'version' => $t->version,
406 ])->values()->all();
407
408 return $snapshot;
409 }

411 /**
412 * Resolve the process templates backing an order, in selection order.
413 * Explicit ids win (deduplicated, order preserved); otherwise fall back to
414 * the single highest-version active template for the product type.
415 *
416 * @param array<int, int|string> $templateIds
417 * @return EloquentCollection<int, ProcessTemplate>
418 *
419 * @throws \InvalidArgumentException When a selected id is unknown or belongs to another product type.
420 */
421 protected function resolveBomTemplates(?int $productTypeId, array $templateIds): EloquentCollection
422 {
423 $ids = $this->normalizeIds($templateIds);
424
425 if (! empty($ids)) {
426 $found = ProcessTemplate::whereIn('id', $ids)
427 ->where('product_type_id', $productTypeId)
428 ->get()
429 ->keyBy('id');
430
431 // Fail loud rather than silently dropping an id that doesn't exist or
432 // belongs to another product type. HTTP requests are already validated,
433 // but this is a public service boundary (non-HTTP callers, deletion races).
434 if ($found->count() !== count($ids)) {
435 throw new \InvalidArgumentException(
436 'Every selected BOM must exist and belong to the work order product type.'
437 );
438 }
439
440 // Preserve the caller's selection order.
441 return new EloquentCollection(
442 collect($ids)->map(fn ($id) => $found->get($id))->values()->all(),
443 );
444 }
445
446 if ($productTypeId === null) {
447 return new EloquentCollection;
448 }
449
450 $active = ProcessTemplate::where('product_type_id', $productTypeId)
451 ->where('is_active', true)
452 ->orderBy('version', 'desc')
453 ->first();
454
455 return new EloquentCollection($active ? [$active] : []);
456 }

458 /**
459 * Merge several snapshot `bom` lists into one, keyed by material so a
460 * material shared across BOMs is summed rather than duplicated (the
461 * allocation engine keys on material per batch, so duplicates would silently
462 * drop the second line). Non-material rows (defensive) pass through verbatim.
463 *
464 * @param array<int, array<int, array<string, mixed>>> $bomLists
465 * @return array<int, array<string, mixed>>
466 */
467 protected function mergeBoms(array $bomLists): array
468 {
469 $merged = [];
470 $passthrough = [];
471
472 foreach ($bomLists as $bom) {
473 foreach ($bom as $row) {
474 $materialId = $row['material_id'] ?? null;
475 if ($materialId === null) {
476 $passthrough[] = $row;
477
478 continue;
479 }
480
481 $merged[$materialId] = isset($merged[$materialId])
482 ? $this->mergeBomRows($merged[$materialId], $row)
483 : $row;
484 }
485 }
486
487 return array_merge(array_values($merged), $passthrough);
488 }

490 /**
491 * Combine two BOM rows for the same material. When the scrap rate matches we
492 * sum the per-unit quantities and keep the rate; when it differs we fold each
493 * line's scrap into an effective per-unit quantity (rate → 0) so the total
494 * required quantity stays exact. Timing collapses to the earliest either BOM
495 * asks for the material.
496 *
497 * @param array<string, mixed> $a
498 * @param array<string, mixed> $b
499 * @return array<string, mixed>
500 */
501 protected function mergeBomRows(array $a, array $b): array
502 {
503 $qtyA = (float) ($a['quantity_per_unit'] ?? 0);
504 $qtyB = (float) ($b['quantity_per_unit'] ?? 0);
505 $scrapA = (float) ($a['scrap_percentage'] ?? 0);
506 $scrapB = (float) ($b['scrap_percentage'] ?? 0);
507
508 if (abs($scrapA - $scrapB) < 1e-9) {
509 $a['quantity_per_unit'] = $qtyA + $qtyB;
510 $a['scrap_percentage'] = $scrapA;
511 } else {
512 $a['quantity_per_unit'] = $qtyA * (1 + $scrapA / 100) + $qtyB * (1 + $scrapB / 100);
513 $a['scrap_percentage'] = 0.0;
514 }
515
516 $aAt = $a['consumed_at'] ?? 'start';
517 $bAt = $b['consumed_at'] ?? 'start';
518 $a['consumed_at'] = $this->earliestConsumedAt($aAt, $bAt);
519
520 $aStep = $a['step_number'] ?? null;
521 $bStep = $b['step_number'] ?? null;
522
523 if ($a['consumed_at'] === 'during') {
524 // Consume at the earliest step that needs the material, considering only
525 // the rows that are themselves `during` (a start/end row's step is
526 // irrelevant to timing), so selection order can't push it to a later step.
527 $duringSteps = array_filter(
528 [$aAt === 'during' ? $aStep : null, $bAt === 'during' ? $bStep : null],
529 fn ($s) => $s !== null,
530 );
531 $a['step_number'] = $duringSteps ? min($duringSteps) : ($aStep ?? $bStep);
532 } elseif ($aStep === null && $bStep !== null) {
533 $a['step_number'] = $bStep;
534 }
535
536 return $a;
537 }

539 /**
540 * Re-time any `during` BOM row whose step isn't part of this order's flow to
541 * `start`, so a material carried in from a secondary BOM is still consumed
542 * rather than silently stranded on a step that never runs.
543 *
544 * @param array<int, array<string, mixed>> $bom
545 * @param array<int, int> $stepNumbers Step numbers present in the primary snapshot.
546 * @return array<int, array<string, mixed>>
547 */
548 protected function degradeOrphanDuringItems(array $bom, array $stepNumbers): array
549 {
550 return array_map(function (array $row) use ($stepNumbers) {
551 if (($row['consumed_at'] ?? null) === 'during'
552 && ! in_array($row['step_number'] ?? null, $stepNumbers, true)) {
553 $row['consumed_at'] = 'start';
554 }
555
556 return $row;
557 }, $bom);
558 }

560 /** Earliest of two consumption timings (start < during < end). */
561 protected function earliestConsumedAt(string $a, string $b): string
562 {
563 $rank = ['start' => 0, 'during' => 1, 'end' => 2];
564
565 return ($rank[$a] ?? 0) <= ($rank[$b] ?? 0) ? $a : $b;
566 }

584 /**
585 * Normalize a raw id list: cast to int, drop empties, dedupe, keep order.
586 *
587 * @param array<int, int|string|null> $ids
588 * @return array<int, int>
589 */
590 protected function normalizeIds(array $ids): array
591 {
592 $seen = [];
593 $out = [];
594 foreach ($ids as $id) {
595 if ($id === null || $id === '') {
596 continue;
597 }
598 $id = (int) $id;
599 if (isset($seen[$id])) {
600 continue;
601 }
602 $seen[$id] = true;
603 $out[] = $id;
604 }
605
606 return $out;
607 }

609 /**
610 * Rebuild a work order's frozen configuration for an approved change (#182).
611 *
612 * Unlike {@see updateBomSelection()}, this DOES re-resolve the revision block
613 * (#180) and the released engineering documents (#179) from live records — that
614 * is the point of an engineering change: the order is deliberately moving onto a
615 * different revision and its current documents. The previous configuration is not
616 * lost, because the caller writes this as a NEW snapshot version and the old one
617 * stays readable as its own row.
618 *
619 * @param array<int, int|string> $templateIds
620 * @return array<string, mixed>|null Null when the product type has no applicable BOM.
621 */
622 public function rebuildProcessSnapshot(?int $productTypeId, array $templateIds, ?int $revisionId): ?array
623 {
624 $snapshot = $this->buildProcessSnapshot($productTypeId, $templateIds);
625
626 // No applicable BOM means there is no process structure to move onto. Report
627 // that as null so the caller keeps the current configuration: attaching the
628 // revision block to an empty snapshot would return a non-null configuration
629 // with no `steps`, and regenerating pending batches from it would delete
630 // their steps and rebuild nothing.
631 if ($snapshot === null) {
632 return null;
633 }
634
635 $snapshot = $this->attachRevisionSnapshot($snapshot, $revisionId);
636
637 return $this->attachEngineeringSnapshot($snapshot, [
638 'product_revision_id' => $revisionId,
639 'product_type_id' => $productTypeId,
640 ]);
641 }

655 /**
656 * Update an existing work order.
657 */
658 public function updateWorkOrder(WorkOrder $workOrder, array $data): WorkOrder
659 {
660 return DB::transaction(function () use ($workOrder, $data) {
661 return $this->updateWorkOrderLocked(WorkOrder::whereKey($workOrder->id)->lockForUpdate()->firstOrFail(), $data);
662 });
663 }
664
665 private function updateWorkOrderLocked(WorkOrder $workOrder, array $data): WorkOrder
666 {
667 // Don't allow updates to completed work orders
668 if ($workOrder->status === WorkOrder::STATUS_DONE) {
669 throw new \Exception('Cannot update completed work order');
670 }
671
672 $workOrder->update([
673 'customer_order_no' => array_key_exists('customer_order_no', $data)
674 ? $data['customer_order_no']
675 : $workOrder->customer_order_no,
676 'customer_id' => array_key_exists('customer_id', $data)
677 ? $data['customer_id']
678 : $workOrder->customer_id,
679 'planned_qty' => $data['planned_qty'] ?? $workOrder->planned_qty,
680 'unit_price' => array_key_exists('unit_price', $data)
681 ? $data['unit_price']
682 : $workOrder->unit_price,
683 'priority' => $data['priority'] ?? $workOrder->priority,
684 'due_date' => $data['due_date'] ?? $workOrder->due_date,
685 'planned_start_at' => array_key_exists('planned_start_at', $data) ? $data['planned_start_at'] : $workOrder->planned_start_at,
686 'description' => $data['description'] ?? $workOrder->description,
687 ]);
688
689 return $workOrder->fresh();
690 }

692 /**
693 * Create a new batch for a work order.
694 */
695 public function createBatch(WorkOrder $workOrder, float $targetQty, ?int $workstationId = null, ?string $lotNumber = null): Batch
696 {
697 return DB::transaction(function () use ($workOrder, $targetQty, $workstationId, $lotNumber) {
698 // Serialize against updateBomSelection() (see there): holding the
699 // work-order row lock guarantees this batch is built from the committed
700 // snapshot, never one that a concurrent BOM change is mid-rewrite.
701 WorkOrder::whereKey($workOrder->getKey())->lockForUpdate()->first();
702 $workOrder->refresh();
703
704 // Calculate next batch number
705 $lastBatch = $workOrder->batches()->reorder('batch_number', 'desc')->first();
706 $batchNumber = $lastBatch ? $lastBatch->batch_number + 1 : 1;
707
708 // Create batch
709 $batch = Batch::create([
710 'work_order_id' => $workOrder->id,
711 // Stamp the configuration version this batch is generated from (#182),
712 // so production before and after an applied change stays attributable
713 // to the configuration it was actually built under.
714 'snapshot_version' => $workOrder->snapshot_version ?? 1,
715 'batch_number' => $batchNumber,
716 'target_qty' => $targetQty,
717 'produced_qty' => 0,
718 'status' => Batch::STATUS_PENDING,
719 'workstation_id' => $workstationId,
720 'lot_number' => $lotNumber,
721 'lot_assigned_at' => $lotNumber ? Batch::LOT_ON_START : null,
722 ]);
723
724 // Create batch steps from process snapshot (skipped if no snapshot)
725 if (! empty($workOrder->process_snapshot)) {
726 $this->createBatchStepsFromSnapshot($batch, $workOrder->process_snapshot);
727 }
728
729 return $batch;
730 });
731 }

733 /**
734 * Create batch steps from work order process snapshot.
735 */
736 protected function createBatchStepsFromSnapshot(Batch $batch, array $processSnapshot): void
737 {
738 $steps = $processSnapshot['steps'] ?? [];
739
740 // Resolve the pre-selected step per variant group: the one flagged as
741 // default, else the lowest step_number in the group. Siblings start
742 // SKIPPED so the operator sees only the chosen path (and can switch).
743 $chosen = [];
744 $explicit = [];
745 foreach ($steps as $s) {
746 $group = $s['variant_group'] ?? null;
747 if ($group === null) {
748 continue;
749 }
750 $num = $s['step_number'];
751 if (! empty($s['is_default_variant'])) {
752 if (empty($explicit[$group])) {
753 $chosen[$group] = $num;
754 $explicit[$group] = true;
755 }
756 } elseif (empty($explicit[$group])) {
757 $chosen[$group] = isset($chosen[$group]) ? min($chosen[$group], $num) : $num;
758 }
759 }
760
761 foreach ($steps as $stepData) {
762 $group = $stepData['variant_group'] ?? null;
763 $status = BatchStep::STATUS_PENDING;
764 if ($group !== null && isset($chosen[$group]) && $stepData['step_number'] !== $chosen[$group]) {
765 $status = BatchStep::STATUS_SKIPPED;
766 }
767
768 BatchStep::create([
769 'batch_id' => $batch->id,
770 'step_number' => $stepData['step_number'],
771 'name' => $stepData['name'],
772 'kind' => $stepData['kind'] ?? \App\Models\TemplateStep::KIND_PRODUCTION,
773 'config' => $stepData['config'] ?? null,
774 'instruction' => $stepData['instruction'] ?? null,
775 'requires_confirmation' => $stepData['requires_confirmation'] ?? false,
776 'workstation_id' => $stepData['workstation_id'] ?? null,
777 // ISA-95 (#52): carry the required Equipment Class + planning times
778 // down from the snapshot so pool dispatch and the actual-vs-standard
779 // comparison have their reference data on the batch step.
780 'workstation_type_id' => $stepData['workstation_type_id'] ?? null,
781 'estimated_duration_minutes' => $stepData['estimated_duration_minutes'] ?? null,
782 'setup_time_minutes' => $stepData['setup_time_minutes'] ?? null,
783 'run_time_per_unit_minutes' => $stepData['run_time_per_unit_minutes'] ?? null,
784 'status' => $status,
785 'is_optional' => $stepData['is_optional'] ?? false,
786 'variant_group' => $group,
787 ]);
788 }
789
790 // Mark the first eligible step(s) READY ("next in line") so the operator
791 // can start straight away; the rest stay PENDING until their turn.
792 $batch->promoteReadySteps();
793 }

795 /**
796 * Update work order status based on batches and issues.
797 */
798 public function updateWorkOrderStatus(WorkOrder $workOrder): void
799 {
800 // Check if blocked by issues
801 if ($workOrder->isBlocked()) {
802 $workOrder->update(['status' => WorkOrder::STATUS_BLOCKED]);
803
804 return;
805 }
806
807 $hasInProgressBatch = $workOrder->batches()
808 ->where('status', Batch::STATUS_IN_PROGRESS)
809 ->exists();
810
811 // Check if complete. In transfer flow the produced quantity is rolled up
812 // live, so it can reach the plan while the last station is still
813 // finishing its step — the order closes with its last batch, not before.
814 // Whole-batch flow keeps closing the moment the plan is reached.
815 $waitForBatches = $hasInProgressBatch && ProductionFlow::isTransfer();
816 if ($workOrder->isComplete() && ! $waitForBatches) {
817 $workOrder->update([
818 'status' => WorkOrder::STATUS_DONE,
819 'completed_at' => now(),
820 ]);
821
822 return;
823 }
824
825 // A finished transfer batch can leave a shortfall (including 100% scrap).
826 // The order has started and still needs replacement production.
827 $hasFinishedTransferBatch = ProductionFlow::isTransfer() && $workOrder->batches()
828 ->where('status', Batch::STATUS_DONE)->exists();
829 if ($hasInProgressBatch || $hasFinishedTransferBatch) {
830 $workOrder->update(['status' => WorkOrder::STATUS_IN_PROGRESS]);
831
832 return;
833 }
834
835 // Otherwise keep as pending
836 if ($workOrder->status !== WorkOrder::STATUS_PENDING &&
837 $workOrder->status !== WorkOrder::STATUS_DONE) {
838 $workOrder->update(['status' => WorkOrder::STATUS_PENDING]);
839 }
840 }

842 /**
843 * Get work orders for a specific user's assigned lines.
844 *
845 * @param \App\Models\User $user
846 * @return \Illuminate\Database\Eloquent\Collection
847 */
848 public function getWorkOrdersForUser($user, array $filters = [])
849 {
850 $query = WorkOrder::forUser($user)
851 ->with(['line', 'productType', 'batches.steps']);
852
853 if ($user->hasRole('Operator') && ! $user->hasAnyRole(['Admin', 'Supervisor'])) {
854 $query->availableForProduction();
855 }
856
857 // Apply filters
858 if (isset($filters['status'])) {
859 $query->status($filters['status']);
860 }
861
862 if (isset($filters['line_id'])) {
863 $query->forLine($filters['line_id']);
864 }
865
866 // Default ordering
867 $query->byPriority();
868
869 return $query->get();
870 }
