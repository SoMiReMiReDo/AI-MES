1344/**
1345 * Confirm-actual-times modal (#52), shown at step completion only for steps with

1350function ConfirmTimesModal({ step, onClose }) {
1351 const startedAt = step.started_at ? new Date(step.started_at).getTime() : null;
1352 const initialElapsed = startedAt ? Math.max(0, Math.round((Date.now() - startedAt) / 60000)) : 0;

1356 const [setup, setSetup] = useState(step.setup_time_minutes != null ? String(initialSetup) : '');
1357 const [run, setRun] = useState(String(Math.max(0, initialElapsed - initialSetup)));

1376 actual_run_minutes: run === '' ? null : Number(run),
1377 }, {
1378 preserveScroll: true,
1379 // Close only on success; keep the modal open (with values intact) so a
1380 // 422 or completion error stays visible instead of being dismissed.
1381 onSuccess: () => onClose(),
1382 onFinish: () => setSaving(false),
1383 });
1384 };

1392 <div>
1393 <div className={labelCls}>{__('Actual elapsed (minutes)')}</div>

1397 <div>
1398 <div className={labelCls}>{__('Actual setup (minutes)')}</div>

1400 </div>
