20/**
21 * Evaluates configurable quality-control triggers (#105) against production
22 * events and creates QualityControlTasks (the "a control is due" prompt). Also
23 * records the result of a task (reusing QualityCheckService) and raises a
24 * non-conformance Issue when a control fails.
25 *
26 * Firing is invoked synchronously from the production lifecycle:
27 * - in_production / every_n_units → BatchService
28 * - after_downtime / after_setup → DowntimeService
29 * - every_n_minutes → FireDueQualityTriggersCommand (scheduler)
30 * - roaming → created manually via the controller
31 */
32class QualityTriggerService
33{
34 public function __construct(
35 private QualityCheckService $qualityCheckService,
36 private IssueService $issueService,
37 ) {}

39 /**
40 * Batch just entered production — fire every active in_production trigger
41 * whose scope matches, once per batch.
42 */
43 public function fireInProduction(Batch $batch): void
44 {
45 $triggers = QualityControlTrigger::active()
46 ->ofType(QualityControlTrigger::TYPE_IN_PRODUCTION)
47 ->get();
48
49 foreach ($triggers as $trigger) {
50 if (! $trigger->matchesBatch($batch)) {
51 continue;
52 }
53 if ($this->batchTaskExists($trigger, $batch)) {
54 continue;
55 }
56 $this->createTaskForBatch($trigger, $batch, __('Batch entered production'));
57 }
58 }

60 /**
61 * Batch produced quantity changed — for every active every_n_units trigger,
62 * ensure one task exists per N cumulative units produced on the work order.
63 */
64 public function fireForUnits(Batch $batch): void
65 {
66 $workOrder = $batch->workOrder;
67 if ($workOrder === null) {
68 return;
69 }
70
71 $triggers = QualityControlTrigger::active()
72 ->ofType(QualityControlTrigger::TYPE_EVERY_N_UNITS)
73 ->where('threshold_n', '>', 0)
74 ->get();
75
76 $produced = (float) $workOrder->produced_qty;
77
78 foreach ($triggers as $trigger) {
79 if (! $trigger->matchesBatch($batch)) {
80 continue;
81 }
82
83 $shouldHave = (int) floor($produced / $trigger->threshold_n);
84 $existing = QualityControlTask::where('quality_control_trigger_id', $trigger->id)
85 ->where('work_order_id', $workOrder->id)
86 ->count();
87
88 // Create one task per N-unit mark still missing. No cap — a later
89 // fireForUnits() may never run (e.g. final batch), so the backlog
90 // must be cleared now or required controls would be lost.
91 $toCreate = max($shouldHave - $existing, 0);
92 for ($i = 0; $i < $toCreate; $i++) {
93 $unitMark = ($existing + $i + 1) * $trigger->threshold_n;
94 $this->createTaskForBatch(
95 $trigger,
96 $batch,
97 __('Every :n units (reached :mark)', ['n' => $trigger->threshold_n, 'mark' => $unitMark]),
98 );
99 }
100 }
101 }

103 /**
104 * A downtime just closed — fire after_setup (changeover) or after_downtime
105 * (unplanned loss) triggers for the line, respecting the minimum-minutes
106 * threshold. Planned downtime never triggers a control.
107 */
108 public function fireAfterDowntime(ProductionDowntime $downtime): void
109 {
110 $downtime->loadMissing('reason');
111 $kind = $downtime->reason?->kind;
112 $kind = $kind instanceof DowntimeKind ? $kind : DowntimeKind::tryFrom((string) $kind);
113
114 $type = match ($kind) {
115 DowntimeKind::Changeover => QualityControlTrigger::TYPE_AFTER_SETUP,
116 DowntimeKind::Unplanned => QualityControlTrigger::TYPE_AFTER_DOWNTIME,
117 default => null, // Planned (or unknown) → no control
118 };
119
120 if ($type === null) {
121 return;
122 }
123
124 $durationMinutes = (int) ($downtime->duration_minutes ?? 0);
125
126 $triggers = QualityControlTrigger::active()->ofType($type)->get();
127
128 // A control is recorded against the batch that was running on this line.
129 // No active batch → nothing to check (and the resulting task couldn't be
130 // completed), so we don't fire.
131 $activeBatch = $this->activeBatchForLine($downtime->line_id, $downtime->workstation_id);
132 if ($activeBatch === null) {
133 return;
134 }
135
136 foreach ($triggers as $trigger) {
137 if ($trigger->line_id !== null && $trigger->line_id !== $downtime->line_id) {
138 continue;
139 }
140 if ($trigger->workstation_id !== null && $trigger->workstation_id !== $downtime->workstation_id) {
141 continue;
142 }
143 if ($trigger->product_type_id !== null
144 && $trigger->product_type_id !== $activeBatch->workOrder?->product_type_id) {
145 continue;
146 }
147 if ($trigger->downtime_min_minutes !== null && $durationMinutes < $trigger->downtime_min_minutes) {
148 continue;
149 }
150
151 $reason = $type === QualityControlTrigger::TYPE_AFTER_SETUP
152 ? __('After setup / changeover')
153 : __('After downtime (:min min)', ['min' => $durationMinutes]);
154
155 $this->createTask($trigger, [
156 'line_id' => $downtime->line_id,
157 'workstation_id' => $downtime->workstation_id ?? $activeBatch->workstation_id,
158 'batch_id' => $activeBatch->id,
159 'work_order_id' => $activeBatch->work_order_id,
160 ], $reason);
161 }
162 }

164 /**
165 * Scheduler tick — for every active every_n_minutes trigger, fire on each
166 * matching in-progress batch whose last task is older than N minutes.
167 *
168 * @return int number of tasks created
169 */
170 public function fireDueTimeTriggers(): int
171 {
172 $triggers = QualityControlTrigger::active()
173 ->ofType(QualityControlTrigger::TYPE_EVERY_N_MINUTES)
174 ->where('threshold_n', '>', 0)
175 ->get();
176
177 if ($triggers->isEmpty()) {
178 return 0;
179 }
180
181 $created = 0;
182 $batches = Batch::where('status', Batch::STATUS_IN_PROGRESS)->with('workOrder')->get();
183
184 foreach ($triggers as $trigger) {
185 foreach ($batches as $batch) {
186 if (! $trigger->matchesBatch($batch)) {
187 continue;
188 }
189
190 $lastFiredAt = QualityControlTask::where('quality_control_trigger_id', $trigger->id)
191 ->where('batch_id', $batch->id)
192 ->max('fired_at');
193
194 $cutoff = Carbon::now()->subMinutes($trigger->threshold_n);
195 if ($lastFiredAt !== null && Carbon::parse($lastFiredAt)->greaterThan($cutoff)) {
196 continue;
197 }
198
199 $this->createTaskForBatch(
200 $trigger,
201 $batch,
202 __('Every :n minutes', ['n' => $trigger->threshold_n]),
203 );
204 $created++;
205 }
206 }
207
208 return $created;
209 }

211 /**
212 * Manually raise a roaming / ad-hoc control task for a roaming trigger.
213 * A batch is required so the control can be recorded and gated like any
214 * other (a control with nothing to record against would be a dead task).
215 */
216 public function createRoamingTask(QualityControlTrigger $trigger, array $links = [], ?string $reason = null): QualityControlTask
217 {
218 if (($links['batch_id'] ?? null) === null) {
219 throw new \DomainException('A roaming quality control must target a batch.');
220 }
221
222 return $this->createTask($trigger, $links, $reason ?? __('Roaming check'));
223 }

225 /**
226 * Record the result of a control task by performing the underlying
227 * QualityCheck against its batch. A failing result raises a non-conformance
228 * Issue (blocking when the trigger is blocking).
229 */
230 public function performTask(QualityControlTask $task, User $user, array $samples, ?float $productionQuantity = null, ?string $notes = null, ?Pallet $pallet = null): QualityControlTask
231 {
232 if ($task->batch_id === null) {
233 throw new \DomainException('This control is not tied to a batch and cannot record a quality check.');
234 }
235
236 return DB::transaction(function () use ($task, $user, $samples, $productionQuantity, $notes, $pallet) {
237 // Lock the row so two concurrent perform/skip calls can't both pass
238 // the open-state check and commit conflicting terminal actions.
239 $task = QualityControlTask::whereKey($task->getKey())->lockForUpdate()->firstOrFail();
240 if (! $task->isOpen()) {
241 throw new \DomainException('This control has already been completed.');
242 }
243 $task->loadMissing(['batch', 'trigger']);
244
245 $check = $this->qualityCheckService->performCheck(
246 $task->batch,
247 $user,
248 $samples,
249 $productionQuantity,
250 $task->trigger->template,
251 $notes,
252 $pallet,
253 );
254
255 $issueId = null;
256 if (! $check->all_passed) {
257 $issueId = $this->raiseFailureIssue($task, $check, $user);
258 }
259
260 $task->update([
261 'status' => QualityControlTask::STATUS_DONE,
262 'quality_check_id' => $check->id,
263 'issue_id' => $issueId,
264 'completed_at' => now(),
265 'completed_by_id' => $user->id,
266 ]);
267
268 return $task->fresh(['qualityCheck', 'issue']);
269 });
270 }
271
272 public function skipTask(QualityControlTask $task, User $user): QualityControlTask
273 {
274 return DB::transaction(function () use ($task, $user) {
275 $task = QualityControlTask::whereKey($task->getKey())->lockForUpdate()->firstOrFail();
276 if (! $task->isOpen()) {
277 throw new \DomainException('This control has already been completed.');
278 }
279
280 $task->update([
281 'status' => QualityControlTask::STATUS_SKIPPED,
282 'completed_at' => now(),
283 'completed_by_id' => $user->id,
284 ]);
285
286 return $task->fresh();
287 });
288 }

290 /** Has this batch already got a task for this trigger? (in_production dedupe) */
291 private function batchTaskExists(QualityControlTrigger $trigger, Batch $batch): bool
292 {
293 return QualityControlTask::where('quality_control_trigger_id', $trigger->id)
294 ->where('batch_id', $batch->id)
295 ->exists();
296 }
297
298 private function activeBatchForLine(?int $lineId, ?int $workstationId): ?Batch
299 {
300 if ($lineId === null) {
301 return null;
302 }
303
304 return Batch::where('status', Batch::STATUS_IN_PROGRESS)
305 ->when($workstationId !== null, fn ($q) => $q->where('workstation_id', $workstationId))
306 ->whereHas('workOrder', fn ($q) => $q->where('line_id', $lineId))
307 ->with('workOrder')
308 ->latest('started_at')
309 ->first();
310 }

312 private function createTaskForBatch(QualityControlTrigger $trigger, Batch $batch, string $reason): QualityControlTask

314 return $this->createTask($trigger, [

317 'workstation_id' => $batch->workstation_id,
318 'line_id' => $batch->workOrder?->line_id,

322 private function createTask(QualityControlTrigger $trigger, array $links, string $reason): QualityControlTask
323 {
324 return QualityControlTask::create(array_merge([
325 'quality_control_trigger_id' => $trigger->id,
326 'status' => QualityControlTask::STATUS_DUE,
327 'due_reason' => $reason,
328 'fired_at' => now(),
329 ], $links));
330 }

332 private function raiseFailureIssue(QualityControlTask $task, QualityCheck $check, User $user): ?int
333 {
334 $blocking = (bool) $task->trigger->is_blocking;
335 $issueTypeId = $this->resolveIssueTypeId($blocking);
336 if ($issueTypeId === null) {
337 return null;
338 }
339
340 $issue = $this->issueService->createIssue([
341 'work_order_id' => $task->work_order_id,
342 'issue_type_id' => $issueTypeId,
343 'source' => Issue::SOURCE_IN_PROCESS,
344 'title' => __('In-process QC fail: :reason', ['reason' => $task->due_reason ?? __('quality control')]),
345 'description' => __('Quality control ":trigger" failed during production (check #:check).', [
346 'trigger' => $task->trigger->name,
347 'check' => $check->id,
348 ]),
349 'reported_by_id' => $user->id,
350 ]);
351
352 return $issue->id;
353 }

355 /**
356 * Failing blocking controls raise a blocking non-conformance (stops the work
357 * order, like inbound QC); non-blocking controls raise an informational one.
358 */
359 private function resolveIssueTypeId(bool $blocking): ?int
360 {
361 $codes = $blocking
362 ? ['IN_PROCESS_QC_FAIL', 'QUALITY_ISSUE', 'OTHER']
363 : ['QUALITY_ISSUE', 'OTHER'];
364
365 foreach ($codes as $code) {
366 $id = IssueType::where('code', $code)->value('id');
367 if ($id !== null) {
368 return $id;
369 }
370 }
371
372 return null;
373 }
