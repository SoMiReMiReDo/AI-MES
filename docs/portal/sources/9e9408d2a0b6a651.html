49 /**
50 * The packing step an order is on right now: running, else the one ready to
51 * run. With the unit's batch known, that batch's step - an order packed in
52 * several batches counts each unit on its own batch.
53 */
54 public function packingStepFor(?int $workOrderId, ?int $batchId = null): ?BatchStep
55 {
56 if (! $workOrderId) {
57 return null;
58 }
59 if ($batchId) {
60 $step = BatchStep::query()
61 ->where('kind', TemplateStep::KIND_PACKING)
62 ->whereIn('status', [BatchStep::STATUS_READY, BatchStep::STATUS_IN_PROGRESS])
63 ->where('batch_id', $batchId)
64 ->orderByRaw("CASE WHEN status = 'IN_PROGRESS' THEN 0 ELSE 1 END")
65 ->orderBy('step_number')
66 ->first();
67 if ($step) {
68 return $step;
69 }
70 }
71
72 return BatchStep::query()
73 ->where('kind', TemplateStep::KIND_PACKING)
74 ->whereIn('status', [BatchStep::STATUS_READY, BatchStep::STATUS_IN_PROGRESS])
75 ->whereHas('batch', fn ($q) => $q->where('work_order_id', $workOrderId))
76 ->orderByRaw("CASE WHEN status = 'IN_PROGRESS' THEN 0 ELSE 1 END")
77 ->orderBy('step_number')
78 ->first();
79 }

81 /**
82 * The packaging the routing's BOM puts on this packing step - cartons,
83 * pallets, foil - with what the batch needs of each, what was reserved and
84 * from which lots, and what has been used so far. `needs_pick` says the
85 * station has to name the lot before packing starts (lot tracking on).
86 *
87 * @return array<int, array<string, mixed>>
88 */
89 public function packingMaterials(BatchStep $step): array
90 {
91 $batch = $step->batch;
92 // The snapshot is read afresh: the station's step list loads the order
93 // with a handful of columns, and a partial relation has no snapshot.
94 $bom = $batch ? (WorkOrder::find($batch->work_order_id)?->process_snapshot['bom'] ?? []) : [];
95 $lotTracking = $this->lotPicking->isLotTrackingEnabled();
96
97 $out = [];
98 foreach ($bom as $line) {
99 if ((int) ($line['step_number'] ?? 0) !== (int) $step->step_number || empty($line['material_id'])) {
100 continue;
101 }
102 $material = Material::find($line['material_id']);
103 if (! $material) {
104 continue;
105 }
106 $required = round((float) $line['quantity_per_unit'] * (float) $batch->target_qty * (1 + (float) ($line['scrap_percentage'] ?? 0) / 100), 4);
107 $allocation = MaterialAllocation::where('batch_id', $batch->id)->where('material_id', $material->id)
108 ->with('lotPicks.lot:id,lot_number')->first();
109
110 $out[] = [
111 'material_id' => $material->id,
112 'code' => $line['material_code'] ?? $material->code,
113 'name' => $line['material_name'] ?? $material->name,
114 'unit_of_measure' => $line['unit_of_measure'] ?? $material->unit_of_measure,
115 'per' => $line['per'] ?? BomItem::PER_UNIT,
116 'quantity_per_basis' => (float) ($line['quantity_per_basis'] ?? $line['quantity_per_unit']),
117 'required_qty' => $required,
118 'status' => $allocation?->status ?? 'pending',
119 'allocated_qty' => (float) ($allocation?->allocated_qty ?? 0),
120 'consumed_qty' => (float) ($allocation?->consumed_qty ?? 0),
121 'lots' => $allocation ? $allocation->lotPicks->map(fn ($pick) => ['lot_number' => $pick->lot?->lot_number, 'qty' => (float) $pick->picked_qty])->values()->all() : [],
122 'needs_pick' => $lotTracking && ! $allocation && $material->tracking_type !== 'serial' && $step->status === BatchStep::STATUS_READY,
123 ];
124 }
125
126 return $out;
127 }

129 /** Why a scan must wait: the packaging lots have not been picked yet. Null when packing may go on. */
130 public function packingMaterialsGate(BatchStep $step): ?string
131 {
132 if ($step->status !== BatchStep::STATUS_READY) {
133 return null;
134 }
135 $waiting = collect($this->packingMaterials($step))->filter(fn ($m) => $m['needs_pick'])->pluck('name');
136 if ($waiting->isEmpty()) {
137 return null;
138 }
139
140 return __('Pick the packaging lots first (:materials), then scan.', ['materials' => $waiting->implode(', ')]);
141 }

144 * Start the packing step from the station with the lots the operator
145 * picked - the same start the operator queue does, so reservations and
146 * genealogy land the same way.

148 * @param array<int|string, array<int, array{material_lot_id: int|string, picked_qty: int|float|string}>> $picks

152 public function startPackingStep(BatchStep $step, User $operator, array $picks = []): BatchStep
153 {
154 if ($step->status !== BatchStep::STATUS_READY) {
155 throw new BindingException(__('Packing step :name is not ready to start.', ['name' => $step->name]));

157 $byMaterial = [];

168 /**
169 * A carton or pallet closed: its packaging (the BOM lines counted per
170 * carton / per pallet) is used up, one more each. Booked on the reservation
171 * so the station's "used" count is live; the batch's completion settles the
172 * rest as it always did. Never in the way of the closing itself.
173 */
174 private function consumePackaging(?int $workOrderId, string $basis): void
175 {
176 $step = $this->packingStepFor($workOrderId);
177 if (! $step || $step->status !== BatchStep::STATUS_IN_PROGRESS) {
178 return;
179 }
180 try {
181 foreach ($this->packingMaterials($step) as $line) {
182 if ($line['per'] !== $basis || $line['status'] !== MaterialAllocation::STATUS_ALLOCATED) {
183 continue;
184 }
185 $allocation = MaterialAllocation::where('batch_id', $step->batch_id)->where('material_id', $line['material_id'])->first();
186 if ($allocation) {
187 $this->allocations->recordConsumption($allocation, min((float) $allocation->allocated_qty, $line['consumed_qty'] + $line['quantity_per_basis']), (float) $allocation->scrap_qty);
188 }
189 }
190 } catch (\Throwable $e) {
191 Log::info('Packaging consumption not booked: '.$e->getMessage(), ['batch_step_id' => $step->id, 'basis' => $basis]);
192 }
193 }

201 /**
202 * The packing step's configuration that governs an order: the frozen batch
203 * step first, the order's snapshot when no batch exists yet, nothing when
204 * the routing has no packing step (then the station's defaults apply).
205 *
206 * @return array<string, mixed>
207 */
208 public function packingConfigFor(?int $workOrderId): array
209 {
210 if (! $workOrderId) {
211 return [];
212 }
213 if (array_key_exists($workOrderId, $this->configCache)) {
214 return $this->configCache[$workOrderId];
215 }
216
217 $step = BatchStep::query()
218 ->where('kind', TemplateStep::KIND_PACKING)
219 ->whereHas('batch', fn ($q) => $q->where('work_order_id', $workOrderId))
220 ->orderBy('step_number')
221 ->first();
222 $config = $step?->config;
223 if ($config === null) {
224 foreach (WorkOrder::find($workOrderId)?->process_snapshot['steps'] ?? [] as $snapshotStep) {
225 if (($snapshotStep['kind'] ?? null) === TemplateStep::KIND_PACKING) {
226 $config = $snapshotStep['config'] ?? [];
227 break;
228 }
229 }
230 }
231
232 return $this->configCache[$workOrderId] = $config ?? [];
233 }

249 /**
250 * A pallet that reached the packing step's size closes itself, the way a
251 * full carton does: counting closed cartons when the step packs into
252 * cartons, units when it packs straight onto pallets. Returns whether it
253 * closed just now, so the station can print its label and drop the bench.
254 */
255 public function closePalletIfFull(?Pallet $pallet): bool
256 {
257 if (! $pallet || ! $pallet->isOpen()) {
258 return false;
259 }
260 $config = $this->packingConfigFor($pallet->work_order_id);
261 $capacity = (int) ($config['pallet_capacity'] ?? 0);
262 if ($capacity < 1) {
263 return false;
264 }
265 $filled = ($config['unit'] ?? 'carton') === 'pallet'
266 ? $pallet->units()->count()
267 : $pallet->cartons()->where('status', UnitCarton::STATUS_CLOSED)->count();
268 if ($filled < $capacity) {
269 return false;
270 }
271
272 // Only the request that actually closes it books the pallet's packaging.
273 $closed = Pallet::whereKey($pallet->id)->where('status', PalletStatus::Open->value)
274 ->update(['status' => PalletStatus::Closed->value, 'active_by_id' => null, 'active_workstation_id' => null, 'activated_at' => null]);
275 if ($closed === 0) {
276 return false;
277 }
278 $pallet->refresh();
279 $this->palletClosed($pallet);
280
281 return true;
282 }

305 /** @param array<string, mixed> $measured what was measured at packing (the weight), kept on the "packed" event */
306 public function packUnit(SerialUnit $unit, User $operator, ?UnitCarton $carton = null, ?Pallet $pallet = null, ?int $workstationId = null, array $measured = []): SerialUnit
307 {
308 [$unit, $firstPacking] = DB::transaction(function () use ($unit, $operator, $carton, $pallet, $workstationId, $measured) {
309 // Read again under lock: two benches, or a scan racing the box's close,
310 // must not both see room in the same carton or an open pallet.
311 $unit = SerialUnit::whereKey($unit->id)->lockForUpdate()->firstOrFail();
312 $carton = $carton ? UnitCarton::whereKey($carton->id)->lockForUpdate()->firstOrFail() : null;
313 $pallet = $pallet ? Pallet::whereKey($pallet->id)->lockForUpdate()->firstOrFail() : null;
314
315 $this->assertPackable($unit);
316 if ($carton && ! $carton->isOpen()) {
317 throw new BindingException(__('Carton :no is closed.', ['no' => $carton->carton_no]));
318 }
319 if ($unit->carton_id && $carton && $unit->carton_id !== $carton->id) {
320 throw new BindingException(__('Unit :sn is already in carton :no.', ['sn' => $unit->display_id, 'no' => $unit->carton?->carton_no]));
321 }
322 // The box is full at the packing step's size: the server holds that
323 // line, whatever the station's auto-close managed to do.
324 $capacity = $this->cartonCapacityFor($unit->work_order_id);
325 if ($carton && $unit->carton_id !== $carton->id && $capacity && $carton->qty >= $capacity) {
326 throw new BindingException(__('Carton :no is full (:count units).', ['no' => $carton->carton_no, 'count' => $capacity]));
327 }
328
329 // A box already on a pallet takes the unit there; another pallet would
330 // split the box. A box with no pallet yet goes onto this one first.
331 if ($carton && $pallet && $carton->pallet_id && $carton->pallet_id !== $pallet->id) {
332 throw new BindingException(__('Carton :no is already on pallet :pallet.', ['no' => $carton->carton_no, 'pallet' => $carton->pallet?->pallet_no]));
333 }
334 if ($carton && $pallet && ! $carton->pallet_id) {
335 $carton = $this->assignCartonToPallet($carton, $pallet, $operator, $workstationId);
336 $unit->refresh();
337 }
338
339 $pallet ??= $carton?->pallet;
340 // A closed or shipped pallet takes nothing more, not even through a box still on it.
341 if ($pallet && ! $pallet->isOpen()) {
342 throw new BindingException(__('Pallet :no is not open.', ['no' => $pallet->pallet_no]));
343 }
344 // A box or pallet belongs to an order; a unit from another order does not go in it.
345 foreach ([$carton?->carton_no => $carton, $pallet?->pallet_no => $pallet] as $no => $container) {
346 if ($container && $unit->work_order_id && $container->work_order_id && $container->work_order_id !== $unit->work_order_id) {
347 throw new BindingException(__('Unit :sn belongs to another work order than :no.', ['sn' => $unit->display_id, 'no' => $no]));
348 }
349 }
350 // A unit already on a pallet stays there; moving it is a supervisor's job, not a rescan.
351 if ($pallet && $unit->pallet_id && $unit->pallet_id !== $pallet->id) {
352 throw new BindingException(__('Unit :sn is already on pallet :no.', ['sn' => $unit->display_id, 'no' => $unit->pallet?->pallet_no]));
353 }
354 $alreadyInCarton = $carton && $unit->carton_id === $carton->id;
355 $newOnPallet = $pallet && $unit->pallet_id !== $pallet->id;
356 $firstPacking = $unit->packed_at === null;
357 // The same label read twice (a wedge scanner, a nervous hand) changes
358 // nothing and writes nothing: the history keeps one "packed" per move.
359 if (! $firstPacking && ($alreadyInCarton || (! $carton && ! $newOnPallet))) {
360 return [$unit, false];
361 }
362
363 $unit->fill([
364 'carton_id' => $carton?->id ?? $unit->carton_id,
365 'pallet_id' => $pallet?->id ?? $unit->pallet_id,
366 'packed_at' => $unit->packed_at ?? now(),
367 'status' => $unit->status === SerialUnit::STATUS_IN_PRODUCTION ? SerialUnit::STATUS_COMPLETED : $unit->status,
368 ])->save();
369
370 if ($carton && ! $alreadyInCarton) {
371 $carton->increment('qty');
372 }
373 // "Pieces on pallet" counts serialised units too, whether they land
374 // there directly or through a carton that is already on it.
375 if ($newOnPallet) {
376 $pallet->increment('qty');
377 $pallet->recomputeQualityStatus();
378 }
379
380 $this->serials->recordStep($unit, $operator, null, [
381 'workstation_id' => $workstationId ?? $operator->workstation_id,
382 'parameters' => array_filter([
383 'event' => 'packed',
384 'psn' => $unit->psn,
385 'carton_no' => $carton?->carton_no,
386 'pallet_no' => $pallet?->pallet_no,
387 ...$measured,
388 ], fn ($v) => $v !== null && $v !== ''),
389 ]);
390
391 return [$unit->fresh(), $firstPacking];
392 });
393
394 // One unit counts once on the packing step, however often its label is
395 // rescanned - and only once the packing itself is safely committed.
396 if ($firstPacking) {
397 $this->countOnPackingStep($unit, $operator);
398 }
399
400 return $unit;
401 }

403 /**
404 * The routing's packing step, if the order has one, gets the piece: that
405 * is what makes packing a step with a station, a duration and a count like
406 * any other, instead of activity beside the routing. A READY step is
407 * started by the first piece. Nothing here may stop the packing itself -
408 * a step the flow rules refuse (not reached yet, a required control still
409 * open, production not yet available) is logged and skipped, whatever the
410 * rule throws.
411 */
412 private function countOnPackingStep(SerialUnit $unit, User $operator): void
413 {
414 $step = $this->packingStepFor($unit->work_order_id, $unit->batch_id);
415 if (! $step) {
416 return;
417 }
418
419 try {
420 if ($step->status === BatchStep::STATUS_READY) {
421 $step = $this->batches->startStep($step, $operator);
422 }
423 $this->batches->recordQuantity($step, $operator, 1, 0);
424 } catch (\Throwable $e) {
425 Log::info('Packing step not counted: '.$e->getMessage(), ['batch_step_id' => $step->id, 'serial_unit_id' => $unit->id]);
426 }
427 }

429 /**
430 * The packing steps a station works on right now: every order with a
431 * packing step that is ready or running, with the step's configuration.
432 * With a workstation selected only steps bound to it (or to none) show.
433 *
434 * @return array<int, array<string, mixed>>
435 */
436 public function openPackingSteps(?int $workstationId = null): array
437 {
438 return BatchStep::query()
439 ->where('kind', TemplateStep::KIND_PACKING)
440 ->whereIn('status', [BatchStep::STATUS_PENDING, BatchStep::STATUS_READY, BatchStep::STATUS_IN_PROGRESS])
441 ->when($workstationId, fn ($q) => $q->where(fn ($w) => $w->whereNull('workstation_id')->orWhere('workstation_id', $workstationId)))
442 ->with(['batch:id,work_order_id,batch_number,target_qty', 'batch.workOrder:id,order_no,product_type_id,planned_qty', 'batch.workOrder.productType:id,name', 'workstation:id,name'])
443 ->whereHas('batch.workOrder', fn ($q) => $q->packable())
444 ->orderBy('status')->orderByDesc('id')
445 ->limit(50)
446 ->get()
447 ->map(fn (BatchStep $s) => [
448 'id' => $s->id,
449 'name' => $s->name,
450 'status' => $s->status,
451 'step_number' => $s->step_number,
452 'config' => $s->config ?? [],
453 'workstation' => $s->workstation?->name,
454 'batch_number' => $s->batch?->batch_number,
455 'target_qty' => (float) ($s->batch?->target_qty ?? 0),
456 'passed_qty' => (float) $s->passed_qty,
457 'work_order_id' => $s->batch?->work_order_id,
458 'order_no' => $s->batch?->workOrder?->order_no,
459 'product' => $s->batch?->workOrder?->productType?->name,
460 'materials' => $this->packingMaterials($s),
461 ])
462 ->values()
463 ->all();
464 }

508 /**
509 * Put a carton (and every unit in it) on a pallet. The pallet's piece count
510 * grows by the carton's, so "pieces on pallet" stays true for serialised
511 * goods that never went through the EAN scan.
512 */
513 public function assignCartonToPallet(UnitCarton $carton, Pallet $pallet, User $operator, ?int $workstationId = null): UnitCarton
514 {
515 return DB::transaction(function () use ($carton, $pallet, $operator, $workstationId) {
516 if (! $pallet->isOpen()) {
517 throw new BindingException(__('Pallet :no is not open.', ['no' => $pallet->pallet_no]));
518 }
519 if ($carton->pallet_id === $pallet->id) {
520 return $carton;
521 }
522 if ($carton->pallet_id !== null) {
523 throw new BindingException(__('Carton :no is already on pallet :pallet.', ['no' => $carton->carton_no, 'pallet' => $carton->pallet?->pallet_no]));
524 }
525 // A pallet belongs to an order, like a carton does: the two must agree.
526 if ($carton->work_order_id && $pallet->work_order_id && $carton->work_order_id !== $pallet->work_order_id) {
527 throw new BindingException(__('Carton :no belongs to another order than pallet :pallet.', ['no' => $carton->carton_no, 'pallet' => $pallet->pallet_no]));
528 }
529 // A unit already counted on another pallet does not move silently.
530 $elsewhere = $carton->units()->whereNotNull('pallet_id')->where('pallet_id', '!=', $pallet->id)->first();
531 if ($elsewhere) {
532 throw new BindingException(__('Unit :sn is already on pallet :no.', ['sn' => $elsewhere->display_id, 'no' => $elsewhere->pallet?->pallet_no]));
533 }
534
535 $carton->update(['pallet_id' => $pallet->id]);
536 // Only the units not yet on this pallet count and get a history row:
537 // one may have been scanned straight onto it before the box was placed.
538 $arriving = $carton->units()->where(fn ($q) => $q->whereNull('pallet_id')->orWhere('pallet_id', '!=', $pallet->id))->get();
539 if ($arriving->isNotEmpty()) {
540 $pallet->increment('qty', $arriving->count());
541 }
542
543 foreach ($arriving as $unit) {
544 $unit->update(['pallet_id' => $pallet->id]);
545 $this->serials->recordStep($unit, $operator, null, [
546 'workstation_id' => $workstationId ?? $operator->workstation_id,
547 'parameters' => ['event' => 'palletised', 'carton_no' => $carton->carton_no, 'pallet_no' => $pallet->pallet_no],
548 ]);
549 }
550 // Only now are the carton's units on the pallet: its quality is theirs.
551 $pallet->recomputeQualityStatus();
552
553 return $carton->fresh();
554 });
555 }

557 /**
558 * The pallet left: every unit on it is shipped, scrap excepted. Called from
559 * the pallet's own status transition so it holds whichever screen ships it.
560 */
561 public function markPalletShipped(Pallet $pallet, ?User $actor = null): int
562 {
563 // A held unit does not leave with the pallet unnoticed: it keeps its hold.
564 $units = SerialUnit::where('pallet_id', $pallet->id)
565 ->whereNotIn('status', [SerialUnit::STATUS_SCRAPPED, SerialUnit::STATUS_SHIPPED, SerialUnit::STATUS_BLOCKED])
566 ->get();
567
568 foreach ($units as $unit) {
569 $unit->update(['status' => SerialUnit::STATUS_SHIPPED, 'shipped_at' => $pallet->shipped_at ?? now()]);
570 if ($actor) {
571 $this->serials->recordStep($unit, $actor, null, [
572 'parameters' => ['event' => 'shipped', 'pallet_no' => $pallet->pallet_no],
573 ]);
574 }
575 }
576
577 return $units->count();
578 }
