20 /**
21 * Calculate and store OEE for a line on a given date.
22 * If shift is provided, calculates per-shift; otherwise whole day.
23 */
24 public function calculateForDate(Line $line, Carbon $date, ?Shift $shift = null): ?OeeRecord
25 {
26 $plannedMinutes = $this->getPlannedMinutes($line, $date, $shift);
27
28 if ($plannedMinutes <= 0) {
29 return null;
30 }
31
32 // Downtimes
33 $lossDowntime = $this->downtimeService->getLossMinutes($line->id, $date, $shift?->id);
34 $plannedDowntime = $this->downtimeService->getPlannedMinutes($line->id, $date, $shift?->id);
35
36 // Subtract planned downtime from planned time (planned breaks don't count as losses)
37 $netPlannedMinutes = $plannedMinutes - $plannedDowntime;
38 $operatingMinutes = max(0, $netPlannedMinutes - $lossDowntime);
39
40 // Production data
41 $productionData = $this->getProductionData($line, $date, $shift);
42 $totalProduced = $productionData['total_produced'];
43 $scrapQty = $productionData['scrap_qty'];
44 $goodProduced = $totalProduced - $scrapQty;
45
46 // Ideal cycle time
47 $idealCycle = $this->getIdealCycleTime($line, $date, $shift);
48
49 // Calculate OEE components
50 $availability = $netPlannedMinutes > 0
51 ? ($operatingMinutes / $netPlannedMinutes) * 100
52 : null;
53
54 $performance = ($operatingMinutes > 0 && $idealCycle > 0)
55 ? (($totalProduced * $idealCycle) / $operatingMinutes) * 100
56 : null;
57
58 $quality = $totalProduced > 0
59 ? ($goodProduced / $totalProduced) * 100
60 : null;
61
62 // Cap performance at 100% to avoid nonsensical values
63 if ($performance !== null && $performance > 100) {
64 $performance = 100;
65 }
66
67 $oee = ($availability !== null && $performance !== null && $quality !== null)
68 ? ($availability / 100) * ($performance / 100) * ($quality / 100) * 100
69 : null;
70
71 // Store/update record
72 return OeeRecord::updateOrCreate(
73 [
74 'line_id' => $line->id,
75 'workstation_id' => null,
76 'shift_id' => $shift?->id,
77 'record_date' => $date->toDateString(),
78 ],
79 [
80 'planned_minutes' => $netPlannedMinutes,
81 'operating_minutes' => $operatingMinutes,
82 'downtime_minutes' => $lossDowntime,
83 'ideal_cycle_minutes' => $idealCycle,
84 'total_produced' => $totalProduced,
85 'good_produced' => $goodProduced,
86 'scrap_qty' => $scrapQty,
87 'availability_pct' => $availability !== null ? round($availability, 2) : null,
88 'performance_pct' => $performance !== null ? round($performance, 2) : null,
89 'quality_pct' => $quality !== null ? round($quality, 2) : null,
90 'oee_pct' => $oee !== null ? round($oee, 2) : null,
91 ]
92 );
93 }

95 /**
96 * Calculate OEE for all active lines for a given date.
97 */
98 public function calculateAll(Carbon $date): array
99 {
100 $lines = Line::where('is_active', true)->get();
101 $records = [];
102
103 foreach ($lines as $line) {
104 $shifts = Shift::where('line_id', $line->id)->where('is_active', true)->get();
105
106 if ($shifts->isEmpty()) {
107 // No shifts defined → calculate whole day (assume 8h default)
108 $record = $this->calculateForDate($line, $date);
109 if ($record) {
110 $records[] = $record;
111 }
112 } else {
113 foreach ($shifts as $shift) {
114 $record = $this->calculateForDate($line, $date, $shift);
115 if ($record) {
116 $records[] = $record;
117 }
118 }
119 }
120 }
121
122 return $records;
123 }

125 /**
126 * Get planned production minutes for a line/shift/date.
127 */
128 private function getPlannedMinutes(Line $line, Carbon $date, ?Shift $shift): int
129 {
130 if ($shift) {
131 return $this->shiftDurationMinutes($shift);
132 }
133
134 // If no specific shift, check if line has shifts defined
135 $shifts = Shift::where('line_id', $line->id)->where('is_active', true)->get();
136
137 if ($shifts->isNotEmpty()) {
138 return $shifts->sum(fn ($s) => $this->shiftDurationMinutes($s));
139 }
140
141 // Default: 8 hours
142 return 480;
143 }

145 private function shiftDurationMinutes(Shift $shift): int
146 {
147 $start = Carbon::parse($shift->start_time);
148 $end = Carbon::parse($shift->end_time);
149
150 // Handle overnight shifts
151 if ($end->lte($start)) {
152 $end->addDay();
153 }
154
155 return (int) $start->diffInMinutes($end);
156 }

158 /**
159 * Get production quantities for a line on a given date/shift.
160 */
161 private function getProductionData(Line $line, Carbon $date, ?Shift $shift): array
162 {
163 $query = Batch::whereHas('workOrder', fn ($q) => $q->where('line_id', $line->id))
164 ->where('status', Batch::STATUS_DONE);
165
166 if ($shift) {
167 // The occurrence, not the calendar day. A night shift's whereDate
168 // and its 22:00–06:00 window intersect to two hours, so everything
169 // produced after midnight was dropped — while the downtime side of
170 // the same record now covers the full eight. Half-open at the end so
171 // a batch landing exactly on the boundary belongs to one shift only.
172 $window = ShiftWindow::startingOn($shift, $date);
173
174 $query->where('completed_at', '>=', $window->start)
175 ->where('completed_at', '<', $window->end);
176 } else {
177 $query->whereDate('completed_at', $date);
178 }
179
180 $totals = $query->select([
181 DB::raw('COALESCE(SUM(produced_qty), 0) as total_produced'),
182 DB::raw('COALESCE(SUM(scrap_qty), 0) as scrap_qty'),
183 ])->first();
184
185 return [
186 'total_produced' => (float) ($totals->total_produced ?? 0),
187 'scrap_qty' => (float) ($totals->scrap_qty ?? 0),
188 ];
189 }

191 /**
192 * Get ideal cycle time for a line.
193 * Priority: 1) Manual on template, 2) Best historical, 3) Estimated from steps
194 */
195 private function getIdealCycleTime(Line $line, Carbon $date, ?Shift $shift): float
196 {
197 // Find most common process template used on this line recently
198 $templateId = Batch::where('batches.status', Batch::STATUS_DONE)
199 ->whereDate('batches.completed_at', '>=', $date->copy()->subDays(30))
200 ->join('work_orders', 'batches.work_order_id', '=', 'work_orders.id')
201 ->where('work_orders.line_id', $line->id)
202 ->whereNotNull('work_orders.process_snapshot')
203 ->selectRaw("work_orders.process_snapshot->>'template_id' as tpl_id")
204 ->groupBy('tpl_id')
205 ->orderByRaw('COUNT(*) DESC')
206 ->value('tpl_id');
207
208 if ($templateId) {
209 $template = ProcessTemplate::find($templateId);
210
211 // 1. Manual ideal cycle time
212 if ($template?->ideal_cycle_minutes) {
213 return (float) $template->ideal_cycle_minutes;
214 }
215 }
216
217 // 2. Best historical cycle time (fastest batch in last 30 days).
218 // Computed in PHP so this query runs on both Postgres and SQLite (tests).
219 $batches = Batch::whereHas('workOrder', fn ($q) => $q->where('line_id', $line->id))
220 ->where('status', Batch::STATUS_DONE)
221 ->where('produced_qty', '>', 0)
222 ->whereDate('completed_at', '>=', $date->copy()->subDays(30))
223 ->whereNotNull('started_at')
224 ->whereNotNull('completed_at')
225 ->get(['started_at', 'completed_at', 'produced_qty']);
226
227 $bestCycle = $batches
228 ->map(fn ($b) => $b->started_at->diffInMinutes($b->completed_at) / max(0.0001, (float) $b->produced_qty))
229 ->filter(fn ($v) => $v > 0)
230 ->min();
231
232 if ($bestCycle !== null && $bestCycle > 0) {
233 return round((float) $bestCycle, 4);
234 }
235
236 // 3. Fallback: sum of estimated step durations
237 if (isset($template)) {
238 $totalEstimated = $template->steps()->sum('estimated_duration_minutes');
239 if ($totalEstimated > 0) {
240 return (float) $totalEstimated;
241 }
242 }
243
244 // Ultimate fallback — 1 minute (to avoid division by zero)
245 return 1.0;
246 }
