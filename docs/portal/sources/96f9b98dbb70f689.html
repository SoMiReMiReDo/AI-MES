72 public function test_allocate_decrements_stock_and_creates_allocation(): void
73 {
74 $allocs = $this->service->allocateForBatch($this->batch, $this->user);

76 $this->assertCount(1, $allocs);

213 public function test_allocate_with_manual_picks_uses_chosen_lots(): void
214 {
215 $this->enableLotTracking();
216 $step = $this->makeStep();
217 $lotA = $this->makeLot('LOT-A', 150);
218 $lotB = $this->makeLot('LOT-B', 100); // required = 210
219
220 $allocs = $this->service->allocateForBatch($this->batch, $this->user, [
221 $this->material->id => [
222 ['material_lot_id' => $lotA->id, 'picked_qty' => 150],
223 ['material_lot_id' => $lotB->id, 'picked_qty' => 60],
224 ],
225 ], attributeStepId: $step->id);
226
227 $this->assertSame(2, AllocationLotPick::count());
228 $this->assertEqualsWithDelta(0.0, (float) $lotA->fresh()->quantity_available, 0.0001);
229 $this->assertEqualsWithDelta(40.0, (float) $lotB->fresh()->quantity_available, 0.0001);
230 AllocationLotPick::all()->each(fn ($p) => $this->assertSame(AllocationLotPick::STRATEGY_MANUAL, $p->picking_strategy));
231 $this->assertSame($step->id, $allocs->first()->batch_step_id);
232 }

245 public function test_consume_writes_genealogy_from_picks_and_is_idempotent(): void
246 {
247 $this->enableLotTracking();
248 $step = $this->makeStep();
249 $lot = $this->makeLot('LOT-A', 300);
250
251 $this->service->allocateForBatch($this->batch, $this->user, [], attributeStepId: $step->id);
252 $this->service->consumeForBatch($this->batch);
253
254 $this->assertSame(1, BatchStepLotConsumption::count());
255 $row = BatchStepLotConsumption::first();
256 $this->assertSame($step->id, $row->batch_step_id);
257 $this->assertSame($lot->id, $row->material_lot_id);
258 $this->assertEqualsWithDelta(210.0, (float) $row->quantity_consumed, 0.0001);
259
260 // Allocations already CONSUMED → a second pass writes nothing more.
261 $this->service->consumeForBatch($this->batch);
262 $this->assertSame(1, BatchStepLotConsumption::count());
263 }
