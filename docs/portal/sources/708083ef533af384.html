23 public function forSource(MachineTag|TopicMapping $source): MachineCounter
24 {
25 $tag = $source instanceof MachineTag;
26 $connection = $tag ? $source->connection : $source->topic?->machineConnection;
27 abort_unless($connection, 404);
28
29 return MachineCounter::firstOrCreate(
30 [$tag ? 'machine_tag_id' : 'topic_mapping_id' => $source->id],
31 ['machine_connection_id' => $connection->id,
32 'workstation_id' => $tag ? $source->workstation_id : null,
33 'mode' => $tag ? ($source->signal_type === 'cycle_complete' ? 'pulse' : 'cumulative')
34 : ($source->action_type === TopicMapping::ACTION_COUNT_STEP ? 'pulse' : (($source->action_params['qty_increment'] ?? false) ? 'increment' : 'cumulative')),
35 'kind' => $tag ? match ($source->signal_type) {
36 'reject_count' => 'reject', 'good_count' => 'good', default => 'total'
37 } : 'good'],
38 );
39 }

70 public function configure(MachineCounter $counter, array $data, int $userId): void
71 {
72 DB::transaction(function () use ($counter, $data, $userId) {
73 $counter = MachineCounter::whereKey($counter->id)->lockForUpdate()->firstOrFail();
74 $ws = Workstation::findOrFail($data['workstation_id']);
75 $connection = $counter->connection;
76 if (! $connection || (string) $connection->tenant_id !== (string) $ws->line?->tenant_id
77 || ($connection->line_id && $connection->line_id !== $ws->line_id)
78 || ($counter->machine_tag_id && $counter->tag?->workstation_id !== $ws->id)) {
79 $this->fail('The counter and workstation must belong to the same machine line.');
80 }
81 if ($counter->topic_mapping_id && strpbrk($counter->mapping?->topic?->topic_pattern ?? '', '+#') !== false) {
82 $this->fail('Use an exact MQTT topic for a production counter; wildcard topics can mix machines.');
83 }
84 if ($counter->machine_tag_id && $connection->modbusConnection && $data['mode'] !== 'cumulative') {
85 $this->fail('The built-in Modbus poller requires cumulative counting. Use an event-aware gateway for pulse or increment counts.');
86 }
87 $step = empty($data['batch_step_id']) ? null : BatchStep::findOrFail($data['batch_step_id']);
88 if ($step) {
89 $this->validateTarget($counter, $step, $ws->id);
90 // Serialize competing assignments using the same order -> step lock order as ingestion.
91 WorkOrder::whereKey($step->batch->work_order_id)->lockForUpdate()->firstOrFail();
92 BatchStep::whereKey($step->id)->lockForUpdate()->firstOrFail();
93 if ($data['kind'] === 'good' && MachineCounter::where('batch_step_id', $step->id)->where('kind', 'good')->whereKeyNot($counter->id)->exists()) {
94 $this->fail('This step already has a good-count channel. Unassign it before connecting another source.');
95 }
96 }
97 $before = $counter->only(['mode', 'kind', 'workstation_id', 'batch_step_id']);
98 $counter->fill([
99 'mode' => $data['mode'], 'kind' => $data['kind'], 'workstation_id' => $ws->id,
100 'batch_step_id' => $step?->id,
101 ]);
102 if (! $counter->isDirty() && $counter->configured_at && $counter->source_fingerprint === $this->fingerprint($counter)) {
103 return;
104 }
105 $counter->fill(['last_raw' => null, 'last_read_at' => null, 'configured_at' => now(),
106 'reset_required' => false, 'source_fingerprint' => $this->fingerprint($counter)])->save();
107 $counter->readings()->create(['status' => 'configured', 'observed_at' => now(), 'batch_step_id' => $step?->id,
108 'payload' => ['before' => $before, 'after' => $counter->only(array_keys($before))],
109 'reviewed_by_id' => $userId, 'reviewed_at' => now(), 'review_note' => $data['note']]);
110 });
111 }

113 public function ingest(MachineCounter $counter, mixed $value, ?Carbon $at = null, ?string $eventId = null, ?array $sourceSnapshot = null): MachineCounterReading
114 {
115 $hasTimestamp = $at !== null;
116 $at = ($at ?? now())->copy()->utc();
117
118 return DB::transaction(function () use ($counter, $value, $at, $eventId, $hasTimestamp, $sourceSnapshot) {
119 // One lock serializes baseline, assignment, deduplication and production.
120 // All subsequent production locks follow order -> step, as BatchService does.
121 $counter = MachineCounter::whereKey($counter->id)->lockForUpdate()->firstOrFail();
122 if ($eventId !== null && $eventId !== '' && ($duplicate = $counter->readings()->where('event_id', $eventId)->first())) {
123 return $duplicate;
124 }
125 $numeric = is_numeric($value) && is_finite((float) $value) && (float) $value >= 0 && (float) $value <= 999999999999.99
126 && abs((float) $value - round((float) $value, 2)) < 0.000001;
127 $raw = $numeric ? round((float) $value, 2) : null;
128 $status = null;
129 $delta = 0.0;
130 $applied = 0.0;
131 if (! $numeric) {
132 $status = 'invalid_value';
133 } elseif (! $counter->configured_at) {
134 $status = 'unconfigured';
135 } elseif ($sourceSnapshot !== null && ! $this->matchesSnapshot($counter, $sourceSnapshot)) {
136 $status = 'source_changed';
137 $counter->reset_required = true;
138 } elseif (! $this->sourceActive($counter) || $counter->source_fingerprint !== $this->fingerprint($counter)) {
139 $status = 'source_changed';
140 $counter->reset_required = true;
141 } elseif ($at->greaterThan(now()->addSeconds(5))) {
142 $status = 'future_reading';
143 } elseif ($at->lessThan($counter->configured_at) || ($counter->last_read_at && ($at->lessThan($counter->last_read_at) || ($counter->mode === 'cumulative' && $at->equalTo($counter->last_read_at))))) {
144 $status = 'out_of_order';
145 } elseif ($counter->reset_required) {
146 $status = 'reset_required';
147 } elseif (! $hasTimestamp) {
148 $status = 'timestamp_required';
149 } elseif ($counter->mode !== 'cumulative' && (! $eventId || strlen($eventId) > 160)) {
150 $status = 'event_id_required';
151 } elseif ($counter->mode === 'pulse' && $raw !== 1.0) {
152 $status = 'invalid_pulse';
153 } else {
154 if ($counter->mode === 'cumulative') {
155 if ($counter->last_raw === null) {
156 $status = 'baseline';
157 } elseif ($raw < (float) $counter->last_raw) {
158 $status = 'reset_required';
159 $counter->reset_required = true;
160 } else {
161 $delta = round($raw - (float) $counter->last_raw, 2);
162 }
163 } else {
164 $delta = $raw;
165 }
166 if (! $counter->reset_required) {
167 $counter->last_raw = $raw;
168 $counter->last_read_at = $at;
169 }
170 if ($status === null) {
171 [$applied, $status] = $this->apply($counter, $counter->step, $delta);
172 }
173 }
174 $counter->save();
175 $reading = $counter->readings()->create([
176 'batch_step_id' => $counter->batch_step_id, 'event_id' => $eventId !== '' && strlen($eventId ?? '') <= 160 ? $eventId : null,
177 'payload' => ['value' => $value, 'kind' => $counter->kind, 'mode' => $counter->mode, 'workstation_id' => $counter->workstation_id],
178 'raw_value' => $raw, 'delta' => $delta, 'applied_qty' => $applied, 'status' => $status, 'observed_at' => $at,
179 ]);
180 if ($delta > 0) {
181 MachineEvent::create(['workstation_id' => $counter->workstation_id, 'machine_connection_id' => $counter->machine_connection_id,
182 'event_type' => MachineEvent::TYPE_COUNTER, 'event_timestamp' => $at, 'correlation_id' => (string) Str::uuid(),
183 'payload' => ['kind' => $counter->kind, 'value' => $raw, 'delta' => $delta, 'applied_qty' => $applied, 'counter_reading_id' => $reading->id]]);
184 }
185
186 return $reading;
187 }, 3);
188 }

190 public function rebaseline(MachineCounter $counter, string $note, int $userId): void
191 {
192 DB::transaction(function () use ($counter, $note, $userId) {
193 $counter = MachineCounter::whereKey($counter->id)->lockForUpdate()->firstOrFail();
194 if ($counter->mode !== 'cumulative') {
195 $this->fail('Only cumulative counters require a baseline.');
196 }
197 if (! $counter->configured_at || $counter->source_fingerprint !== $this->fingerprint($counter)) {
198 $this->fail('Save the counter configuration after changing its source.');
199 }
200 // The next fresh reading establishes the baseline; never manufacture a delta.
201 $counter->update(['last_raw' => null, 'last_read_at' => null, 'configured_at' => now(), 'reset_required' => false,
202 'source_fingerprint' => $this->fingerprint($counter)]);
203 $counter->readings()->create(['status' => 'rebaseline', 'observed_at' => now(), 'review_note' => $note,
204 'reviewed_by_id' => $userId, 'reviewed_at' => now(), 'batch_step_id' => $counter->batch_step_id]);
205 });

208 public function review(MachineCounter $counter, int $readingId, array $data, int $userId): void
209 {
210 DB::transaction(function () use ($counter, $readingId, $data, $userId) {
211 $counter = MachineCounter::whereKey($counter->id)->lockForUpdate()->firstOrFail();
212 $reading = $counter->readings()->whereKey($readingId)->lockForUpdate()->firstOrFail();
213 if ($reading->reviewed_at || ! in_array($reading->status, ['unassigned', 'blocked', 'partial', 'quality_unknown'], true)) {
214 $this->fail('This reading cannot be reconciled again.');
215 }
216 $applied = 0.0;
217 if ($data['decision'] === 'apply') {
218 if (($reading->payload['kind'] ?? null) !== 'good') {
219 $this->fail('Only confirmed good counts can be applied to production.');
220 }
221 $step = BatchStep::findOrFail($data['batch_step_id']);
222 $this->validateTarget($counter, $step, $counter->workstation_id);
223 $remaining = round((float) $reading->delta - (float) $reading->applied_qty, 2);
224 [$applied] = $this->apply($counter, $step, $remaining);
225 if ($applied !== $remaining || $applied <= 0) {
226 $this->fail('The selected step cannot accept the full remaining quantity.');
227 }
228 }
229 $reading->update(['reviewed_by_id' => $userId, 'reviewed_at' => now(), 'review_note' => $data['note']]);
230 // Preserve the original attribution and append the reconciliation separately.
231 $counter->readings()->create(['status' => $data['decision'] === 'apply' ? 'reconciled' : 'dismissed',
232 'observed_at' => now(), 'batch_step_id' => isset($step) ? $step->id : null, 'applied_qty' => $applied,
233 'payload' => ['reading_id' => $reading->id], 'reviewed_by_id' => $userId, 'reviewed_at' => now(), 'review_note' => $data['note']]);
234 });
235 }

237 private function apply(MachineCounter $counter, ?BatchStep $step, float $delta): array
238 {
239 if ($delta <= 0) {
240 return [0.0, 'unchanged'];
241 }
242 if (! $step) {
243 return [0.0, 'unassigned'];
244 }
245 if ($counter->kind !== 'good') {
246 return [0.0, 'quality_unknown'];
247 }
248 $order = $step->batch?->workOrder;
249 if (! $order) {
250 return [0.0, 'blocked'];
251 }
252 $order = WorkOrder::whereKey($order->id)->lockForUpdate()->firstOrFail();
253 $step = BatchStep::whereKey($step->id)->lockForUpdate()->firstOrFail();
254 if (! $order->isMachineCounted() || $step->workstation_id !== $counter->workstation_id
255 || (string) $order->tenant_id !== (string) $counter->connection?->tenant_id
256 || $order->line_id !== $counter->workstation?->line_id
257 || ($counter->connection?->line_id && $order->line_id !== $counter->connection->line_id)
258 || in_array($order->status, WorkOrder::TERMINAL_STATUSES, true)
259 || $step->status !== BatchStep::STATUS_IN_PROGRESS || $step->productionBlocker()) {
260 return [0.0, 'blocked'];
261 }
262 $applied = app(BatchService::class)->recordMachinePass($step, min($delta, $step->availableQty()));
263 // Whole-batch mode keeps its legacy order total, only at the final step.
264 if (! ProductionFlow::isTransfer() && $step->batch->lastEffectiveStep()?->is($step) && $applied > 0) {
265 app(MachineProductionService::class)->recordGoodCount($order, $applied);
266 }
267
268 return [$applied, $applied <= 0 ? 'blocked' : ($applied < $delta ? 'partial' : 'applied')];
269 }

271 private function validateTarget(MachineCounter $counter, BatchStep $step, int $workstationId): void
272 {
273 $order = $step->batch?->workOrder;
274 if (! $order || ! $order->isMachineCounted() || $step->workstation_id !== $workstationId
275 || (string) $order->tenant_id !== (string) $counter->connection?->tenant_id
276 || $order->line_id !== Workstation::find($workstationId)?->line_id
277 || in_array($step->status, [BatchStep::STATUS_DONE, BatchStep::STATUS_SKIPPED], true)
278 || in_array($order->status, WorkOrder::TERMINAL_STATUSES, true)) {
279 $this->fail('Choose an open machine-counted step at this workstation.');
280 }

296 return ['attributes' => $source->only($source instanceof MachineTag

298 : ['field_path', 'action_type', 'action_params', 'condition_expr'])];

301 private function matchesSnapshot(MachineCounter $counter, array $snapshot): bool
302 {
303 $source = $counter->machine_tag_id ? $counter->tag : $counter->mapping;
304 if (! $source || $source->only(array_keys($snapshot['attributes'])) != $snapshot['attributes']) {
305 return false;
306 }
307
308 return ! isset($snapshot['modbus']) || $counter->connection?->modbusConnection?->only(array_keys($snapshot['modbus'])) == $snapshot['modbus'];
309 }

311 private function fingerprint(MachineCounter $counter): string
312 {
313 $source = $counter->machine_tag_id ? $counter->tag : $counter->mapping;
314
315 $connection = $counter->connection;
316 $transport = match ($connection?->protocol) {
317 'modbus' => $connection->modbusConnection?->only(['host', 'port', 'unit_id', 'byte_order', 'word_order']),
318 'opcua' => $connection->opcuaConnection?->only(['endpoint_url']),
319 'mqtt' => $connection->mqttConnection?->only(['broker_host', 'broker_port']),
320 default => null,
321 };
322
323 return hash('sha256', json_encode([
324 'source' => $source?->only($counter->machine_tag_id
325 ? ['address', 'transform', 'signal_type', 'workstation_id', 'data_type', 'register_type', 'unit']
326 : ['field_path', 'action_type', 'action_params', 'condition_expr']),
327 'topic' => $counter->mapping?->topic?->topic_pattern,
328 'connection' => $connection?->only(['protocol', 'line_id']),
329 'transport' => $transport,
330 ]));
331 }
